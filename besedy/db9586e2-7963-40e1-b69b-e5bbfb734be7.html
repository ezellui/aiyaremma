// Функция для извлечения категорий из обоих файлов
function extractCategories(content) {
  // Выделяем все термины в кавычках
  const matches = content.match(/"([^"]+)"/g);
  if (matches) {
    // Убираем кавычки и создаем множество уникальных категорий
    return new Set(matches.map(match => match.replace(/"/g, '')));
  }
  return new Set();
}

// Функция для извлечения связей из обоих файлов
function extractRelations(content) {
  // Ищем строки после "Типы фундаментальных философских связей:"
  const relationsSection = content.split(/Типы фундаментальных философских связей:?/)[1];
  if (!relationsSection) return new Set();
  
  // Выделяем термины в кавычках или без кавычек перед скобками
  const relations = [];
  const lines = relationsSection.split('\n');
  
  for (const line of lines) {
    if (line.includes('(А')) {
      let relation;
      // Проверяем, есть ли кавычки
      const matchWithQuotes = line.match(/"([^"]+)"/);
      if (matchWithQuotes) {
        relation = matchWithQuotes[1];
      } else {
        const matchWithoutQuotes = line.match(/^([^(]+)/);
        if (matchWithoutQuotes) {
          relation = matchWithoutQuotes[1].trim();
        }
      }
      if (relation) relations.push(relation);
    }
  }
  
  return new Set(relations);
}

function extractStructuredGroups(content) {
  const groups = {};
  const lines = content.split('\n');
  let currentGroup = '';
  
  for (const line of lines) {
    if (line.match(/^\/\/|^###/)) {
      currentGroup = line.replace(/\/\/|###/g, '').trim();
      groups[currentGroup] = new Set();
    } 
    else if (currentGroup && line.includes('"')) {
      const matches = line.match(/"([^"]+)"/g);
      if (matches) {
        matches.forEach(match => {
          const term = match.replace(/"/g, '');
          groups[currentGroup].add(term);
        });
      }
    }
  }
  
  // Преобразуем множества в массивы для удобства
  Object.keys(groups).forEach(key => {

  });
  
  return groups;
}
