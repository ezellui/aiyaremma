  const traditions = [];

    traditions.push(...Object.values(category));
  }
  return traditions;
}

/**
 * Get tradition category for a specific tradition
 * @param {string} tradition - The tradition to check
 * @returns {string|null} - The category name or null if not found
 */
function getTraditionCategory(tradition) {
  for (const [category, traditions] of Object.entries(PHILOSOPHY_CONSTANTS.PHILOSOPHICAL_TRADITIONS)) {
    if (Object.values(traditions).includes(tradition)) {
      return category;
    }
  }
  return null;
}

  1. SYNTHESIS_TYPES - полная версия с 10 типами (диалектический, интегративный, творческий, сравнительный, исторический, концептуальный, прагматический, методологический, герменевтический, кросс-культурный)

  2. SYNTHESIS_METHODS - упрощенная версия с 4 методами (диалектический, дополнительный, иерархический, гибридный)

Судя по архитектуре системы, они используются в разных контекстах:

  • SYNTHESIS_TYPES - для детальной классификации в сервисах синтеза
  • SYNTHESIS_METHODS - для простого выбора в UI компонентах

Я добавил комментарий к SYNTHESIS_METHODS, что это упрощенная версия для обратной совместимости и используется в некоторых UI компонентах для простого выбора.

Это позволяет сохранить обратную совместимость и при этом иметь более детальную классификацию там, где это необходимо.
