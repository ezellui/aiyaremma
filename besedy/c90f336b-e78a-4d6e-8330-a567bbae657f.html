Я должен проанализировать приложенный HTML-файл и выявить причину, по которой в режиме "Доменной карты" корректно отображаются только функции z^2 и 1/z, а другие функции не вычисляются.

Для начала найду код, отвечающий за рендеринг доменной карты. В файле есть функция renderDomainColoring(), которая отвечает за этот режим визуализации.

В этой функции я вижу использование Web Worker для рендеринга доменной карты в фоновом режиме. Вот ключевая часть кода:

// Используем Web Worker для рендеринга доменной карты в фоновом режиме
if (window.Worker) {
    const worker = new Worker(URL.createObjectURL(new Blob([`
        // Определение функций внутри Web Worker
        ${hslToRgb.toString()}
        
        self.onmessage = function(e) {
            const { width, height, functionStr, viewScale, viewOffsetX, viewOffsetY, centerX, centerY } = e.data;
            
            const imageData = new ImageData(width, height);
            const data = imageData.data;
            
            // Размер шага для вычисления значений
            const stepSize = 1 / viewScale;
            
            // Верхняя левая точка видимой области в комплексной плоскости
            const startRe = -width / (2 * 100 * viewScale) - viewOffsetX;
            const startIm = height / (2 * 100 * viewScale) - viewOffsetY;
            
            // Итерация по всем пикселям
            for (let j = 0; j < height; j++) {
                for (let i = 0; i < width; i++) {
                    // Преобразуем координаты пикселя в комплексную плоскость
                    const re = startRe + i * stepSize / 100;
                    const im = startIm - j * stepSize / 100;
                    
                    try {
                        // Вычисляем значение функции (упрощенно для Worker)
                        // Используем строковую функцию для базовых операций
                        let result;
                        const z = {re: re, im: im};
                        
                        // Очень упрощенная версия расчета для часто используемых функций
                        if (functionStr === 'z^2') {
                            result = {
                                re: z.re * z.re - z.im * z.im,
                                im: 2 * z.re * z.im
                            };
                        } else if (functionStr === '1/z') {
                            const denom = z.re * z.re + z.im * z.im;
                            result = {
                                re: z.re / denom,
                                im: -z.im / denom
                            };
                        } else {
                            // Для других функций используем простую реализацию
                            result = {re: re, im: im};
                        }
                        
                        // Получаем аргумент и модуль
                        const arg = Math.atan2(result.im, result.re);
                        const mag = Math.sqrt(result.re * result.re + result.im * result.im);
                        
                        // Преобразуем в цвет
                        const hue = ((arg + Math.PI) / (2 * Math.PI)) * 360;
                        const norm = 2 * (Math.atan(mag / 2) / Math.PI);
                        const sat = 80 + 20 * Math.sin(norm * Math.PI * 2);
                        const light = 60 - 30 * Math.cos(norm * Math.PI);
                        
                        // Преобразуем HSL в RGB
                        const rgb = hslToRgb(hue / 360, sat / 100, light / 100);
                        
                        // Индекс в массиве данных изображения
                        const idx = (j * width + i) * 4;
                        
                        // Устанавливаем RGBA значения
                        data[idx] = rgb[0];
                        data[idx + 1] = rgb[1];
                        data[idx + 2] = rgb[2];
                        data[idx + 3] = 255; // Альфа-канал
                    } catch (e) {
                        // Ошибка при вычислении (e.g. деление на ноль)
                        // Окрашиваем пиксель в черный цвет
                        const idx = (j * width + i) * 4;
                        data[idx] = 0;
                        data[idx + 1] = 0;
                        data[idx + 2] = 0;
                        data[idx + 3] = 255;
                    }
                }
            }
            
            self.postMessage({imageData: imageData});
        };
