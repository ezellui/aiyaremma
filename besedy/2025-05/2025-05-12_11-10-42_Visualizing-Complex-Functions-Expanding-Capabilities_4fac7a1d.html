// Функция Бесселя первого рода
besselJ: function(nu, z) {
    // Приближение для целых nu >= 0 и произвольного z
    if (Math.abs(Math.round(math.re(nu)) - math.re(nu)) < 1e-10 && math.re(nu) >= 0 && math.im(nu) === 0) {
        const n = Math.round(math.re(nu));
        let sum = math.complex(0);
        const maxIter = 50;
        
        for (let m = 0; m < maxIter; m++) {
            const term = math.multiply(
                math.pow(math.complex(-1), m),
                math.divide(
                    math.pow(math.divide(z, 2), math.add(n, math.multiply(2, m))),
                    math.multiply(
                        ComplexMath.gamma(math.add(m, 1)),
                        ComplexMath.gamma(math.add(n, m, 1))
                    )
                )
            );
            sum = math.add(sum, term);
            
            // Проверка сходимости
            if (math.abs(term) < 1e-10) break;
        }
        
        return sum;
    }
    
    // Общий случай для произвольных nu и z
    let sum = math.complex(0);
    const maxIter = 50;
    
    for (let m = 0; m < maxIter; m++) {
        const term = math.multiply(
            math.pow(math.complex(-1), m),
            math.divide(
                math.pow(math.divide(z, 2), math.add(nu, math.multiply(2, m))),
                math.multiply(
                    ComplexMath.gamma(math.add(m, 1)),
                    ComplexMath.gamma(math.add(nu, m, 1))
                )
            )
        );
        sum = math.add(sum, term);
        
        // Проверка сходимости
        if (math.abs(term) < 1e-10) break;
    }
    
    return sum;
},

// Функция Бесселя второго рода
besselY: function(nu, z) {
    // Y_nu(z) = (J_nu(z)cos(nu*pi) - J_{-nu}(z))/sin(nu*pi)
    const nuPi = math.multiply(nu, math.pi);
    return math.divide(
        math.subtract(
            math.multiply(ComplexMath.besselJ(nu, z), math.cos(nuPi)),
            ComplexMath.besselJ(math.multiply(nu, -1), z)
        ),
        math.sin(nuPi)
    );
},

// W-функция Ламберта (главная ветвь)
lambertW: function(z) {
    // Начальное приближение
    let w;
    if (math.abs(z) < 1) {
        w = z;
    } else {
        const lnz = math.log(z);
        w = math.subtract(lnz, math.log(lnz));
    }
    
    // Итерационный метод Ньютона
    const maxIter = 20;
    for (let i = 0; i < maxIter; i++) {
        const ew = math.exp(w);
        const wew = math.multiply(w, ew);
        const delta = math.divide(
            math.subtract(wew, z),
            math.add(ew, math.multiply(w, ew))
        );
        w = math.subtract(w, delta);
        
        // Проверка сходимости

theta: function(z, tau, type) {
    const q = math.exp(math.multiply(math.pi, math.multiply(math.i, tau)));
    let sum = math.complex(0);
    const maxIter = 20;
    
    if (type === 1) {
        // θ₁(z, τ)
        for (let n = 0; n < maxIter; n++) {
            const qn2 = math.pow(q, math.pow(n + 0.5, 2));
            const term = math.multiply(
                math.pow(math.complex(-1), n),
                math.multiply(
                    qn2,
                    math.sin(math.multiply(math.complex(2*n + 1), z))
                )
            );
            sum = math.add(sum, term);
        }
        return math.multiply(2, sum);
    } else if (type === 2) {
        // θ₂(z, τ)
        for (let n = 0; n < maxIter; n++) {
            const qn2 = math.pow(q, math.pow(n + 0.5, 2));
            const term = math.multiply(
                qn2,
                math.cos(math.multiply(math.complex(2*n + 1), z))
            );
            sum = math.add(sum, term);
        }
        return math.multiply(2, sum);
    } else if (type === 3) {
        // θ₃(z, τ)
        sum = math.complex(1);
        for (let n = 1; n < maxIter; n++) {
            const qn2 = math.pow(q, math.pow(n, 2));
            const term = math.multiply(
                math.multiply(2, qn2),
                math.cos(math.multiply(math.complex(2*n), z))
            );
            sum = math.add(sum, term);
        }
        return sum;
    } else if (type === 4) {
        // θ₄(z, τ)
        sum = math.complex(1);
        for (let n = 1; n < maxIter; n++) {
            const qn2 = math.pow(q, math.pow(n, 2));
            const term = math.multiply(
                math.multiply(2, math.pow(math.complex(-1), n), qn2),
                math.cos(math.multiply(math.complex(2*n), z))
            );
            sum = math.add(sum, term);
        }
        return sum;
    }
    
    return math.complex(NaN, NaN);
},

// Полином Лежандра
legendreP: function(n, z) {
    if (n === 0) return math.complex(1);
    if (n === 1) return z;
    
    // Рекуррентная формула: (n+1)P_{n+1}(z) = (2n+1)zP_n(z) - nP_{n-1}(z)
    let p_prev = math.complex(1);  // P_0(z)
    let p_curr = z;                 // P_1(z)
    
    for (let k = 1; k < n; k++) {
        const p_next = math.divide(
            math.subtract(
                math.multiply(math.multiply(2*k + 1, z), p_curr),
                math.multiply(k, p_prev)
            ),
            k + 1
        );
        p_prev = p_curr;
        p_curr = p_next;
    }
    
    return p_curr;
},

        }

        return math.complex(NaN, NaN);

function parseExpressionValue(expression) {
    try {
        // Заменяем pi и e на их значения
        const processedExpression = expression
            .replace(/pi/gi, 'Math.PI')
            .replace(/e/gi, 'Math.E');
        
        // Используем функцию eval с предосторожностями
        // eslint-disable-next-line no-new-func
        const result = new Function('return ' + processedExpression)();
        
        if (typeof result !== 'number' || isNaN(result)) {
            throw new Error('Результат не является числом');
        }
        
        return result;
    } catch (error) {
        console.error('Ошибка при парсинге выражения:', error);
        throw error;
    }
}

                !isFinite(math.re(w)) || !isFinite(math.im(w))) {
                // Устанавливаем цвет для undefined
                const pixelIndex = (y * width + x) * 4;
                gridBuffer[pixelIndex] = 0;     // R
                gridBuffer[pixelIndex + 1] = 0; // G
                gridBuffer[pixelIndex + 2] = 0; // B
                gridBuffer[pixelIndex + 3] = 255; // A
                continue;
            }
            
            // Разделяем на действительную и мнимую части
            const wReal = math.re(w);
            const wImag = math.im(w);
            
            // Проверяем, находится ли точка вблизи линии решетки на преобразованной плоскости
            const isOnGridReal = (Math.abs(Math.round(wReal / gridSpacing) * gridSpacing - wReal) < gridThickness);
            const isOnGridImag = (Math.abs(Math.round(wImag / gridSpacing) * gridSpacing - wImag) < gridThickness);
            
            // Получаем модуль и аргумент
            const modulus = math.abs(w);
            const phase = math.arg(w);
            
            // Нормализуем фазу в диапазон [0, 1]
            const normalizedPhase = (phase + Math.PI) / (2 * Math.PI);
            
            // Вычисляем яркость на основе модуля
            const brightness = 0.5 + 0.5 * Math.tanh(modulus - 1);
            
            // Определяем базовый цвет в зависимости от выбранной цветовой схемы
            let r, g, b;
            
            switch (state.colorScheme) {
                case 'standard':
                    // HSV to RGB с фазой как оттенком и модулем как яркостью
                    const hue = normalizedPhase * 360;
                    [r, g, b] = hsvToRgb(hue, 0.8, brightness);
                    break;
                    
                // ... другие цветовые схемы
                
                default:
                    // В случае неизвестной схемы используем стандартную
                    const defaultHue = normalizedPhase * 360;
                    [r, g, b] = hsvToRgb(defaultHue, 0.8, brightness);
            }
            
            // Подсвечиваем линии решетки
            if (isOnGridReal || isOnGridImag) {
                // Делаем линии решетки более светлыми
                r = Math.min(255, r + 70);
                g = Math.min(255, g + 70);
                b = Math.min(255, b + 70);
                
                // Добавляем градиент для линий решетки
                const gradientFactor = Math.sin(wReal * 0.5) * Math.sin(wImag * 0.5);
                const gradientAmount = Math.abs(gradientFactor) * 50;
                
                if (gradientFactor > 0) {
                    r = Math.min(255, r + gradientAmount);
                    g = Math.min(255, g + gradientAmount);
                    b = Math.min(255, b - gradientAmount);
                } else {
                    r = Math.min(255, r - gradientAmount);
                    g = Math.min(255, g + gradientAmount);
                    b = Math.min(255, b + gradientAmount);
                }
            }
            
            // Устанавливаем цвет пикселя
            const pixelIndex = (y * width + x) * 4;
            gridBuffer[pixelIndex] = r;
            gridBuffer[pixelIndex + 1] = g;
            gridBuffer[pixelIndex + 2] = b;
            gridBuffer[pixelIndex + 3] = 255; // Альфа-канал (непрозрачность)
        }
    }
    
    return gridBuffer;
}
