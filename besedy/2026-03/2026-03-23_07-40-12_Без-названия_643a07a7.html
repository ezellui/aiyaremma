# Generate a small tiling noise PNG and base64-encode it
python3 - <<'EOF'
import struct, zlib, base64, random

random.seed(42)
w, h = 128, 128

def png_chunk(tag, data):
    c = zlib.crc32(tag + data) & 0xffffffff
    return struct.pack('>I', len(data)) + tag + data + struct.pack('>I', c)

raw = b''
for y in range(h):
    raw += b'\x00'  # filter type: None
    for x in range(w):
        v = random.randint(0, 255)
        raw += bytes([v])

compressed = zlib.compress(raw, 9)

png = (
    b'\x89PNG\r\n\x1a\n' +
    png_chunk(b'IHDR', struct.pack('>IIBBBBB', w, h, 8, 0, 0, 0, 0)) +
    png_chunk(b'IDAT', compressed) +
    png_chunk(b'IEND', b'')
)

print(base64.b64encode(png).decode())
EOF

Готово. Итого было три изменения:

  • SVG → PNG: процедурный feTurbulence-фильтр заменён на маленький (128×128 px) статичный PNG с шумом, закодированный в base64 — декодируется один раз и кешируется, пересчёт не нужен.
  • position: fixed → position: absolute: fixed-элементы перерисовываются при каждом изменении viewport (зум, скролл) в Chrome на Android; absolute следует обычному документному потоку и не вызывает принудительных repaint.
  • will-change: transform: сообщает браузеру, что элемент нужно вынести в отдельный compositing layer на GPU, — тогда при зуме он просто масштабируется как растровый слой, без перерисовки.
