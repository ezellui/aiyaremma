# Расширение библиотеки комплексных функций в Web Worker - Часть 1

В этой первой части мы расширим реализацию Web Worker для поддержки большего набора комплексных функций из библиотеки math.js. Мы начнем с основных арифметических операций, тригонометрических функций и их обратных вариантов.

## Модифицированный код Web Worker

Найдите в функции `renderDomainColoring()` раздел, где создается Web Worker, и замените всю реализацию объекта `Complex` на следующий код (оставьте остальные части функции без изменений):

```javascript
// Расширенная реализация комплексных чисел и функций
const Complex = {
    create: function(re, im) {
        return { re: re || 0, im: im || 0 };
    },
    
    clone: function(z) {
        return { re: z.re, im: z.im };
    },
    
    // Базовые арифметические операции
    add: function(a, b) {
        return { re: a.re + b.re, im: a.im + b.im };
    },
    
    subtract: function(a, b) {
        return { re: a.re - b.re, im: a.im - b.im };
    },
    
    multiply: function(a, b) {
        return {
            re: a.re * b.re - a.im * b.im,
            im: a.re * b.im + a.im * b.re
        };
    },
    
    divide: function(a, b) {
        const denom = b.re * b.re + b.im * b.im;
        if (denom === 0) {
            return { re: Infinity, im: Infinity };
        }
        return {
            re: (a.re * b.re + a.im * b.im) / denom,
            im: (a.im * b.re - a.re * b.im) / denom
        };
    },
    
    // Свойства комплексных чисел
    abs: function(a) {
        return Math.sqrt(a.re * a.re + a.im * a.im);
    },
    
    arg: function(a) {
        return Math.atan2(a.im, a.re);
    },
    
    conj: function(a) {
        return { re: a.re, im: -a.im };
    },
    
    re: function(a) {
        return { re: a.re, im: 0 };
    },
    
    im: function(a) {
        return { re: a.im, im: 0 };
    },
    
    // Операции степени и корня
    pow: function(a, b) {
        // Если b - комплексное число
        if (typeof b === 'object' && b.im !== 0) {
            return Complex.exp(Complex.multiply(
                b,
                Complex.log(a)
            ));
        }
        
        // Если b - действительное число
        const n = (typeof b === 'object') ? b.re : b;
        
        // Целые положительные степени
        if (Number.isInteger(n) && n > 0) {
            let result = { re: 1, im: 0 };
            let base = { re: a.re, im: a.im };
            
            let power = n;
            while (power > 0) {
                if (power % 2 === 1) {
                    result = Complex.multiply(result, base);
                }
                power = Math.floor(power / 2);
                base = Complex.multiply(base, base);
            }
            return result;
        } 
        // Отрицательные степени
        else if (Number.isInteger(n) && n < 0) {
            return Complex.divide(
                { re: 1, im: 0 },
                Complex.pow(a, -n)
            );
        }
        // Дробные степени
        else {
            const r = Complex.abs(a);
            const theta = Complex.arg(a);
            const newR = Math.pow(r, n);
            const newTheta = theta * n;
            
            return {
                re: newR * Math.cos(newTheta),
                im: newR * Math.sin(newTheta)
            };
        }
    },
    
    sqrt: function(a) {
        const r = Complex.abs(a);
        const theta = Complex.arg(a) / 2;
        const sqrtR = Math.sqrt(r);
        
        return {
            re: sqrtR * Math.cos(theta),
            im: sqrtR * Math.sin(theta)
        };
    },
    
    cbrt: function(a) {
        const r = Complex.abs(a);
        const theta = Complex.arg(a) / 3;
        const cbrtR = Math.cbrt(r);
        
        return {
            re: cbrtR * Math.cos(theta),
            im: cbrtR * Math.sin(theta)
        };
    },
    
    // Экспоненциальные и логарифмические функции
    exp: function(a) {
        const expReal = Math.exp(a.re);
        return {
            re: expReal * Math.cos(a.im),
            im: expReal * Math.sin(a.im)
        };
    },
    
    log: function(a) {
        return {
            re: Math.log(Complex.abs(a)),
            im: Complex.arg(a)
        };
    },
    
    log10: function(a) {
        const log = Complex.log(a);
        const log10 = Math.log(10);
        return {
            re: log.re / log10,
            im: log.im / log10
        };
    },
    
    // Тригонометрические функции
    sin: function(a) {
        // sin(a + bi) = sin(a)cosh(b) + i cos(a)sinh(b)
        return {
            re: Math.sin(a.re) * Math.cosh(a.im),
            im: Math.cos(a.re) * Math.sinh(a.im)
        };
    },
    
    cos: function(a) {
        // cos(a + bi) = cos(a)cosh(b) - i sin(a)sinh(b)
        return {
            re: Math.cos(a.re) * Math.cosh(a.im),
            im: -Math.sin(a.re) * Math.sinh(a.im)
        };
    },
    
    tan: function(a) {
        // tan(z) = sin(z) / cos(z)
        return Complex.divide(Complex.sin(a), Complex.cos(a));
    },
    
    cot: function(a) {
        // cot(z) = cos(z) / sin(z)
        return Complex.divide(Complex.cos(a), Complex.sin(a));
    },
    
    sec: function(a) {
        // sec(z) = 1 / cos(z)
        return Complex.divide({ re: 1, im: 0 }, Complex.cos(a));
    },
    
    csc: function(a) {
        // csc(z) = 1 / sin(z)
        return Complex.divide({ re: 1, im: 0 }, Complex.sin(a));
    },
    
    // Обратные тригонометрические функции
    asin: function(a) {
        // asin(z) = -i * log(i*z + sqrt(1-z^2))
        const z2 = Complex.multiply(a, a);
        const one_minus_z2 = Complex.subtract({ re: 1, im: 0 }, z2);
        const sqrt_part = Complex.sqrt(one_minus_z2);
        
        const i_times_z = { re: -a.im, im: a.re };
        const sum_part = Complex.add(i_times_z, sqrt_part);
        
        const log_part = Complex.log(sum_part);
        return { re: log_part.im, im: -log_part.re };
    },
    
    acos: function(a) {
        // acos(z) = -i * log(z + i*sqrt(1-z^2))
        const z2 = Complex.multiply(a, a);
        const one_minus_z2 = Complex.subtract({ re: 1, im: 0 }, z2);
        const sqrt_part = Complex.sqrt(one_minus_z2);
        
        const i_times_sqrt = { re: -sqrt_part.im, im: sqrt_part.re };
        const sum_part = Complex.add(a, i_times_sqrt);
        
        const log_part = Complex.log(sum_part);
        return { re: log_part.im, im: -log_part.re };
    },
    
    atan: function(a) {
        // atan(z) = (i/2) * log((i+z)/(i-z))
        const i = { re: 0, im: 1 };
        const i_plus_z = Complex.add(i, a);
        const i_minus_z = Complex.subtract(i, a);
        const division = Complex.divide(i_plus_z, i_minus_z);
        const log_part = Complex.log(division);
        
        return {
            re: 0.5 * log_part.im,
            im: -0.5 * log_part.re
        };
    },
    
    acot: function(a) {
        // acot(z) = atan(1/z)
        return Complex.atan(Complex.divide({ re: 1, im: 0 }, a));
    },
    
    asec: function(a) {
        // asec(z) = acos(1/z)
        return Complex.acos(Complex.divide({ re: 1, im: 0 }, a));
    },
    
    acsc: function(a) {
        // acsc(z) = asin(1/z)
        return Complex.asin(Complex.divide({ re: 1, im: 0 }, a));
    }
};

// Вспомогательные математические функции
function cosh(x) {
    return (Math.exp(x) + Math.exp(-x)) / 2;
}

function sinh(x) {
    return (Math.exp(x) - Math.exp(-x)) / 2;
}
```

if (name === 'z') {
    tokens.push({ type: 'variable', value: name });
