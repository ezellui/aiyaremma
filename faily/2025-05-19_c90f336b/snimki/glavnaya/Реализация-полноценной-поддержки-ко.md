# Оптимизированная реализация Web Worker для визуализации комплексных функций

В этом документе приведено полное решение для исправления визуализации комплексных функций в режиме Доменной карты с сохранением преимуществ Web Worker для производительности. Это более сложный, но более эффективный подход, который позволит ускорить рендеринг сложных функций без потери качества.

## Основная проблема

Web Worker, используемый для рендеринга Доменной карты, имеет реализацию только для двух конкретных функций (`z^2` и `1/z`), а для всех остальных вместо вычисления используется тождественная функция `f(z) = z`.

## Решение

Добавить в Web Worker полную реализацию всех необходимых комплексных функций и парсер выражений, позволяющий вычислять произвольные формулы.

            const Complex = {
                add: function(a, b) {
                    return { re: a.re + b.re, im: a.im + b.im };
                },
                subtract: function(a, b) {
                    return { re: a.re - b.re, im: a.im - b.im };
                },
                multiply: function(a, b) {
                    return {
                        re: a.re * b.re - a.im * b.im,
                        im: a.re * b.im + a.im * b.re
                    };
                },
                divide: function(a, b) {
                    const denom = b.re * b.re + b.im * b.im;
                    if (denom === 0) {
                        return { re: Infinity, im: Infinity };
                    }
                    return {
                        re: (a.re * b.re + a.im * b.im) / denom,
                        im: (a.im * b.re - a.re * b.im) / denom
                    };
                },
                power: function(a, n) {
                    if (n === 0) return { re: 1, im: 0 };
                    if (n === 1) return { re: a.re, im: a.im };
                    if (Number.isInteger(n) && n > 0) {
                        let result = { re: 1, im: 0 };
                        let base = { re: a.re, im: a.im };
                        
                        while (n > 0) {
                            if (n % 2 === 1) {
                                result = Complex.multiply(result, base);
                            }
                            n = Math.floor(n / 2);
                            base = Complex.multiply(base, base);
                        }
                        return result;
                    } else {
                        // Для дробных и отрицательных степеней используем exp и log
                        return Complex.exp(Complex.multiply(
                            { re: n, im: 0 },
                            Complex.log(a)
                        ));
                    }
                },
                abs: function(a) {
                    return Math.sqrt(a.re * a.re + a.im * a.im);
                },
                arg: function(a) {
                    return Math.atan2(a.im, a.re);
                },
                exp: function(a) {
                    const expReal = Math.exp(a.re);
                    return {
                        re: expReal * Math.cos(a.im),
                        im: expReal * Math.sin(a.im)
                    };

                    };

                    const theta = Complex.arg(a) / 2;
                    const sqrtR = Math.sqrt(r);
                    return {
                        re: sqrtR * Math.cos(theta),
                        im: sqrtR * Math.sin(theta)
                    };
                }

                return (Math.exp(x) + Math.exp(-x)) / 2;
            }

                return (Math.exp(x) - Math.exp(-x)) / 2;

                const tokens = tokenize(funcStr);

            // Токенизатор
            function tokenize(expr) {
                const tokens = [];
                let i = 0;
                
                while (i < expr.length) {
                    let char = expr[i];
                    
                    // Пропуск пробелов
                    if (char === ' ') {
                        i++;
                        continue;
                    }
                    
                    // Числа

                        while (i < expr.length && /[0-9.]/.test(expr[i])) {
                            num += expr[i++];
                        }
                        tokens.push({ type: 'number', value: parseFloat(num) });
                        continue;
                    }
                    
                    // Операторы
                    if (/[+\\-*/^()]/.test(char)) {
                        tokens.push({ type: 'operator', value: char });
                        i++;
                        continue;
                    }
                    
                    // Функции и переменные
                    if (/[a-zA-Z]/.test(char)) {
                        let name = '';
                        while (i < expr.length && /[a-zA-Z0-9_]/.test(expr[i])) {
                            name += expr[i++];
                        }
                        
                        if (name === 'z') {
                            tokens.push({ type: 'variable', value: name });
                        } else if (['sin', 'cos', 'tan', 'exp', 'log', 'sqrt'].includes(name)) {
                            tokens.push({ type: 'function', value: name });
                        } else if (CONSTANTS[name]) {
                            tokens.push({ type: 'constant', value: name });
                        } else {
                            throw new Error('Неизвестная функция или переменная: ' + name);
                        }
                        continue;

                    // Неизвестный символ

                }

            // Алгоритм сортировочной станции (Shunting-yard)
            function shuntingYard(tokens) {
                const output = [];
                const operatorStack = [];
                const precedence = {
                    '+': 1, '-': 1,
                    '*': 2, '/': 2,
                    '^': 3
                };
                
                for (let i = 0; i < tokens.length; i++) {
                    const token = tokens[i];
                    
                    if (token.type === 'number' || token.type === 'variable' || token.type === 'constant') {
                        output.push(token);
                    } else if (token.type === 'function') {
                        operatorStack.push(token);
                    } else if (token.type === 'operator') {
                        if (token.value === '(') {
                            operatorStack.push(token);
                        } else if (token.value === ')') {
                            while (operatorStack.length > 0 && operatorStack[operatorStack.length - 1].value !== '(') {
                                output.push(operatorStack.pop());
                            }
                            if (operatorStack.length > 0 && operatorStack[operatorStack.length - 1].value === '(') {
                                operatorStack.pop(); // Удаляем открывающую скобку
                            }
                            if (operatorStack.length > 0 && operatorStack[operatorStack.length - 1].type === 'function') {
                                output.push(operatorStack.pop());
                            }
                        } else {
                            while (operatorStack.length > 0 &&
                                   operatorStack[operatorStack.length - 1].type === 'operator' &&
                                   operatorStack[operatorStack.length - 1].value !== '(' &&
                                   precedence[operatorStack[operatorStack.length - 1].value] >= precedence[token.value]) {
                                output.push(operatorStack.pop());
                            }
                            operatorStack.push(token);
                        }
                    }
                }
                
                // Перемещаем оставшиеся операторы в выходной стек
                while (operatorStack.length > 0) {
                    if (operatorStack[operatorStack.length - 1].value === '(') {
                        throw new Error('Несоответствие скобок');
                    }
                    output.push(operatorStack.pop());
                }
                
                return output;
            }
            
            // Вычисление выражения в обратной польской записи
            function evaluate(rpn, z) {
                const stack = [];
                
                for (let i = 0; i < rpn.length; i++) {
                    const token = rpn[i];
                    
                    if (token.type === 'number') {
                        stack.push({ re: token.value, im: 0 });
                    } else if (token.type === 'variable') {
                        stack.push(z);
                    } else if (token.type === 'constant') {
                        stack.push(CONSTANTS[token.value]);
                    } else if (token.type === 'function') {
                        const arg = stack.pop();
                        switch (token.value) {
                            case 'sin': stack.push(Complex.sin(arg)); break;
                            case 'cos': stack.push(Complex.cos(arg)); break;
                            case 'tan': stack.push(Complex.tan(arg)); break;
                            case 'exp': stack.push(Complex.exp(arg)); break;
                            case 'log': stack.push(Complex.log(arg)); break;
                            case 'sqrt': stack.push(Complex.sqrt(arg)); break;
                            default:
                                throw new Error('Неизвестная функция: ' + token.value);
                        }
                    } else if (token.type === 'operator') {
                        const b = stack.pop();
                        // Обработка унарных операторов
                        if (stack.length === 0 && (token.value === '+' || token.value === '-')) {
                            if (token.value === '-') {
                                stack.push({ re: -b.re, im: -b.im });
                            } else {
                                stack.push(b); // Унарный плюс ничего не делает
                            }
                            continue;
                        }
                        
                        const a = stack.pop();
                        switch (token.value) {
                            case '+': stack.push(Complex.add(a, b)); break;
                            case '-': stack.push(Complex.subtract(a, b)); break;
                            case '*': stack.push(Complex.multiply(a, b)); break;
                            case '/': stack.push(Complex.divide(a, b)); break;
                            case '^': stack.push(Complex.power(a, b.re)); break;
                            default:
                                throw new Error('Неизвестный оператор: ' + token.value);
                        }
                    }
                }
                
                if (stack.length !== 1) {
                    throw new Error('Некорректное выражение');
                }
                
                return stack[0];
            }
            
            // Обработчик сообщений от основного потока
            self.onmessage = function(e) {
                const { width, height, functionStr, viewScale, viewOffsetX, viewOffsetY, centerX, centerY } = e.data;
                
                // Предварительная обработка функции - замена ^ на pow и т.д.
                let processedFuncStr = functionStr
                    .replace(/\\s+/g, '')      // Удаляем все пробелы
                    .replace(/\\^(\\d+)/g, '^$1')  // Упрощаем степени
                    .toLowerCase();              // Приводим к нижнему регистру
                
                const imageData = new ImageData(width, height);
                const data = imageData.data;
                
                // Размер шага для вычисления значений
                const stepSize = 1 / viewScale;
                
                // Верхняя левая точка видимой области в комплексной плоскости
                const startRe = -width / (2 * 100 * viewScale) - viewOffsetX;
                const startIm = height / (2 * 100 * viewScale) - viewOffsetY;
                
                // Итерация по всем пикселям
                for (let j = 0; j < height; j++) {
                    for (let i = 0; i < width; i++) {
                        // Преобразуем координаты пикселя в комплексную плоскость
                        const re = startRe + i * stepSize / 100;
                        const im = startIm - j * stepSize / 100;
                        
                        try {
                            // Вычисляем значение функции
                            const z = { re, im };
                            const w = evaluateFunction(z, processedFuncStr);
                            
                            // Получаем аргумент и модуль
                            const arg = Complex.arg(w);
                            const mag = Complex.abs(w);
                            
                            // Преобразуем в цвет
                            const hue = ((arg + Math.PI) / (2 * Math.PI)) * 360;
                            const norm = 2 * (Math.atan(mag / 2) / Math.PI);
                            const sat = 80 + 20 * Math.sin(norm * Math.PI * 2);
                            const light = 60 - 30 * Math.cos(norm * Math.PI);

## Подробное описание решения

### 1. Реализация комплексных чисел

Внутри Web Worker реализована библиотека для работы с комплексными числами, включающая:
- Базовые операции (сложение, умножение, деление)
- Вычисление модуля и аргумента комплексного числа
- Тригонометрические функции (sin, cos, tan)
- Экспоненциальные и логарифмические функции (exp, log)
- Корень квадратный (sqrt)
- Возведение в произвольную степень (power)

### 2. Парсер выражений

Реализован полноценный парсер выражений, который:
1. Разбивает входную строку на токены (токенизация)
2. Преобразует инфиксную запись в обратную польскую запись (ОПЗ) с помощью алгоритма сортировочной станции (Shunting-yard)
3. Вычисляет результат по ОПЗ с использованием стека

### 3. Поддерживаемые функции и операции

- Арифметические операции: +, -, *, /, ^
- Тригонометрические функции: sin, cos, tan
- Экспоненциальная и логарифмическая функции: exp, log
- Квадратный корень: sqrt
- Константы: pi, e, i
- Переменная: z (комплексное число)

### 4. Оптимизации

- Предварительная обработка функциональной строки для устранения пробелов и нормализации записи
- Оптимизация повторяющихся вычислений в наиболее частых операциях
- Эффективная обработка ошибок для устойчивости к особым точкам

### 5. Преимущества данного решения

1. **Производительность**: Вычисления выполняются в отдельном потоке, что предотвращает "зависание" интерфейса при рендеринге сложных функций
2. **Универсальность**: Поддерживаются все основные математические функции и произвольные комбинации
3. **Масштабируемость**: Решение легко расширить для поддержки дополнительных функций или оптимизаций

### 6. Ограничения и возможные улучшения

1. **Кэширование результатов**: Для часто встречающихся функций можно ввести кэширование значений
2. **Адаптивное разрешение**: Можно добавить интеллектуальное изменение плотности выборки в зависимости от сложности функции
3. **Дополнительные функции**: Можно добавить поддержку гиперболических функций (sinh, cosh, tanh) и других специальных математических функций
4. **Оптимизация парсера**: Предварительная компиляция выражения в более эффективное представление для повторных вычислений

## Инструкция по внедрению

1. Найдите функцию `renderDomainColoring()` в коде JavaScript (около строк 2132-2300)
2. Замените весь блок функции на код, приведённый выше
3. Проверьте, что функция `hslToRgb()` доступна в том же файле (она должна быть, поскольку используется в текущем коде)
4. Сохраните изменения и проверьте работу визуализатора

Реализованное решение обеспечит корректное отображение всех комплексных функций в режиме Доменной карты с сохранением высокой производительности.
