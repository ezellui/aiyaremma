## Добавление гиперболических и других функций

Добавьте следующие методы в объект `Complex` в Web Worker:

```javascript
// Гиперболические функции
sinh: function(a) {
    // sinh(a + bi) = sinh(a)cos(b) + i cosh(a)sin(b)
    return {
        re: Math.sinh(a.re) * Math.cos(a.im),
        im: Math.cosh(a.re) * Math.sin(a.im)
    };
},

cosh: function(a) {
    // cosh(a + bi) = cosh(a)cos(b) + i sinh(a)sin(b)
    return {
        re: Math.cosh(a.re) * Math.cos(a.im),
        im: Math.sinh(a.re) * Math.sin(a.im)
    };
},

tanh: function(a) {
    // tanh(z) = sinh(z) / cosh(z)
    return Complex.divide(Complex.sinh(a), Complex.cosh(a));
},

coth: function(a) {
    // coth(z) = cosh(z) / sinh(z)
    return Complex.divide(Complex.cosh(a), Complex.sinh(a));
},

sech: function(a) {
    // sech(z) = 1 / cosh(z)
    return Complex.divide({ re: 1, im: 0 }, Complex.cosh(a));
},

csch: function(a) {
    // csch(z) = 1 / sinh(z)
    return Complex.divide({ re: 1, im: 0 }, Complex.sinh(a));
},

// Обратные гиперболические функции
asinh: function(a) {
    // asinh(z) = log(z + sqrt(z^2 + 1))
    const z2_plus_1 = Complex.add(Complex.multiply(a, a), { re: 1, im: 0 });
    const sqrt_part = Complex.sqrt(z2_plus_1);
    const sum_part = Complex.add(a, sqrt_part);
    return Complex.log(sum_part);
},

acosh: function(a) {
    // acosh(z) = log(z + sqrt(z^2 - 1))
    const z2_minus_1 = Complex.subtract(Complex.multiply(a, a), { re: 1, im: 0 });
    const sqrt_part = Complex.sqrt(z2_minus_1);
    const sum_part = Complex.add(a, sqrt_part);
    return Complex.log(sum_part);
},

atanh: function(a) {
    // atanh(z) = (1/2) * log((1+z)/(1-z))
    const one_plus_z = Complex.add({ re: 1, im: 0 }, a);
    const one_minus_z = Complex.subtract({ re: 1, im: 0 }, a);
    const division = Complex.divide(one_plus_z, one_minus_z);
    const log_part = Complex.log(division);
    return {
        re: log_part.re * 0.5,
        im: log_part.im * 0.5
    };
},

acoth: function(a) {
    // acoth(z) = (1/2) * log((z+1)/(z-1))
    const z_plus_1 = Complex.add(a, { re: 1, im: 0 });
    const z_minus_1 = Complex.subtract(a, { re: 1, im: 0 });
    const division = Complex.divide(z_plus_1, z_minus_1);
    const log_part = Complex.log(division);
    return {
        re: log_part.re * 0.5,
        im: log_part.im * 0.5
    };
},

asech: function(a) {
    // asech(z) = acosh(1/z)
    return Complex.acosh(Complex.divide({ re: 1, im: 0 }, a));
},

acsch: function(a) {
    // acsch(z) = asinh(1/z)
    return Complex.asinh(Complex.divide({ re: 1, im: 0 }, a));
},

// Функции округления
floor: function(a) {
    return {
        re: Math.floor(a.re),
        im: Math.floor(a.im)
    };
},

ceil: function(a) {
    return {
        re: Math.ceil(a.re),
        im: Math.ceil(a.im)
    };
},

round: function(a) {
    return {
        re: Math.round(a.re),
        im: Math.round(a.im)
    };
},

fix: function(a) {
    return {
        re: (a.re > 0) ? Math.floor(a.re) : Math.ceil(a.re),
        im: (a.im > 0) ? Math.floor(a.im) : Math.ceil(a.im)
    };
},

// Дополнительные функции
sign: function(a) {
    const abs = Complex.abs(a);
    if (abs === 0) {
        return { re: 0, im: 0 };
    }
    return {
        re: a.re / abs,
        im: a.im / abs
    };
},

abs2: function(a) {
    // |z|²
    return a.re * a.re + a.im * a.im;
},

arg2: function(a) {
    // Возвращает arg(z) в диапазоне [0, 2π]
    let arg = Math.atan2(a.im, a.re);
    if (arg < 0) {
        arg += 2 * Math.PI;
    }
    return arg;
},

factorial: function(a) {
    // Факториал для целых неотрицательных действительных чисел
    if (a.im !== 0) {
        return { re: NaN, im: NaN }; // Не определен для комплексных чисел
    }
    
    const n = Math.round(a.re);
    if (n < 0) {
        return { re: NaN, im: NaN }; // Не определен для отрицательных чисел
    }
    
    let result = 1;
    for (let i = 2; i <= n; i++) {
        result *= i;
    }
    return { re: result, im: 0 };
},

gamma: function(a) {
    // Приближение гамма-функции через формулу Стирлинга
    // Примечание: это приближение, для полной точной реализации требуется больше кода
    // Работает лучше всего для действительных чисел > 0.5
    if (a.im === 0 && a.re > 0) {
        // Для действительных положительных чисел используем рекурсивное определение
        // Γ(n+1) = n * Γ(n)
        if (a.re === Math.floor(a.re)) {
            return Complex.factorial({ re: a.re - 1, im: 0 });
        }
        
        // Приближение Стирлинга: Γ(z) ≈ sqrt(2π) * (z/e)^(z-0.5) * (1 + 1/(12z) + ...)
        const z = a.re;
        const sqrt2pi = Math.sqrt(2 * Math.PI);
        const result = sqrt2pi * Math.pow(z / Math.E, z - 0.5) * (1 + 1/(12*z));
        return { re: result, im: 0 };
    }
    
    // Для комплексных чисел требуется более сложная реализация
    // Используем очень приближенную формулу
    return Complex.exp(Complex.subtract(
        Complex.multiply(
            Complex.log(a),
            Complex.subtract(a, { re: 0.5, im: 0 })
        ),
        a
    ));
},

// Специальные функции для анализа и отображения
clip: function(a, limit) {
    // Ограничивает модуль комплексного числа до limit
    const abs = Complex.abs(a);
    if (abs <= limit) {
        return { re: a.re, im: a.im };
    }
    const factor = limit / abs;
    return { re: a.re * factor, im: a.im * factor };
},

isFinite: function(a) {
    return isFinite(a.re) && isFinite(a.im);
},

isNaN: function(a) {
    return isNaN(a.re) || isNaN(a.im);
},

isPolar: function(a) {
    return {
        r: Complex.abs(a),
        phi: Complex.arg(a)
    };
},

fromPolar: function(r, phi) {
    return {
        re: r * Math.cos(phi),
        im: r * Math.sin(phi)
    };
},

toJSON: function(a) {
    return { re: a.re, im: a.im };
},

fromJSON: function(json) {
    return { re: json.re || 0, im: json.im || 0 };
}
```

## Обновление определения математических вспомогательных функций

```javascript
// Вспомогательные математические функции
function cosh(x) {
    return (Math.exp(x) + Math.exp(-x)) / 2;
}

function sinh(x) {
    return (Math.exp(x) - Math.exp(-x)) / 2;
}

// Функция для расчёта модуля факториала для больших чисел (используется приближение Стирлинга)
function stirlingApprox(n) {
    if (n < 10) {
        let result = 1;
        for (let i = 2; i <= n; i++) {
            result *= i;
        }
        return result;
    }
    
    // Формула Стирлинга: n! ≈ sqrt(2πn) * (n/e)^n
    const sqrt2pi = Math.sqrt(2 * Math.PI * n);
    const power = Math.pow(n / Math.E, n);
    return sqrt2pi * power;
}
```

## Что добавлено в этой части

В этой части мы расширили функциональность библиотеки комплексных чисел:

### 1. Гиперболические функции
- `sinh(z)`, `cosh(z)`, `tanh(z)` - синус, косинус, тангенс гиперболический
- `coth(z)`, `sech(z)`, `csch(z)` - котангенс, секанс, косеканс гиперболический

### 2. Обратные гиперболические функции
- `asinh(z)`, `acosh(z)`, `atanh(z)` - арксинус, арккосинус, арктангенс гиперболический
- `acoth(z)`, `asech(z)`, `acsch(z)` - арккотангенс, арксеканс, арккосеканс гиперболический

### 3. Функции округления
- `floor(z)`, `ceil(z)`, `round(z)`, `fix(z)` - округление в разных направлениях

### 4. Дополнительные операции
- `sign(z)` - знак комплексного числа (нормализованное направление)
- `abs2(z)` - квадрат модуля (для оптимизации вычислений)
- `arg2(z)` - аргумент в диапазоне [0, 2π]
- `factorial(z)` - факториал
- `gamma(z)` - гамма-функция (приближение)

### 5. Специальные утилиты для визуализации
- `clip(z, limit)` - ограничение модуля для визуализации
- `isFinite(z)`, `isNaN(z)` - проверки состояния
- `isPolar(z)`, `fromPolar(r, phi)` - преобразования из/в полярную форму

### 6. Дополнительные константы
- `phi` - золотое сечение
- `inf` - бесконечность
- `nan` - не число
