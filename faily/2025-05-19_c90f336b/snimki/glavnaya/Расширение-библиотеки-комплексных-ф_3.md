## Оптимизации производительности и кэширование

Сначала добавим механизм кэширования, чтобы улучшить производительность при повторных вычислениях:

```javascript
// Добавьте в начало блока Web Worker после определения констант:

// Система кэширования для оптимизации производительности
const FunctionCache = {
    // Максимальный размер кэша
    maxCacheSize: 10000,
    
    // Структуры кэша для разных типов функций
    caches: {
        exp: new Map(),
        log: new Map(),
        sin: new Map(),
        cos: new Map(),
        pow: new Map(),
        // Добавьте кэш для других функций по необходимости
    },
    
    // Получение значения из кэша
    get: function(funcName, z, precision = 5) {
        if (!this.caches[funcName]) return null;
        
        // Округляем до заданной точности для кэширования
        const key = this.makeKey(z, precision);
        return this.caches[funcName].get(key);
    },
    
    // Сохранение значения в кэше
    set: function(funcName, z, result, precision = 5) {
        if (!this.caches[funcName]) return;
        
        const cache = this.caches[funcName];
        
        // Если кэш переполнен, удаляем старые записи
        if (cache.size >= this.maxCacheSize) {
            // Удаляем первые 20% записей для эффективности
            const keysToDelete = Array.from(cache.keys()).slice(0, this.maxCacheSize / 5);
            keysToDelete.forEach(key => cache.delete(key));
        }
        
        const key = this.makeKey(z, precision);
        cache.set(key, { re: result.re, im: result.im });
    },
    
    // Создание ключа кэша из комплексного числа
    makeKey: function(z, precision) {
        const re = parseFloat(z.re.toFixed(precision));
        const im = parseFloat(z.im.toFixed(precision));
        return `${re},${im}`;
    },
    
    // Очистка кэша
    clear: function(funcName) {
        if (funcName && this.caches[funcName]) {
            this.caches[funcName].clear();
        } else {
            Object.values(this.caches).forEach(cache => cache.clear());
        }
    }
};

// Оптимизация рендеринга через фрагментирование вычислений
let renderingProgress = {
    total: 0,
    completed: 0,
    isRendering: false,
    startTime: 0
};

// Теперь модифицируем некоторые методы Complex для использования кэша
const ComplexOptimized = {
    // Добавляем эти методы к существующему объекту Complex
    
    // Оптимизированная экспонента с кэшированием
    expCached: function(a, precision = 5) {
        const cached = FunctionCache.get('exp', a, precision);
        if (cached) return cached;
        
        const result = Complex.exp(a);
        FunctionCache.set('exp', a, result, precision);
        return result;
    },
    
    // Оптимизированный логарифм с кэшированием
    logCached: function(a, precision = 5) {
        const cached = FunctionCache.get('log', a, precision);
        if (cached) return cached;
        
        const result = Complex.log(a);
        FunctionCache.set('log', a, result, precision);
        return result;
    },
    
    // Оптимизированный синус с кэшированием
    sinCached: function(a, precision = 5) {
        const cached = FunctionCache.get('sin', a, precision);
        if (cached) return cached;
        
        const result = Complex.sin(a);
        FunctionCache.set('sin', a, result, precision);
        return result;
    },
    
    // Оптимизированный косинус с кэшированием
    cosCached: function(a, precision = 5) {
        const cached = FunctionCache.get('cos', a, precision);
        if (cached) return cached;
        
        const result = Complex.cos(a);
        FunctionCache.set('cos', a, result, precision);
        return result;
    },
    
    // Оптимизированное возведение в степень с кэшированием
    powCached: function(a, b, precision = 5) {
        // Для степеней кэшируем только при целых показателях
        if (typeof b === 'number' && Number.isInteger(b)) {
            const key = { re: a.re, im: a.im, power: b };
            const keyStr = this.makeKey(key, precision);
            
            const cached = FunctionCache.get('pow', keyStr, precision);
            if (cached) return cached;
            
            const result = Complex.pow(a, b);
            FunctionCache.set('pow', keyStr, result, precision);
            return result;
        }
        
        return Complex.pow(a, b);
    },
    
    // Вспомогательный метод для ключа кэша степени

        return `${re},${im},${obj.power}`;
    }

// Функция Бесселя первого рода порядка n
besselJ: function(n, z) {
    // Используем разложение в ряд для J_n(z)
    // J_n(z) = (z/2)^n * sum_k=0^inf (-1)^k / (k! * (n+k)!) * (z/2)^(2k)
    
    if (n !== Math.floor(n) || n < 0) {
        // Для не целых или отрицательных n нужны более сложные формулы
        // Возвращаем NaN для упрощения
        return { re: NaN, im: NaN };
    }
    
    // Для маленьких |z| сумма быстро сходится
    const maxTerms = 30; // Ограничиваем число слагаемых
    let sum = { re: 0, im: 0 };
    
    // Первый множитель (z/2)^n
    let term = { re: 1, im: 0 };
    const z_half = Complex.divide(z, { re: 2, im: 0 });
    for (let i = 0; i < n; i++) {
        term = Complex.multiply(term, z_half);
    }
    
    // Суммирование ряда
    for (let k = 0; k < maxTerms; k++) {
        // Коэффициент (-1)^k / (k! * (n+k)!)
        const coef = (k % 2 === 0 ? 1 : -1) / (factorial(k) * factorial(n + k));
        
        // Степень (z/2)^(2k)
        let power = { re: 1, im: 0 };
        const z_half_squared = Complex.multiply(z_half, z_half);
        for (let i = 0; i < 2*k; i++) {
            power = Complex.multiply(power, z_half);
        }
        
        // Слагаемое
        const add = Complex.multiply({ re: coef, im: 0 }, power);
        
        // Добавляем к сумме
        sum = Complex.add(sum, add);
        
        // Проверка сходимости
        if (Complex.abs(add) < 1e-10) break;
    }
    
    // Умножаем сумму на (z/2)^n
    return Complex.multiply(term, sum);
},

// Функция Бесселя второго рода порядка n
besselY: function(n, z) {
    // Для целых n используем соотношение 
    // Y_n(z) = (J_n(z)cos(nπ) - J_{-n}(z)) / sin(nπ)
    // Для n = 0 используем другую формулу
    
    if (n !== Math.floor(n) || n < 0) {
        return { re: NaN, im: NaN };
    }
    
    if (n === 0) {
        // Y_0(z) аппроксимация
        const ln_half_z = Complex.log(Complex.divide(z, { re: 2, im: 0 }));
        const j0 = Complex.besselJ(0, z);
        
        const gamma = 0.57721566490153; // Константа Эйлера-Маскерони
        
        // Y_0(z) ≈ (2/π) * (ln(z/2) + γ) * J_0(z) + additional_terms
        const term1 = Complex.multiply(
            { re: 2/Math.PI * (ln_half_z.re + gamma), im: 2/Math.PI * ln_half_z.im },
            j0
        );
        
        // Приближенное значение
        return term1;
    }
    
    // Для n > 0
    const jn = Complex.besselJ(n, z);
    const j_minus_n = Complex.besselJ(-n, z);
    
    // Y_n(z) = (J_n(z)cos(nπ) - J_{-n}(z)) / sin(nπ)
    const cos_n_pi = Math.cos(n * Math.PI);
    const sin_n_pi = Math.sin(n * Math.PI);
    
    if (sin_n_pi === 0) {
        return { re: Infinity, im: Infinity }; // Особая точка
    }
    
    return Complex.divide(
        Complex.subtract(
            Complex.multiply(jn, { re: cos_n_pi, im: 0 }),
            j_minus_n
        ),
        { re: sin_n_pi, im: 0 }
    );
},

// Функция ошибок (Error function)
erf: function(z) {
    // erf(z) = (2/√π) * ∫_0^z e^(-t^2) dt
    // Аппроксимация рядом Тейлора
    
    // Для действительного аргумента используем прямую формулу
    if (z.im === 0) {
        // Аппроксимация для действительного аргумента
        const x = z.re;
        const sign = Math.sign(x);
        const a = Math.abs(x);
        
        // Абрамовиц и Стеган: Handbook of Mathematical Functions, формула 7.1.26
        if (a < 3.5) {
            // Ряд Тейлора
            let sum = 0;
            let term = x;
            for (let n = 0; n < 50; n++) {
                const prev = sum;
                sum += term;
                if (Math.abs(sum - prev) < 1e-10) break;
                
                term *= -x*x / (n + 1.5);
            }
            
            return { re: sum * 2 / Math.sqrt(Math.PI), im: 0 };
        } else {
            // Асимптотическое разложение для больших |x|
            const t = 1.0 / (1.0 + 0.5 * a);
            
            // Коэффициенты из аппроксимации Chebyshev
            const c = [
                -1.26551223, 1.00002368, 0.37409196, 0.09678418, 
                -0.18628806, 0.27886807, -1.13520398, 1.48851587, 
                -0.82215223, 0.17087277
            ];
            
            let sum = c[0] * t;
            const t2 = t * t;
            let v = t;
            
            for (let i = 1; i < c.length; i++) {
                v *= t2;
                sum += c[i] * v;
            }
            
            const expf = Math.exp(-a * a);
            const result = sign * (1.0 - expf * t * sum);
            
            return { re: result, im: 0 };
        }
    }
    
    // Для комплексного аргумента используем ряд Тейлора
    // erf(z) = (2/√π) * z * sum_{n=0}^∞ (-1)^n * z^(2n) / [n! * (2n+1)]
    
    const maxTerms = 30;
    let sum = { re: 0, im: 0 };
    let term = { re: 1, im: 0 }; // Начальный член равен 1
    
    for (let n = 0; n < maxTerms; n++) {
        // Коэффициент (-1)^n / [n! * (2n+1)]
        const coef = (n % 2 === 0 ? 1 : -1) / (factorial(n) * (2*n + 1));
        
        // Степень z^(2n)
        let z_power = { re: 1, im: 0 };
        const z_squared = Complex.multiply(z, z);
        
        for (let i = 0; i < n; i++) {
            z_power = Complex.multiply(z_power, z_squared);
        }
        
        // Текущий член ряда
        term = Complex.multiply({ re: coef, im: 0 }, z_power);
        
        // Добавляем к сумме
        sum = Complex.add(sum, term);
        
        // Проверка сходимости
        if (Complex.abs(term) < 1e-10) break;
    }
    
    // Умножаем на (2/√π) * z
    return Complex.multiply(
        Complex.multiply(
            { re: 2 / Math.sqrt(Math.PI), im: 0 },
            z
        ),
        sum
    );
},

// Эллиптический интеграл первого рода (неполный)
ellipticF: function(phi, m) {
    // F(φ|m) = ∫_0^φ dt/√(1-m*sin²(t))
    // Аппроксимация через ряды для небольших m и φ
    
    // Для действительных параметров (упрощенная реализация)
    if (typeof phi === 'object') {
        phi = phi.re; // Используем только действительную часть
    }
    if (typeof m === 'object') {
        m = m.re; // Используем только действительную часть
    }
    
    // Проверки входных параметров
    if (isNaN(phi) || isNaN(m)) {
        return { re: NaN, im: 0 };
    }
    
    // Для m > 1 или φ за пределами [0, π/2] нужны специальные преобразования
    if (m > 1 || phi < 0 || phi > Math.PI/2) {
        // Возвращаем приближение для простоты
        return { re: NaN, im: 0 };
    }
    
    // Вычисление с помощью арифметико-геометрического среднего
    const sinPhi = Math.sin(phi);
    const cosPhi = Math.cos(phi);
    
    if (sinPhi === 0) {
        return { re: 0, im: 0 };
    }
    
    if (cosPhi === 0) {
        if (m === 1) {
            return { re: Infinity, im: 0 };
        }
        return { re: 1 / Math.sqrt(1 - m), im: 0 };
    }
    
    // Приближение через разложение в ряд
    let sum = phi;
    let term = phi;
    let n = 1;
    
    while (true) {
        // Коэффициенты ряда
        let coef = 1;
        for (let i = 1; i <= n; i++) {
            coef *= (2*i - 1) / (2*i);
        }
        coef *= m**n;
        
        // Sin^(2n+1)(φ)
        let sinTerm = sinPhi;
        for (let i = 1; i <= 2*n; i++) {
            sinTerm *= sinPhi;
        }
        
        term = coef * sinTerm * cosPhi / (2*n + 1);
        sum += term;
        
        if (Math.abs(term) < 1e-10 || n > 20) break;
        n++;
    }
    
    return { re: sum, im: 0 };
},

ellipticE: function(phi, m) {

    // Аппроксимация через ряды для небольших m и φ

    }

    if (isNaN(phi) || isNaN(m)) {
        return { re: NaN, im: 0 };
    }
    
    // Для m > 1 или φ за пределами [0, π/2] нужны специальные преобразования
    if (m > 1 || phi < 0 || phi > Math.PI/2) {
        // Возвращаем приближение для простоты
        return { re: NaN, im: 0 };
    }

    let term = phi;
    let n = 1;
    
    while (true) {
        // Коэффициенты ряда
        let coef = 1;
        for (let i = 1; i <= n; i++) {
            coef *= (2*i - 1) / (2*i);
        }
        coef *= -m**n / (2*n - 1);

        let sinTerm = 1;
        for (let i = 1; i <= 2*n; i++) {
            sinTerm *= Math.sin(phi);
        }
        
        term = coef * sinTerm * Math.cos(phi);
        sum += term;
        
        if (Math.abs(term) < 1e-10 || n > 20) break;
        n++;
    }
    
    return { re: sum, im: 0 };
},

// Дзета-функция Римана
zeta: function(s) {
    // Реализация для действительного s > 1
    if (s.im === 0 && s.re > 1) {
        // Приближение через прямое суммирование ряда
        let sum = 0;
        const maxTerms = 10000;
        
        for (let n = 1; n <= maxTerms; n++) {
            const term = 1 / Math.pow(n, s.re);
            sum += term;
            
            // Проверка сходимости
            if (term < 1e-10) break;
        }
        
        return { re: sum, im: 0 };
    }
    
    // Для произвольного комплексного s требуется аналитическое продолжение
    // Используем приближенную формулу для области Re(s) > 0.5
    
    if (s.re <= 0.5) {
        // Функциональное уравнение для s с Re(s) <= 0.5
        // ζ(s) = 2^s * π^(s-1) * sin(πs/2) * Γ(1-s) * ζ(1-s)
        
        const oneMinusS = Complex.subtract({ re: 1, im: 0 }, s);
        const zetaOneMinusS = Complex.zeta(oneMinusS);
        
        const piPow = Math.pow(Math.PI, s.re - 1);
        const twoPow = Math.pow(2, s.re);
        
        const piS_div_2 = Complex.multiply(s, { re: Math.PI/2, im: 0 });
        const sinPiS_div_2 = Complex.sin(piS_div_2);
        
        const gamma1MinusS = Complex.gamma(oneMinusS);
        
        return Complex.multiply(
            { re: twoPow * piPow, im: 0 },
            Complex.multiply(
                sinPiS_div_2,
                Complex.multiply(gamma1MinusS, zetaOneMinusS)
            )
        );
    }
    
    // Для Re(s) > 0.5 используем формулу суммирования Эйлера-Маклорена
    const maxTerms = 100;
    let sum = 0;
    
    for (let n = 1; n <= maxTerms; n++) {
        sum += 1 / Math.pow(n, s.re) * Math.cos(s.im * Math.log(n));
        
        // Проверка сходимости
        if (1 / Math.pow(n, s.re) < 1e-10) break;
    }
    
    // Приближенное значение для не слишком больших Im(s)
    return { re: sum, im: 0 };
},

// Функция Ламберта W (обратная к x*e^x)
lambertW: function(z) {
    // Функция Ламберта W(z) решает уравнение W*e^W = z
    // Используем метод Халли для численного решения
    
    // Для действительного z ≥ -1/e
    if (z.im === 0 && z.re >= -0.36787944117144) {
        const x = z.re;
        
        // Начальное приближение
        let w;
        if (x < 0) {
            // Для x близкого к -1/e
            w = -1 + Math.sqrt(2 * (Math.E * x + 1));
        } else if (x < 1) {
            // Для x от 0 до 1
            w = x * (1 - x + 3*x*x/2);
        } else {
            // Для x > 1
            w = Math.log(x);
        }
        
        // Метод Халли (улучшенный метод Ньютона)
        for (let i = 0; i < 10; i++) {
            const ew = Math.exp(w);
            const wew = w * ew;
            const wewz = wew - x;
            
            if (Math.abs(wewz) < 1e-10) {
                break;
            }
            
            const w1 = w + 1;
            w = w - wewz / (ew * w1 - (w + 2) * wewz / (2 * w1));
        }
        
        return { re: w, im: 0 };
    }
    
    // Для комплексного z требуется более сложная реализация
    // Возвращаем приближение основной ветви
    
    // Начальное приближение
    let w = { re: 0, im: 0 };
    
    if (Complex.abs(z) < 1) {
        // Разложение в ряд для малых |z|
        // W(z) ≈ z - z^2 + 3z^3/2 - ...
        w = Complex.subtract(
            z,
            Complex.multiply(z, z)
        );
    } else {
        // Асимптотическое приближение для больших |z|
        // W(z) ≈ log(z) - log(log(z)) + ...
        const logZ = Complex.log(z);
        const logLogZ = Complex.log(logZ);
        w = Complex.subtract(logZ, logLogZ);
    }
    
    // Итерационный метод Халли
    for (let i = 0; i < 20; i++) {
        const ew = Complex.exp(w);
        const wew = Complex.multiply(w, ew);
        const wewz = Complex.subtract(wew, z);
        
        if (Complex.abs(wewz) < 1e-10) {
            break;
        }
        
        const w1 = Complex.add(w, { re: 1, im: 0 });
        const eww1 = Complex.multiply(ew, w1);
        
        const term1 = Complex.divide(wewz, eww1);
        
        const w2 = Complex.add(w, { re: 2, im: 0 });
        const wewzw2 = Complex.multiply(wewz, w2);
        const term2 = Complex.divide(wewzw2, Complex.multiply({ re: 2, im: 0 }, Complex.multiply(w1, w1)));
        
        const correction = Complex.add(term1, term2);
        
        w = Complex.subtract(w, correction);
    }
    
    return w;
}

## Расширенная обработка ошибок и граничных случаев

Добавим расширенную обработку ошибок и граничных случаев для повышения стабильности:

// Добавьте в объект Complex методы для работы с ошибками и особыми случаями

// Проверка и безопасное выполнение функции
safeCompute: function(func, fallback = { re: NaN, im: NaN }) {
    try {
        const result = func();
        if (this.isNaN(result) || !this.isFinite(result)) {
            return fallback;
        }
        return result;
    } catch (e) {
        return fallback;
    }
},

// Обработка сингулярности (полюса) с возможностью визуализации
handlePole: function(z, polePosition, poleOrder = 1, visualization = 'clip') {
    const distance = this.abs(this.subtract(z, polePosition));
    
    // Если точка находится слишком близко к полюсу
    if (distance < 1e-6) {
        switch (visualization) {
            case 'clip':
                // Возвращаем очень большое, но конечное значение
                return this.fromPolar(1e10, this.arg(this.subtract(z, polePosition)));
            case 'color':
                // Возвращаем специальное значение для особой раскраски
                return { re: Infinity, im: Infinity, pole: true };
            case 'invert':
                // Инвертируем направление 
                return this.divide({ re: 1, im: 0 }, this.multiply(
                    this.subtract(z, polePosition),
                    { re: Math.pow(10, poleOrder), im: 0 }
                ));
            default:
                return { re: NaN, im: NaN };
        }
    }
    
    // Обычное вычисление, если точка не на полюсе
    return null; // Сигнал о необходимости обычного вычисления
},

// Обработка разрывов и ветвлений
handleBranchCut: function(z, line, visualization = 'jump') {
    // line = { start: complex, end: complex, type: 'log'|'sqrt'|'asin'|etc }
    
    // Проверка, находится ли точка вблизи линии разреза
    const t = this.projectPointToLine(z, line.start, line.end);
    const projection = this.add(line.start, this.multiply(
        this.subtract(line.end, line.start),
        { re: t, im: 0 }
    ));
    const distance = this.abs(this.subtract(z, projection));
    
    // Если точка находится вблизи линии разреза
    if (distance < 1e-3 && t >= 0 && t <= 1) {
        // Определяем, с какой стороны разреза находится точка
        const normal = this.multiply(
            { re: 0, im: 1 },
            this.normalize(this.subtract(line.end, line.start))
        );
        const side = Math.sign(this.dotProduct(
            this.subtract(z, projection),
            normal
        ));
        
        switch (visualization) {
            case 'jump':
                // Возвращаем значение на соответствующей стороне разреза
                switch (line.type) {
                    case 'log':
                        // Для логарифма скачок равен 2πi
                        return side > 0 ? 
                            { re: 0, im: 2 * Math.PI * side } : 
                            { re: 0, im: 0 };
                    case 'sqrt':
                        // Для квадратного корня меняем знак
                        return { re: 0, im: Math.PI * side };
                    default:
                        return { re: 0, im: 0 };
                }
            case 'color':
                // Возвращаем специальное значение для раскраски разреза
                return { re: 0, im: 0, branchCut: true, side: side };
            default:
                return null;
        }
    }
    
    // Обычное вычисление, если точка не на разрезе
    return null;
},

// Проекция точки на линию
projectPointToLine: function(point, lineStart, lineEnd) {
    const line = this.subtract(lineEnd, lineStart);
    const pointLine = this.subtract(point, lineStart);
    
    const lineLength2 = this.abs2(line);
    if (lineLength2 === 0) return 0;
    
    const t = this.dotProduct(pointLine, line) / lineLength2;
    return t;
},

// Скалярное произведение комплексных чисел как 2D векторов
dotProduct: function(a, b) {
    return a.re * b.re + a.im * b.im;
},

// Нормализация комплексного числа (единичный вектор того же направления)
normalize: function(z) {
    const abs = this.abs(z);
    if (abs === 0) return { re: 0, im: 0 };
    return { re: z.re / abs, im: z.im / abs };
}
```

// Преобразование комплексного числа в параметрический цвет
toColor: function(z, colorMapping = 'default') {
    // Обработка особых значений
    if (!this.isFinite(z)) {
        if (z.pole) return 'rgba(255, 0, 0, 1)'; // Красный для полюсов
        if (z.branchCut) return z.side > 0 ? 'rgba(255, 200, 0, 1)' : 'rgba(0, 255, 255, 1)';
        return 'rgba(0, 0, 0, 1)'; // Черный для бесконечностей и NaN
    }
    
    const arg = this.arg(z);
    const abs = this.abs(z);
    
    // Нормализуем модуль с помощью гиперболического тангенса
    const normAbs = 2 * Math.atan(abs) / Math.PI;
    
    switch (colorMapping) {
        case 'hsv':
            // Фаза -> оттенок, модуль -> насыщенность и яркость
            const hue = ((arg + Math.PI) / (2 * Math.PI)) * 360;
            const sat = Math.min(1, normAbs) * 100;
            const val = 50 + 50 * Math.min(1, normAbs);
            return `hsla(${hue}, ${sat}%, ${val}%, 1)`;
            
        case 'rainbow':
            // Радужное отображение с акцентом на фазе
            const h = ((arg + Math.PI) / (2 * Math.PI)) * 360;
            const s = 80 + 20 * Math.sin(normAbs * Math.PI);
            const l = 50 + 25 * Math.sin(normAbs * Math.PI / 2);
            return `hsla(${h}, ${s}%, ${l}%, 1)`;
            
        case 'magnetic':
            // Магнитные линии, модуль влияет на интенсивность
            const lines = 12; // Количество линий
            const lineWidth = 0.1; // Относительная ширина линии
            
            const angle = ((arg + Math.PI) / (2 * Math.PI)) * lines;
            const distFromLine = Math.min(
                Math.abs(angle - Math.floor(angle)),
                Math.abs(Math.ceil(angle) - angle)
            );
            
            const intensity = Math.exp(-distFromLine / lineWidth) * (0.5 + 0.5 * normAbs);
            const phase = ((arg + Math.PI) / (2 * Math.PI)) * 360;
            
            return `hsla(${phase}, 100%, ${50 + 50 * intensity}%, ${0.5 + 0.5 * intensity})`;
            
        case 'blackhole':
            // Визуализация в стиле черной дыры
            const blackHoleRadius = 0.1;
            const accretionDiskWidth = 0.2;
            
            if (abs < blackHoleRadius) {
                return 'rgba(0, 0, 0, 1)'; // Черная дыра
            }
            
            const relativeDistance = (abs - blackHoleRadius) / accretionDiskWidth;
            
            if (relativeDistance < 1) {
                // Аккреционный диск
                const temp = 2000 + 8000 * (1 - relativeDistance); // Температура от 2000K до 10000K
                return this.temperatureToColor(temp, 0.8 + 0.2 * relativeDistance);
            } else {
                // Окружающее пространство
                const phaseColor = ((arg + Math.PI) / (2 * Math.PI)) * 360;
                const brightness = Math.max(0, 0.8 - (relativeDistance - 1) * 0.2);
                return `hsla(${phaseColor}, 50%, ${brightness * 100}%, ${brightness})`;
            }
            
        case 'default':
        default:
            // Стандартное отображение: фаза -> оттенок, модуль -> яркость
            const hueAngle = ((arg + Math.PI) / (2 * Math.PI)) * 360;
            const saturation = 80 + 20 * Math.sin(normAbs * Math.PI * 2);
            const lightness = 50 - 30 * Math.cos(normAbs * Math.PI);
            
            return `hsla(${hueAngle}, ${saturation}%, ${lightness}%, 1)`;
    }
},

// Преобразование температуры в цвет (для метода визуализации 'blackhole')
temperatureToColor: function(temperature, alpha = 1) {
    // Аппроксимация цвета черного тела
    // Формула из: http://www.tannerhelland.com/4435/convert-temperature-rgb-algorithm-code/
    
    temperature = Math.min(40000, Math.max(1000, temperature)) / 100;
    
    let r, g, b;
    
    if (temperature <= 66) {
        r = 255;
        g = 99.4708025861 * Math.log(temperature) - 161.1195681661;
        if (temperature <= 19) {
            b = 0;
        } else {
            b = 138.5177312231 * Math.log(temperature - 10) - 305.0447927307;
        }
    } else {
        r = 329.698727446 * Math.pow(temperature - 60, -0.1332047592);
        g = 288.1221695283 * Math.pow(temperature - 60, -0.0755148492);
        b = 255;
    }
    
    r = Math.min(255, Math.max(0, Math.round(r)));
    g = Math.min(255, Math.max(0, Math.round(g)));
    b = Math.min(255, Math.max(0, Math.round(b)));
    
    return `rgba(${r}, ${g}, ${b}, ${alpha})`;
},

// Создание текстуры фрактала Жюлиа для точки комплексной плоскости
juliaTexture: function(z, c, iterations = 20, escapeRadius = 2) {
    let w = { re: z.re, im: z.im };
    let i = 0;
    
    while (i < iterations && this.abs(w) < escapeRadius) {
        w = this.add(this.multiply(w, w), c);
        i++;
    }
    
    if (i === iterations) {
        return { re: 0, im: 0, julia: true, inside: true };
    }
    
    // Плавная окраска с использованием логарифма
    const smoothed = i + 1 - Math.log(Math.log(this.abs(w))) / Math.log(2);
    return { re: smoothed / iterations, im: 0, julia: true, value: i };
}
