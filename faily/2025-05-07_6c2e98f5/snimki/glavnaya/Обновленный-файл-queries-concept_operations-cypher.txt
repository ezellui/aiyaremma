// queries/concept_operations.cypher

// Создание концепции с категориями
// Параметры: $conceptData, $categories
UNWIND $categories AS category
CREATE (c:Concept {
  concept_id: $conceptData.concept_id,
  name: $conceptData.name,
  description: $conceptData.description,
  creation_date: datetime($conceptData.creation_date),
  is_synthesis: $conceptData.is_synthesis,
  innovation_degree: $conceptData.innovation_degree
})
CREATE (cat:Category {
  category_id: category.category_id,
  concept_id: $conceptData.concept_id,
  name: category.name,
  definition: category.definition,
  type: category.type,
  centrality: category.characteristics.centrality,
  certainty: category.characteristics.certainty,
  historical_significance: category.characteristics.historical_significance,
  innovation_degree: category.characteristics.innovation_degree,
  clarity: category.characteristics.clarity,
  breadth: category.characteristics.breadth,
  depth: category.characteristics.depth,
  applicability: category.characteristics.applicability
})
CREATE (c)-[:INCLUDES]->(cat)
RETURN c, collect(cat) as categories;

// Получение полного графа концепции
// Параметры: $concept_id
MATCH (c:Concept {concept_id: $concept_id})
OPTIONAL MATCH (c)-[:INCLUDES]->(cat:Category)
OPTIONAL MATCH (cat)-[r:RELATED_TO]-(related:Category)
WHERE related.concept_id = $concept_id
RETURN c, collect(DISTINCT cat) as categories, 
       collect(DISTINCT r) as relationships, 
       collect(DISTINCT related) as relatedCategories;

// Обновление характеристик категории
// Параметры: $category_id, $updates
MATCH (cat:Category {category_id: $category_id})
SET cat += $updates
SET cat.updated_at = datetime()
RETURN cat;

// Создание связи между категориями
// Параметры: $source_id, $target_id, $relationshipData
MATCH (source:Category {category_id: $source_id})
MATCH (target:Category {category_id: $target_id})
CREATE (source)-[r:RELATED_TO {
  relationship_id: $relationshipData.relationship_id,
  type: $relationshipData.type,
  direction: $relationshipData.direction,
  strength: $relationshipData.characteristics.strength,
  certainty: $relationshipData.characteristics.certainty,
  innovation_degree: $relationshipData.characteristics.innovation_degree,
  historical_support: $relationshipData.characteristics.historical_support,
  logical_necessity: $relationshipData.characteristics.logical_necessity,
  context_dependence: $relationshipData.characteristics.context_dependence,
  created_at: datetime()
}]->(target)
RETURN r;

// Удаление категории и её связей
// Параметры: $category_id
MATCH (cat:Category {category_id: $category_id})
DETACH DELETE cat;

// Валидация структуры графа
// Параметры: $concept_id
MATCH (c:Concept {concept_id: $concept_id})-[:INCLUDES]->(cat:Category)
OPTIONAL MATCH (cat)-[r:RELATED_TO]-(related:Category)
WHERE related.concept_id = $concept_id
WITH c, count(DISTINCT cat) as categoryCount, count(r) as relationshipCount,
     collect(cat.type) as categoryTypes
// Проверка наличия категорий различных типов
WITH c, categoryCount, relationshipCount, categoryTypes,
     size([type IN categoryTypes WHERE type = 'ontological']) as ontologicalCount,
     size([type IN categoryTypes WHERE type = 'epistemological']) as epistemologicalCount
RETURN c.concept_id as conceptId,
       categoryCount,
       relationshipCount,
       CASE 
         WHEN categoryCount < 3 THEN 'Too few categories'
         WHEN relationshipCount < categoryCount - 1 THEN 'Graph might be disconnected'
         WHEN ontologicalCount = 0 THEN 'Missing ontological categories'
         WHEN epistemologicalCount = 0 THEN 'Missing epistemological categories'
         ELSE 'Valid structure'
       END as validationResult;

// Анализ центральности категорий
// Параметры: $concept_id
MATCH (c:Concept {concept_id: $concept_id})-[:INCLUDES]->(cat:Category)
OPTIONAL MATCH (cat)-[r:RELATED_TO]-(related:Category)
WHERE related.concept_id = $concept_id
WITH cat, count(r) as connectionCount
ORDER BY connectionCount DESC
RETURN cat.category_id as categoryId,
       cat.name as name,
       cat.type as type,
       connectionCount,
       cat.centrality as definedCentrality,
       toFloat(connectionCount) / toFloat((
          SELECT count(*) FROM (MATCH (c:Concept {concept_id: $concept_id})-[:INCLUDES]->(ca:Category) RETURN ca)
       ) - 1) as calculatedCentrality;

// Анализ характеристик связей
// Параметры: $concept_id
MATCH (c:Concept {concept_id: $concept_id})-[:INCLUDES]->(cat:Category)
MATCH (cat)-[r:RELATED_TO]-(related:Category)
WHERE related.concept_id = $concept_id
RETURN r.relationship_id as relationshipId,
       startNode(r).name as sourceName,
       endNode(r).name as targetName,
       r.type as type,
       r.direction as direction,
       r.strength as strength,
       r.certainty as certainty,
       r.innovation_degree as innovationDegree,
       r.logical_necessity as logicalNecessity,
       r.context_dependence as contextDependence,
       r.historical_support as historicalSupport;

// Поиск путей между категориями
// Параметры: $start_category_id, $end_category_id
MATCH path = shortestPath(
  (start:Category {category_id: $start_category_id})-[:RELATED_TO*]-(end:Category {category_id: $end_category_id})
)
WHERE all(cat IN nodes(path) WHERE cat.concept_id = start.concept_id)
RETURN path;

// Экспорт графа для визуализации
// Параметры: $concept_id
MATCH (c:Concept {concept_id: $concept_id})-[:INCLUDES]->(cat:Category)
OPTIONAL MATCH (cat)-[r:RELATED_TO]-(related:Category)
WHERE related.concept_id = $concept_id
WITH collect(DISTINCT cat) as nodes, collect(DISTINCT r) as edges
RETURN 
  [node IN nodes | {
    id: node.category_id,
    label: node.name,
    type: node.type,
    centrality: node.centrality,
    certainty: node.certainty,
    historical_significance: node.historical_significance,
    innovation_degree: node.innovation_degree,
    clarity: node.clarity,
    breadth: node.breadth,
    depth: node.depth,
    applicability: node.applicability
  }] as nodes,
  [edge IN edges | {
    id: edge.relationship_id,
    source: startNode(edge).category_id,
    target: endNode(edge).category_id,
    type: edge.type,
    direction: edge.direction,
    strength: edge.strength,
    certainty: edge.certainty,
    innovation_degree: edge.innovation_degree,
    logical_necessity: edge.logical_necessity,
    context_dependence: edge.context_dependence,
    historical_support: edge.historical_support
  }] as edges;

// Обнаружение циклов в графе
// Параметры: $concept_id
MATCH (c:Concept {concept_id: $concept_id})-[:INCLUDES]->(cat:Category)
MATCH path = (cat)-[:RELATED_TO*3..]-(cat)
WHERE all(node IN nodes(path) WHERE node.concept_id = $concept_id)
RETURN path
LIMIT 10;

// Анализ типов категорий
// Параметры: $concept_id
MATCH (c:Concept {concept_id: $concept_id})-[:INCLUDES]->(cat:Category)
RETURN cat.type as categoryType, count(cat) as count
ORDER BY count DESC;

// Анализ типов связей
// Параметры: $concept_id
MATCH (c:Concept {concept_id: $concept_id})-[:INCLUDES]->(cat:Category)
MATCH (cat)-[r:RELATED_TO]-(related:Category)
WHERE related.concept_id = $concept_id
RETURN r.type as relationshipType, count(r) as count
ORDER BY count DESC;

// Поиск категорий с высокой инновационностью
// Параметры: $concept_id, $min_innovation_level
MATCH (c:Concept {concept_id: $concept_id})-[:INCLUDES]->(cat:Category)
WHERE cat.innovation_degree >= $min_innovation_level
RETURN cat.category_id as categoryId,
       cat.name as name,
       cat.type as type,
       cat.innovation_degree as innovationDegree
ORDER BY cat.innovation_degree DESC;

// Поиск связей с высокой логической необходимостью
// Параметры: $concept_id, $min_logical_necessity
MATCH (c:Concept {concept_id: $concept_id})-[:INCLUDES]->(cat:Category)
MATCH (cat)-[r:RELATED_TO]-(related:Category)
WHERE related.concept_id = $concept_id AND r.logical_necessity >= $min_logical_necessity
RETURN r.relationship_id as relationshipId,
       startNode(r).name as sourceName,
       endNode(r).name as targetName,
       r.type as type,
       r.logical_necessity as logicalNecessity
ORDER BY r.logical_necessity DESC;

// Создание узла философской традиции
// Параметры: $tradition_data
CREATE (t:Tradition {
  tradition_id: $tradition_data.tradition_id,
  name: $tradition_data.name,
  time_period: $tradition_data.time_period,
  created_at: datetime()
})
RETURN t;

// Связывание категории с философской традицией
// Параметры: $category_id, $tradition_id, $strength
MATCH (cat:Category {category_id: $category_id})
MATCH (t:Tradition {tradition_id: $tradition_id})
CREATE (cat)-[r:BELONGS_TO_TRADITION {
  strength: $strength,
  created_at: datetime()
}]->(t)
RETURN r;
