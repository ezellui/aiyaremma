// Функция для создания математического потока мыслей
function generateMathematicalThoughtFlow() {
    // Выбираем категорию
    const categories = Object.keys(mathematicalConcepts.categories);
    const category = categories[Math.floor(Math.random() * categories.length)];
    
    // Выбираем концепты из выбранной категории
    const concepts = mathematicalConcepts.categories[category];
    const concept1 = concepts[Math.floor(Math.random() * concepts.length)];
    
    let concept2;
    do {
        concept2 = concepts[Math.floor(Math.random() * concepts.length)];
    } while (concept2 === concept1);
    
    // Выбираем паттерн
    const pattern = mathematicalConcepts.patternStream[Math.floor(Math.random() * mathematicalConcepts.patternStream.length)];
    
    // Выбираем метафору
    const metaphor = mathematicalConcepts.visualMetaphors[Math.floor(Math.random() * mathematicalConcepts.visualMetaphors.length)];
    
    // Генерируем глубинную математическую мысль
    const probabilityStructure = Math.random();
    
    if (probabilityStructure < 0.25) {
        // Концепт и паттерн
        return `${concept1}: ${pattern}...`;
    } else if (probabilityStructure < 0.5) {
        // Комбинация концептов
        return `${concept1} ⟹ ${concept2}...`;
    } else if (probabilityStructure < 0.75) {
        // Метафора и концепт
        return `${metaphor} через ${concept1}...`;
    } else {
        // Интеграция концептов
        return `${concept1} ∩ ${concept2} в пространстве ${metaphor}...`;
    }
}

// Функция для генерации звуков математических формул
function playMathFormulaSound(x, y) {
    if (!audioContext || !audioEnabled) return;
    
    // Выбираем звуковой паттерн
    const pattern = mathSoundPatterns[Math.floor(Math.random() * mathSoundPatterns.length)];
    
    // Модифицируем базовую частоту в зависимости от положения на экране
    const freqModifier = 0.8 + (x / window.innerWidth) * 0.4;
    const baseFreq = pattern.baseFreq * freqModifier;
    
    // Создаем последовательность нот на основе гармоник и ритма
    for (let i = 0; i < pattern.harmonics.length; i++) {
        const harmonic = pattern.harmonics[i];
        const delay = i > 0 ? pattern.rhythm.slice(0, i).reduce((a, b) => a + b, 0) : 0;
        
        setTimeout(() => {
            // Создаем осциллятор для этой ноты
            const oscillator = audioContext.createOscillator();
            
            // Выбираем тип волны в зависимости от гармоники
            if (harmonic % 1 !== 0) {
                oscillator.type = 'sine'; // Чистый синус для иррациональных гармоник
            } else if (harmonic % 2 === 0) {
                oscillator.type = 'triangle'; // Треугольник для четных гармоник
            } else {
                oscillator.type = Math.random() < 0.7 ? 'sine' : 'sawtooth'; // Преимущественно синус для нечетных
            }
            
            oscillator.frequency.value = baseFreq * harmonic;
            
            // Создаем усилитель с огибающей ADSR
            const gainNode = audioContext.createGain();
            gainNode.gain.value = 0;
            
            // Создаем фильтр для формирования тембра
            const filter = audioContext.createBiquadFilter();
            filter.type = 'bandpass';
            filter.frequency.value = oscillator.frequency.value * 1.5;
            filter.Q.value = 1 + Math.random() * 5;
            
            // Создаем панорамирование в зависимости от позиции клика
            const panner = audioContext.createStereoPanner();
            panner.pan.value = (x / window.innerWidth) * 2 - 1; // От -1 до 1
            
            // Соединяем компоненты
            oscillator.connect(filter);
            filter.connect(gainNode);
            gainNode.connect(panner);
            panner.connect(analyser);
            
            // Настраиваем огибающую (attack-release)
            const attackTime = 0.01 + Math.random() * 0.1;
            const releaseTime = pattern.decayRate * (0.1 + Math.random() * 0.3);
            const sustainTime = 0.1 + Math.random() * 0.2;
            const volume = 0.2 - (i * 0.03); // Уменьшение громкости для высоких гармоник
            
            const now = audioContext.currentTime;
            
            // Форма огибающей
            gainNode.gain.setValueAtTime(0, now);
            gainNode.gain.linearRampToValueAtTime(volume, now + attackTime);
            gainNode.gain.setValueAtTime(volume, now + attackTime + sustainTime);
            gainNode.gain.exponentialRampToValueAtTime(0.001, now + attackTime + sustainTime + releaseTime);
            
            // Модуляция частоты для "живости" звука
            const freqModAmount = harmonic < 2 ? 0.01 : 0.03;
            oscillator.frequency.setValueAtTime(oscillator.frequency.value, now);
            oscillator.frequency.exponentialRampToValueAtTime(
                oscillator.frequency.value * (1 + (Math.random() * 2 - 1) * freqModAmount),
                now + attackTime + sustainTime + releaseTime
            );
            
            // Запускаем и останавливаем
            oscillator.start(now);
            oscillator.stop(now + attackTime + sustainTime + releaseTime + 0.1);
            
            // Удаляем ноду после окончания звучания
            setTimeout(() => {
                oscillator.disconnect();
                filter.disconnect();
                gainNode.disconnect();
                panner.disconnect();
            }, (attackTime + sustainTime + releaseTime + 0.2) * 1000);
            
        }, delay * 1000);
    }
}
