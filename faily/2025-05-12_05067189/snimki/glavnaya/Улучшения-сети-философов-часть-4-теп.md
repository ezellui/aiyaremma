  /**
   * Создает тепловую карту влияния между школами

  /**

   * @param {string} school - ID школы

  /**
   * Создает тепловую карту по историческим периодам
   * @param {number} periodSize - Размер периода в годах
   * @param {string} linkType - Тип связи для анализа
   */
  function createPeriodHeatmap(periodSize, linkType) {
    const container = document.getElementById('period-heatmap-container');
    container.innerHTML = '';
    
    // Находим временной диапазон всей сети
    const yearsData = data.nodes
      .map(node => extractYearsFromData(node))
      .filter(years => years.birthYear !== null && years.deathYear !== null);
    
    const allYears = yearsData.flatMap(years => [years.birthYear, years.deathYear]);
    const minYear = Math.min(...allYears);
    const maxYear = Math.max(...allYears);
    
    // Разбиваем на периоды
    const periods = [];
    for (let year = minYear; year < maxYear; year += periodSize) {
      periods.push({
        start: year,
        end: year + periodSize - 1,
        label: `${year}-${year + periodSize - 1}`
      });
    }
    
    // Группируем философов по периодам (философ может относиться к нескольким периодам)
    const periodPhilosophers = periods.map(period => {
      return {
        period: period,
        philosophers: data.nodes.filter(node => {
          const years = extractYearsFromData(node);
          return years.birthYear !== null && years.deathYear !== null && 
                 isInTimeRange(years, period.start, period.end);
        })
      };
    });
    
    // Создаем матрицу влияния между периодами
    const matrix = [];
    const maxValue = { value: 0 };
    
    periodPhilosophers.forEach(sourcePeriod => {
      const row = [];
      
      periodPhilosophers.forEach(targetPeriod => {
        // Считаем вес влияния от источника к цели
        let totalWeight = 0;
        let connectionCount = 0;
        
        sourcePeriod.philosophers.forEach(source => {
          targetPeriod.philosophers.forEach(target => {
            if (source.id !== target.id) {  // Исключаем саму с собой
              const weight = calculateInfluenceWeight(source, target, linkType);
              if (weight > 0) {
                totalWeight += weight;
                connectionCount++;
              }
            }
          });
        });
        
        const avgWeight = connectionCount > 0 ? totalWeight / connectionCount : 0;
        if (avgWeight > maxValue.value) maxValue.value = avgWeight;
        
        row.push({
          source: sourcePeriod.period,
          target: targetPeriod.period,
          value: avgWeight,
          connectionCount: connectionCount,
          sourcePhilosophers: sourcePeriod.philosophers,
          targetPhilosophers: targetPeriod.philosophers
        });
      });
      
      matrix.push(row);
    });
    
    // Рисуем тепловую карту
    drawPeriodMatrix(container, matrix, periods, maxValue.value);
  }

  /**
   * Применяет тепловую карту школ к сетевой визуализации
   * @param {string} school1 - ID первой школы
   * @param {string} school2 - ID второй школы
   * @param {string} linkType - Тип связи для анализа
   */
  function applySchoolHeatmapToNetwork(school1, school2, linkType) {
    // Сбрасываем предыдущие выделения
    resetNetworkHighlighting();
    
    // Находим философов из каждой школы
    const school1Ids = new Set(data.nodes.filter(node => node.group === school1).map(n => n.id));
    const school2Ids = new Set(data.nodes.filter(node => node.group === school2).map(n => n.id));
    
    // Выделяем узлы в сети
    d3.selectAll(".node").each(function(d) {
      const node = d3.select(this);
      const inSchool1 = school1Ids.has(d.id);
      const inSchool2 = school2Ids.has(d.id);
      
      if (inSchool1 || inSchool2) {
        node.classed("heat-active", true);
      } else {
        node.classed("heat-inactive", true);
      }
    });
    
    // Выделяем связи в сети
    d3.selectAll(".links line").each(function(d) {
      const link = d3.select(this);
      const sourceId = typeof d.source === 'object' ? d.source.id : d.source;
      const targetId = typeof d.target === 'object' ? d.target.id : d.target;
      
      const sourceInSchool1 = school1Ids.has(sourceId);
      const targetInSchool2 = school2Ids.has(targetId);
      
      // Проверяем тип связи
      const isCorrectType = linkType === 'all' || d.type === linkType;
      
      if (isCorrectType && ((sourceInSchool1 && targetInSchool2) || (school1 === school2 && sourceInSchool1 && targetInSchool2))) {
        link.classed("heat-highlighted", true);
      } else if ((sourceInSchool1 || sourceInSchool2) && (targetInSchool1 || targetInSchool2)) {
        link.classed("heat-active", true);
      } else {
        link.classed("heat-inactive", true);
      }
    });
  }
  
  /**
   * Применяет тепловую карту периодов к сетевой визуализации
   * @param {number} periodSize - Размер периода в годах
   * @param {string} linkType - Тип связи для анализа
   */
  function applyPeriodHeatmapToNetwork(periodSize, linkType) {
    // Сбрасываем предыдущие выделения
    resetNetworkHighlighting();
    
    // Находим временной диапазон всей сети
    const yearsData = data.nodes
      .map(node => extractYearsFromData(node))
      .filter(years => years.birthYear !== null && years.deathYear !== null);
    
    const allYears = yearsData.flatMap(years => [years.birthYear, years.deathYear]);
    const minYear = Math.min(...allYears);
    const maxYear = Math.max(...allYears);
    
    // Создаем периоды
    const periods = [];
    for (let year = minYear; year < maxYear; year += periodSize) {
      periods.push({
        start: year,
        end: year + periodSize - 1
      });
    }
    
    // Ищем два периода с максимальным взаимодействием
    let maxInteractionPeriods = [0, 0];
    let maxInteractionValue = 0;
    
    for (let i = 0; i < periods.length; i++) {
      for (let j = 0; j < periods.length; j++) {
        if (i === j) continue;
        
        const sourcePeriod = periods[i];
        const targetPeriod = periods[j];
        
        // Находим философов в каждом периоде
        const sourcePhilosophers = data.nodes.filter(node => {
          const years = extractYearsFromData(node);
          return years.birthYear !== null && years.deathYear !== null && 
                 isInTimeRange(years, sourcePeriod.start, sourcePeriod.end);
        });
        
        const targetPhilosophers = data.nodes.filter(node => {
          const years = extractYearsFromData(node);
          return years.birthYear !== null && years.deathYear !== null && 
                 isInTimeRange(years, targetPeriod.start, targetPeriod.end);
        });
        
        // Считаем силу взаимодействия
        let interactionValue = 0;
        
        sourcePhilosophers.forEach(source => {
          targetPhilosophers.forEach(target => {
            if (source.id !== target.id) {
              interactionValue += calculateInfluenceWeight(source, target, linkType);
            }
          });
        });
        
        if (interactionValue > maxInteractionValue) {
          maxInteractionValue = interactionValue;
          maxInteractionPeriods = [i, j];
        }
      }
    }
    
    // Выделяем узлы из периодов с максимальным взаимодействием
    const period1 = periods[maxInteractionPeriods[0]];
    const period2 = periods[maxInteractionPeriods[1]];
    
    d3.selectAll(".node").each(function(d) {
      const node = d3.select(this);
      const years = extractYearsFromData(d);
      
      const inPeriod1 = years.birthYear !== null && years.deathYear !== null && 
                       isInTimeRange(years, period1.start, period1.end);
      const inPeriod2 = years.birthYear !== null && years.deathYear !== null && 
                       isInTimeRange(years, period2.start, period2.end);
      
      if (inPeriod1) {
        node.classed("heat-active", true);
      } else if (inPeriod2) {
        node.classed("heat-active", true);
      } else {
        node.classed("heat-inactive", true);
      }
    });
    
    // Выделяем связи между периодами
    d3.selectAll(".links line").each(function(d) {
      const link = d3.select(this);
      const sourceYears = extractYearsFromData(d.source);
      const targetYears = extractYearsFromData(d.target);
      
      const sourceInPeriod1 = sourceYears.birthYear !== null && sourceYears.deathYear !== null && 
                             isInTimeRange(sourceYears, period1.start, period1.end);
      const targetInPeriod2 = targetYears.birthYear !== null && targetYears.deathYear !== null && 
                             isInTimeRange(targetYears, period2.start, period2.end);
      
      const sourceInPeriod2 = sourceYears.birthYear !== null && sourceYears.deathYear !== null && 
                             isInTimeRange(sourceYears, period2.start, period2.end);
      const targetInPeriod1 = targetYears.birthYear !== null && targetYears.deathYear !== null && 
                             isInTimeRange(targetYears, period1.start, period1.end);
      
      // Проверяем тип связи
      const isCorrectType = linkType === 'all' || d.type === linkType;
      
      if (isCorrectType && ((sourceInPeriod1 && targetInPeriod2) || (sourceInPeriod2 && targetInPeriod1))) {
        link.classed("heat-highlighted", true);
      } else if ((sourceInPeriod1 || sourceInPeriod2) && (targetInPeriod1 || targetInPeriod2)) {
        link.classed("heat-active", true);
      } else {
        link.classed("heat-inactive", true);
      }
    });
    
    // Отображаем информацию о выбранных периодах
    alert(`Выделены периоды с наиболее интенсивным взаимодействием:\n${period1.start}-${period1.end} и ${period2.start}-${period2.end}`);
  }
  
  /**
   * Применяет тепловую карту центральности к сетевой визуализации
   * @param {string} metric - Метрика центральности
   */
  function applyCustomHeatmapToNetwork(metric) {
    // Сбрасываем предыдущие выделения
    resetNetworkHighlighting();
    
    // Рассчитываем метрики для всех узлов
    const nodeMetrics = calculateNodeMetrics(metric);
    
    // Находим максимальное и минимальное значение метрики
    const metricValues = nodeMetrics.map(m => m.value);
    const maxMetric = Math.max(...metricValues);
    const minMetric = Math.min(...metricValues);
    
    // Цветовая шкала для метрики
    const colorScale = d3.scaleLinear()
      .domain([minMetric, maxMetric])
      .range(["#f7fbff", "#08306b"]);
    
    // Применяем выделение к узлам
    d3.selectAll(".node").each(function(d) {
      const node = d3.select(this);
      const metricData = nodeMetrics.find(m => m.id === d.id);
      
      if (metricData) {
        // Устанавливаем цвет в зависимости от значения метрики
        node.select("circle").style("fill", colorScale(metricData.value));
        
        // Выделяем узлы с высокими значениями метрики
        if (metricData.value > maxMetric * 0.7) {
          node.classed("heat-highlighted", true);
        }
      }
    });
    
    // Выделяем связи для узлов с высокими значениями метрики
    const topNodeIds = new Set(
      nodeMetrics
        .sort((a, b) => b.value - a.value)
        .slice(0, Math.ceil(nodeMetrics.length * 0.2))
        .map(m => m.id)
    );
    
    d3.selectAll(".links line").each(function(d) {
      const link = d3.select(this);
      const sourceId = typeof d.source === 'object' ? d.source.id : d.source;
      const targetId = typeof d.target === 'object' ? d.target.id : d.target;
      
      if (topNodeIds.has(sourceId) && topNodeIds.has(targetId)) {
        link.classed("heat-highlighted", true);
      } else if (topNodeIds.has(sourceId) || topNodeIds.has(targetId)) {
        link.classed("heat-active", true);
      } else {
        link.classed("heat-inactive", true);
      }
    });
  }
  
  /**
   * Рассчитывает метрики центральности для узлов
   * @param {string} metric - Тип метрики центральности
   * @return {Array} - Массив объектов с ID узла и значением метрики
   */
  function calculateNodeMetrics(metric) {
    const metrics = [];
    
    if (metric === 'degree') {
      // Рассчитываем степень центральности (количество связей)
      data.nodes.forEach(node => {
        const connections = data.links.filter(link => {
          const sourceId = typeof link.source === 'object' ? link.source.id : link.source;
          const targetId = typeof link.target === 'object' ? link.target.id : link.target;
          return sourceId === node.id || targetId === node.id;
        });
        
        metrics.push({
          id: node.id,
          name: node.name,
          value: connections.length
        });
      });
    }
    else if (metric === 'betweenness') {
      // Упрощенный расчет центральности по посредничеству
      
      // Создаем счетчик для каждого узла (сколько раз он лежит на кратчайшем пути)
      const betweennessCount = {};
      data.nodes.forEach(node => {
        betweennessCount[node.id] = 0;
      });
      
      // Проходим по всем парам узлов
      for (let i = 0; i < data.nodes.length; i++) {
        const sourceId = data.nodes[i].id;
        
        for (let j = i + 1; j < data.nodes.length; j++) {
          const targetId = data.nodes[j].id;
          
          // Находим кратчайший путь
          const path = findShortestPath(sourceId, targetId);
          
          // Увеличиваем счетчик для всех промежуточных узлов на пути
          if (path.length > 2) {
            for (let k = 1; k < path.length - 1; k++) {
              betweennessCount[path[k]]++;
            }
          }
        }
      }
      
      // Нормализуем значения
      const maxValue = Math.max(...Object.values(betweennessCount));
      
      data.nodes.forEach(node => {
        metrics.push({
          id: node.id,
          name: node.name,
          value: betweennessCount[node.id]
        });
      });
    }
    else if (metric === 'closeness') {
      // Упрощенный расчет центральности по близости
      
      data.nodes.forEach(sourceNode => {
        let sumDistances = 0;
        let reachableNodes = 0;
        
        data.nodes.forEach(targetNode => {
          if (sourceNode.id !== targetNode.id) {
            const path = findShortestPath(sourceNode.id, targetNode.id);
            
            if (path.length > 0) {
              sumDistances += path.length - 1;  // Вычитаем 1, т.к. путь включает исходный узел
              reachableNodes++;
            }
          }
        });
        
        // Близость равна 1 / средняя длина пути до других узлов
        const closeness = reachableNodes > 0 ? reachableNodes / sumDistances : 0;
        
        metrics.push({
          id: sourceNode.id,
          name: sourceNode.name,
          value: closeness
        });
      });
    }
    
    return metrics;
  }
  
  /**
   * Находит кратчайший путь между двумя узлами
   * @param {string} startId - ID начального узла
   * @param {string} endId - ID конечного узла
   * @return {Array} - Массив ID узлов, составляющих кратчайший путь
   */
  function findShortestPath(startId, endId) {
    // Используем алгоритм поиска в ширину для нахождения кратчайшего пути
    const queue = [startId];
    const visited = new Set([startId]);
    const predecessors = {};
    let found = false;
    
    while (queue.length > 0 && !found) {
      const current = queue.shift();
      
      // Перебираем всех соседей текущего узла
      const neighbors = [];
      data.links.forEach(link => {
        if (typeof link.source === 'object') {
          if (link.source.id === current && !visited.has(link.target.id)) {
            neighbors.push(link.target.id);
          } else if (link.target.id === current && !visited.has(link.source.id)) {
            neighbors.push(link.source.id);
          }
        } else {
          if (link.source === current && !visited.has(link.target)) {
            neighbors.push(link.target);
          } else if (link.target === current && !visited.has(link.source)) {
            neighbors.push(link.source);
          }
        }
      });
      
      for (const neighbor of neighbors) {
        if (!visited.has(neighbor)) {
          visited.add(neighbor);
          predecessors[neighbor] = current;
          queue.push(neighbor);
          
          // Если нашли конечный узел, прерываем поиск
          if (neighbor === endId) {
            found = true;
            break;
          }
        }
      }
    }
    
    // Если путь не найден, возвращаем пустой массив
    if (!found) {
      return [];
    }
    
    // Восстанавливаем путь
    const path = [endId];
    let current = endId;
    
    while (current !== startId) {
      current = predecessors[current];
      path.unshift(current);
    }
    
    return path;
  }
  
  /**
   * Рассчитывает вес влияния между двумя философами
   * @param {Object} source - Философ-источник
   * @param {Object} target - Философ-цель
   * @param {string} linkType - Тип связи для анализа
   * @param {Object} maxValue - Объект с максимальным значением (для обновления)
   * @return {number} - Вес влияния
   */
  function calculateInfluenceWeight(source, target, linkType, maxValue = null) {
    // Если философ сам с собой, возвращаем 0
    if (source.id === target.id) {
      return 0;
    }
    
    // Ищем прямые связи между философами
    const directLinks = data.links.filter(link => {
      const sourceId = typeof link.source === 'object' ? link.source.id : link.source;
      const targetId = typeof link.target === 'object' ? link.target.id : link.target;
      
      // Проверяем направление от source к target
      const isDirected = sourceId === source.id && targetId === target.id;
      
      // Проверяем тип связи, если указан
      const isCorrectType = linkType === 'all' || link.type === linkType;
      
      return isDirected && isCorrectType;
    });
    
    // Если есть прямые связи, суммируем их вес
    if (directLinks.length > 0) {
      const weight = directLinks.length;
      
      // Обновляем максимальное значение, если передан объект
      if (maxValue !== null && weight > maxValue.value) {
        maxValue.value = weight;
      }
      
      return weight;
    }
    
    return 0;
  }
  
  /**
   * Рисует матрицу тепловой карты
   * @param {HTMLElement} container - Контейнер для рисования
   * @param {Array} matrix - Матрица данных
   * @param {Array} rows - Элементы для строк
   * @param {Array} columns - Элементы для столбцов
   * @param {number} maxValue - Максимальное значение в матрице
   * @param {string} title - Заголовок тепловой карты
   */
  function drawMatrix(container, matrix, rows, columns, maxValue, title) {
    const width = container.clientWidth;
    const height = 500;
    const margin = { top: 80, right: 100, bottom: 50, left: 100 };
    
    // Размер ячейки
    const cellSize = Math.min(
      (width - margin.left - margin.right) / columns.length,
      (height - margin.top - margin.bottom) / rows.length
    );
    
    // Создаем SVG
    const svg = d3.select(container)
      .append("svg")
      .attr("width", width)
      .attr("height", height);
    
    // Добавляем заголовок
    svg.append("text")
      .attr("x", width / 2)
      .attr("y", margin.top / 2)
      .attr("text-anchor", "middle")
      .style("font-size", "16px")
      .style("font-weight", "bold")
      .text(title);
    
    // Создаем группу для матрицы
    const g = svg.append("g")
      .attr("transform", `translate(${margin.left},${margin.top})`);
    
    // Цветовая шкала
    const color = d3.scaleLinear()
      .domain([0, maxValue])
      .range(["white", "#69b3a2"]);
    
    // Создаем подписи для столбцов (философы из второй школы)
    g.selectAll(".column-label")
      .data(columns)
      .enter()
      .append("text")
      .attr("class", "column-label")
      .attr("x", (d, i) => i * cellSize + cellSize / 2)
      .attr("y", -10)
      .attr("text-anchor", "end")
      .attr("transform", (d, i) => `rotate(-45, ${i * cellSize + cellSize / 2}, -10)`)
      .style("font-size", "10px")
      .text(d => d.name);
    
    // Создаем подписи для строк (философы из первой школы)
    g.selectAll(".row-label")
      .data(rows)
      .enter()
      .append("text")
      .attr("class", "row-label")
      .attr("x", -5)
      .attr("y", (d, i) => i * cellSize + cellSize / 2)
      .attr("text-anchor", "end")
      .attr("dominant-baseline", "middle")
      .style("font-size", "10px")
      .text(d => d.name);
    
    // Создаем ячейки матрицы
    const rows_g = g.selectAll(".row")
      .data(matrix)
      .enter()
      .append("g")
      .attr("class", "row")
      .attr("transform", (d, i) => `translate(0, ${i * cellSize})`);
    
    // Создаем ячейки
    const cells = rows_g.selectAll(".cell")
      .data(d => d)
      .enter()
      .append("rect")
      .attr("class", "cell")
      .attr("x", (d, i) => i * cellSize)
      .attr("width", cellSize)
      .attr("height", cellSize)
      .style("fill", d => color(d.value))
      .style("stroke", "#ccc")
      .style("stroke-width", 0.5);
    
    // Добавляем всплывающую подсказку
    const tooltip = d3.select(container)
      .append("div")
      .attr("class", "heatmap-tooltip");
    
    cells.on("mouseover", function(event, d) {
      // Показываем подсказку
      tooltip.style("display", "block")
        .html(`
          <strong>От:</strong> ${d.source.name}<br>
          <strong>К:</strong> ${d.target.name}<br>
          <strong>Влияние:</strong> ${d.value.toFixed(2)}
        `)
        .style("left", (event.pageX - container.getBoundingClientRect().left + 10) + "px")
        .style("top", (event.pageY - container.getBoundingClientRect().top - 40) + "px");
      
      // Выделяем ячейку
      d3.select(this)
        .style("stroke", "#FF5722")
        .style("stroke-width", 2);
    })
    .on("mouseout", function() {
      // Скрываем подсказку
      tooltip.style("display", "none");
      
      // Убираем выделение
      d3.select(this)
        .style("stroke", "#ccc")
        .style("stroke-width", 0.5);
    });
    
    // Добавляем легенду
    const legendWidth = 200;
    const legendHeight = 20;
    
    const legendX = width - margin.right - legendWidth;
    const legendY = margin.top / 2;
    
    // Градиент для легенды
    const defs = svg.append("defs");
    const linearGradient = defs.append("linearGradient")
      .attr("id", "linear-gradient")
      .attr("x1", "0%")
      .attr("y1", "0%")
      .attr("x2", "100%")
      .attr("y2", "0%");
    
    linearGradient.append("stop")
      .attr("offset", "0%")
      .attr("stop-color", "white");
    
    linearGradient.append("stop")
      .attr("offset", "100%")
      .attr("stop-color", "#69b3a2");
    
    // Создаем легенду
    svg.append("rect")
      .attr("x", legendX)
      .attr("y", legendY)
      .attr("width", legendWidth)
      .attr("height", legendHeight)
      .style("fill", "url(#linear-gradient)");
    
    // Подписи для легенды
    svg.append("text")
      .attr("x", legendX)
      .attr("y", legendY - 5)
      .style("text-anchor", "start")
      .style("font-size", "12px")
      .text("0");
    
    svg.append("text")
      .attr("x", legendX + legendWidth)
      .attr("y", legendY - 5)
      .style("text-anchor", "end")
      .style("font-size", "12px")
      .text(maxValue.toFixed(2));
    
    svg.append("text")
      .attr("x", legendX + legendWidth / 2)
      .attr("y", legendY - 5)
      .style("text-anchor", "middle")
      .style("font-size", "12px")
      .text("Интенсивность влияния");
  }

  /**
   * Рисует тепловую карту на основе метрик центральности
   * @param {HTMLElement} container - Контейнер для рисования
   * @param {Array} metrics - Массив с метриками узлов
   * @param {string} metricType - Тип метрики
   */
  function drawCustomHeatmap(container, metrics, metricType) {
    const width = container.clientWidth;
    const height = 500;
    const margin = { top: 80, right: 30, bottom: 50, left: 50 };
    
    // Создаем SVG
    const svg = d3.select(container)
      .append("svg")
      .attr("width", width)
      .attr("height", height);
    
    // Получаем метрику в читаемом виде
    const metricName = {
      'degree': 'Количество связей',
      'betweenness': 'Посредничество',
      'closeness': 'Близость'
    }[metricType] || metricType;
    
    // Добавляем заголовок
    svg.append("text")
      .attr("x", width / 2)
      .attr("y", margin.top / 2)
      .attr("text-anchor", "middle")
      .style("font-size", "16px")
      .style("font-weight", "bold")
      .text(`Тепловая карта центральности: ${metricName}`);
    
    // Создаем группу для графика
    const g = svg.append("g")
      .attr("transform", `translate(${margin.left},${margin.top})`);
    
    // Сортируем метрики по значению
    const sortedMetrics = [...metrics].sort((a, b) => b.value - a.value);
    
    // Максимальное значение метрики
    const maxValue = sortedMetrics[0].value;
    
    // Ограничиваем количество отображаемых философов
    const displayCount = Math.min(sortedMetrics.length, 25);
    const displayMetrics = sortedMetrics.slice(0, displayCount);
    
    // Шкалы
    const x = d3.scaleBand()
      .domain(displayMetrics.map(d => d.name))
      .range([0, width - margin.left - margin.right])
      .padding(0.1);
    
    const y = d3.scaleLinear()
      .domain([0, maxValue])
      .nice()
      .range([height - margin.top - margin.bottom, 0]);
    
    // Цветовая шкала
    const color = d3.scaleLinear()
      .domain([0, maxValue])
      .range(["#f7fbff", "#08306b"]);
    
    // Создаем оси
    g.append("g")
      .attr("transform", `translate(0,${height - margin.top - margin.bottom})`)
      .call(d3.axisBottom(x))
      .selectAll("text")
      .attr("transform", "translate(-10,0)rotate(-45)")
      .style("text-anchor", "end");
    
    g.append("g")
      .call(d3.axisLeft(y));
    
    // Создаем столбцы
    const bars = g.selectAll(".bar")
      .data(displayMetrics)
      .enter()
      .append("rect")
      .attr("class", "bar")
      .attr("x", d => x(d.name))
      .attr("y", d => y(d.value))
      .attr("width", x.bandwidth())
      .attr("height", d => height - margin.top - margin.bottom - y(d.value))
      .attr("fill", d => color(d.value));
    
    // Добавляем всплывающую подсказку
    const tooltip = d3.select(container)
      .append("div")
      .attr("class", "heatmap-tooltip");
    
    bars.on("mouseover", function(event, d) {
      // Показываем подсказку
      tooltip.style("display", "block")
        .html(`
          <strong>Философ:</strong> ${d.name}<br>
          <strong>${metricName}:</strong> ${d.value.toFixed(2)}
        `)
        .style("left", (event.pageX - container.getBoundingClientRect().left + 10) + "px")
        .style("top", (event.pageY - container.getBoundingClientRect().top - 40) + "px");
      
      // Выделяем столбец
      d3.select(this)
        .attr("stroke", "#FF5722")
        .attr("stroke-width", 2);
    })
    .on("mouseout", function() {
      // Скрываем подсказку
      tooltip.style("display", "none");
      
      // Убираем выделение
      d3.select(this)
        .attr("stroke", "none");
    });
    
    // Добавляем подписи осей
    svg.append("text")
      .attr("transform", `translate(${width/2}, ${height - 10})`)
      .style("text-anchor", "middle")
      .text("Философы");
    
    svg.append("text")
      .attr("transform", "rotate(-90)")
      .attr("y", 15)
      .attr("x", -(height / 2))
      .style("text-anchor", "middle")
      .text(metricName);
  }
  
  /**
   * Сбрасывает выделение узлов и связей на сети
   */
  function resetNetworkHighlighting() {
    d3.selectAll(".node")
      .classed("heat-highlighted", false)
      .classed("heat-active", false)
      .classed("heat-inactive", false)
      .select("circle")
      .style("fill", d => getNodeColor(d.group));
    
    d3.selectAll(".links line")
      .classed("heat-highlighted", false)
      .classed("heat-active", false)
      .classed("heat-inactive", false);
  }
  
  // Показывает панель тепловых карт
  function showPanel() {
    panel.style.display = 'block';
    backdrop.style.display = 'block';
  }
  
  // Возвращаем контроллер тепловых карт
  return {
    show: showPanel,
    hide: closePanel,
    resetHighlighting: resetNetworkHighlighting
  };

    });
    
    // Формируем хронологический порядок школ
    const schools = Object.entries(schoolGroups).map(([group, philosophers]) => {
      // Находим средний год активности для школы
      const years = philosophers
        .map(p => extractYearsFromData(p))
        .filter(y => y.birthYear !== null && y.deathYear !== null);
      
      let avgYear = 1900; // Значение по умолчанию
      
      if (years.length > 0) {
        avgYear = years.reduce((sum, y) => sum + (y.birthYear + y.deathYear) / 2, 0) / years.length;
      }
      
      return {
        group,
        name: getGroupDisplayName(group),
        philosophers,
        avgYear
      };
    }).sort((a, b) => a.avgYear - b.avgYear);
    
    // Генерируем заголовок
    const title = `Российская философская традиция: ${
      focus === 'overview' ? 'общий обзор' : 
      focus === 'schools' ? 'школы и направления' : 
      focus === 'chronology' ? 'хронологическое развитие' : 
      'цепочки влияния'}`;
    
    text += `<div class="narrative-heading">${title}</div>\n\n`;
    
    // Вступление
    if (style === 'academic') {
      text += `Эволюция российской философской мысли представляет собой сложную систему взаимосвязанных интеллектуальных традиций, развивавшихся в контексте уникального культурно-исторического опыта. Данное исследование представляет аналитический обзор ключевых мыслителей, идейных течений и концептуальных связей, составляющих историю отечественной философии.\n\n`;
    } else if (style === 'journalistic') {
      text += `Российская философия – это захватывающая история идей, противостояний и неожиданных союзов. Представьте себе карту, где вместо городов – великие мыслители, а вместо дорог – влияния и споры, которые связывают их воедино. Давайте совершим увлекательное путешествие по этой карте, чтобы узнать, как формировалась интеллектуальная история России.\n\n`;
    } else if (style === 'educational') {
      text += `Философская мысль России имеет богатую историю, наполненную яркими мыслителями и оригинальными идеями. В этом обзоре мы познакомимся с основными философскими течениями, узнаем о ключевых философах и увидим, как их идеи связаны между собой. Это поможет нам лучше понять развитие российской философии и её вклад в мировую мысль.\n\n`;
    }
    
    // Основное содержание в зависимости от фокуса
    if (focus === 'overview' || focus === 'schools') {
      text += generateSchoolsNarrative(schools, style);
    }
    
    if (focus === 'chronology') {
      text += generateChronologyNarrative(schools, style);
    }
    
    if (focus === 'influences') {
      text += generateInfluencesNarrative(style);
    }
    
    // Заключение
    if (style === 'academic') {
      text += `\n<div class="narrative-heading">Заключение</div>\n\n`;
      text += `Таким образом, российская философская традиция представляет собой сложную сеть взаимодействий, влияний и полемики, отражающую многогранный характер национальной культуры и её диалог с мировой философской мыслью. Диалектическое взаимодействие религиозно-метафизических, социально-политических и естественнонаучных концепций создало уникальный философский ландшафт, продолжающий развиваться в современную эпоху.\n\n`;
    } else if (style === 'journalistic') {
      text += `\n<div class="narrative-heading">И что дальше?</div>\n\n`;
      text += `Карта российской философии продолжает расширяться. Новые мыслители вступают в диалог с великими предшественниками, переосмысливая их наследие для современности. Возможно, именно сейчас формируются идеи, которые изменят наше понимание мира и человека. Ведь философия – это не застывший музейный экспонат, а живой разговор, продолжающийся через века.\n\n`;
    } else if (style === 'educational') {
      text += `\n<div class="narrative-heading">Что мы узнали</div>\n\n`;
      text += `Мы познакомились с основными школами российской философии и увидели, как они развивались во времени. Важно помнить, что каждый философ внёс свой уникальный вклад в развитие мысли, а взаимное влияние идей создало богатую интеллектуальную традицию. Российская философия продолжает развиваться и сегодня, отвечая на вопросы современности и переосмысливая наследие прошлого.\n\n`;
    }
    
    // Отображаем текст
    contentElement.innerHTML = text;
  }
  
  /**
   * Генерирует нарративное описание школ философии
   * @param {Array} schools - Массив школ философии
   * @param {string} style - Стиль описания
   * @return {string} - HTML-текст с описанием
   */
  function generateSchoolsNarrative(schools, style) {
    let text = `<div class="narrative-heading">Основные философские школы</div>\n\n`;
    
    schools.forEach(school => {
      if (school.philosophers.length === 0) return;
      
      text += `<div class="narrative-subheading">${school.name}</div>\n\n`;
      
      // Сортируем философов по годам (если доступны)
      const sortedPhilosophers = [...school.philosophers].sort((a, b) => {
        const yearsA = extractYearsFromData(a);
        const yearsB = extractYearsFromData(b);
        
        if (yearsA.birthYear && yearsB.birthYear) {
          return yearsA.birthYear - yearsB.birthYear;
        }
        return 0;
      });
      
      // Формируем описание школы в зависимости от стиля
      if (style === 'academic') {
        text += `
