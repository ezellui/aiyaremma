```javascript
  /**
   * Подсчитывает количество связей между философами одной школы
   * @param {string} schoolGroup - ID группы/школы
   * @return {number} - Количество связей
   */
  function countIntraSchoolLinks(schoolGroup) {
    let count = 0;
    
    data.links.forEach(link => {
      const sourceGroup = typeof link.source === 'object' ? 
        link.source.group : 
        data.nodes.find(n => n.id === link.source)?.group;
      
      const targetGroup = typeof link.target === 'object' ? 
        link.target.group : 
        data.nodes.find(n => n.id === link.target)?.group;
      
      if (sourceGroup === schoolGroup && targetGroup === schoolGroup) {
        count++;
      }
    });
    
    return count;
  }
  
  /**
   * Подсчитывает количество связей между философами разных школ
   * @param {string} schoolGroup - ID группы/школы
   * @return {number} - Количество связей
   */
  function countExternalSchoolLinks(schoolGroup) {
    let count = 0;
    
    data.links.forEach(link => {
      const sourceGroup = typeof link.source === 'object' ? 
        link.source.group : 
        data.nodes.find(n => n.id === link.source)?.group;
      
      const targetGroup = typeof link.target === 'object' ? 
        link.target.group : 
        data.nodes.find(n => n.id === link.target)?.group;
      
      if ((sourceGroup === schoolGroup && targetGroup !== schoolGroup) || 
          (sourceGroup !== schoolGroup && targetGroup === schoolGroup)) {
        count++;
      }
    });
    
    return count;
  }
  
  /**
   * Подсчитывает количество связей у философа
   * @param {string} philosopherId - ID философа
   * @return {number} - Количество связей
   */
  function countPhilosopherConnections(philosopherId) {
    let count = 0;
    
    data.links.forEach(link => {
      const sourceId = typeof link.source === 'object' ? link.source.id : link.source;
      const targetId = typeof link.target === 'object' ? link.target.id : link.target;
      
      if (sourceId === philosopherId || targetId === philosopherId) {
        count++;
      }
    });
    
    return count;
  }

        text += `Даты жизни философов в этой цепочке:\n\n`;
        chain.forEach((philosopherId) => {
          const philosopher = data.nodes.find(n => n.id === philosopherId);
          if (philosopher) {
            text += `• <span class="philosopher-name">${philosopher.name}</span>: ${philosopher.years || 'даты неизвестны'}\n`;
          }
        });
        
        text += `\n`;
      });
    }
    
    return text;
  }
  
  /**
   * Находит значимые цепочки влияния в сети философов
   * @return {Array} - Массив цепочек влияния (массивов ID философов)
   */
  function findSignificantInfluenceChains() {
    const chains = [];
    
    // Строим граф только с отношениями влияния
    const influenceGraph = {};
    
    data.links.forEach(link => {
      if (link.type === 'influence' || link.type === 'mutual') {
        const sourceId = typeof link.source === 'object' ? link.source.id : link.source;
        const targetId = typeof link.target === 'object' ? link.target.id : link.target;
        
        if (!influenceGraph[sourceId]) influenceGraph[sourceId] = [];
        influenceGraph[sourceId].push(targetId);
        
        if (link.type === 'mutual') {
          if (!influenceGraph[targetId]) influenceGraph[targetId] = [];
          influenceGraph[targetId].push(sourceId);
        }
      }
    });
    
    // Находим философов с наибольшим количеством входящих связей (влиятельные)
    const influentialPhilosophers = data.nodes.map(node => {
      const incomingLinks = data.links.filter(link => {
        const targetId = typeof link.target === 'object' ? link.target.id : link.target;
        return targetId === node.id && (link.type === 'influence' || link.type === 'mutual');
      });
      
      return {
        id: node.id,
        incomingCount: incomingLinks.length
      };
    }).sort((a, b) => b.incomingCount - a.incomingCount).slice(0, 5);
    
    // Находим философов с наибольшим количеством исходящих связей (восприимчивые)
    const receptivePhilosophers = data.nodes.map(node => {
      const outgoingLinks = data.links.filter(link => {
        const sourceId = typeof link.source === 'object' ? link.source.id : link.source;
        return sourceId === node.id && (link.type === 'influence' || link.type === 'mutual');
      });
      
      return {
        id: node.id,
        outgoingCount: outgoingLinks.length
      };
    }).sort((a, b) => b.outgoingCount - a.outgoingCount).slice(0, 5);
    
    // Ищем цепочки между влиятельными и восприимчивыми философами
    influentialPhilosophers.forEach(influentialPhilosopher => {
      receptivePhilosophers.forEach(receptivePhilosopher => {
        // Ищем до 2 разных путей между каждой парой
        for (let maxChains = 0; maxChains < 2; maxChains++) {
          const path = findPath(receptivePhilosopher.id, influentialPhilosopher.id, 6, []);
          
          if (path && path.length > 2) {
            chains.push(path);
            
            // Модифицируем граф, чтобы найти альтернативный путь
            const sourceId = path[0];
            const targetId = path[1];
            
            // Временно удаляем связь для поиска альтернативного пути
            if (influenceGraph[sourceId]) {
              influenceGraph[sourceId] = influenceGraph[sourceId].filter(id => id !== targetId);
            }
          }
        }
      });
    });
    
    // Если цепочек недостаточно, ищем дополнительные
    if (chains.length < 3) {
      // Выбираем случайные пары философов из разных школ
      const schools = {};
      
      data.nodes.forEach(node => {
        if (!schools[node.group]) {
          schools[node.group] = [];
        }
        schools[node.group].push(node.id);
      });
      
      // Берем по одному философу из каждой школы
      const schoolPairs = [];
      const schoolGroups = Object.keys(schools);
      
      for (let i = 0; i < schoolGroups.length; i++) {
        for (let j = i + 1; j < schoolGroups.length; j++) {
          if (schools[schoolGroups[i]].length > 0 && schools[schoolGroups[j]].length > 0) {
            schoolPairs.push({
              source: schools[schoolGroups[i]][0],
              target: schools[schoolGroups[j]][0]
            });
          }
        }
      }
      
      // Ищем пути между философами из разных школ
      schoolPairs.forEach(pair => {
        const path = findPath(pair.source, pair.target, 6, []);
        if (path && path.length > 2) {
          chains.push(path);
        }
      });
    }
    
    // Возвращаем уникальные цепочки (не больше 5)
    return chains.slice(0, 5);
    
    // Вспомогательная функция для поиска пути
    function findPath(start, end, maxLength, visited) {
      if (start === end) {
        return [start];
      }
      
      if (maxLength <= 0 || visited.includes(start)) {
        return null;
      }
      
      visited = [...visited, start];
      
      const neighbors = influenceGraph[start] || [];
      
      for (const neighbor of neighbors) {
        const path = findPath(neighbor, end, maxLength - 1, visited);
        if (path) {
          return [start, ...path];
        }
      }
      
      return null;
    }
  }

  /**
   * Получает временной промежуток цепочки философов
   * @param {Array} chain - Массив ID философов
   * @param {boolean} readable - Вернуть читаемую строку (например, "почти 100 лет")
   * @return {string} - Временной промежуток
   */
  function getPhilosopherYearsSpan(chain, readable = false) {
    // Извлекаем годы жизни для философов в цепочке
    const yearsData = chain.map(id => {
      const philosopher = data.nodes.find(n => n.id === id);
      if (philosopher) {
        return extractYearsFromData(philosopher);
      }
      return { birthYear: null, deathYear: null };
    }).filter(y => y.birthYear !== null && y.deathYear !== null);
    
    if (yearsData.length < 2) return readable ? "неизвестный период" : "";
    
    // Находим мин. год рождения и макс. год смерти
    const minBirth = Math.min(...yearsData.map(y => y.birthYear));
    const maxDeath = Math.max(...yearsData.map(y => y.deathYear));
    
    const yearsDiff = maxDeath - minBirth;
    
    if (readable) {
      if (yearsDiff > 200) {
        return `более ${Math.floor(yearsDiff / 100) * 100} лет`;
      } else if (yearsDiff > 100) {
        return `почти ${Math.floor(yearsDiff / 100) * 100} лет`;
      } else if (yearsDiff > 50) {
        return `более полувека`;
      } else {
        return `${yearsDiff} лет`;
      }
    }
    
    return `с ${minBirth} по ${maxDeath} годы (${yearsDiff} лет)`;
  }
  
  /**
   * Генерирует описание цепочки влияния
   * @param {Array} chain - Массив ID философов
   * @param {string} style - Стиль описания
   * @return {string} - Описание цепочки
   */
  function getChainDescription(chain, style) {
    // Анализируем принадлежность философов к школам
    const schoolGroups = new Set();
    
    chain.forEach(id => {
      const philosopher = data.nodes.find(n => n.id === id);
      if (philosopher) {
        schoolGroups.add(philosopher.group);
      }
    });
    
    const schoolsCount = schoolGroups.size;
    
    if (style === 'academic') {
      if (schoolsCount > 1) {
        return `трансформацию идей через различные интеллектуальные парадигмы (${schoolsCount} школы)`;
      } else {
        return `развитие идей внутри одной интеллектуальной традиции`;
      }
    } else if (style === 'journalistic') {
      if (schoolsCount > 1) {
        return `идеи преодолевали границы между разными философскими школами, обогащаясь в каждой новой интерпретации`;
      } else {
        return `передавалось и развивалось уникальное философское наследие одной школы`;
      }
    } else if (style === 'educational') {
      if (schoolsCount > 1) {
        return `идеи переходили от одной философской школы к другой, показывая связи между разными направлениями мысли`;
      } else {
        return `развивалась и углублялась философская традиция внутри одного направления`;
      }
    }
    
    return "";
  }
  
  /**
   * Генерирует структурированный текст о философской сети
   * @param {string} format - Формат представления (список, таблица, иерархия)
   * @param {string} grouping - Тип группировки (по школам, периодам, влиянию)
   */
  function generateStructuredText(format, grouping) {
    const contentElement = document.getElementById('structured-content');
    contentElement.innerHTML = ''; // Очищаем содержимое
    
    // Подготавливаем данные в зависимости от типа группировки
    let structuredData;
    
    if (grouping === 'schools') {
      structuredData = groupDataBySchools();
    } else if (grouping === 'periods') {
      structuredData = groupDataByPeriods();
    } else if (grouping === 'influence') {
      structuredData = groupDataByInfluence();
    }
    
    // Форматируем данные в соответствии с выбранным форматом
    let text = '';
    
    if (format === 'list') {
      text = formatAsLists(structuredData);
    } else if (format === 'table') {
      text = formatAsTables(structuredData);
    } else if (format === 'hierarchy') {
      text = formatAsHierarchy(structuredData);
    }
    
    contentElement.innerHTML = text;
  }

  /**
   * Группирует данные философов по историческим периодам
   * @return {Object} - Сгруппированные данные
   */
  function groupDataByPeriods() {
    // Определяем исторические периоды
    const periods = [
      { id: "ancient", name: "Древнерусский период (до 1700)", start: 0, end: 1700 },
      { id: "enlightenment", name: "Эпоха Просвещения (1700-1800)", start: 1700, end: 1800 },
      { id: "19century", name: "XIX век", start: 1800, end: 1900 },
      { id: "silver", name: "Серебряный век (1900-1922)", start: 1900, end: 1922 },
      { id: "soviet", name: "Советский период (1922-1991)", start: 1922, end: 1991 },
      { id: "modern", name: "Современный период (с 1991)", start: 1991, end: 2030 }
    ];
    
    // Создаем группы для периодов
    const groups = {};
    periods.forEach(period => {
      groups[period.id] = {
        id: period.id,
        name: period.name,
        start: period.start,
        end: period.end,
        philosophers: []
      };
    });
    
    // Распределяем философов по периодам
    data.nodes.forEach(node => {
      const years = extractYearsFromData(node);
      
      // Если годы жизни не определены, пропускаем
      if (years.birthYear === null || years.deathYear === null) return;
      
      // Находим все периоды, к которым относится философ
      const relevantPeriods = periods.filter(period => 
        isInTimeRange(years, period.start, period.end)
      );
      
      // Находим связи философа
      const connections = data.links.filter(link => {
        const sourceId = typeof link.source === 'object' ? link.source.id : link.source;
        const targetId = typeof link.target === 'object' ? link.target.id : link.target;
        return sourceId === node.id || targetId === node.id;
      });
      
      // Добавляем философа во все соответствующие периоды
      relevantPeriods.forEach(period => {
        groups[period.id].philosophers.push({
          id: node.id,
          name: node.name,
          years: node.years,
          group: node.group,
          groupName: node.groupName || getGroupDisplayName(node.group),
          connections: connections.length,
          influences: connections.filter(link => {
            const sourceId = typeof link.source === 'object' ? link.source.id : link.source;
            return sourceId === node.id && (link.type === 'influence' || link.type === 'mutual');
          }).length,
          influencedBy: connections.filter(link => {
            const targetId = typeof link.target === 'object' ? link.target.id : link.target;
            return targetId === node.id && (link.type === 'influence' || link.type === 'mutual');
          }).length
        });
      });
    });
    
    // Сортируем философов по годам рождения в каждом периоде
    Object.values(groups).forEach(group => {
      group.philosophers.sort((a, b) => {
        const yearsA = extractYearsFromData({ years: a.years });
        const yearsB = extractYearsFromData({ years: b.years });
        
        if (yearsA.birthYear !== null && yearsB.birthYear !== null) {
          return yearsA.birthYear - yearsB.birthYear;
        }
        
        return 0;
      });
    });
    
    // Оставляем только группы с философами
    const result = Object.values(groups).filter(group => group.philosophers.length > 0);
    
    return {
      title: "Философы по историческим периодам",
      groups: result
    };
  }
  
  /**
   * Группирует данные философов по влиянию
   * @return {Object} - Сгруппированные данные
   */
  function groupDataByInfluence() {
    // Группы по влиянию
    const influenceGroups = [
      { id: "most_influential", name: "Наиболее влиятельные философы", philosophers: [] },
      { id: "most_influenced", name: "Философы, испытавшие наибольшее влияние", philosophers: [] },
      { id: "balanced", name: "Философы с сбалансированным влиянием", philosophers: [] },
      { id: "isolated", name: "Философы с малым количеством связей", philosophers: [] }
    ];
    
    // Подготавливаем данные о философах
    const philosophersData = data.nodes.map(node => {
      // Находим связи философа
      const outgoingLinks = data.links.filter(link => {
        const sourceId = typeof link.source === 'object' ? link.source.id : link.source;
        return sourceId === node.id && (link.type === 'influence' || link.type === 'mutual');
      });
      
      const incomingLinks = data.links.filter(link => {
        const targetId = typeof link.target === 'object' ? link.target.id : link.target;
        return targetId === node.id && (link.type === 'influence' || link.type === 'mutual');
      });
      
      return {
        id: node.id,
        name: node.name,
        years: node.years,
        group: node.group,
        groupName: node.groupName || getGroupDisplayName(node.group),
        outgoing: outgoingLinks.length,
        incoming: incomingLinks.length,
        total: outgoingLinks.length + incomingLinks.length,
        ratio: incomingLinks.length > 0 ? outgoingLinks.length / incomingLinks.length : Infinity
      };
    });
    
    // Сортируем философов по количеству исходящих связей (влияние на других)
    const sortedByOutgoing = [...philosophersData].sort((a, b) => b.outgoing - a.outgoing);
    
    // Сортируем философов по количеству входящих связей (влияние от других)
    const sortedByIncoming = [...philosophersData].sort((a, b) => b.incoming - a.incoming);
    
    // Сортируем по соотношению исходящих к входящим (баланс влияния)
    const sortedByRatio = [...philosophersData].sort((a, b) => {
      // Если соотношения равны или одно из них бесконечно
      if (Math.abs(a.ratio - b.ratio) < 0.01 || !isFinite(a.ratio) || !isFinite(b.ratio)) {
        // Сортируем по общему количеству связей
        return b.total - a.total;
      }
      
      // Чем ближе к 1.0, тем более сбалансирован философ
      return Math.abs(a.ratio - 1.0) - Math.abs(b.ratio - 1.0);
    });
    
    // Сортируем философов по общему количеству связей (изолированные философы)
    const sortedByTotal = [...philosophersData].sort((a, b) => a.total - b.total);
    
    // Заполняем группы (берем топ-15 для каждой)
    influenceGroups[0].philosophers = sortedByOutgoing.slice(0, 15).map(p => ({
      ...p,
      description: `Повлиял на ${p.outgoing} философов, испытал влияние ${p.incoming} философов`
    }));
    
    influenceGroups[1].philosophers = sortedByIncoming.slice(0, 15).map(p => ({
      ...p,
      description: `Испытал влияние ${p.incoming} философов, повлиял на ${p.outgoing} философов`
    }));
    
    influenceGroups[2].philosophers = sortedByRatio.slice(0, 15)
      .filter(p => p.total >= 2) // Должно быть хотя бы 2 связи для баланса
      .map(p => ({
        ...p,
        description: `Баланс влияния: ${p.ratio.toFixed(2)} (исходящих: ${p.outgoing}, входящих: ${p.incoming})`
      }));
    
    influenceGroups[3].philosophers = sortedByTotal.slice(0, 15)
      .filter(p => p.total < 3) // Только с малым количеством связей
      .map(p => ({
        ...p,
        description: `Всего ${p.total} философских связей (исходящих: ${p.outgoing}, входящих: ${p.incoming})`
      }));
    
    // Оставляем только группы с философами
    const result = influenceGroups.filter(group => group.philosophers.length > 0);
    
    return {
      title: "Философы по характеристикам влияния",
      groups: result
    };
  }
  
  /**
   * Форматирует данные в виде списков
   * @param {Object} data - Сгруппированные данные
   * @return {string} - HTML-текст со списками
   */
  function formatAsLists(data) {
    let text = `<h3>${data.title}</h3>\n\n`;
    
    data.groups.forEach(group => {
      text += `<h4>${group.name}</h4>\n\n`;
      text += `<ul>\n`;
      
      group.philosophers.forEach(philosopher => {
        text += `<li class="list-item">`;
        text += `<strong>${philosopher.name}</strong> ${philosopher.years ? `(${philosopher.years})` : ''}`;
        
        // Дополнительная информация в зависимости от типа группировки
        if (philosopher.groupName && !group.name.includes(philosopher.groupName)) {
          text += ` - ${philosopher.groupName}`;
        }
        
        if (philosopher.connections !== undefined) {
          text += ` - ${philosopher.connections} связей`;
        }
        
        if (philosopher.description) {
          text += `<br>${philosopher.description}`;
        }
        
        text += `</li>\n`;
      });
      
      text += `</ul>\n\n`;
    });
    
    return text;
  }
  
  /**
   * Форматирует данные в виде таблиц
   * @param {Object} data - Сгруппированные данные
   * @return {string} - HTML-текст с таблицами
   */
  function formatAsTables(data) {
    let text = `<h3>${data.title}</h3>\n\n`;
    
    data.groups.forEach(group => {
      text += `<h4>${group.name}</h4>\n\n`;
      text += `<table class="text-table">\n`;
      
      // Заголовок таблицы
      text += `<thead>\n<tr>`;
      text += `<th>Философ</th>`;
      text += `<th>Годы жизни</th>`;
      
      // Динамические колонки в зависимости от доступных данных
      if (group.philosophers[0] && group.philosophers[0].groupName && !group.name.includes('школам')) {
        text += `<th>Школа</th>`;
      }
      
      if (group.philosophers[0] && group.philosophers[0].connections !== undefined) {
        text += `<th>Всего связей</th>`;
      }
      
      if (group.philosophers[0] && group.philosophers[0].influences !== undefined) {
        text += `<th>Исходящее влияние</th>`;
      }
      
      if (group.philosophers[0] && group.philosophers[0].influencedBy !== undefined) {
        text += `<th>Входящее влияние</th>`;
      }
      
      if (group.philosophers[0] && group.philosophers[0].ratio !== undefined) {
        text += `<th>Баланс влияния</th>`;
      }
      
      text += `</tr>\n</thead>\n`;
      
      // Тело таблицы
      text += `<tbody>\n`;
      
      group.philosophers.forEach(philosopher => {
        text += `<tr>`;
        text += `<td><strong>${philosopher.name}</strong></td>`;
        text += `<td>${philosopher.years || '-'}</td>`;
        
        // Динамические колонки
        if (group.philosophers[0] && group.philosophers[0].groupName && !group.name.includes('школам')) {
          text += `<td>${philosopher.groupName || '-'}</td>`;
        }
        
        if (group.philosophers[0] && group.philosophers[0].connections !== undefined) {
          text += `<td>${philosopher.connections}</td>`;
        }
        
        if (group.philosophers[0] && group.philosophers[0].influences !== undefined) {
          text += `<td>${philosopher.influences}</td>`;
        }
        
        if (group.philosophers[0] && group.philosophers[0].influencedBy !== undefined) {
          text += `<td>${philosopher.influencedBy}</td>`;
        }
        
        if (group.philosophers[0] && group.philosophers[0].ratio !== undefined) {
          text += `<td>${
            isFinite(philosopher.ratio) ? philosopher.ratio.toFixed(2) : '∞'
          }</td>`;
        }
        
        text += `</tr>\n`;
      });
      
      text += `</tbody>\n</table>\n\n`;
    });
    
    return text;
  }
  
  /**
   * Форматирует данные в виде иерархии
   * @param {Object} data - Сгруппированные данные
   * @return {string} - HTML-текст с иерархическим представлением
   */
  function formatAsHierarchy(data) {
    let text = `<h3>${data.title}</h3>\n\n`;
    
    data.groups.forEach(group => {
      text += `<details open>\n`;
      text += `<summary><strong>${group.name}</strong> (${group.philosophers.length} философов)</summary>\n\n`;
      
      // Подгруппы (создаем их по определенному критерию в зависимости от типа группировки)
      const subgroups = createSubgroups(group);
      
      subgroups.forEach(subgroup => {
        text += `<details${subgroup.open ? ' open' : ''}>\n`;
        text += `<summary>${subgroup.name} (${subgroup.philosophers.length})</summary>\n\n`;
        
        text += `<ul>\n`;
        subgroup.philosophers.forEach(philosopher => {
          text += `<li class="list-item">`;
          text += `<strong>${philosopher.name}</strong> ${philosopher.years ? `(${philosopher.years})` : ''}`;
          
          // Дополнительная информация
          if (philosopher.description) {
            text += `<br>${philosopher.description}`;
          } else if (philosopher.connections !== undefined) {
            text += ` - ${philosopher.connections} связей`;
            
            if (philosopher.influences !== undefined && philosopher.influencedBy !== undefined) {
              text += ` (влияние на других: ${philosopher.influences}, влияние от других: ${philosopher.influencedBy})`;
            }
          }
          
          text += `</li>\n`;
        });
        text += `</ul>\n\n`;
        
        text += `</details>\n\n`;
      });
      
      text += `</details>\n\n`;
    });
    
    return text;
    
    // Вспомогательная функция для создания подгрупп
    function createSubgroups(group) {
      // Проверяем тип группировки по первой группе
      if (data.title.includes('школам')) {
        // Группируем по хронологии
        const timeGroups = [
          { id: "early", name: "Ранний период", philosophers: [], open: true },
          { id: "middle", name: "Средний период", philosophers: [], open: true },
          { id: "late", name: "Поздний период", philosophers: [], open: true }
        ];
        
        // Сортируем философов по годам
        const sortedPhilosophers = [...group.philosophers].sort((a, b) => {
          const yearsA = extractYearsFromData({ years: a.years });
          const yearsB = extractYearsFromData({ years: b.years });
          
          if (yearsA.birthYear !== null && yearsB.birthYear !== null) {
            return yearsA.birthYear - yearsB.birthYear;
          }
          
          return 0;
        });
        
        // Распределяем на три равные части
        const chunkSize = Math.ceil(sortedPhilosophers.length / 3);
        
        timeGroups[0].philosophers = sortedPhilosophers.slice(0, chunkSize);
        timeGroups[1].philosophers = sortedPhilosophers.slice(chunkSize, 2 * chunkSize);
        timeGroups[2].philosophers = sortedPhilosophers.slice(2 * chunkSize);
        
        return timeGroups.filter(g => g.philosophers.length > 0);
      } else if (data.title.includes('периодам')) {
        // Группируем по школам
        const schoolGroups = {};
        
        group.philosophers.forEach(philosopher => {
          if (!schoolGroups[philosopher.group]) {
            schoolGroups[philosopher.group] = {
              id: philosopher.group,
              name: philosopher.groupName,
              philosophers: []
            };
          }
          
          schoolGroups[philosopher.group].philosophers.push(philosopher);
        });
        
        // Сортируем подгруппы по количеству философов
        return Object.values(schoolGroups)
          .sort((a, b) => b.philosophers.length - a.philosophers.length)
          .map(g => ({ ...g, open: g.philosophers.length <= 5 }));
      } else {
        // Группируем по интенсивности влияния
        const influenceGroups = [
          { id: "high", name: "Высокая интенсивность влияния", philosophers: [], open: true },
          { id: "medium", name: "Средняя интенсивность влияния", philosophers: [], open: true },
          { id: "low", name: "Низкая интенсивность влияния", philosophers: [], open: false }
        ];
        
        // Для группировки используем общее количество связей
        group.philosophers.forEach(philosopher => {
          const connectionCount = philosopher.connections || 
                                  philosopher.total || 
                                  (philosopher.influences + philosopher.influencedBy) || 0;
          
          if (connectionCount >= 7) {
            influenceGroups[0].philosophers.push(philosopher);
          } else if (connectionCount >= 3) {
            influenceGroups[1].philosophers.push(philosopher);
          } else {
            influenceGroups[2].philosophers.push(philosopher);
          }
        });
        
        return influenceGroups.filter(g => g.philosophers.length > 0);
      }
    }
  }
  
  /**
   * Генерирует матрицу смежности для сети философов
   * @param {string} size - Размер матрицы (10, 20, full)
   * @param {string} linkType - Тип связи для анализа
   */
  function generateAdjacencyMatrix(size, linkType) {
    const contentElement = document.getElementById('adjacency-content');
    contentElement.innerHTML = ''; // Очищаем содержимое
    contentElement.className = 'text-view-content adjacency-matrix';
    
    // Определяем количество узлов для матрицы
    const nodeCount = size === 'full' ? data.nodes.length : parseInt(size, 10);
    
    // Выбираем наиболее связанные узлы
    const selectedNodes = data.nodes
      .map(node => {
        const connections = data.links.filter(link => {
          const sourceId = typeof link.source === 'object' ? link.source.id : link.source;
          const targetId = typeof link.target === 'object' ? link.target.id : link.target;
          
          const isCorrectType = linkType === 'all' || link.type === linkType;
          
          return isCorrectType && (sourceId === node.id || targetId === node.id);
        });
        
        return {
          node,
          connections: connections.length
        };
      })
      .sort((a, b) => b.connections - a.connections)
      .slice(0, nodeCount)
      .map(item => item.node);
    
    // Создаем матрицу смежности
    const matrix = [];
    const nodeIds = selectedNodes.map(node => node.id);
    
    for (let i = 0; i < selectedNodes.length; i++) {
      const row = [];
      
      for (let j = 0; j < selectedNodes.length; j++) {
        if (i === j) {
          // Диагональные элементы - ноль
          row.push(0);
        } else {
          // Ищем связь между узлами i и j
          const connection = data.links.find(link => {
            const sourceId = typeof link.source === 'object' ? link.source.id : link.source;
            const targetId = typeof link.target === 'object' ? link.target.id : link.target;
            
            const isCorrectType = linkType === 'all' || link.type === linkType;
            
            return isCorrectType && (
              (sourceId === nodeIds[i] && targetId === nodeIds[j]) ||
              (link.type === 'mutual' && sourceId === nodeIds[j] && targetId === nodeIds[i])
            );
          });
          
          row.push(connection ? 1 : 0);
        }
      }
      
      matrix.push(row);
    }
    
    // Формируем текстовое представление матрицы
    let text = `Матрица смежности для ${selectedNodes.length} философов\n`;
    text += `Тип связи: ${linkType === 'all' ? 'все типы' : linkType}\n\n`;
    
    // Заголовок таблицы с номерами
    text += '    ';
    for (let i = 0; i < selectedNodes.length; i++) {
      text += ` ${(i + 1).toString().padStart(2, ' ')}`;
    }
    text += '\n';
    
    // Горизонтальная линия
    text += '    ' + '---'.repeat(selectedNodes.length) + '\n';
    
    // Содержимое матрицы
    for (let i = 0; i < matrix.length; i++) {
      text += `${(i + 1).toString().padStart(2, ' ')} |`;
      
      for (let j = 0; j < matrix[i].length; j++) {
        text += ` ${matrix[i][j]} `;
      }
      
      text += '\n';
    }
    
    // Список философов с номерами
    text += '\nСписок философов:\n';
    for (let i = 0; i < selectedNodes.length; i++) {
      text += `${i + 1}. ${selectedNodes[i].name} ${selectedNodes[i].years ? `(${selectedNodes[i].years})` : ''}\n`;
    }
    
    // Отображаем текст
    contentElement.textContent = text;
  }
  
  /**
   * Генерирует описание путей влияния между философами
   * @param {string} sourceId - ID философа-источника
   * @param {string} targetId - ID философа-цели
   * @param {number} maxLength - Максимальная длина пути
   */
  function generatePathsDescription(sourceId, targetId, maxLength) {
    const contentElement = document.getElementById('paths-content');
    contentElement.innerHTML = ''; // Очищаем содержимое
    
    // Получаем имена философов
    const sourceName = getPhilosopherName(sourceId);
    const targetName = getPhilosopherName(targetId);
    
    // Находим все пути между философами
    const paths = findAllPaths(sourceId, targetId, maxLength);
    
    let text = `<h3>Пути влияния от ${sourceName} к ${targetName}</h3>\n\n`;
    
    if (paths.length === 0) {
      text += `<p>Не найдено путей влияния между этими философами (с максимальной длиной ${maxLength}).</p>\n\n`;

      text += `<p>Возможные причины:</p>\n<ul>\n`;

      text += `<li>Путь существует, но его длина превышает указанный максимум (${maxLength})</li>\n`;
      text += `<li>В данных нет достаточной информации о связях</li>\n`;
      text += `</ul>\n\n`;
      
      // Предлагаем связанных философов
      const sourceConnections = findRelatedPhilosophers(sourceId);
      const targetConnections = findRelatedPhilosophers(targetId);
      
      if (sourceConnections.length > 0) {
        text += `<p><strong>Философы, связанные с ${sourceName}:</strong></p>\n<ul>\n`;
        sourceConnections.slice(0, 5).forEach(connection => {
          text += `<li>${connection.name} (${connection.type})</li>\n`;
        });
        text += `</ul>\n\n`;
      }
      
      if (targetConnections.length > 0) {
        text += `<p><strong>Философы, связанные с ${targetName}:</strong></p>\n<ul>\n`;
        targetConnections.slice(0, 5).forEach(connection => {
          text += `<li>${connection.name} (${connection.type})</li>\n`;
        });
        text += `</ul>\n\n`;
      }
    } else {
      // Сортируем пути по длине (сначала короткие)
      paths.sort((a, b) => a.length - b.length);
      
      text += `<p>Найдено ${paths.length} путей влияния.</p>\n\n`;
      
      // Выводим пути
      paths.forEach((path, index) => {
        text += `<h4>Путь ${index + 1} (длина: ${path.length - 1})</h4>\n\n`;
        
        text += `<p class="highlight-text">`;
        path.forEach((philosopherId, i) => {
          text += `<strong>${getPhilosopherName(philosopherId)}</strong>`;
          
          if (i < path.length - 1) {
            text += ` → `;
          }
        });
        text += `</p>\n\n`;
        
        // Детальное описание для каждого шага
        if (path.length > 2) {
          text += `<p>Детали по шагам:</p>\n<ol>\n`;
          
          for (let i = 0; i < path.length - 1; i++) {
            const fromId = path[i];
            const toId = path[i + 1];
            
            const fromName = getPhilosopherName(fromId);
            const toName = getPhilosopherName(toId);
            
            // Находим тип связи

              const targetId = typeof link.target === 'object' ? link.target.id : link.target;
              
              return (sourceId === fromId && targetId === toId) || 
                     (link.type === 'mutual' && sourceId === toId && targetId === fromId);
            });
            
            let linkType = link ? link.type : 'unknown';
            let linkDescription = getRelationDescription(linkType);
            
            text += `<li><strong>${fromName}</strong> → <strong>${toName}</strong>: ${linkDescription}</li>\n`;
          }

        }
        
        // Информация о хронологии
        const years = path.map(id => {
          const philosopher = data.nodes.find(n => n.id === id);
          return philosopher ? extractYearsFromData(philosopher) : { birthYear: null, deathYear: null };

        if (years.length >= 2) {
          const firstBirth = years[0].birthYear;

  }
  
  /**
   * Находит все пути между двумя философами
   * @param {string} startId - ID начального философа
   * @param {string} endId - ID конечного философа
   * @param {number} maxLength - Максимальная длина пути
   * @return {Array} - Массив путей (каждый путь - массив ID философов)
   */
  function findAllPaths(startId, endId, maxLength) {
    const paths = [];
    const visited = new Set();
    
    // Функция для поиска в глубину
    function dfs(currentId, path) {
      // Если путь достиг максимальной длины, прерываем поиск
      if (path.length > maxLength) {
        return;
      }
      
      // Если достигли цели, добавляем путь
      if (currentId === endId) {
        paths.push([...path]);
        return;
      }
      
      // Отмечаем узел как посещенный
      visited.add(currentId);
      
      // Находим всех соседей
      const neighbors = [];
      data.links.forEach(link => {
        const sourceId = typeof link.source === 'object' ? link.source.id : link.source;
        const targetId = typeof link.target === 'object' ? link.target.id : link.target;
        
        // Проверяем типы связей (только влияние и взаимное влияние)
        const isInfluenceType = link.type === 'influence' || link.type === 'mutual';
        
        if (sourceId === currentId && !visited.has(targetId) && isInfluenceType) {
          neighbors.push(targetId);
        } else if (link.type === 'mutual' && targetId === currentId && !visited.has(sourceId)) {
          neighbors.push(sourceId);
        }
      });
      
      // Рекурсивно обходим всех соседей
      for (const neighbor of neighbors) {
        path.push(neighbor);
        dfs(neighbor, path);
        path.pop();
      }
      
      // Снимаем отметку при возврате (для поиска других путей)
      visited.delete(currentId);
    }
    
    // Запускаем поиск
    dfs(startId, [startId]);
    
    return paths;
  }
  
  /**
   * Находит философов, связанных с указанным
   * @param {string} philosopherId - ID философа
   * @return {Array} - Массив объектов со связанными философами
   */
  function findRelatedPhilosophers(philosopherId) {
    const related = [];
    
    data.links.forEach(link => {
      const sourceId = typeof link.source === 'object' ? link.source.id : link.source;
      const targetId = typeof link.target === 'object' ? link.target.id : link.target;
      
      let connectedId, relationDirection;
      
      if (sourceId === philosopherId) {
        connectedId = targetId;
        relationDirection = 'from';
      } else if (targetId === philosopherId) {
        connectedId = sourceId;
        relationDirection = 'to';
      } else {
        return;
      }
      
      const connectedPhilosopher = data.nodes.find(n => n.id === connectedId);
      if (!connectedPhilosopher) return;
      
      const relationDescription = getRelationDescription(link.type, relationDirection);
      
      related.push({
        id: connectedId,
        name: connectedPhilosopher.name,
        type: relationDescription
      });
    });
    
    return related;
  }
  
  /**
   * Получает описание типа связи
   * @param {string} linkType - Тип связи
   * @param {string} direction - Направление связи ('from', 'to', undefined)
   * @return {string} - Описание связи
   */
  function getRelationDescription(linkType, direction) {
    if (linkType === 'influence') {
      if (direction === 'from') {
        return 'повлиял на';
      } else if (direction === 'to') {
        return 'испытал влияние от';
      } else {
        return 'влияние';
      }
    } else if (linkType === 'mutual') {
      return 'взаимное влияние';
    } else if (linkType === 'criticism') {
      if (direction === 'from') {
        return 'критиковал';
      } else if (direction === 'to') {
        return 'подвергался критике от';
      } else {
        return 'критика';
      }
    } else if (linkType === 'polemic') {
      return 'полемика';
    } else if (linkType === 'acquaintance') {
      return 'знакомство';
    } else if (linkType === 'false') {
      return 'ложно приписываемая связь';
    } else if (linkType === 'indirect') {
      return 'опосредованное влияние';
    } else {
      return 'связь неизвестного типа';
    }
  }
  
  // Показывает панель текстовых представлений
  function showPanel() {
    panel.style.display = 'block';
    backdrop.style.display = 'block';
  }
  
  // Возвращаем контроллер
  return {
    show: showPanel,
    hide: closePanel
  };
}
```

### Итоговая интеграция

Чтобы обеспечить ошибкоустойчивость при работе с разными наборами данных, добавьте следующие функции-обертки для извлечения и проверки данных:

```javascript
/**
 * Безопасно извлекает информацию из базы данных философов
 * @param {string} philosopherId - ID философа
 * @param {string} field - Поле данных (bio, works, ideas, influence)
 * @return {any} - Значение поля или null
 */
function getPhilosopherData(philosopherId, field) {
  if (!window.philosophersDetails) return null;
  if (!window.philosophersDetails[philosopherId]) return null;
  
  return window.philosophersDetails[philosopherId][field] || null;
}

/**
 * Безопасно извлекает цитату для связи
 * @param {string} sourceId - ID философа-источника
 * @param {string} targetId - ID философа-цели
 * @param {string} linkType - Тип связи
 * @return {Object|null} - Объект с цитатой или null
 */
function getConnectionQuote(sourceId, targetId, linkType) {
  if (!window.philosopherQuotes) return null;
  
  const directKey = `${sourceId}-${targetId}-${linkType}`;
  const reverseKey = `${targetId}-${sourceId}-${linkType}`;
  
  return window.philosopherQuotes[directKey] || window.philosopherQuotes[reverseKey] || null;
}
```

### Система проверки данных

Чтобы обеспечить работу со всеми наборами данных, добавьте функцию проверки наличия необходимых данных:

```javascript
/**
 * Проверяет наличие необходимых данных и адаптирует интерфейс
 * @param {Object} data - Объект с узлами и связями
 */
function adaptInterfaceToAvailableData(data) {
  // Проверяем наличие годов жизни
  const hasYearsData = data.nodes.some(node => {
    const years = extractYearsFromData(node);
    return years.birthYear !== null && years.deathYear !== null;
  });
  
  // Проверяем наличие групп/школ
  const hasGroupData = data.nodes.some(node => node.group);
  
  // Проверяем наличие типов связей
  const hasLinkTypes = data.links.some(link => link.type);
  
  // Проверяем наличие расширенной информации
  const hasExtendedInfo = window.philosophersDetails !== undefined && Object.keys(window.philosophersDetails).length > 0;
  
  // Проверяем наличие цитат
  const hasQuotes = window.philosopherQuotes !== undefined && Object.keys(window.philosopherQuotes).length > 0;
  
  // Адаптируем временную шкалу
  const timelineBtn = document.getElementById("animated-timeline-btn");
  if (timelineBtn) {
    timelineBtn.disabled = !hasYearsData;
    timelineBtn.title = hasYearsData ? "" : "Недостаточно данных о годах жизни";
  }
  
  // Адаптируем тепловые карты
  const heatmapBtn = document.getElementById("heatmap-btn");
  if (heatmapBtn) {
    heatmapBtn.disabled = !hasGroupData && !hasYearsData;
    heatmapBtn.title = hasGroupData || hasYearsData ? "" : "Недостаточно данных о группах или годах жизни";
  }
  
  // Добавляем информацию о доступных данных в панель информации
  const infoPanel = document.querySelector(".info-panel");
  if (infoPanel) {
    const dataAvailability = document.createElement("div");
    dataAvailability.className = "data-availability";
    dataAvailability.innerHTML = `
      <h4>Доступность данных:</h4>
      <ul>
        <li>Временная информация: ${hasYearsData ? '✓' : '✗'}</li>
        <li>Информация о школах: ${hasGroupData ? '✓' : '✗'}</li>
        <li>Типы связей: ${hasLinkTypes ? '✓' : '✗'}</li>
        <li>Расширенная информация: ${hasExtendedInfo ? '✓' : '✗'}</li>
        <li>Цитаты: ${hasQuotes ? '✓' : '✗'}</li>
      </ul>
    `;
    
    infoPanel.appendChild(dataAvailability);
  }
}
```
