  /**
   * Отображает анализ групп/школ философов
   */
  function displayGroupAnalysis() {
    const container = document.getElementById('group-analysis');
    
    // Группируем философов по школам
    const groups = {};
    data.nodes.forEach(node => {
      const group = node.group;
      const groupName = node.groupName || getGroupDisplayName(group);
      
      if (!groups[group]) {
        groups[group] = {
          id: group,
          name: groupName,
          nodes: [],
          internalLinks: 0,
          externalLinks: 0
        };
      }
      
      groups[group].nodes.push(node);
    });
    
    // Подсчитываем связи внутри и между группами
    data.links.forEach(link => {
      const sourceGroup = typeof link.source === 'object' ? link.source.group : data.nodes.find(n => n.id === link.source)?.group;
      const targetGroup = typeof link.target === 'object' ? link.target.group : data.nodes.find(n => n.id === link.target)?.group;
      
      if (sourceGroup === targetGroup) {
        groups[sourceGroup].internalLinks++;
      } else {
        if (sourceGroup && groups[sourceGroup]) groups[sourceGroup].externalLinks++;
        if (targetGroup && groups[targetGroup]) groups[targetGroup].externalLinks++;
      }
    });
    
    // Рассчитываем коэффициент кластеризации для каждой группы
    Object.values(groups).forEach(group => {
      const n = group.nodes.length;
      // Максимальное количество связей внутри группы
      const maxLinks = n * (n - 1) / 2;
      
      // Плотность связей внутри группы
      group.density = maxLinks > 0 ? group.internalLinks / maxLinks : 0;
      
      // Соотношение внутренних и внешних связей
      group.inOutRatio = group.externalLinks > 0 ? group.internalLinks / group.externalLinks : Infinity;
    });
    
    // Сортируем группы по количеству узлов
    const sortedGroups = Object.values(groups).sort((a, b) => b.nodes.length - a.nodes.length);
    
    let html = `
      <table class="stats-table">
        <thead>
          <tr>
            <th>Школа/группа</th>
            <th>Философы</th>
            <th>Внутр. связи</th>
            <th>Внеш. связи</th>
            <th>Плотность</th>
            <th>Замкнутость</th>
          </tr>
        </thead>
        <tbody>
    `;
    
    sortedGroups.forEach(group => {
      html += `
        <tr>
          <td>${group.name}</td>
          <td>${group.nodes.length}</td>
          <td>${group.internalLinks}</td>
          <td>${group.externalLinks}</td>
          <td>${group.density.toFixed(2)}</td>
          <td>${group.inOutRatio === Infinity ? '∞' : group.inOutRatio.toFixed(2)}</td>
        </tr>
      `;
    });
    
    html += `
        </tbody>
      </table>
      
      <p style="margin-top: 15px; font-size: 0.9em;">
        <strong>Плотность</strong> - отношение существующих связей внутри группы к максимально возможному их количеству.<br>
        <strong>Замкнутость</strong> - отношение внутренних связей к внешним. Высокие значения указывают на изолированные школы.
      </p>
    `;
    
    container.innerHTML = html;
  }
  
  /**
   * Отображает анализ типов связей
   */
  function displayConnectionTypes() {
    const container = document.getElementById('connection-types');
    
    // Подсчитываем типы связей
    const connectionTypes = {};
    data.links.forEach(link => {
      const type = link.type || 'normal';
      if (!connectionTypes[type]) {
        connectionTypes[type] = 0;
      }
      connectionTypes[type]++;
    });
    
    // Формируем данные для диаграммы
    const connectionData = Object.entries(connectionTypes).map(([type, count]) => {
      let typeName = type;
      
      // Получаем понятное название типа связи, если доступно
      if (networkRelationTypes) {
        const relationType = networkRelationTypes.find(rt => rt.id === type);
        if (relationType) {
          typeName = relationType.name;
        }
      }
      
      return {
        type,
        typeName,
        count,
        percentage: (count / data.links.length * 100).toFixed(1)
      };
    }).sort((a, b) => b.count - a.count);
    
    let html = `
      <div class="stats-graph" id="connection-types-chart"></div>
      
      <table class="stats-table">
        <thead>
          <tr>
            <th>Тип связи</th>
            <th>Количество</th>
            <th>Процент</th>
          </tr>
        </thead>
        <tbody>
    `;
    
    connectionData.forEach(item => {
      html += `
        <tr>
          <td>${item.typeName}</td>
          <td>${item.count}</td>
          <td>${item.percentage}%</td>
        </tr>
      `;
    });
    
    html += `
        </tbody>
      </table>
    `;
    
    container.innerHTML = html;
    
    // Создаем диаграмму распределения типов связей
    setTimeout(() => {
      createConnectionTypesChart(connectionData);
    }, 100);
  }

      .text(d => d.count);
  }
  
  /**
   * Рассчитывает плотность сети
   * @param {number} nodeCount - Количество узлов
   * @param {number} linkCount - Количество связей
   * @return {number} - Плотность сети
   */
  function calculateNetworkDensity(nodeCount, linkCount) {
    // Максимальное количество связей в ориентированном графе
    const maxLinks = nodeCount * (nodeCount - 1);
    
    // Плотность - отношение существующих связей к максимально возможному
    return maxLinks > 0 ? linkCount / maxLinks : 0;
  }
  
  /**
   * Находит компоненты связности в графе
   * @return {Object} - Объект с компонентами связности
   */
  function findConnectedComponents() {
    const visited = new Set();
    const components = [];
    
    // Функция обхода в глубину
    function dfs(nodeId, component) {
      visited.add(nodeId);
      component.push(nodeId);
      
      // Находим всех соседей
      const neighbors = [];
      data.links.forEach(link => {
        if (typeof link.source === 'object') {
          if (link.source.id === nodeId && !visited.has(link.target.id)) {
            neighbors.push(link.target.id);
          } else if (link.target.id === nodeId && !visited.has(link.source.id)) {
            neighbors.push(link.source.id);
          }
        } else {
          if (link.source === nodeId && !visited.has(link.target)) {
            neighbors.push(link.target);
          } else if (link.target === nodeId && !visited.has(link.source)) {
            neighbors.push(link.source);
          }
        }
      });
      
      // Рекурсивно обходим соседей
      neighbors.forEach(neighbor => {
        if (!visited.has(neighbor)) {
          dfs(neighbor, component);
        }
      });
    }
    
    // Обходим все узлы
    data.nodes.forEach(node => {
      if (!visited.has(node.id)) {
        const component = [];
        dfs(node.id, component);
        components.push(component);
      }
    });
    
    // Находим наибольшую компоненту
    let largestComponent = components[0] || [];
    for (let i = 1; i < components.length; i++) {
      if (components[i].length > largestComponent.length) {
        largestComponent = components[i];
      }
    }
    
    return { components, largestComponent };
  }
  
  /**
   * Находит мосты в графе (связи, удаление которых увеличивает число компонент)
   * @return {Array} - Массив мостов (пары узлов)
   */
  function findBridges() {
    // Упрощенная реализация для примера
    // Для полной реализации нужно использовать алгоритм поиска мостов
    // на основе поиска в глубину и времени посещения узлов
    
    const bridges = [];
    
    // Проверяем каждую связь
    data.links.forEach(link => {
      // Сохраняем связь
      const sourceId = typeof link.source === 'object' ? link.source.id : link.source;
      const targetId = typeof link.target === 'object' ? link.target.id : link.target;
      
      // Временно удаляем связь
      const tempLinks = data.links.filter(l => {
        const lSourceId = typeof l.source === 'object' ? l.source.id : l.source;
        const lTargetId = typeof l.target === 'object' ? l.target.id : l.target;
        
        return !(lSourceId === sourceId && lTargetId === targetId) && 
               !(lSourceId === targetId && lTargetId === sourceId);
      });
      
      // Проверяем, увеличилось ли число компонент
      const tempData = { nodes: data.nodes, links: tempLinks };
      const { components: originalComponents } = findConnectedComponents();
      const { components: newComponents } = findConnectedComponentsForData(tempData);
      
      if (newComponents.length > originalComponents.length) {
        bridges.push([sourceId, targetId]);
      }
    });
    
    return bridges;
  }
  
  /**
   * Находит компоненты связности в заданном графе
   * @param {Object} graphData - Объект с узлами и связями
   * @return {Object} - Объект с компонентами связности
   */
  function findConnectedComponentsForData(graphData) {
    const visited = new Set();
    const components = [];
    
    // Функция обхода в глубину
    function dfs(nodeId, component) {
      visited.add(nodeId);
      component.push(nodeId);
      
      // Находим всех соседей
      const neighbors = [];
      graphData.links.forEach(link => {
        if (typeof link.source === 'object') {
          if (link.source.id === nodeId && !visited.has(link.target.id)) {
            neighbors.push(link.target.id);
          } else if (link.target.id === nodeId && !visited.has(link.source.id)) {
            neighbors.push(link.source.id);
          }
        } else {
          if (link.source === nodeId && !visited.has(link.target)) {
            neighbors.push(link.target);
          } else if (link.target === nodeId && !visited.has(link.source)) {
            neighbors.push(link.source);
          }
        }
      });
      
      // Рекурсивно обходим соседей
      neighbors.forEach(neighbor => {
        if (!visited.has(neighbor)) {
          dfs(neighbor, component);
        }
      });
    }
    
    // Обходим все узлы
    graphData.nodes.forEach(node => {
      if (!visited.has(node.id)) {
        const component = [];
        dfs(node.id, component);
        components.push(component);
      }
    });
    
    // Находим наибольшую компоненту
    let largestComponent = components[0] || [];
    for (let i = 1; i < components.length; i++) {
      if (components[i].length > largestComponent.length) {
        largestComponent = components[i];
      }
    }
    
    return { components, largestComponent };
  }
  
  /**
   * Находит точки сочленения в графе (узлы, удаление которых увеличивает число компонент)
   * @return {Array} - Массив ID узлов-точек сочленения
   */
  function findArticulationPoints() {
    // Упрощенная реализация для примера
    // Для полной реализации нужно использовать алгоритм поиска точек сочленения
    
    const articulationPoints = [];
    
    // Проверяем каждый узел
    data.nodes.forEach(node => {
      // Сохраняем ID узла
      const nodeId = node.id;
      
      // Временно удаляем узел и все его связи
      const tempNodes = data.nodes.filter(n => n.id !== nodeId);
      const tempLinks = data.links.filter(l => {
        const sourceId = typeof l.source === 'object' ? l.source.id : l.source;
        const targetId = typeof l.target === 'object' ? l.target.id : l.target;
        
        return sourceId !== nodeId && targetId !== nodeId;
      });
      
      // Проверяем, увеличилось ли число компонент
      const tempData = { nodes: tempNodes, links: tempLinks };
      const { components: originalComponents } = findConnectedComponents();
      const { components: newComponents } = findConnectedComponentsForData(tempData);
      
      if (newComponents.length > originalComponents.length) {
        articulationPoints.push(nodeId);
      }
    });
    
    return articulationPoints;
  }
  
  /**
   * Рассчитывает метрики центральности для узлов сети
   * @return {Array} - Массив объектов с метриками для каждого узла
   */
  function calculateCentralityMetrics() {
    const metrics = [];
    
    // Подготовка данных (инициализация метрик для каждого узла)
    data.nodes.forEach(node => {
      metrics.push({
        id: node.id,
        name: node.name,
        group: node.group,
        groupName: node.groupName || getGroupDisplayName(node.group),
        degree: 0,
        betweenness: 0,
        closeness: 0
      });
    });
    
    // Рассчитываем степень центральности (количество связей)
    data.links.forEach(link => {
      const sourceId = typeof link.source === 'object' ? link.source.id : link.source;
      const targetId = typeof link.target === 'object' ? link.target.id : link.target;
      
      const sourceMetric = metrics.find(m => m.id === sourceId);
      const targetMetric = metrics.find(m => m.id === targetId);
      
      if (sourceMetric) sourceMetric.degree++;
      if (targetMetric) targetMetric.degree++;
    });
    
    // Упрощенный расчет близости (closeness centrality)
    // Для каждого узла рассчитываем среднюю длину кратчайших путей до других узлов
    const { distances } = calculateAllPairsShortestPaths();
    
    // Для каждого узла
    metrics.forEach(metric => {
      const nodeDistances = distances[metric.id];
      if (!nodeDistances) return;
      
      // Сумма всех расстояний от этого узла до других
      let sum = 0;
      let count = 0;
      
      Object.values(nodeDistances).forEach(distance => {
        if (distance !== Infinity) {
          sum += distance;
          count++;
        }
      });
      
      // Близость равна 1 / средняя длина пути
      metric.closeness = count > 0 ? (count - 1) / sum : 0;
    });
    
    // Упрощенный расчет центральности по посредничеству (betweenness centrality)
    // Для полного расчета нужно использовать алгоритм Брандеса
    // Мы используем упрощенный подход для демонстрации
    
    // Создаем счетчик для каждого узла (сколько раз он лежит на кратчайшем пути)
    const betweennessCount = {};
    data.nodes.forEach(node => {
      betweennessCount[node.id] = 0;
    });
    
    // Проходим по всем парам узлов и считаем кратчайшие пути
    for (let i = 0; i < data.nodes.length; i++) {
      const sourceId = data.nodes[i].id;
      
      for (let j = i + 1; j < data.nodes.length; j++) {
        const targetId = data.nodes[j].id;
        
        // Находим кратчайший путь
        const path = findShortestPath(sourceId, targetId);
        
        // Увеличиваем счетчик для всех промежуточных узлов на пути
        if (path.length > 2) {
          for (let k = 1; k < path.length - 1; k++) {
            betweennessCount[path[k]]++;
          }
        }
      }
    }
    
    // Нормализуем значения центральности по посредничеству
    const maxPossiblePaths = (data.nodes.length - 1) * (data.nodes.length - 2) / 2;
    
    metrics.forEach(metric => {
      metric.betweenness = maxPossiblePaths > 0 ? 
        betweennessCount[metric.id] / maxPossiblePaths : 0;
    });
    
    return metrics;
  }
  
  /**
   * Рассчитывает кратчайшие пути между всеми парами узлов
   * @return {Object} - Объект с расстояниями и предшественниками
   */
  function calculateAllPairsShortestPaths() {
    const distances = {};
    const predecessors = {};
    
    // Инициализация
    data.nodes.forEach(node => {
      distances[node.id] = {};
      predecessors[node.id] = {};
      
      data.nodes.forEach(target => {
        if (node.id === target.id) {
          distances[node.id][target.id] = 0;
        } else {
          distances[node.id][target.id] = Infinity;
        }
        predecessors[node.id][target.id] = null;
      });
    });
    
    // Заполняем расстояния для напрямую связанных узлов
    data.links.forEach(link => {
      const sourceId = typeof link.source === 'object' ? link.source.id : link.source;
      const targetId = typeof link.target === 'object' ? link.target.id : link.target;
      
      distances[sourceId][targetId] = 1;
      predecessors[sourceId][targetId] = sourceId;
      
      // Для неориентированного графа
      distances[targetId][sourceId] = 1;
      predecessors[targetId][sourceId] = targetId;
    });
    
    // Алгоритм Флойда-Уоршелла
    data.nodes.forEach(k => {
      data.nodes.forEach(i => {
        data.nodes.forEach(j => {
          const ikDistance = distances[i.id][k.id];
          const kjDistance = distances[k.id][j.id];
          
          if (ikDistance !== Infinity && kjDistance !== Infinity) {
            const newDistance = ikDistance + kjDistance;
            
            if (newDistance < distances[i.id][j.id]) {
              distances[i.id][j.id] = newDistance;
              predecessors[i.id][j.id] = predecessors[k.id][j.id];
            }
          }
        });
      });
    });
    
    return { distances, predecessors };
  }
  
  /**
   * Находит кратчайший путь между двумя узлами
   * @param {string} startId - ID начального узла
   * @param {string} endId - ID конечного узла
   * @return {Array} - Массив ID узлов, составляющих кратчайший путь
   */
  function findShortestPath(startId, endId) {
    // Используем алгоритм поиска в ширину для нахождения кратчайшего пути
    const queue = [startId];
    const visited = new Set([startId]);
    const predecessors = {};
    let found = false;
    
    while (queue.length > 0 && !found) {
      const current = queue.shift();
      
      // Перебираем всех соседей текущего узла
      const neighbors = [];
      data.links.forEach(link => {
        if (typeof link.source === 'object') {
          if (link.source.id === current && !visited.has(link.target.id)) {
            neighbors.push(link.target.id);
          } else if (link.target.id === current && !visited.has(link.source.id)) {
            neighbors.push(link.source.id);
          }
        } else {
          if (link.source === current && !visited.has(link.target)) {
            neighbors.push(link.target);
          } else if (link.target === current && !visited.has(link.source)) {
            neighbors.push(link.source);
          }
        }
      });
      
      for (const neighbor of neighbors) {
        if (!visited.has(neighbor)) {
          visited.add(neighbor);
          predecessors[neighbor] = current;
          queue.push(neighbor);
          
          // Если нашли конечный узел, прерываем поиск
          if (neighbor === endId) {
            found = true;
            break;
          }
        }
      }
    }
    
    // Если путь не найден, возвращаем пустой массив
    if (!found) {
      return [];
    }
    
    // Восстанавливаем путь
    const path = [endId];
    let current = endId;
    
    while (current !== startId) {
      current = predecessors[current];
      path.unshift(current);
    }
    
    return path;
  }
  
  /**
   * Экспортирует статистику в файл
   */
  function exportStatistics() {
    // Собираем все данные статистики
    const stats = {
      overview: {
        nodes: data.nodes.length,
        links: data.links.length,
        density: calculateNetworkDensity(data.nodes.length, data.links.length),
        groups: new Set(data.nodes.map(n => n.group)).size
      },
      connectivity: {
        components: findConnectedComponents().components.length,
        largestComponent: findConnectedComponents().largestComponent.length,
        bridges: findBridges().length,
        articulationPoints: findArticulationPoints()
      },
      centrality: calculateCentralityMetrics(),
      groups: calculateGroupStats(),
      connectionTypes: calculateConnectionStats()
    };
    
    // Экспортируем в JSON
    const statsJson = JSON.stringify(stats, null, 2);
    const blob = new Blob([statsJson], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    
    const downloadLink = document.createElement('a');
    downloadLink.href = url;
    downloadLink.download = 'philosophy-network-stats.json';
    document.body.appendChild(downloadLink);
    downloadLink.click();
    
    // Очищаем
    setTimeout(() => {
      document.body.removeChild(downloadLink);
      URL.revokeObjectURL(url);
    }, 100);
    
    /**
     * Рассчитывает статистику по группам
     * @return {Array} - Массив статистик по группам
     */
    function calculateGroupStats() {
      // Группируем философов по школам
      const groups = {};
      data.nodes.forEach(node => {
        const group = node.group;
        const groupName = node.groupName || getGroupDisplayName(group);
        
        if (!groups[group]) {
          groups[group] = {
            id: group,
            name: groupName,
            nodes: [],
            internalLinks: 0,
            externalLinks: 0
          };
        }
        
        groups[group].nodes.push(node.id);
      });
      
      // Подсчитываем связи внутри и между группами
      data.links.forEach(link => {
        const sourceGroup = typeof link.source === 'object' ? link.source.group : data.nodes.find(n => n.id === link.source)?.group;
        const targetGroup = typeof link.target === 'object' ? link.target.group : data.nodes.find(n => n.id === link.target)?.group;
        
        if (sourceGroup === targetGroup) {
          groups[sourceGroup].internalLinks++;
        } else {
          if (sourceGroup && groups[sourceGroup]) groups[sourceGroup].externalLinks++;
          if (targetGroup && groups[targetGroup]) groups[targetGroup].externalLinks++;
        }
      });
      
      // Рассчитываем коэффициент кластеризации для каждой группы
      Object.values(groups).forEach(group => {
        const n = group.nodes.length;
        // Максимальное количество связей внутри группы
        const maxLinks = n * (n - 1) / 2;
        
        // Плотность связей внутри группы
        group.density = maxLinks > 0 ? group.internalLinks / maxLinks : 0;
        
        // Соотношение внутренних и внешних связей
        group.inOutRatio = group.externalLinks > 0 ? group.internalLinks / group.externalLinks : Infinity;
      });
      
      return Object.values(groups);
    }
    
    /**
     * Рассчитывает статистику по типам связей
     * @return {Array} - Массив статистик по типам связей
     */
    function calculateConnectionStats() {
      // Подсчитываем типы связей
      const connectionTypes = {};
      data.links.forEach(link => {
        const type = link.type || 'normal';
        if (!connectionTypes[type]) {
          connectionTypes[type] = {
            type,
            count: 0,
            typeName: type
          };
        }
        connectionTypes[type].count++;
      });
      
      // Добавляем имена типов связей, если доступны
      if (networkRelationTypes) {
        Object.values(connectionTypes).forEach(connection => {
          const relationType = networkRelationTypes.find(rt => rt.id === connection.type);
          if (relationType) {
            connection.typeName = relationType.name;
          }
        });
      }
      
      // Добавляем процентное соотношение
      Object.values(connectionTypes).forEach(connection => {
        connection.percentage = (connection.count / data.links.length * 100).toFixed(1);
      });
      
      return Object.values(connectionTypes);
    }
  }
  
  // Возвращаем контроллер
  return {
    show: showPanel
  };
}

### Шаг 2: Добавить функцию для анимированной временной линии

Добавьте следующую функцию в блок `<script>` перед функцией `initApplication()`:

```javascript
/**
 * Создает инструмент анимированной временной линии
 * @param {Object} data - Объект с узлами и связями
 * @param {string} containerId - ID контейнера визуализации
 * @return {Object} - Контроллер временной линии
 */
function createAnimatedTimeline(data, containerId) {
  // Извлекаем годы жизни философов
  const philosophers = data.nodes.map(node => {
    const years = extractYearsFromData(node);
    return {
      id: node.id,
      name: node.name,
      birthYear: years.birthYear,
      deathYear: years.deathYear,
      group: node.group
    };
  }).filter(p => p.birthYear !== null && p.deathYear !== null);
  
  // Находим временной диапазон
  const allYears = philosophers.flatMap(p => [p.birthYear, p.deathYear]);
  const minYear = Math.min(...allYears);
  const maxYear = Math.max(...allYears);
  
  // Создаем панель анимации
  const panel = document.createElement('div');
  panel.className = 'timeline-animation-panel';
  panel.innerHTML = `
    <button class="timeline-animation-close">&times;</button>
    
    <div class="timeline-animation-controls">
      <div class="timeline-year-display" id="timeline-year">1800</div>
      
      <div class="timeline-animation-slider">
        <input type="range" class="timeline-slider" id="timeline-slider" 
          min="${minYear}" max="${maxYear}" value="${minYear}" step="1">
        <div class="timeline-markers" id="timeline-markers"></div>
      </div>
      
      <div class="timeline-animation-buttons">
        <button class="timeline-animation-button" id="timeline-play-btn">▶ Играть</button>
        <button class="timeline-animation-button" id="timeline-stop-btn" disabled>■ Стоп</button>
      </div>
      
      <div class="timeline-animation-speed">
        <label for="timeline-speed">Скорость:</label>
        <select id="timeline-speed">
          <option value="10">Медленно</option>
          <option value="20" selected>Средне</option>
          <option value="40">Быстро</option>
        </select>
      </div>
    </div>
    
    <div class="timeline-info-panel" id="timeline-info">
      <div class="timeline-info-title">Активные философы: <span id="active-count">0</span></div>
      <ul class="timeline-info-list" id="active-philosophers"></ul>
    </div>
  `;
  
  document.body.appendChild(panel);
  
  // Получаем элементы управления
  const yearDisplay = document.getElementById('timeline-year');
  const timelineSlider = document.getElementById('timeline-slider');
  const playButton = document.getElementById('timeline-play-btn');
  const stopButton = document.getElementById('timeline-stop-btn');
  const speedSelect = document.getElementById('timeline-speed');
  const closeButton = panel.querySelector('.timeline-animation-close');
  const activeCountElement = document.getElementById('active-count');
  const activePhilosophersElement = document.getElementById('active-philosophers');
  
  // Создаем маркеры исторических эпох
  const markers = document.getElementById('timeline-markers');
  const historicalEras = [
    { start: -500, end: 476, name: "Античность" },
    { start: 476, end: 1453, name: "Средние века" },
    { start: 1453, end: 1789, name: "Ренессанс и Новое время" },
    { start: 1789, end: 1914, name: "Долгий XIX век" },
    { start: 1914, end: 1991, name: "XX век" },
    { start: 1991, end: maxYear, name: "Современность" }
  ];
  
  // Добавляем маркеры эпох
  historicalEras.forEach(era => {
    if (era.start > maxYear || era.end < minYear) return;
    
    const actualStart = Math.max(era.start, minYear);
    const actualEnd = Math.min(era.end, maxYear);
    
    // Рассчитываем позицию и ширину маркера
    const startPercent = (actualStart - minYear) / (maxYear - minYear) * 100;
    const endPercent = (actualEnd - minYear) / (maxYear - minYear) * 100;
    const width = endPercent - startPercent;
    
    // Создаем индикатор эпохи
    const indicator = document.createElement('div');
    indicator.className = 'timeline-era-indicator';
    indicator.style.left = `${startPercent}%`;
    indicator.style.width = `${width}%`;
    markers.appendChild(indicator);
    
    // Добавляем метку эпохи
    const label = document.createElement('div');
    label.className = 'timeline-era-label';
    label.textContent = era.name;
    label.style.left = `${startPercent + width / 2}%`;
    markers.appendChild(label);
  });
  
  // Добавляем индикатор текущего года
  const currentIndicator = document.createElement('div');
  currentIndicator.className = 'timeline-current-indicator';
  markers.appendChild(currentIndicator);
  
  // Состояние анимации
  let isPlaying = false;
  let animationInterval = null;
  let currentYear = minYear;
  
  // Обработчики событий
  timelineSlider.addEventListener('input', function() {
    currentYear = parseInt(this.value, 10);
    updateYearDisplay();
    updateVisualization();
  });
  
  playButton.addEventListener('click', function() {
    startAnimation();
  });
  
  stopButton.addEventListener('click', function() {
    stopAnimation();
  });
  
  closeButton.addEventListener('click', function() {
    hidePanel();
  });
  
  // Функции анимации
  function startAnimation() {
    if (isPlaying) return;
    
    isPlaying = true;
    playButton.disabled = true;
    stopButton.disabled = false;
    
    const speed = parseInt(speedSelect.value, 10);
    
    animationInterval = setInterval(() => {
      currentYear++;
      
      if (currentYear > maxYear) {
        stopAnimation();
        return;
      }
      
      timelineSlider.value = currentYear;
      updateYearDisplay();
      updateVisualization();
    }, 1000 / speed);
  }
  
  function stopAnimation() {
    if (!isPlaying) return;
    
    isPlaying = false;
    clearInterval(animationInterval);
    playButton.disabled = false;
    stopButton.disabled = true;
  }
  
  function updateYearDisplay() {
    yearDisplay.textContent = currentYear;
    
    // Обновляем позицию индикатора
    const percent = (currentYear - minYear) / (maxYear - minYear) * 100;
    currentIndicator.style.left = `${percent}%`;
  }
  
  function updateVisualization() {
    const year = currentYear;
    
    // Отображаем активных философов
    const activePhilosophers = philosophers.filter(p => 
      p.birthYear <= year && p.deathYear >= year
    );
    
    // Обновляем информационную панель
    activeCountElement.textContent = activePhilosophers.length;
    activePhilosophersElement.innerHTML = '';
    
    activePhilosophers
      .sort((a, b) => (a.birthYear - b.birthYear))
      .forEach(philosopher => {
        const li = document.createElement('li');
        li.textContent = `${philosopher.name} (${philosopher.birthYear}-${philosopher.deathYear})`;
        activePhilosophersElement.appendChild(li);
      });
    
    // Обновляем узлы на визуализации
    const activeIds = new Set(activePhilosophers.map(p => p.id));
    
    d3.selectAll(".node").each(function(d) {
      const years = extractYearsFromData(d);
      const isActive = years.birthYear <= year && years.deathYear >= year;
      
      d3.select(this)
        .classed("time-active", isActive)
        .classed("time-inactive", !isActive);
    });
    
    // Обновляем связи на визуализации
    d3.selectAll(".links line").each(function(d) {
      const sourceYears = extractYearsFromData(d.source);
      const targetYears = extractYearsFromData(d.target);
      
      const isSourceActive = sourceYears.birthYear <= year && sourceYears.deathYear >= year;
      const isTargetActive = targetYears.birthYear <= year && targetYears.deathYear >= year;
      
      const isActive = isSourceActive && isTargetActive;
      
      d3.select(this)
        .classed("time-active", isActive)
        .classed("time-inactive", !isActive);
    });
  }
  
  // Функции для открытия и закрытия панели
  function showPanel() {
    panel.style.display = 'block';
    
    // Инициализируем визуализацию с начальным годом
    currentYear = minYear;
    timelineSlider.value = currentYear;
    updateYearDisplay();
    updateVisualization();
  }
  
  function hidePanel() {
    stopAnimation();
    panel.style.display = 'none';
    
    // Восстанавливаем исходную визуализацию
    d3.selectAll(".node")
      .classed("time-active", false)
      .classed("time-inactive", false);
    
    d3.selectAll(".links line")
      .classed("time-active", false)
      .classed("time-inactive", false);
  }
  
  // Возвращаем контроллер
  return {
    show: showPanel,
    hide: hidePanel
  };
}
```
