### Шаг 2: Добавить функции для анализа и визуализации цепочек влияния

Добавьте следующую функцию в блок `<script>` перед функцией `initApplication()`:

```javascript
/**
 * Создает инструмент для анализа и визуализации цепочек влияния
 * @param {Object} data - Объект с узлами и связями
 * @param {string} containerId - ID контейнера для отображения
 * @return {Object} - Контроллер для цепочек влияния
 */
function createInfluenceChainTool(data, containerId) {
  // Получаем контейнер
  const container = document.getElementById(containerId);
  
  // Создаем панель для цепочек влияния
  const panel = document.createElement('div');
  panel.className = 'influence-chain-panel';
  panel.innerHTML = `
    <h3>Анализ цепочек влияния</h3>
    <div class="influence-form">
      <label for="source-philosopher">От философа:</label>
      <select id="source-philosopher"></select>
      
      <label for="target-philosopher">К философу:</label>
      <select id="target-philosopher"></select>
      
      <label for="max-chain-depth">Максимальная длина цепочки:</label>
      <select id="max-chain-depth">
        <option value="2">2</option>
        <option value="3" selected>3</option>
        <option value="4">4</option>
        <option value="5">5</option>
      </select>
      
      <div>
        <button id="find-chains-btn">Найти цепочки</button>
        <button id="close-chains-panel" style="margin-left: 10px;">Закрыть</button>
      </div>
    </div>
    
    <div class="chain-results" id="chain-results">
      <p>Выберите философов и нажмите "Найти цепочки"</p>
    </div>
  `;
  
  container.appendChild(panel);
  
  // Заполняем выпадающие списки философов
  const sourceSelect = document.getElementById('source-philosopher');
  const targetSelect = document.getElementById('target-philosopher');
  
  // Сортируем философов по имени
  const sortedPhilosophers = [...data.nodes].sort((a, b) => a.name.localeCompare(b.name));
  
  // Добавляем опции
  sortedPhilosophers.forEach(philosopher => {
    const sourceOption = document.createElement('option');
    sourceOption.value = philosopher.id;
    sourceOption.textContent = philosopher.name;
    sourceSelect.appendChild(sourceOption);
    
    const targetOption = document.createElement('option');
    targetOption.value = philosopher.id;
    targetOption.textContent = philosopher.name;
    targetSelect.appendChild(targetOption);
  });
  
  // Обработчик для закрытия панели
  document.getElementById('close-chains-panel').addEventListener('click', function() {
    panel.style.display = 'none';
    
    // Сбрасываем выделение путей
    resetChainHighlighting();
  });
  
  // Обработчик для поиска цепочек
  document.getElementById('find-chains-btn').addEventListener('click', function() {
    const sourceId = sourceSelect.value;
    const targetId = targetSelect.value;
    const maxDepth = parseInt(document.getElementById('max-chain-depth').value, 10);
    
    if (sourceId === targetId) {
      alert('Пожалуйста, выберите разных философов для начала и конца цепочки');
      return;
    }
    
    // Находим цепочки влияния
    const chains = findInfluenceChains(sourceId, targetId, maxDepth);
    
    // Отображаем результаты
    displayChains(chains, sourceId, targetId);
  });
  
  /**
   * Находит все возможные цепочки влияния между двумя философами
   * @param {string} sourceId - ID философа-источника
   * @param {string} targetId - ID философа-цели
   * @param {number} maxDepth - Максимальная длина цепочки
   * @return {Array} - Массив цепочек влияния
   */
  function findInfluenceChains(sourceId, targetId, maxDepth) {
    const paths = [];
    
    // Функция для глубокого поиска
    function dfs(currentId, path, depth) {
      // Проверка достижения максимальной глубины
      if (depth > maxDepth) return;
      
      // Проверка достижения цели
      if (currentId === targetId) {
        paths.push([...path]);
        return;
      }
      
      // Поиск всех философов, на которых текущий оказал влияние
      const influencedNodes = data.links.filter(link => 
        (link.source.id === currentId || link.source === currentId) && 
        (link.type === "influence" || link.type === "mutual" || link.type === "normal")
      ).map(link => typeof link.target === 'object' ? link.target.id : link.target);
      
      // Проходим по каждому повлиявшему философу
      for (const nextId of influencedNodes) {
        // Избегаем циклов
        if (!path.includes(nextId)) {
          path.push(nextId);
          dfs(nextId, path, depth + 1);
          path.pop();
        }
      }
    }
    
    // Запускаем поиск
    dfs(sourceId, [sourceId], 0);
    
    return paths;
  }
  
  /**
   * Отображает найденные цепочки влияния
   * @param {Array} chains - Массив цепочек
   * @param {string} sourceId - ID философа-источника
   * @param {string} targetId - ID философа-цели
   */
  function displayChains(chains, sourceId, targetId) {
    const resultsContainer = document.getElementById('chain-results');
    
    // Очищаем контейнер
    resultsContainer.innerHTML = '';
    
    if (chains.length === 0) {
      resultsContainer.innerHTML = '<p>Цепочек влияния не найдено</p>';
      return;
    }
    
    // Сортируем цепочки по длине (сначала короткие)
    chains.sort((a, b) => a.length - b.length);
    
    // Заголовок с количеством найденных цепочек
    const header = document.createElement('h4');
    header.textContent = `Найдено цепочек: ${chains.length}`;
    resultsContainer.appendChild(header);
    
    // Добавляем каждую цепочку
    chains.forEach((chain, index) => {
      const pathElement = document.createElement('div');
      pathElement.className = 'chain-path';
      
      // Формируем строку цепочки
      let pathText = '';
      
      chain.forEach((philosopherId, i) => {
        // Находим имя философа
        const philosopher = data.nodes.find(n => n.id === philosopherId);
        const philosopherName = philosopher ? philosopher.name : philosopherId;
        
        pathText += philosopherName;
        
        if (i < chain.length - 1) {
          pathText += ' → ';
        }
      });
      
      pathElement.textContent = pathText;
      
      // Добавляем действие для подсветки цепочки
      const action = document.createElement('span');
      action.className = 'chain-path-action';
      action.textContent = 'Показать на схеме';
      action.addEventListener('click', () => {
        // Сбрасываем предыдущие выделения
        resetChainHighlighting();
        
        // Выделяем узлы и связи цепочки
        highlightChain(chain);
      });
      
      pathElement.appendChild(action);
      resultsContainer.appendChild(pathElement);
    });
  }
  
  /**
   * Выделяет цепочку влияния на схеме
   * @param {Array} chain - Массив ID философов в цепочке
   */
  function highlightChain(chain) {
    // Сначала сбрасываем все выделения
    resetChainHighlighting();
    
    // Выделяем узлы цепочки
    d3.selectAll(".node").each(function(d) {
      const node = d3.select(this);
      
      if (chain.includes(d.id)) {
        node.classed("chain-node", true);
        
        // Особо выделяем начало и конец цепочки
        if (d.id === chain[0]) {
          node.classed("chain-start", true);
        }
        
        if (d.id === chain[chain.length - 1]) {
          node.classed("chain-end", true);
        }
      }
    });
    
    // Выделяем связи цепочки
    d3.selectAll(".links line").each(function(d) {
      for (let i = 0; i < chain.length - 1; i++) {
        const sourceId = chain[i];
        const targetId = chain[i + 1];
        
        // Проверяем, является ли текущая связь частью цепочки
        if ((d.source.id === sourceId && d.target.id === targetId) ||
            (d.source === sourceId && d.target === targetId)) {
          d3.select(this).classed("chain-link", true);
          break;
        }
      }
    });
  }
  
  /**
   * Сбрасывает выделение цепочки влияния
   */
  function resetChainHighlighting() {
    d3.selectAll(".node").classed("chain-node", false)
                         .classed("chain-start", false)
                         .classed("chain-end", false);
                         
    d3.selectAll(".links line").classed("chain-link", false);
  }
  
  // Возвращаем контроллер
  return {
    show: () => {
      panel.style.display = 'block';
    },
    hide: () => {
      panel.style.display = 'none';
      resetChainHighlighting();
    },
    isVisible: () => panel.style.display === 'block'
  };
}
```

/**
 * Экспортирует данные сети в различных форматах
 * @param {string} format - Формат экспорта (json, csv, graphml, gexf)
 * @param {Object} data - Объект с узлами и связями
 * @param {Object} options - Параметры экспорта
 */
function exportData(format, data, options = {}) {
  const { includePositions = true, includeDetails = true } = options;
  
  let exportContent, filename, mimeType;
  
  switch (format) {
    case 'json':
      exportContent = prepareJsonExport(data, includePositions, includeDetails);
      filename = 'philosophy-network.json';
      mimeType = 'application/json';
      break;
      
    case 'csv':
      exportContent = prepareCsvExport(data, includePositions, includeDetails);
      filename = 'philosophy-network.zip';
      mimeType = 'application/zip';
      downloadCsvZip(exportContent, filename);
      return; // Ранний выход после создания ZIP-архива
      
    case 'graphml':
      exportContent = prepareGraphMLExport(data, includePositions, includeDetails);
      filename = 'philosophy-network.graphml';
      mimeType = 'application/xml';
      break;
      
    case 'gexf':
      exportContent = prepareGexfExport(data, includePositions, includeDetails);
      filename = 'philosophy-network.gexf';
      mimeType = 'application/xml';
      break;
      
    default:
      alert('Неизвестный формат экспорта');
      return;
  }
  
  // Создаем ссылку для скачивания
  const blob = new Blob([exportContent], { type: mimeType });
  const url = URL.createObjectURL(blob);
  
  const downloadLink = document.createElement('a');
  downloadLink.href = url;
  downloadLink.download = filename;
  document.body.appendChild(downloadLink);
  downloadLink.click();
  
  // Очищаем
  setTimeout(() => {
    document.body.removeChild(downloadLink);
    URL.revokeObjectURL(url);
  }, 100);
}

/**
 * Подготавливает данные для экспорта в JSON
 * @param {Object} data - Объект с узлами и связями
 * @param {boolean} includePositions - Включать ли координаты
 * @param {boolean} includeDetails - Включать ли расширенные данные
 * @return {string} - Строка JSON
 */
function prepareJsonExport(data, includePositions, includeDetails) {
  // Создаем глубокую копию данных
  const exportData = {
    metadata: {
      title: "Сеть российских философов",
      exportDate: new Date().toISOString(),
      nodesCount: data.nodes.length,
      linksCount: data.links.length
    },
    nodes: data.nodes.map(node => {
      const nodeData = {
        id: node.id,
        name: node.name,
        years: node.years,
        group: node.group
      };
      
      // Добавляем координаты, если требуется
      if (includePositions && node.x !== undefined && node.y !== undefined) {
        nodeData.x = node.x;
        nodeData.y = node.y;
      }
      
      // Добавляем расширенные данные, если требуется и они есть
      if (includeDetails && window.philosophersDetails && window.philosophersDetails[node.id]) {
        const details = window.philosophersDetails[node.id];
        nodeData.works = details.works;
        nodeData.ideas = details.ideas;
        nodeData.bio = details.bio;
        nodeData.influence = details.influence;
      }
      
      return nodeData;
    }),
    links: data.links.map(link => ({
      source: typeof link.source === 'object' ? link.source.id : link.source,
      target: typeof link.target === 'object' ? link.target.id : link.target,
      type: link.type,
      direction: link.direction
    }))
  };
  
  // Добавляем информацию о группах и типах связей, если она доступна
  if (window.networkGroups) {
    exportData.groups = window.networkGroups;
  }
  
  if (window.networkRelationTypes) {
    exportData.relationTypes = window.networkRelationTypes;
  }
  
  return JSON.stringify(exportData, null, 2);
}

/**
 * Подготавливает данные для экспорта в CSV
 * @param {Object} data - Объект с узлами и связями
 * @param {boolean} includePositions - Включать ли координаты
 * @param {boolean} includeDetails - Включать ли расширенные данные
 * @return {Object} - Объект с CSV-представлениями узлов и связей
 */
function prepareCsvExport(data, includePositions, includeDetails) {
  // Функция для преобразования массива объектов в CSV
  function objectsToCsv(objects, fields) {
    const header = fields.map(field => field.name).join(',');
    
    const rows = objects.map(obj => {
      return fields.map(field => {
        let value = field.getter(obj);
        
        // Экранируем кавычки и добавляем кавычки, если в значении есть запятая
        if (typeof value === 'string') {
          value = value.replace(/"/g, '""');
          if (value.includes(',')) {
            value = `"${value}"`;
          }
        }
        
        return value;
      }).join(',');
    });
    
    return [header, ...rows].join('\n');
  }
  
  // Определяем поля для узлов
  const nodeFields = [
    { name: 'id', getter: node => node.id },
    { name: 'name', getter: node => node.name },
    { name: 'years', getter: node => node.years },
    { name: 'group', getter: node => node.group },
    { name: 'groupName', getter: node => node.groupName || getGroupDisplayName(node.group) }
  ];
  
  // Добавляем координаты, если требуется
  if (includePositions) {
    nodeFields.push(
      { name: 'x', getter: node => node.x !== undefined ? node.x : '' },
      { name: 'y', getter: node => node.y !== undefined ? node.y : '' }
    );
  }
  
  // Определяем поля для связей
  const linkFields = [
    { name: 'source', getter: link => typeof link.source === 'object' ? link.source.id : link.source },
    { name: 'target', getter: link => typeof link.target === 'object' ? link.target.id : link.target },
    { name: 'type', getter: link => link.type },
    { name: 'direction', getter: link => link.direction || 'one-way' }
  ];
  
  // Создаем CSV для узлов и связей
  const nodesCsv = objectsToCsv(data.nodes, nodeFields);
  const linksCsv = objectsToCsv(data.links, linkFields);
  
  // Дополнительный CSV с расширенной информацией, если требуется
  let detailsCsv = '';
  
  if (includeDetails && window.philosophersDetails) {
    const detailsFields = [
      { name: 'id', getter: ([id, _]) => id },
      { name: 'name', getter: ([_, details]) => details.name },
      { name: 'years', getter: ([_, details]) => details.years },
      { name: 'bio', getter: ([_, details]) => details.bio || '' },
      { name: 'influence', getter: ([_, details]) => details.influence || '' },
      { name: 'works', getter: ([_, details]) => details.works ? details.works.join('|') : '' },
      { name: 'ideas', getter: ([_, details]) => details.ideas ? details.ideas.join('|') : '' }
    ];
    
    const detailsData = Object.entries(window.philosophersDetails);
    detailsCsv = objectsToCsv(detailsData, detailsFields);
  }
  
  return {
    nodesCsv,
    linksCsv,
    detailsCsv
  };
}

/**
 * Создает ZIP-архив с CSV-файлами и скачивает его
 * @param {Object} csvData - Объект с CSV-строками
 * @param {string} filename - Имя файла для скачивания
 */
function downloadCsvZip(csvData, filename) {
  // Для создания ZIP-архива требуется библиотека JSZip
  // Загружаем ее динамически, если она не существует
  if (typeof JSZip === 'undefined') {
    const script = document.createElement('script');
    script.src = 'https://cdnjs.cloudflare.com/ajax/libs/jszip/3.7.1/jszip.min.js';
    script.onload = () => createZipAndDownload(csvData, filename);
    document.head.appendChild(script);
  } else {
    createZipAndDownload(csvData, filename);
  }
  
  // Функция для создания ZIP-архива
  function createZipAndDownload(csvData, filename) {
    const zip = new JSZip();
    
    // Добавляем файлы в архив
    zip.file("nodes.csv", csvData.nodesCsv);
    zip.file("links.csv", csvData.linksCsv);
    
    if (csvData.detailsCsv) {
      zip.file("philosophers_details.csv", csvData.detailsCsv);
    }
    
    // Добавляем readme с описанием
    const readme = `Сеть российских философов - Данные в формате CSV
Экспортировано: ${new Date().toLocaleString()}

Файлы в архиве:
- nodes.csv: Данные об узлах сети (философах)
- links.csv: Данные о связях между философами
${csvData.detailsCsv ? '- philosophers_details.csv: Расширенная информация о философах\n' : ''}

Для импорта в программы анализа сетей используйте nodes.csv и links.csv.
`;
    
    zip.file("readme.txt", readme);
    
    // Генерируем архив и скачиваем его
    zip.generateAsync({ type: "blob" })
      .then(function(content) {
        const url = URL.createObjectURL(content);
        const downloadLink = document.createElement('a');
        downloadLink.href = url;
        downloadLink.download = filename;
        document.body.appendChild(downloadLink);
        downloadLink.click();
        
        // Очищаем
        setTimeout(() => {
          document.body.removeChild(downloadLink);
          URL.revokeObjectURL(url);
        }, 100);
      });
  }
}

/**
 * Подготавливает данные для экспорта в формате GraphML
 * @param {Object} data - Объект с узлами и связями
 * @param {boolean} includePositions - Включать ли координаты
 * @param {boolean} includeDetails - Включать ли расширенные данные
 * @return {string} - Строка GraphML
 */
function prepareGraphMLExport(data, includePositions, includeDetails) {
  // XML-прелюдия и открывающий тег graphml
  let output = `<?xml version="1.0" encoding="UTF-8"?>
<graphml xmlns="http://graphml.graphdrawing.org/xmlns"
  xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
  xsi:schemaLocation="http://graphml.graphdrawing.org/xmlns
  http://graphml.graphdrawing.org/xmlns/1.0/graphml.xsd">
  
  <!-- Определения атрибутов для узлов -->
  <key id="name" for="node" attr.name="name" attr.type="string"/>
  <key id="years" for="node" attr.name="years" attr.type="string"/>
  <key id="group" for="node" attr.name="group" attr.type="string"/>
  <key id="groupName" for="node" attr.name="groupName" attr.type="string"/>`;
  
  // Добавляем определения для координат, если требуется
  if (includePositions) {
    output += `
  <key id="x" for="node" attr.name="x" attr.type="double"/>
  <key id="y" for="node" attr.name="y" attr.type="double"/>`;
  }
  
  // Добавляем определения для расширенной информации, если требуется
  if (includeDetails) {
    output += `
  <key id="bio" for="node" attr.name="bio" attr.type="string"/>
  <key id="influence" for="node" attr.name="influence" attr.type="string"/>
  <key id="works" for="node" attr.name="works" attr.type="string"/>
  <key id="ideas" for="node" attr.name="ideas" attr.type="string"/>`;
  }
  
  // Определения атрибутов для связей
  output += `
  
  <!-- Определения атрибутов для связей -->
  <key id="type" for="edge" attr.name="type" attr.type="string"/>
  <key id="direction" for="edge" attr.name="direction" attr.type="string"/>
  
  <!-- Граф -->
  <graph id="philosophy-network" edgedefault="directed">`;
  
  // Добавляем узлы
  data.nodes.forEach(node => {
    // Открывающий тег узла
    output += `
    <node id="${escapeXml(node.id)}">
      <data key="name">${escapeXml(node.name)}</data>
      <data key="years">${escapeXml(node.years)}</data>
      <data key="group">${escapeXml(node.group)}</data>
      <data key="groupName">${escapeXml(node.groupName || getGroupDisplayName(node.group))}</data>`;
    
    // Добавляем координаты, если требуется
    if (includePositions && node.x !== undefined && node.y !== undefined) {
      output += `
      <data key="x">${node.x}</data>
      <data key="y">${node.y}</data>`;
    }
    
    // Добавляем расширенную информацию, если требуется и она есть
    if (includeDetails && window.philosophersDetails && window.philosophersDetails[node.id]) {
      const details = window.philosophersDetails[node.id];
      
      if (details.bio) {
        output += `
      <data key="bio">${escapeXml(details.bio)}</data>`;
      }
      
      if (details.influence) {
        output += `
      <data key="influence">${escapeXml(details.influence)}</data>`;
      }
      
      if (details.works && details.works.length > 0) {
        output += `
      <data key="works">${escapeXml(details.works.join('|'))}</data>`;
      }
      
      if (details.ideas && details.ideas.length > 0) {
        output += `
      <data key="ideas">${escapeXml(details.ideas.join('|'))}</data>`;
      }
    }
    
    // Закрывающий тег узла
    output += `
    </node>`;
  });
  
  // Добавляем связи
  data.links.forEach((link, index) => {
    const sourceId = typeof link.source === 'object' ? link.source.id : link.source;
    const targetId = typeof link.target === 'object' ? link.target.id : link.target;
    
    output += `
    <edge id="e${index}" source="${escapeXml(sourceId)}" target="${escapeXml(targetId)}">
      <data key="type">${escapeXml(link.type)}</data>
      <data key="direction">${escapeXml(link.direction || 'one-way')}</data>
    </edge>`;
  });
  
  // Закрывающие теги
  output += `
  </graph>
</graphml>`;
  
  return output;
}

/**
 * Экранирует специальные символы для XML
 * @param {string} str - Исходная строка
 * @return {string} - Экранированная строка
 */
function escapeXml(str) {
  if (typeof str !== 'string') return str;
  
  return str
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&apos;');
}

/**
 * Преобразует HEX-цвет в объект RGB
 * @param {string} hex - Цвет в формате HEX
 * @return {Object|null} - Объект с компонентами RGB или null
 */
function hexToRgb(hex) {
  // Если цвет передан в формате переменной CSS
  if (hex.startsWith('var(--')) {
    // Получаем значение переменной CSS
    const varName = hex.match(/var\(([^)]+)\)/)[1];
    const computedStyle = getComputedStyle(document.documentElement);
    hex = computedStyle.getPropertyValue(varName).trim();
  }
  
  // Если цвет не начинается с #, предполагаем, что это уже RGB
  if (!hex.startsWith('#')) {
    return null;
  }
  
  // Убираем # из начала строки
  hex = hex.substring(1);
  
  // Проверяем длину строки
  let r, g, b;
  if (hex.length === 3) {
    // Краткая запись (#RGB)
    r = parseInt(hex[0] + hex[0], 16);
    g = parseInt(hex[1] + hex[1], 16);
    b = parseInt(hex[2] + hex[2], 16);
  } else if (hex.length === 6) {
    // Полная запись (#RRGGBB)
    r = parseInt(hex.substring(0, 2), 16);
    g = parseInt(hex.substring(2, 4), 16);
    b = parseInt(hex.substring(4, 6), 16);
  } else {
    // Некорректный формат
    return null;
  }
  
  return { r, g, b };
}
