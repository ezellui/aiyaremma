### Шаг 2: Добавить функции режима погружения

Добавьте следующие функции в блок `<script>` перед функцией `initApplication()`:

```javascript
/**
 * Реализует режим погружения с фокусом на конкретном философе
 * @param {Object} data - Объект с узлами и связями
 * @param {string} containerId - ID контейнера для отображения
 * @return {Object} - Контроллер режима погружения
 */
function createFocusMode(data, containerId) {
  // Получаем контейнер и элементы SVG
  const container = document.getElementById(containerId);
  const focusControls = document.createElement('div');
  focusControls.className = 'focus-controls';
  focusControls.style.display = 'none';
  
  // Добавляем элементы управления
  focusControls.innerHTML = `
    <h3 style="margin-top: 0;">Режим погружения</h3>
    <button id="exit-focus-mode">Выйти из режима</button>
    <div class="focus-depth-control">
      <label>Глубина связей: <span id="depth-value">1</span></label>
      <input type="range" id="focus-depth" min="1" max="3" value="1" />
    </div>
    <div class="focus-info">
      <div id="focus-connections">Связанных философов: <span>0</span></div>
    </div>
  `;
  
  container.appendChild(focusControls);
  
  // Получаем элементы управления
  const exitButton = document.getElementById('exit-focus-mode');
  const depthSlider = document.getElementById('focus-depth');
  const depthValue = document.getElementById('depth-value');
  const connectionsInfo = document.getElementById('focus-connections').querySelector('span');
  
  // Состояние режима погружения
  let currentFocusNode = null;
  let currentDepth = 1;
  
  // Находит связанные узлы на определенной глубине
  function findConnectedNodes(startNode, depth) {
    const connected = new Set();
    const visited = new Set([startNode.id]);
    
    function traverse(node, currentDepth) {
      if (currentDepth > depth) return;
      
      // Добавляем все соседние узлы
      data.links.forEach(link => {
        let nextNode = null;
        
        if (link.source.id === node.id) {
          nextNode = link.target;
        } else if (link.target.id === node.id) {
          nextNode = link.source;
        }
        
        if (nextNode && !visited.has(nextNode.id)) {
          connected.add(nextNode);
          visited.add(nextNode.id);
          // Рекурсивно добавляем соседей соседей
          traverse(nextNode, currentDepth + 1);
        }
      });
    }
    
    traverse(startNode, 1);
    return Array.from(connected);
  }
  
  // Активирует режим погружения для узла
  function activateFocusMode(node) {
    if (!node) return;
    
    currentFocusNode = node;
    currentDepth = parseInt(depthSlider.value, 10);
    
    // Показываем панель управления
    focusControls.style.display = 'block';
    
    // Применяем фокус
    applyFocus();
  }
  
  // Применяет визуальный фокус на основе текущих настроек
  function applyFocus() {
    if (!currentFocusNode) return;
    
    // Находим связанные узлы
    const connectedNodes = findConnectedNodes(currentFocusNode, currentDepth);
    connectionsInfo.textContent = connectedNodes.length;
    
    // Сбрасываем все классы фокуса
    d3.selectAll(".node").classed("focus-main", false)
      .classed("focus-connected", false)
      .classed("focus-dimmed", false);
    
    d3.selectAll(".links line").classed("focus-connection", false)
      .classed("focus-dimmed", false);
    
    // Применяем фокус к основному узлу
    d3.selectAll(".node").filter(d => d.id === currentFocusNode.id)
      .classed("focus-main", true);
    
    // Выделяем связанные узлы
    d3.selectAll(".node").filter(d => connectedNodes.includes(d))
      .classed("focus-connected", true);
    
    // Затемняем остальные узлы
    d3.selectAll(".node").filter(d => 
      d.id !== currentFocusNode.id && !connectedNodes.includes(d)
    ).classed("focus-dimmed", true);
    
    // Выделяем связи
    d3.selectAll(".links line").each(function(d) {
      const isMainNodeConnection = 
        d.source.id === currentFocusNode.id || 
        d.target.id === currentFocusNode.id;
      
      const isConnectionBetweenHighlighted = 
        connectedNodes.includes(d.source) && 
        connectedNodes.includes(d.target);
      
      if (isMainNodeConnection || isConnectionBetweenHighlighted) {
        d3.select(this).classed("focus-connection", true);
      } else {
        d3.select(this).classed("focus-dimmed", true);
      }
    });
  }
  
  // Выход из режима погружения
  function exitFocusMode() {
    // Сбрасываем все классы
    d3.selectAll(".node").classed("focus-main", false)
      .classed("focus-connected", false)
      .classed("focus-dimmed", false);
    
    d3.selectAll(".links line").classed("focus-connection", false)
      .classed("focus-dimmed", false);
    
    // Скрываем панель управления
    focusControls.style.display = 'none';
    
    // Сбрасываем текущий узел
    currentFocusNode = null;
  }
  
  // Обработчики событий
  exitButton.addEventListener('click', exitFocusMode);
  
  depthSlider.addEventListener('input', function() {
    currentDepth = parseInt(this.value, 10);
    depthValue.textContent = currentDepth;
    applyFocus();
  });
  
  // Возвращаем контроллер режима погружения
  return {
    activate: activateFocusMode,
    exit: exitFocusMode
  };
}
```

### Шаг 3: Добавить функции для инструмента лассо

Добавьте следующую функцию в блок `<script>` перед функцией `initApplication()`:

```javascript
/**
 * Создает и настраивает инструмент лассо для выделения групп философов
 * @param {Object} data - Объект с узлами и связями
 * @param {string} containerId - ID контейнера для отображения
 * @return {Object} - Контроллер инструмента лассо
 */
function createLassoTool(data, containerId) {
  // Получаем контейнер и элементы SVG
  const container = document.getElementById(containerId);
  const svg = d3.select(container).select("svg");
  const g = svg.select("g");
  
  // Создаем панель управления для выделенных узлов
  const lassoPanel = document.createElement('div');
  lassoPanel.className = 'lasso-panel';
  lassoPanel.innerHTML = `
    <h3>Выделено узлов: <span id="selected-count">0</span></h3>
    <button id="lasso-highlight">Выделить связи</button>
    <button id="lasso-filter">Фильтровать</button>
    <button id="lasso-export">Экспорт группы</button>
    <button id="lasso-cancel">Отмена</button>
    <div class="lasso-info" id="lasso-info"></div>
  `;
  
  container.appendChild(lassoPanel);
  
  // Переключатель для активации инструмента
  let isLassoActive = false;
  let selectedNodes = [];
  
  // Инициализация инструмента лассо
  const lasso = d3.lasso()
    .closePathSelect(true)
    .closePathDistance(100)
    .targetArea(svg)
    .on("start", lassoStart)
    .on("draw", lassoDraw)
    .on("end", lassoEnd);
  
  // Функции обработки событий лассо
  function lassoStart() {
    lasso.items().classed("selected", false);
    selectedNodes = [];
    updateLassoPanel();
  }
  
  function lassoDraw() {
    // Опционально: можно добавить предварительный просмотр выделения
  }
  
  function lassoEnd() {
    // Получаем выделенные узлы
    selectedNodes = lasso.selectedItems().data();
    
    // Обновляем информацию на панели
    updateLassoPanel();
    
    // Показываем панель управления, если выделены узлы
    if (selectedNodes.length > 0) {
      lassoPanel.style.display = 'block';
      
      // Позиционируем панель возле центра выделения
      const bounds = calculateBounds(selectedNodes);
      const transform = d3.zoomTransform(svg.node());
      
      lassoPanel.style.left = `${bounds.centerX * transform.k + transform.x + 20}px`;
      lassoPanel.style.top = `${bounds.centerY * transform.k + transform.y - 20}px`;
    } else {
      lassoPanel.style.display = 'none';
    }
  }
  
  // Рассчитывает границы выделенных узлов
  function calculateBounds(nodes) {
    let minX = Infinity, minY = Infinity;
    let maxX = -Infinity, maxY = -Infinity;
    
    nodes.forEach(node => {
      minX = Math.min(minX, node.x);
      minY = Math.min(minY, node.y);
      maxX = Math.max(maxX, node.x);
      maxY = Math.max(maxY, node.y);
    });
    
    return {
      minX, minY, maxX, maxY,
      centerX: (minX + maxX) / 2,
      centerY: (minY + maxY) / 2
    };
  }
  
  // Обновляет информацию на панели управления лассо
  function updateLassoPanel() {
    document.getElementById('selected-count').textContent = selectedNodes.length;
    
    // Дополнительная информация о выделенных узлах
    const infoElement = document.getElementById('lasso-info');
    
    if (selectedNodes.length > 0) {
      // Считаем статистику по группам
      const groupCounts = {};
      selectedNodes.forEach(node => {
        groupCounts[node.groupName] = (groupCounts[node.groupName] || 0) + 1;
      });
      
      // Формируем текст
      let infoText = '<strong>Группы:</strong><br>';
      Object.entries(groupCounts).forEach(([group, count]) => {
        infoText += `${group}: ${count}<br>`;
      });
      
      infoElement.innerHTML = infoText;
    } else {
      infoElement.innerHTML = '';
    }
  }
  
  // Активация/деактивация инструмента лассо
  function toggleLassoTool() {
    isLassoActive = !isLassoActive;
    
    if (isLassoActive) {
      // Деактивируем другие интерактивные элементы
      g.selectAll(".node").on(".drag", null);
      
      // Настраиваем лассо
      lasso.items(g.selectAll(".node"));
      
      // Активируем инструмент
      svg.call(lasso);
    } else {
      // Удаляем лассо
      svg.on(".dragstart", null)
         .on(".drag", null)
         .on(".dragend", null);
      
      // Скрываем панель
      lassoPanel.style.display = 'none';
      
      // Восстанавливаем перетаскивание узлов
      g.selectAll(".node").call(d3.drag()
        .on("start", dragstarted)
        .on("drag", dragged)
        .on("end", dragended));
      
      // Возвращаем оригинальные обработчики событий
      // (требуется воссоздать из оригинального кода)
      function dragstarted(event, d) {
        if (!event.active) window.simulationManager.simulation.alphaTarget(0.3).restart();
        d.fx = d.x;
        d.fy = d.y;
      }
      
      function dragged(event, d) {
        d.fx = event.x;
        d.fy = event.y;
      }
      
      function dragended(event, d) {
        if (!event.active) window.simulationManager.simulation.alphaTarget(0);
        d.fx = null;
        d.fy = null;
      }
    }
    
    return isLassoActive;
  }
  
  // Настройка кнопок панели управления
  document.getElementById('lasso-highlight').addEventListener('click', function() {
    // Выделяем связи между выделенными узлами
    const selectedIds = new Set(selectedNodes.map(n => n.id));
    
    // Сбрасываем предыдущие выделения
    d3.selectAll(".links line").classed("highlight-related", false);
    
    // Выделяем связи между выделенными узлами
    d3.selectAll(".links line").filter(link => 
      selectedIds.has(link.source.id) && selectedIds.has(link.target.id)
    ).classed("highlight-related", true);
    
    // Дополнительная информация о связях
    const connections = d3.selectAll(".links line").filter(link => 
      selectedIds.has(link.source.id) && selectedIds.has(link.target.id)
    ).size();
    
    document.getElementById('lasso-info').innerHTML += 
      `<br><strong>Связей внутри группы:</strong> ${connections}`;
  });
  
  document.getElementById('lasso-filter').addEventListener('click', function() {
    // Фильтрация: отображаем только выделенные узлы и связи между ними
    const selectedIds = new Set(selectedNodes.map(n => n.id));
    
    // Фильтруем узлы
    d3.selectAll(".node").style("opacity", node => 
      selectedIds.has(node.__data__.id) ? 1 : 0.1
    );
    
    // Фильтруем связи
    d3.selectAll(".links line").style("opacity", link => 
      (selectedIds.has(link.source.id) && selectedIds.has(link.target.id)) ? 1 : 0.1
    );
    
    // Скрываем панель
    lassoPanel.style.display = 'none';
  });
  
  document.getElementById('lasso-export').addEventListener('click', function() {
    // Экспорт выделенной группы в JSON
    const exportData = {
      nodes: selectedNodes.map(n => ({
        id: n.id,
        name: n.name,
        years: n.years,
        group: n.group
      })),
      links: data.links.filter(link => 
        selectedNodes.includes(link.source) && selectedNodes.includes(link.target)
      ).map(l => ({
        source: l.source.id,
        target: l.target.id,
        type: l.type
      }))
    };
    
    // Создаем и скачиваем файл
    const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(exportData, null, 2));
    const downloadLink = document.createElement("a");
    downloadLink.setAttribute("href", dataStr);
    downloadLink.setAttribute("download", "selected-philosophers.json");
    document.body.appendChild(downloadLink);
    downloadLink.click();
    document.body.removeChild(downloadLink);
  });
  
  document.getElementById('lasso-cancel').addEventListener('click', function() {
    // Отменяем выделение
    lasso.items().classed("selected", false);
    selectedNodes = [];
    
    // Возвращаем исходную отрисовку
    d3.selectAll(".node").style("opacity", 1);
    d3.selectAll(".links line").style("opacity", 1)
                               .classed("highlight-related", false);
    
    // Скрываем панель
    lassoPanel.style.display = 'none';
  });
  
  // Возвращаем контроллер инструмента
  return {
    toggle: toggleLassoTool,
    isActive: () => isLassoActive
  };
}
```
