```javascript
class ElementValidationService {
  constructor(categoryTemplateRepository, relationshipTypeRepository, generatedElementsRepository) {
    this.categoryTemplateRepo = categoryTemplateRepository;
    this.relationshipTypeRepo = relationshipTypeRepository;
    this.generatedElementsRepo = generatedElementsRepository;
  }

  async validateGeneratedGraph(graphData, conceptId, generationContext) {
    const validationResult = {
      newCategories: [],
      newRelationshipTypes: [],
      existingCategories: [],
      existingRelationshipTypes: []
    };

    // Проверяем категории
    for (const category of graphData.categories) {
      const exists = await this.categoryTemplateRepo.findByName(category.name);
      if (!exists) {
        validationResult.newCategories.push({
          name: category.name,
          definition: category.definition,
          conceptId,
          generationContext
        });
      } else {
        validationResult.existingCategories.push(category);
      }
    }

    // Проверяем типы связей
    for (const relationship of graphData.relationships) {
      const exists = await this.relationshipTypeRepo.findByType(relationship.type);
      if (!exists) {
        validationResult.newRelationshipTypes.push({
          type: relationship.type,
          direction: relationship.direction,
          conceptId,
          generationContext
        });
      } else {
        validationResult.existingRelationshipTypes.push(relationship);
      }
    }

    // Сохраняем информацию о новых элементах
    await this.saveNewElements(validationResult, conceptId, generationContext);

    return validationResult;
  }

  async saveNewElements(validationResult, conceptId, generationContext) {
    const newElements = [];

    // Сохраняем новые категории
    for (const category of validationResult.newCategories) {
      newElements.push({
        concept_id: conceptId,
        generation_context: generationContext,
        element_type: 'category',
        element_name: category.name,
        element_definition: category.definition,
        is_new: true,
        is_approved: false,
        created_at: new Date()
      });
    }

    // Сохраняем новые типы связей
    for (const relType of validationResult.newRelationshipTypes) {
      newElements.push({
        concept_id: conceptId,
        generation_context: generationContext,
        element_type: 'relationship_type',
        element_name: relType.type,
        element_definition: `${relType.direction} relationship`,
        is_new: true,
        is_approved: false,
        created_at: new Date()
      });
    }

    if (newElements.length > 0) {
      await this.generatedElementsRepo.insertMany(newElements);
    }
  }

  async approveNewElements(elementIds, userId) {
    const approvedElements = await this.generatedElementsRepo.findByIds(elementIds);
    
    for (const element of approvedElements) {
      if (element.element_type === 'category') {
        await this.categoryTemplateRepo.create({
          name: element.element_name,
          description: element.element_definition,
          default_definition: element.element_definition,
          created_by: userId,
          is_system: false
        });
      } else if (element.element_type === 'relationship_type') {
        await this.relationshipTypeRepo.create({
          type_name: element.element_name,
          description: element.element_definition,
          created_by: userId,
          is_system: false
        });
      }

      // Обновляем статус элемента
      await this.generatedElementsRepo.updateApprovalStatus(
        element.element_id,
        true,
        userId,
        new Date()
      );
    }
  }
}

module.exports = ElementValidationService;
```

```javascript
async createGraphFromTheses(conceptId, thesesData) {
  try {
    // Существующая логика получения данных и запроса к Claude...
    
    // После получения структуры графа от Claude
    const generatedGraph = await this.claudeServiceClient.generateGraphFromTheses(thesesData);
    
    // НОВОЕ: Валидация и обработка новых элементов
    const validationResult = await this.elementValidationService.validateGeneratedGraph(
      generatedGraph,
      conceptId,
      'thesis_to_graph'
    );
    
    // Если есть новые элементы, возвращаем информацию для пользователя
    if (validationResult.newCategories.length > 0 || validationResult.newRelationshipTypes.length > 0) {
      return {
        graph: generatedGraph,
        validationResult,
        requiresApproval: true,
        message: 'Граф содержит новые категории или типы связей, требующие одобрения'
      };
    }
    
    // Если все элементы существуют, сохраняем граф как обычно
    await this.saveGraph(conceptId, generatedGraph);
    
    return {
      graph: generatedGraph,
      validationResult,
      requiresApproval: false
    };
  } catch (error) {
    throw new Error(`Failed to create graph from theses: ${error.message}`);
  }
}
```

```javascript
function processGraphGenerationResponse(claudeResponse) {
  const processedGraph = {
    categories: [],
    relationships: []
  };
  
  // Парсинг ответа Claude
  const parsedResponse = parseClaudeGraphResponse(claudeResponse);
  
  // Обработка категорий
  for (const category of parsedResponse.categories) {
    processedGraph.categories.push({
      name: category.name,
      definition: category.definition,
      // НОВОЕ: Флаг для индикации, что это предложенная Claude категория
      isGenerated: true,
      sourceTheses: category.sourceTheses || []
    });
  }
  
  // Обработка связей
  for (const relationship of parsedResponse.relationships) {
    processedGraph.relationships.push({
      source: relationship.source,
      target: relationship.target,
      type: relationship.type,
      direction: relationship.direction || 'unidirectional',
      // НОВОЕ: Флаг для индикации, что это предложенный Claude тип связи
      isGenerated: true,
      justification: relationship.justification
    });
  }
  
  return processedGraph;
}
```

```typescript
import React, { useState } from 'react';
import { Check, X, AlertCircle } from 'lucide-react';

interface NewElement {
  element_id: string;
  element_type: 'category' | 'relationship_type';
  element_name: string;
  element_definition: string;
  is_new: boolean;
  is_approved: boolean;
}

interface NewElementsApprovalProps {
  newElements: NewElement[];
  onApprove: (elementIds: string[]) => void;
  onReject: (elementIds: string[]) => void;
  onModify: (elementId: string, newDefinition: string) => void;
}

const NewElementsApproval: React.FC<NewElementsApprovalProps> = ({
  newElements,
  onApprove,
  onReject,
  onModify
}) => {
  const [selectedElements, setSelectedElements] = useState<string[]>([]);
  const [editingElement, setEditingElement] = useState<string | null>(null);
  const [editedDefinitions, setEditedDefinitions] = useState<Record<string, string>>({});

  const categories = newElements.filter(el => el.element_type === 'category');
  const relationshipTypes = newElements.filter(el => el.element_type === 'relationship_type');

  const handleSelectElement = (elementId: string) => {
    setSelectedElements(prev => 
      prev.includes(elementId) 
        ? prev.filter(id => id !== elementId)
        : [...prev, elementId]
    );
  };

  const handleApproveSelected = () => {
    onApprove(selectedElements);
    setSelectedElements([]);
  };

  const handleRejectSelected = () => {
    onReject(selectedElements);
    setSelectedElements([]);
  };

  const handleEditDefinition = (elementId: string, newDefinition: string) => {
    setEditedDefinitions(prev => ({
      ...prev,
      [elementId]: newDefinition
    }));
    onModify(elementId, newDefinition);
  };

  return (
    <div className="bg-yellow-50 border border-yellow-200 rounded-lg p-4">
      <div className="flex items-center mb-4">
        <AlertCircle className="text-yellow-600 mr-2" size={20} />
        <h3 className="text-lg font-medium">Новые элементы требуют одобрения</h3>
      </div>
      
      {categories.length > 0 && (
        <div className="mb-4">
          <h4 className="font-medium mb-2">Новые категории:</h4>
          <div className="space-y-2">
            {categories.map(category => (
              <div key={category.element_id} className="bg-white p-3 rounded border">
                <div className="flex items-start">
                  <input
                    type="checkbox"
                    checked={selectedElements.includes(category.element_id)}
                    onChange={() => handleSelectElement(category.element_id)}
                    className="mt-1 mr-3"
                  />
                  <div className="flex-1">
                    <div className="font-medium">{category.element_name}</div>
                    {editingElement === category.element_id ? (
                      <textarea
                        value={editedDefinitions[category.element_id] || category.element_definition}
                        onChange={(e) => handleEditDefinition(category.element_id, e.target.value)}
                        onBlur={() => setEditingElement(null)}
                        className="w-full mt-1 p-2 border rounded"
                        rows={3}
                      />
                    ) : (
                      <div 
                        className="text-sm text-gray-600 cursor-pointer hover:bg-gray-50 p-1 rounded"
                        onClick={() => setEditingElement(category.element_id)}
                      >
                        {editedDefinitions[category.element_id] || category.element_definition}
                      </div>
                    )}
                  </div>
                </div>
              </div>
            ))}
          </div>
        </div>
      )}
      
      {relationshipTypes.length > 0 && (
        <div className="mb-4">
          <h4 className="font-medium mb-2">Новые типы связей:</h4>
          <div className="space-y-2">
            {relationshipTypes.map(relType => (
              <div key={relType.element_id} className="bg-white p-3 rounded border">
                <div className="flex items-center">
                  <input
                    type="checkbox"
                    checked={selectedElements.includes(relType.element_id)}
                    onChange={() => handleSelectElement(relType.element_id)}
                    className="mr-3"
                  />
                  <div className="flex-1">
                    <span className="font-medium">{relType.element_name}</span>
                    <span className="text-sm text-gray-600 ml-2">
                      ({relType.element_definition})
                    </span>
                  </div>
                </div>
              </div>
            ))}
          </div>
        </div>
      )}
      
      <div className="flex justify-between mt-4">
        <div className="text-sm text-gray-600">
          Выбрано: {selectedElements.length} элементов
        </div>
        <div className="space-x-2">
          <button
            onClick={handleRejectSelected}
            disabled={selectedElements.length === 0}
            className="px-4 py-2 bg-red-500 text-white rounded hover:bg-red-600 disabled:bg-gray-300"
          >
            <X size={16} className="inline mr-1" />
            Отклонить выбранные
          </button>
          <button
            onClick={handleApproveSelected}
            disabled={selectedElements.length === 0}
            className="px-4 py-2 bg-green-500 text-white rounded hover:bg-green-600 disabled:bg-gray-300"
          >
            <Check size={16} className="inline mr-1" />
            Одобрить выбранные
          </button>
        </div>
      </div>
    </div>
  );
};
```

```markdown
### 9.4 Двунаправленное преобразование (тезисы ↔ граф) с обработкой новых элементов

1. Пользователь создает тезисы через UI (напрямую, без графа)
2. Пользователь запрашивает генерацию графа на основе тезисов
3. UI отправляет запрос в Graph Service
4. Graph Service получает тезисы из Thesis Service
5. Graph Service формирует запрос к Claude Service для генерации графа
6. Claude Service отправляет запрос к Claude
7. Claude возвращает структуру графа
8. **НОВОЕ: Graph Service проверяет наличие категорий и типов связей в базе**
9. **НОВОЕ: Если обнаружены новые элементы:**
   - Graph Service сохраняет информацию о новых элементах в MongoDB
   - Graph Service возвращает граф с флагом requiresApproval
   - UI отображает компонент одобрения новых элементов
   - Пользователь просматривает, редактирует и одобряет/отклоняет новые элементы
   - При одобрении Graph Service добавляет новые элементы в соответствующие таблицы
10. Graph Service преобразует ответ в граф и сохраняет в Neo4j
11. Пользователь видит сгенерированный граф
12. Пользователь может редактировать граф и снова генерировать тезисы
```

Этот функционал обеспечивает:
1. Автоматическое обнаружение новых категорий и типов связей при генерации графа
2. Сохранение информации о новых элементах для последующего одобрения
3. Интерфейс для просмотра, редактирования и одобрения новых элементов
4. Добавление одобренных элементов в базу данных для повторного использования
5. Поддержку системных (предустановленных) категорий и типов связей
