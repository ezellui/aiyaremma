        // Улучшенная функция склонения существительных
        function decline(wordObj, caseType) {
            const word = wordObj.word;
            const gender = wordObj.gender;
            const animate = wordObj.animate || false;
            
            // Несклоняемые слова
            if (wordObj.indeclinable) return word;
            
            // Особые случаи для эльфийских имен
            if (['Ауле', 'Ороме', 'Манвэ', 'Ульмо', 'Вайрэ', 'Эстэ', 'Фродо'].includes(word)) return word;
            
            switch(caseType) {
                case 'gen': // Родительный падеж
                    if (gender === 'мн') {
                        if (word === 'Люди') return 'Людей';
                        if (word === 'Дети') return 'Детей';
                        if (word.endsWith('ы')) return word.slice(0, -1) + (word.includes('ц') ? 'ев' : 'ов');
                        if (word.endsWith('и')) return word.slice(0, -1) + 'ей';
                        if (word.endsWith('ар') || word.endsWith('айн')) return word;
                        return word + 'ов';
                    }
                    if (gender === 'м') {
                        if (word === 'Мандос') return 'Мандоса';
                        if (word.endsWith('ь')) return word.slice(0, -1) + 'я';
                        if (word.endsWith('й')) return word.slice(0, -1) + 'я';
                        if (word.endsWith('ор') || word.endsWith('ир') || word.endsWith('ар')) return word + 'а';
                        return word + 'а';
                    }
                    if (gender === 'ж') {
                        if (word === 'Галадриэль') return 'Галадриэли';
                        if (word.endsWith('ость')) return word.slice(0, -4) + 'ости';
                        if (word.endsWith('ия')) return word.slice(0, -2) + 'ии';
                        if (word.endsWith('ка')) return word.slice(0, -2) + 'ки';
                        if (word.endsWith('ь')) return word.slice(0, -1) + 'и';
                        if (word.endsWith('а')) return word.slice(0, -1) + 'ы';
                        if (word.endsWith('я')) return word.slice(0, -1) + 'и';
                    }
                    if (gender === 'с') {
                        if (word.endsWith('ие')) return word.slice(0, -2) + 'ия';
                        if (word.endsWith('ье')) return word.slice(0, -2) + 'ья';
                        if (word.endsWith('о')) return word.slice(0, -1) + 'а';
                        if (word.endsWith('е')) return word.slice(0, -1) + 'я';
                        if (word.endsWith('мя')) return word.slice(0, -2) + 'мени';
                    }
                    break;
                    
                case 'dat': // Дательный падеж
                    if (gender === 'мн') {
                        if (word === 'Люди') return 'Людям';
                        if (word.endsWith('ы')) return word.slice(0, -1) + 'ам';
                        if (word.endsWith('и')) return word.slice(0, -1) + 'ям';
                        if (word.endsWith('ар') || word.endsWith('айн')) return word + 'ам';
                        return word + 'ам';
                    }
                    if (gender === 'м') {
                        if (word === 'Мандос') return 'Мандосу';
                        if (word.endsWith('ь')) return word.slice(0, -1) + 'ю';
                        if (word.endsWith('й')) return word.slice(0, -1) + 'ю';
                        return word + 'у';
                    }
                    if (gender === 'ж') {
                        if (word === 'Галадриэль') return 'Галадриэли';
                        if (word.endsWith('ость')) return word.slice(0, -4) + 'ости';
                        if (word.endsWith('ия')) return word.slice(0, -2) + 'ии';
                        if (word.endsWith('ь')) return word.slice(0, -1) + 'и';
                        if (word.endsWith('а')) return word.slice(0, -1) + 'е';
                        if (word.endsWith('я')) return word.slice(0, -1) + 'е';
                    }
                    if (gender === 'с') {
                        if (word.endsWith('ие')) return word.slice(0, -2) + 'ию';
                        if (word.endsWith('ье')) return word.slice(0, -2) + 'ью';
                        if (word.endsWith('о')) return word.slice(0, -1) + 'у';
                        if (word.endsWith('е')) return word.slice(0, -1) + 'ю';
                        if (word.endsWith('мя')) return word.slice(0, -2) + 'мени';
                    }
                    break;
                    
                case 'acc': // Винительный падеж
                    if (gender === 'мн') return animate ? decline(wordObj, 'gen') : word;
                    if (gender === 'м') return animate ? decline(wordObj, 'gen') : word;
                    if (gender === 'ж') {
                        if (word === 'Галадриэль') return 'Галадриэль';
                        if (word.endsWith('ость')) return word;
                        if (word.endsWith('ь')) return word;
                        if (word.endsWith('а')) return word.slice(0, -1) + 'у';
                        if (word.endsWith('я')) return word.slice(0, -1) + 'ю';
                        if (word.endsWith('ия')) return word.slice(0, -2) + 'ию';
                    }
                    if (gender === 'с') return word;
                    break;
                    
                case 'inst': // Творительный падеж
                    if (gender === 'мн') {
                        if (word === 'Люди') return 'Людьми';
                        if (word.endsWith('ы')) return word.slice(0, -1) + 'ами';
                        if (word.endsWith('и')) return word.slice(0, -1) + 'ями';
                        return word + 'ами';
                    }
                    if (gender === 'м') {
                        if (word === 'Мандос') return 'Мандосом';
                        if (word.endsWith('ь')) return word.slice(0, -1) + 'ем';
                        if (word.endsWith('й')) return word.slice(0, -1) + 'ем';
                        return word + 'ом';
                    }
                    if (gender === 'ж') {
                        if (word === 'Галадриэль') return 'Галадриэлью';
                        if (word.endsWith('ость')) return word.slice(0, -4) + 'остью';
                        if (word.endsWith('ь')) return word + 'ю';
                        if (word.endsWith('а')) return word.slice(0, -1) + 'ой';
                        if (word.endsWith('я')) return word.slice(0, -1) + 'ей';
                        if (word.endsWith('ия')) return word.slice(0, -2) + 'ией';
                    }
                    if (gender === 'с') {
                        if (word.endsWith('ие')) return word.slice(0, -2) + 'ием';
                        if (word.endsWith('о')) return word.slice(0, -1) + 'ом';
                        if (word.endsWith('е')) return word.slice(0, -1) + 'ем';
                        if (word.endsWith('мя')) return word.slice(0, -2) + 'менем';
                    }
                    break;
                    
                case 'prep': // Предложный падеж
                    if (gender === 'мн') {
                        if (word === 'Люди') return 'Людях';
                        if (word.endsWith('ы')) return word.slice(0, -1) + 'ах';
                        if (word.endsWith('и')) return word.slice(0, -1) + 'ях';
                        return word + 'ах';
                    }
                    if (gender === 'м') {
                        if (word === 'Мандос') return 'Мандосе';
                        if (word === 'Шир') return 'Шире';
                        if (word.endsWith('ий')) return word.slice(0, -2) + 'ии';
                        if (word.endsWith('ь')) return word.slice(0, -1) + 'е';
                        if (word.endsWith('й')) return word.slice(0, -1) + 'е';
                        return word + 'е';
                    }
                    if (gender === 'ж') {
                        if (word === 'Галадриэль') return 'Галадриэли';
                        if (word.endsWith('ость')) return word.slice(0, -4) + 'ости';
                        if (word.endsWith('ия')) return word.slice(0, -2) + 'ии';
                        if (word.endsWith('ь')) return word.slice(0, -1) + 'и';
                        if (word.endsWith('а')) return word.slice(0, -1) + 'е';
                        if (word.endsWith('я')) return word.slice(0, -1) + 'е';
                    }
                    if (gender === 'с') {
                        if (word === 'Средиземье') return 'Средиземье';
                        if (word.endsWith('ие')) return word.slice(0, -2) + 'ии';
                        if (word.endsWith('ье')) return word.slice(0, -2) + 'ье';
                        if (word.endsWith('о')) return word.slice(0, -1) + 'е';
                        if (word.endsWith('е')) return word.slice(0, -1) + 'и';
                        if (word.endsWith('мя')) return word.slice(0, -2) + 'мени';
                    }
                    break;
            }
            
            return word;
        }

        // Улучшенная функция согласования прилагательного
        function agreeAdjective(adjBase, gender, caseType = 'nom', plural = false) {
            const genderKey = plural ? 'мн' : gender;
            
            // Мягкие основы
            if (adjBase.endsWith('н')) {
                const softEndings = {
                    nom: {м: 'ий', ж: 'яя', с: 'ее', мн: 'ие'},
                    gen: {м: 'его', ж: 'ей', с: 'его', мн: 'их'},
                    dat: {м: 'ему', ж: 'ей', с: 'ему', мн: 'им'},
                    acc: {м: 'ий', ж: 'юю', с: 'ее', мн: 'ие'},
                    inst: {м: 'им', ж: 'ей', с: 'им', мн: 'ими'},
                    prep: {м: 'ем', ж: 'ей', с: 'ем', мн: 'их'}
                };
                return adjBase + softEndings[caseType][genderKey];
            }
            
            // Основы на -ическ
            if (adjBase.endsWith('ическ')) {
                const endings = {
                    nom: {м: 'ий', ж: 'ая', с: 'ое', мн: 'ие'},
                    gen: {м: 'ого', ж: 'ой', с: 'ого', мн: 'их'},
                    dat: {м: 'ому', ж: 'ой', с: 'ому', мн: 'им'},
                    acc: {м: 'ий', ж: 'ую', с: 'ое', мн: 'ие'},
                    inst: {м: 'им', ж: 'ой', с: 'им', мн: 'ими'},
                    prep: {м: 'ом', ж: 'ой', с: 'ом', мн: 'их'}
                };
                return adjBase + endings[caseType][genderKey];
            }
            
            // Твердые основы
            const hardEndings = {
                nom: {м: 'ый', ж: 'ая', с: 'ое', мн: 'ые'},
                gen: {м: 'ого', ж: 'ой', с: 'ого', мн: 'ых'},
                dat: {м: 'ому', ж: 'ой', с: 'ому', мн: 'ым'},
                acc: {м: 'ый', ж: 'ую', с: 'ое', мн: 'ые'},
                inst: {м: 'ым', ж: 'ой', с: 'ым', мн: 'ыми'},
                prep: {м: 'ом', ж: 'ой', с: 'ом', мн: 'ых'}
            };
            
            return adjBase + hardEndings[caseType][genderKey];
        }

        // Функция для получения случайного элемента

            return array[Math.floor(Math.random() * array.length)];
        }

            quote = quote.charAt(0).toUpperCase() + quote.slice(1) + '.';
            
            // Генерируем псевдо-автора
            const authorTemplates = [
                () => {
                    const char = getRandomElement(tolkienElements.characters);
                    const concept = getRandomElement(philosophicalConcepts.subjects);
                    return `— ${char.word}, "${getRandomElement(['Трактат', 'Размышления', 'Диалоги', 'Записки', 'Хроники'])} о ${decline(concept, 'prep')}"`;
                },
                () => {
                    const place = getRandomElement(tolkienElements.places);
                    return `— Из "${getRandomElement(['Анналов', 'Летописей', 'Свитков', 'Книги', 'Песни'])} ${decline(place, 'gen')}"`;
                },
                () => {

                    return `— ${getRandomElement(['Древняя мудрость', 'Изречение', 'Пословица', 'Поговорка'])} ${decline(race, 'gen')}`;
                },
                () => {

            }, 10);
        }
