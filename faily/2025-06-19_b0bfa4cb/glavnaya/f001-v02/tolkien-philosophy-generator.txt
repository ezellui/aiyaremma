        // Функция склонения существительных
        function decline(word, caseType, gender) {
            // Специальные случаи для имен собственных из Толкина
            const indeclinable = ['Ауле', 'Ороме', 'Тулкас', 'Мандос', 'Ульмо', 'Манвэ', 'Лориэн', 'Дориат'];
            if (indeclinable.includes(word)) return word;
            
            // Родительный падеж
            if (caseType === 'gen') {
                if (gender === 'мн') {
                    if (word.endsWith('ы')) return word.slice(0, -1);
                    if (word.endsWith('и')) return word.slice(0, -1) + 'ей';
                    if (word.endsWith('ар')) return word;
                    return word + 'ов';
                }
                if (gender === 'м') {
                    if (word.endsWith('ь')) return word.slice(0, -1) + 'я';
                    if (word.endsWith('й')) return word.slice(0, -1) + 'я';
                    if (word.endsWith('изм')) return word + 'а';
                    return word + 'а';
                }
                if (gender === 'ж') {
                    if (word.endsWith('ость')) return word.slice(0, -4) + 'ости';
                    if (word.endsWith('ия')) return word.slice(0, -2) + 'ии';
                    if (word.endsWith('ка')) return word.slice(0, -2) + 'ки';
                    if (word.endsWith('ь')) return word.slice(0, -1) + 'и';
                    if (word.endsWith('а')) return word.slice(0, -1) + 'ы';
                    if (word.endsWith('я')) return word.slice(0, -1) + 'и';
                }
                if (gender === 'с') {
                    if (word.endsWith('ие')) return word.slice(0, -2) + 'ия';
                    if (word.endsWith('о')) return word.slice(0, -1) + 'а';
                    if (word.endsWith('е')) return word.slice(0, -1) + 'я';
                    if (word.endsWith('мя')) return word.slice(0, -2) + 'мени';
                }
            }
            
            // Дательный падеж
            if (caseType === 'dat') {
                if (gender === 'мн') {
                    if (word.endsWith('ы')) return word.slice(0, -1) + 'ам';
                    if (word.endsWith('и')) return word.slice(0, -1) + 'ям';
                    if (word.endsWith('ар')) return word + 'ам';
                    return word + 'ам';
                }
                if (gender === 'м') {
                    if (word.endsWith('ь')) return word.slice(0, -1) + 'ю';
                    if (word.endsWith('й')) return word.slice(0, -1) + 'ю';
                    return word + 'у';
                }
                if (gender === 'ж') {
                    if (word.endsWith('ость')) return word.slice(0, -4) + 'ости';
                    if (word.endsWith('ия')) return word.slice(0, -2) + 'ии';
                    if (word.endsWith('ь')) return word.slice(0, -1) + 'и';
                    if (word.endsWith('а')) return word.slice(0, -1) + 'е';
                    if (word.endsWith('я')) return word.slice(0, -1) + 'е';
                }
                if (gender === 'с') {
                    if (word.endsWith('ие')) return word.slice(0, -2) + 'ию';
                    if (word.endsWith('о')) return word.slice(0, -1) + 'у';
                    if (word.endsWith('е')) return word.slice(0, -1) + 'ю';
                    if (word.endsWith('мя')) return word.slice(0, -2) + 'мени';
                }
            }
            
            // Винительный падеж
            if (caseType === 'acc') {
                if (gender === 'мн') return word;
                if (gender === 'м') return word; // для неодушевленных
                if (gender === 'ж') {
                    if (word.endsWith('ость')) return word;
                    if (word.endsWith('ь')) return word;
                    if (word.endsWith('а')) return word.slice(0, -1) + 'у';
                    if (word.endsWith('я')) return word.slice(0, -1) + 'ю';
                }
                if (gender === 'с') return word;
            }
            
            // Предложный падеж
            if (caseType === 'prep') {
                if (gender === 'мн') {
                    if (word.endsWith('ы')) return word.slice(0, -1) + 'ах';
                    if (word.endsWith('и')) return word.slice(0, -1) + 'ях';
                    return word + 'ах';
                }
                if (gender === 'м') {
                    if (word.endsWith('ь')) return word.slice(0, -1) + 'е';
                    if (word.endsWith('й')) return word.slice(0, -1) + 'е';
                    return word + 'е';
                }
                if (gender === 'ж') {
                    if (word.endsWith('ость')) return word.slice(0, -4) + 'ости';
                    if (word.endsWith('ия')) return word.slice(0, -2) + 'ии';
                    if (word.endsWith('ь')) return word.slice(0, -1) + 'и';
                    if (word.endsWith('а')) return word.slice(0, -1) + 'е';
                    if (word.endsWith('я')) return word.slice(0, -1) + 'е';
                }
                if (gender === 'с') {
                    if (word.endsWith('ие')) return word.slice(0, -2) + 'ии';
                    if (word.endsWith('о')) return word.slice(0, -1) + 'е';
                    if (word.endsWith('е')) return word.slice(0, -1) + 'и';
                    if (word.endsWith('мя')) return word.slice(0, -2) + 'мени';
                }
            }
            
            return word;
        }

        // Функция согласования прилагательного с существительным
        function agreeAdjective(adjBase, gender, caseType = 'nom') {
            const endings = {
                nom: {м: 'ый', ж: 'ая', с: 'ое', мн: 'ые'},
                gen: {м: 'ого', ж: 'ой', с: 'ого', мн: 'ых'},
                dat: {м: 'ому', ж: 'ой', с: 'ому', мн: 'ым'},
                acc: {м: 'ый', ж: 'ую', с: 'ое', мн: 'ые'},
                prep: {м: 'ом', ж: 'ой', с: 'ом', мн: 'ых'}
            };
            
            if (adjBase.endsWith('ическ')) {
                const softEndings = {
                    nom: {м: 'ий', ж: 'ая', с: 'ое', мн: 'ие'},
                    gen: {м: 'ого', ж: 'ой', с: 'ого', мн: 'их'},
                    dat: {м: 'ому', ж: 'ой', с: 'ому', мн: 'им'},
                    acc: {м: 'ий', ж: 'ую', с: 'ое', мн: 'ие'},
                    prep: {м: 'ом', ж: 'ой', с: 'ом', мн: 'их'}
                };
                return adjBase + softEndings[caseType][gender];
            }
            
            return adjBase + endings[caseType][gender];
        }

                template: '{race1:nom} постигают {subj:acc} иначе, чем {race2:nom}, ибо их {adj:nom} природа {pred}',
                structure: ['race1', 'race2', 'subj', 'adj', 'pred']
            },
            {
                template: 'В {place:prep} {subj:nom} {pred} подобно тому, как {artifact:nom} воплощает {concept:acc}',
                structure: ['place', 'subj', 'pred', 'artifact', 'concept']
            },
            {
                template: '{char:nom} учил, что {adj:nom} {subj:nom} {concept:gen} {pred} только через {concept2:acc}',
                structure: ['char', 'adj', 'subj', 'concept', 'pred', 'concept2']
            },
            {
                template: 'Когда {artifact:nom} был создан в {place:prep}, {subj:nom} навсегда изменило свою {adj:acc} сущность',
                structure: ['artifact', 'place', 'subj', 'adj']
            },
            {
                template: '{concept:nom} {race:gen} {pred} в {subj:acc}, тогда как {race2:nom} ищут {concept2:acc}',
                structure: ['concept', 'race', 'pred', 'subj', 'race2', 'concept2']
            },
            {
                template: 'Мудрость {char:gen} гласит: "{subj:nom} и {concept:nom} — две стороны {artifact:gen}"',
                structure: ['char', 'subj', 'concept', 'artifact']
            },
            {
                template: 'В эпоху {concept:gen} {char:nom} познал, что {adj:nom} {subj:nom} {pred} в {place:prep}',
                structure: ['concept', 'char', 'adj', 'subj', 'pred', 'place']
            },
            {
                template: '{race:nom}, в отличие от {race2:gen}, понимают, что {subj:nom} {pred} через {concept:acc}',
                structure: ['race', 'race2', 'subj', 'pred', 'concept']
            },
            {
                template: 'Подобно {artifact:dat}, {adj:nom} {subj:nom} {char:gen} трансформирует {concept:acc} в {place:prep}',
                structure: ['artifact', 'adj', 'subj', 'char', 'concept', 'place']
            },
            {
                template: 'Не {subj:nom} определяет {concept:acc}, но {concept2:nom} {pred} через {char:acc}',
                structure: ['subj', 'concept', 'concept2', 'pred', 'char']
            }
        ];

        // Генерация псевдофилософского высказывания
        function generateQuote() {
            const template = getRandomElement(templates);
            let quote = template.template;
            
            // Заполняем шаблон
            const matches = quote.match(/{([^}]+)}/g);
            if (matches) {
                matches.forEach(match => {
                    const parts = match.slice(1, -1).split(':');
                    const type = parts[0];
                    const caseType = parts[1] || 'nom';
                    
                    let replacement = '';
                    let gender = '';
                    
                    // Выбираем элемент и определяем его род
                    switch(type) {
                        case 'adj':
                            const adj = getRandomElement(philosophicalConcepts.attributes);
                            // Найдем род связанного существительного
                            const subjMatch = quote.match(/{subj:([^}]+)}/);
                            if (subjMatch) {
                                const subj = getRandomElement(philosophicalConcepts.subjects);
                                gender = subj.gender;
                                replacement = agreeAdjective(adj, gender, caseType);
                            }
                            break;
                            
                        case 'subj':
                            const subj = getRandomElement(philosophicalConcepts.subjects);
                            replacement = decline(subj.word, caseType, subj.gender);
                            break;
                            
                        case 'pred':
                            const pred = getRandomElement(philosophicalConcepts.predicates);
                            replacement = pred.word;
                            break;
                            
                        case 'char':
                        case 'char2':
                            const char = getRandomElement(tolkienElements.characters);
                            replacement = decline(char.word, caseType, char.gender);
                            break;
                            
                        case 'place':
                            const place = getRandomElement(tolkienElements.places);
                            replacement = decline(place.word, caseType, place.gender);
                            break;
                            
                        case 'artifact':
                            const artifact = getRandomElement(tolkienElements.artifacts);
                            replacement = decline(artifact.word, caseType, artifact.gender);
                            break;
                            
                        case 'concept':
                        case 'concept2':
                            const concept = getRandomElement(tolkienElements.concepts);
                            replacement = decline(concept.word, caseType, concept.gender);
                            break;
                            
                        case 'race':
                        case 'race1':
                        case 'race2':
                            const race = getRandomElement(tolkienElements.races);
                            replacement = decline(race.word, caseType, race.gender);
                            break;
                            
                        case 'obj':
                            // Объект зависит от управления глагола
                            const predMatch = quote.match(/{pred}/);
                            if (predMatch) {
                                const predObj = getRandomElement(philosophicalConcepts.predicates);
                                const objConcept = getRandomElement(tolkienElements.concepts);
                                replacement = decline(objConcept.word, predObj.case === 'case' ? 'acc' : predObj.case, objConcept.gender);
                            }
                            break;
                    }
                    
                    quote = quote.replace(match, replacement);
                });
            }
            
            // Делаем первую букву заглавной
            quote = quote.charAt(0).toUpperCase() + quote.slice(1);
            
            // Генерируем псевдо-автора
            const authorTemplates = [
                () => {
                    const char = getRandomElement(tolkienElements.characters);
                    const concept = getRandomElement(philosophicalConcepts.subjects);
                    return `— ${char.word}, "${getRandomElement(['Трактат', 'Размышления', 'Диалоги', 'Записки', 'Хроники'])} о ${decline(concept.word, 'prep', concept.gender)}"`;
                },
                () => {
                    const place = getRandomElement(tolkienElements.places);
                    return `— Из "${getRandomElement(['Анналов', 'Летописей', 'Свитков', 'Книги', 'Песни'])} ${decline(place.word, 'gen', place.gender)}"`;
                },
                () => {
                    const race = getRandomElement(tolkienElements.races);
                    return `— ${getRandomElement(['Древняя мудрость', 'Изречение', 'Пословица', 'Поговорка'])} ${decline(race.word, 'gen', race.gender)}`;
                },
                () => {
                    const char = getRandomElement(tolkienElements.characters);
                    return `— ${char.word}, ${getRandomElement(['Первая', 'Вторая', 'Третья', 'Четвертая'])} Эпоха`;
                },
                () => {
                    const place = getRandomElement(tolkienElements.places);
                    return `— Надпись на ${getRandomElement(['стенах', 'вратах', 'скрижалях', 'камне'])} ${decline(place.word, 'gen', place.gender)}`;
                }
            ];
            
            document.getElementById('quote').textContent = quote;
            document.getElementById('author').textContent = getRandomElement(authorTemplates)();
            
            // Анимация появления
            const quoteContainer = document.querySelector('.quote-container');
            quoteContainer.style.animation = 'none';
            setTimeout(() => {
                quoteContainer.style.animation = 'fadeIn 0.5s ease-in';
            }, 10);
        }
        
        // Генерируем первое высказывание при загрузке
        window.onload = () => generateQuote();
    </script>
