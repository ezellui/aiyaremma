        function createNetwork(container, nodes, edges) {
            const data = {
                nodes: nodes,
                edges: edges
            };

                nodes: {
                    shape: 'box',
                    margin: 10,
                    widthConstraint: {
                        maximum: 120
                    },
                    font: {
                        size: 14
                    }
                },
                edges: {
                    arrows: {
                        to: { enabled: true, scaleFactor: 1 }
                    },
                    font: {
                        size: 12,
                        align: 'middle'
                    },
                    length: 200
                },
                physics: {
                    stabilization: true,
                    barnesHut: {
                        gravitationalConstant: -5000,
                        springLength: 130,
                        springConstant: 0.04
                    }
                },
                layout: {
                    improvedLayout: true
                }
            };
