# 3.2 Семантические связи

## Описание
Система семантических связей визуализирует смысловые отношения между фрагментами сознания, создавая динамическую сеть взаимосвязанных концептов.

### 3. JavaScript для семантических связей (добавить после системы многомерной навигации)

```javascript
// === СИСТЕМА СЕМАНТИЧЕСКИХ СВЯЗЕЙ ===

class SemanticConnections {
    constructor() {
        this.connections = new Map();
        this.clusters = new Map();
        this.semanticGraph = new Map();
        this.activeConnections = new Set();
        
        this.semanticGroups = {
            philosophical: ['бытие', 'сущность', 'феномен', 'ноумен', 'трансцендент', 'имманент', 'экзистенция', 'онтология', 'эпистемология'],
            quantum: ['квант', 'суперпозиция', 'запутанность', 'коллапс', 'вероятность', 'неопределенность', 'наблюдатель', 'волна', 'частица'],
            psychological: ['сознание', 'подсознание', 'эго', 'ид', 'архетип', 'тень', 'персона', 'анима', 'анимус', 'самость'],
            mathematical: ['функция', 'интеграл', 'производная', 'предел', 'бесконечность', 'множество', 'топология', 'фрактал'],
            temporal: ['время', 'прошлое', 'будущее', 'настоящее', 'вечность', 'мгновение', 'длительность', 'момент'],
            linguistic: ['язык', 'слово', 'смысл', 'знак', 'символ', 'означающее', 'означаемое', 'семиотика', 'грамматика'],
            mystical: ['пустота', 'единство', 'дао', 'медитация', 'просветление', 'нирвана', 'сатори', 'самадхи'],
            chaos: ['хаос', 'порядок', 'энтропия', 'бифуркация', 'аттрактор', 'турбулентность', 'диссипация']
        };
        
        this.connectionStrength = {
            strong: 0.8,
            medium: 0.5,
            weak: 0.2
        };
        
        this.settings = {
            maxConnections: 50,
            connectionThreshold: 0.3,
            clusterThreshold: 3,
            evolutionInterval: 5000,
            visualizeClusters: true
        };
        
        this.init();
    }
    
    init() {
        this.analyzeExistingFragments();
        this.setupFragmentObserver();
        this.startConnectionEvolution();
        this.setupInteractionHandlers();
        this.createSemanticPanel();
    }
    
    analyzeExistingFragments() {
        const fragments = document.querySelectorAll('.consciousness-fragment');
        
        fragments.forEach(fragment => {
            this.analyzeFragment(fragment);
        });
        
        this.updateConnections();
        this.identifyClusters();
    }
    
    analyzeFragment(fragment) {
        const text = fragment.textContent.toLowerCase();
        const semanticTags = [];
        
        // Определяем семантические теги
        for (const [group, keywords] of Object.entries(this.semanticGroups)) {
            for (const keyword of keywords) {
                if (text.includes(keyword)) {
                    semanticTags.push(group);
                    break;
                }
            }
        }
        
        // Сохраняем семантическую информацию
        fragment.dataset.semanticTags = semanticTags.join(',');
        
        // Добавляем в граф
        this.semanticGraph.set(fragment, {
            tags: semanticTags,
            text: text,
            position: this.getElementPosition(fragment),
            connections: new Set()
        });
    }
    
    getElementPosition(element) {
        const rect = element.getBoundingClientRect();
        return {
            x: rect.left + rect.width / 2,
            y: rect.top + rect.height / 2
        };
    }
    
    updateConnections() {
        // Очищаем старые соединения
        this.clearConnections();
        
        const fragments = Array.from(this.semanticGraph.keys());
        
        // Создаем связи между семантически близкими элементами
        for (let i = 0; i < fragments.length; i++) {
            for (let j = i + 1; j < fragments.length; j++) {
                const strength = this.calculateSemanticSimilarity(fragments[i], fragments[j]);
                
                if (strength > this.settings.connectionThreshold) {
                    this.createConnection(fragments[i], fragments[j], strength);
                }
            }
        }
        
        this.updateSemanticPanel();
    }
    
    calculateSemanticSimilarity(fragment1, fragment2) {
        const data1 = this.semanticGraph.get(fragment1);
        const data2 = this.semanticGraph.get(fragment2);
        
        if (!data1 || !data2) return 0;
        
        // Сравнение по тегам
        const commonTags = data1.tags.filter(tag => data2.tags.includes(tag));
        const tagSimilarity = commonTags.length / Math.max(data1.tags.length, data2.tags.length, 1);
        
        // Сравнение по словам
        const words1 = data1.text.split(/\s+/);
        const words2 = data2.text.split(/\s+/);
        const commonWords = words1.filter(word => words2.includes(word) && word.length > 3);
        const wordSimilarity = commonWords.length / Math.max(words1.length, words2.length, 1);
        
        // Учитываем расстояние между элементами
        const distance = Math.sqrt(
            Math.pow(data1.position.x - data2.position.x, 2) +
            Math.pow(data1.position.y - data2.position.y, 2)
        );
        const distanceFactor = 1 / (1 + distance / 500);
        
        return (tagSimilarity * 0.5 + wordSimilarity * 0.3 + distanceFactor * 0.2);
    }
    
    createConnection(fragment1, fragment2, strength) {
        const line = document.createElement('div');
        line.className = 'semantic-connection';
        
        // Определяем силу связи
        if (strength > 0.7) {
            line.classList.add('strong');
        } else if (strength < 0.4) {
            line.classList.add('weak');
        }
        
        // Позиционируем линию
        this.positionConnection(line, fragment1, fragment2);
        
        document.getElementById('container').appendChild(line);
        
        // Сохраняем информацию о связи
        const connectionId = `${fragment1.id || Math.random()}_${fragment2.id || Math.random()}`;
        this.connections.set(connectionId, {
            line: line,
            fragment1: fragment1,
            fragment2: fragment2,
            strength: strength
        });
        
        // Обновляем граф
        this.semanticGraph.get(fragment1).connections.add(fragment2);
        this.semanticGraph.get(fragment2).connections.add(fragment1);
        
        // Добавляем интерактивность
        this.setupConnectionInteraction(line, fragment1, fragment2);
    }
    
    positionConnection(line, element1, element2) {
        const pos1 = this.getElementPosition(element1);
        const pos2 = this.getElementPosition(element2);
        
        const distance = Math.sqrt(
            Math.pow(pos2.x - pos1.x, 2) +
            Math.pow(pos2.y - pos1.y, 2)
        );
        
        const angle = Math.atan2(pos2.y - pos1.y, pos2.x - pos1.x);
        
        line.style.width = distance + 'px';
        line.style.left = pos1.x + 'px';
        line.style.top = pos1.y + 'px';
        line.style.transform = `rotate(${angle}rad)`;
    }
    
    clearConnections() {
        this.connections.forEach(connection => {
            connection.line.remove();
        });
        this.connections.clear();
        
        // Очищаем связи в графе
        this.semanticGraph.forEach(data => {
            data.connections.clear();
        });
    }
    
    identifyClusters() {
        this.clusters.clear();
        const visited = new Set();
        let clusterIndex = 0;
        
        // Поиск в глубину для определения кластеров
        this.semanticGraph.forEach((data, fragment) => {
            if (!visited.has(fragment)) {
                const cluster = new Set();
                this.dfs(fragment, visited, cluster);
                
                if (cluster.size >= this.settings.clusterThreshold) {
                    this.clusters.set(clusterIndex++, cluster);
                    
                    if (this.settings.visualizeClusters) {
                        this.visualizeCluster(cluster);
                    }
                }
            }
        });
    }
    
    dfs(fragment, visited, cluster) {
        visited.add(fragment);
        cluster.add(fragment);
        
        const data = this.semanticGraph.get(fragment);
        if (data) {
            data.connections.forEach(connected => {
                if (!visited.has(connected)) {
                    this.dfs(connected, visited, cluster);
                }
            });
        }
    }
    
    visualizeCluster(cluster) {
        const positions = Array.from(cluster).map(fragment => 
            this.getElementPosition(fragment)
        );
        
        // Находим центр кластера
        const center = {
            x: positions.reduce((sum, pos) => sum + pos.x, 0) / positions.length,
            y: positions.reduce((sum, pos) => sum + pos.y, 0) / positions.length
        };
        
        // Находим радиус
        const radius = Math.max(...positions.map(pos => 
            Math.sqrt(Math.pow(pos.x - center.x, 2) + Math.pow(pos.y - center.y, 2))
        )) + 50;
        
        // Создаем визуализацию кластера
        const clusterEl = document.createElement('div');
        clusterEl.className = 'semantic-cluster';
        clusterEl.style.width = radius * 2 + 'px';
        clusterEl.style.height = radius * 2 + 'px';
        clusterEl.style.left = (center.x - radius) + 'px';
        clusterEl.style.top = (center.y - radius) + 'px';
        
        document.getElementById('container').appendChild(clusterEl);
        
        // Создаем семантическое поле
        const field = document.createElement('div');
        field.className = 'semantic-field';
        field.style.width = radius * 2.5 + 'px';
        field.style.height = radius * 2.5 + 'px';
        field.style.left = (center.x - radius * 1.25) + 'px';
        field.style.top = (center.y - radius * 1.25) + 'px';
        
        document.getElementById('container').appendChild(field);
    }
    
    setupFragmentObserver() {
        // Отслеживаем новые фрагменты
        const observer = new MutationObserver((mutations) => {
            mutations.forEach(mutation => {
                mutation.addedNodes.forEach(node => {
                    if (node.classList && node.classList.contains('consciousness-fragment')) {
                        setTimeout(() => {
                            this.analyzeFragment(node);
                            this.updateConnections();
                            this.identifyClusters();
                        }, 100);
                    }
                });
                
                mutation.removedNodes.forEach(node => {
                    if (node.classList && node.classList.contains('consciousness-fragment')) {
                        this.semanticGraph.delete(node);
                        this.updateConnections();
                        this.identifyClusters();
                    }
                });
            });
        });
        
        observer.observe(document.getElementById('container'), {
            childList: true,
            subtree: true
        });
        
        // Отслеживаем изменения позиций
        setInterval(() => {
            let needsUpdate = false;
            
            this.semanticGraph.forEach((data, fragment) => {
                const newPos = this.getElementPosition(fragment);
                if (Math.abs(newPos.x - data.position.x) > 5 || 
                    Math.abs(newPos.y - data.position.y) > 5) {
                    data.position = newPos;
                    needsUpdate = true;
                }
            });
            
            if (needsUpdate) {
                this.updateConnectionPositions();
            }
        }, 500);
    }
    
    updateConnectionPositions() {
        this.connections.forEach(connection => {
            this.positionConnection(
                connection.line,
                connection.fragment1,
                connection.fragment2
            );
        });
    }
    
    startConnectionEvolution() {
        setInterval(() => {
            // Эволюция связей - усиление или ослабление
            this.connections.forEach((connection, id) => {
                // Случайная мутация силы связи
                if (Math.random() < 0.2) {
                    const mutation = (Math.random() - 0.5) * 0.2;
                    connection.strength = Math.max(0, Math.min(1, connection.strength + mutation));
                    
                    // Обновляем визуализацию
                    if (connection.strength < this.settings.connectionThreshold) {
                        connection.line.style.opacity = '0';
                        setTimeout(() => {
                            connection.line.remove();
                            this.connections.delete(id);
                        }, 500);
                    } else {
                        connection.line.style.opacity = connection.strength;
                    }
                }
            });
            
            // Создание новых связей
            if (Math.random() < 0.3 && this.connections.size < this.settings.maxConnections) {
                this.discoverNewConnections();
            }
        }, this.settings.evolutionInterval);
    }
    
    discoverNewConnections() {
        const fragments = Array.from(this.semanticGraph.keys());
        
        // Выбираем случайные пары для проверки
        for (let i = 0; i < 5; i++) {
            const fragment1 = fragments[Math.floor(Math.random() * fragments.length)];
            const fragment2 = fragments[Math.floor(Math.random() * fragments.length)];
            
            if (fragment1 !== fragment2) {
                const data1 = this.semanticGraph.get(fragment1);
                const data2 = this.semanticGraph.get(fragment2);
                
                if (data1 && data2 && !data1.connections.has(fragment2)) {
                    const strength = this.calculateSemanticSimilarity(fragment1, fragment2);
                    
                    if (strength > this.settings.connectionThreshold * 0.8) {
                        this.createConnection(fragment1, fragment2, strength);
                        
                        // Визуализация эволюции
                        const pos1 = data1.position;
                        const pos2 = data2.position;
                        const midX = (pos1.x + pos2.x) / 2;
                        const midY = (pos1.y + pos2.y) / 2;
                        
                        const evolution = document.createElement('div');
                        evolution.className = 'connection-evolution';
                        evolution.style.left = (midX - 10) + 'px';
                        evolution.style.top = (midY - 10) + 'px';
                        
                        document.getElementById('container').appendChild(evolution);
                        setTimeout(() => evolution.remove(), 3000);
                    }
                }
            }
        }
    }
    
    setupInteractionHandlers() {
        // Подсветка связанных элементов при наведении
        document.addEventListener('mouseover', (e) => {
            if (e.target.classList.contains('consciousness-fragment')) {
                this.highlightConnections(e.target);
            }
        });
        
        document.addEventListener('mouseout', (e) => {
            if (e.target.classList.contains('consciousness-fragment')) {
                this.unhighlightConnections();
            }
        });
        
        // Показ панели при клике
        document.addEventListener('click', (e) => {
            if (e.target.classList.contains('consciousness-fragment')) {
                this.showSemanticInfo(e.target);
            }
        });
    }
    
    highlightConnections(fragment) {
        const data = this.semanticGraph.get(fragment);
        if (!data) return;
        
        // Подсвечиваем сам элемент
        fragment.classList.add('semantically-active');
        
        // Подсвечиваем связанные элементы
        data.connections.forEach(connected => {
            connected.classList.add('semantically-related');
        });
        
        // Активируем линии связей
        this.connections.forEach(connection => {
            if (connection.fragment1 === fragment || connection.fragment2 === fragment) {
                connection.line.classList.add('active');
                this.activeConnections.add(connection);
            }
        });
    }
    
    unhighlightConnections() {
        // Убираем подсветку элементов
        document.querySelectorAll('.semantically-active, .semantically-related').forEach(el => {
            el.classList.remove('semantically-active', 'semantically-related');
        });
        
        // Деактивируем линии
        this.activeConnections.forEach(connection => {
            connection.line.classList.remove('active');
        });
        this.activeConnections.clear();
    }
    
    setupConnectionInteraction(line, fragment1, fragment2) {
        // Создаем узлы на концах связи
        const node1 = document.createElement('div');
        const node2 = document.createElement('div');
        
        node1.className = 'semantic-node';
        node2.className = 'semantic-node';
        
        const pos1 = this.getElementPosition(fragment1);
        const pos2 = this.getElementPosition(fragment2);
        
        node1.style.left = (pos1.x - 3) + 'px';
        node1.style.top = (pos1.y - 3) + 'px';
        node2.style.left = (pos2.x - 3) + 'px';
        node2.style.top = (pos2.y - 3) + 'px';
        
        // Определяем хабы (элементы с множеством связей)
        const data1 = this.semanticGraph.get(fragment1);
        const data2 = this.semanticGraph.get(fragment2);
        
        if (data1 && data1.connections.size > 5) {
            node1.classList.add('hub');
        }
        if (data2 && data2.connections.size > 5) {
            node2.classList.add('hub');
        }
        
        document.getElementById('container').appendChild(node1);
        document.getElementById('container').appendChild(node2);
        
        // Сохраняем узлы для обновления позиций
        line.dataset.node1 = node1;
        line.dataset.node2 = node2;
    }
    
    createSemanticPanel() {
        // Панель уже создана в HTML, настраиваем обработчики
        const panel = document.getElementById('semantic-panel');
        
        // Закрытие панели при клике вне её
        document.addEventListener('click', (e) => {
            if (!panel.contains(e.target) && 
                !e.target.classList.contains('consciousness-fragment')) {
                panel.classList.remove('visible');
            }
        });
    }
    
    showSemanticInfo(fragment) {
        const panel = document.getElementById('semantic-panel');
        const data = this.semanticGraph.get(fragment);
        
        if (!data) return;
        
        // Показываем теги
        const tagsContainer = document.getElementById('semantic-tags');
        tagsContainer.innerHTML = '';
        
        data.tags.forEach(tag => {
            const tagEl = document.createElement('span');
            tagEl.className = 'semantic-tag';
            tagEl.textContent = tag;
            tagsContainer.appendChild(tagEl);
        });
        
        // Добавляем информацию о связях
        const connectedTags = new Set();
        data.connections.forEach(connected => {
            const connectedData = this.semanticGraph.get(connected);
            if (connectedData) {
                connectedData.tags.forEach(tag => connectedTags.add(tag));
            }
        });
        
        if (connectedTags.size > 0) {
            const relatedHeader = document.createElement('div');
            relatedHeader.style.marginTop = '10px';
            relatedHeader.style.marginBottom = '5px';
            relatedHeader.textContent = 'Связанные концепты:';
            tagsContainer.appendChild(relatedHeader);
            
            connectedTags.forEach(tag => {
                if (!data.tags.includes(tag)) {
                    const tagEl = document.createElement('span');
                    tagEl.className = 'semantic-tag';
                    tagEl.style.opacity = '0.6';
                    tagEl.textContent = tag;
                    tagsContainer.appendChild(tagEl);
                }
            });
        }
        
        // Показываем панель
        panel.classList.add('visible');
    }
    
    updateSemanticPanel() {
        // Обновляем статистику
        document.getElementById('active-connections').textContent = this.connections.size;
        document.getElementById('cluster-count').textContent = this.clusters.size;
        
        // Рассчитываем семантическую плотность
        const totalPossibleConnections = (this.semanticGraph.size * (this.semanticGraph.size - 1)) / 2;
        const density = totalPossibleConnections > 0 ? 
            (this.connections.size / totalPossibleConnections * 100).toFixed(1) : 0;
        
        document.getElementById('semantic-density').textContent = density + '%';
    }
    
    // Метод для программного создания связи
    forceConnection(fragment1, fragment2, strength = 0.8) {
        if (fragment1 && fragment2 && fragment1 !== fragment2) {
            this.createConnection(fragment1, fragment2, strength);
        }
    }
    
    // Метод для анализа семантической сети
    analyzeNetwork() {
        const analysis = {
            totalNodes: this.semanticGraph.size,
            totalConnections: this.connections.size,
            clusters: this.clusters.size,
            averageConnections: 0,
            mostConnected: null,
            semanticGroups: {}
        };
        
        let maxConnections = 0;
        let totalConnections = 0;
        
        this.semanticGraph.forEach((data, fragment) => {
            totalConnections += data.connections.size;
            
            if (data.connections.size > maxConnections) {
                maxConnections = data.connections.size;
                analysis.mostConnected = fragment;
            }
            
            // Подсчет по группам
            data.tags.forEach(tag => {
                analysis.semanticGroups[tag] = (analysis.semanticGroups[tag] || 0) + 1;
            });
        });
        
        analysis.averageConnections = totalConnections / this.semanticGraph.size;
        
        return analysis;
    }
}

// Инициализируем систему семантических связей
const semanticConnections = new SemanticConnections();

// Экспортируем для глобального доступа
window.semanticConnections = semanticConnections;
```

## Использование

После добавления кода:

1. Между семантически связанными элементами появятся линии связи
2. При наведении на элемент подсветятся все связанные с ним фрагменты
3. Кластеры близких по смыслу элементов визуализируются окружностями
4. Связи эволюционируют со временем - усиливаются или ослабевают
5. Клик по элементу покажет панель с семантической информацией
