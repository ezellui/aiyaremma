# 3.3 Темпоральные аномалии

## Описание
Система темпоральных аномалий создает нелинейные временные эффекты: петли времени, обращение причинности, временные парадоксы и хроноклазмы.

### 3. JavaScript для темпоральных аномалий (добавить после системы семантических связей)

```javascript
// === СИСТЕМА ТЕМПОРАЛЬНЫХ АНОМАЛИЙ ===

class TemporalAnomalies {
    constructor() {
        this.timeLoops = new Map();
        this.paradoxes = new Set();
        this.temporalEchoes = new Map();
        this.causalChains = new Map();
        this.timeRate = 1.0;
        this.timeDirection = 1; // 1 = вперед, -1 = назад
        this.currentTimePosition = 0.5; // Позиция на временной линии
        
        this.activeEffects = {
            loops: 0,
            paradoxes: 0,
            rifts: 0,
            dilations: new Map()
        };
        
        this.settings = {
            maxLoops: 5,
            maxParadoxes: 3,
            echoDelay: 500,
            echoCount: 3,
            causalityRadius: 200
        };
        
        this.init();
    }
    
    init() {
        this.setupControls();
        this.startTimeFlow();
        this.setupCausalityTracking();
        this.createTemporalField();
    }
    
    setupControls() {
        // Кнопка создания временной петли
        document.getElementById('create-time-loop').addEventListener('click', () => {
            this.createTimeLoop();
        });
        
        // Кнопка обращения времени
        document.getElementById('reverse-time').addEventListener('click', () => {
            this.reverseTimeFlow();
        });
        
        // Кнопка создания парадокса
        document.getElementById('create-paradox').addEventListener('click', () => {
            this.createParadox();
        });
        
        // Кнопка временного разрыва
        document.getElementById('temporal-rift').addEventListener('click', () => {
            this.createTemporalRift();
        });
        
        // Кнопка хроноклазма
        document.getElementById('chronoclasm').addEventListener('click', () => {
            this.triggerChronoclasm();
        });
        
        // Отслеживание взаимодействий для создания темпоральных эхо
        document.addEventListener('click', (e) => {
            if (e.target.classList.contains('consciousness-fragment')) {
                this.createTemporalEcho(e.target);
            }
        });
    }
    
    createTimeLoop() {
        if (this.activeEffects.loops >= this.settings.maxLoops) {
            console.log('Достигнут максимум временных петель');
            return;
        }
        
        // Выбираем случайные элементы для петли
        const fragments = Array.from(document.querySelectorAll('.consciousness-fragment'));
        const loopSize = 2 + Math.floor(Math.random() * 4);
        const loopElements = [];
        
        for (let i = 0; i < loopSize && i < fragments.length; i++) {
            const randomFragment = fragments[Math.floor(Math.random() * fragments.length)];
            if (!loopElements.includes(randomFragment)) {
                loopElements.push(randomFragment);
            }
        }
        
        // Создаем петлю
        const loopId = Date.now();
        const loop = {
            elements: loopElements,
            startTime: Date.now(),
            duration: 5000 + Math.random() * 5000,
            iterations: 0
        };
        
        this.timeLoops.set(loopId, loop);
        
        // Применяем эффект петли
        loopElements.forEach((element, index) => {
            element.classList.add('time-loop');
            element.style.animationDelay = `${index * 0.5}s`;
            
            // Сохраняем исходное состояние
            if (!element.dataset.originalState) {
                element.dataset.originalState = JSON.stringify({
                    text: element.textContent,
                    position: element.style.transform,
                    opacity: element.style.opacity
                });
            }
        });
        
        // Визуализация петли
        this.visualizeTimeLoop(loopElements);
        
        this.activeEffects.loops++;
        this.updateTemporalPanel();
        
        // Автоматическое разрушение петли
        setTimeout(() => {
            this.destroyTimeLoop(loopId);
        }, loop.duration);
    }
    
    visualizeTimeLoop(elements) {
        if (elements.length < 2) return;
        
        // Создаем визуальные связи между элементами петли
        for (let i = 0; i < elements.length; i++) {
            const current = elements[i];
            const next = elements[(i + 1) % elements.length];
            
            const line = document.createElement('div');
            line.className = 'causal-chain';
            line.style.borderStyle = 'dashed';
            line.style.borderColor = 'rgba(255, 100, 255, 0.5)';
            
            this.positionLine(line, current, next);
            document.getElementById('container').appendChild(line);
            
            // Стрелка направления
            const arrow = document.createElement('div');
            arrow.className = 'causal-arrow';
            line.appendChild(arrow);
            
            // Анимация движения по петле
            const particle = document.createElement('div');
            particle.style.cssText = `
                position: absolute;
                width: 10px;
                height: 10px;
                background: rgba(255, 100, 255, 0.8);
                border-radius: 50%;
                animation: moveAlongLoop 3s linear infinite;
            `;
            
            line.appendChild(particle);
            
            // Сохраняем для последующего удаления
            current.dataset.loopLine = line;
        }
    }
    
    destroyTimeLoop(loopId) {
        const loop = this.timeLoops.get(loopId);
        if (!loop) return;
        
        // Удаляем эффекты петли
        loop.elements.forEach(element => {
            element.classList.remove('time-loop');
            
            // Удаляем визуализацию
            if (element.dataset.loopLine) {
                element.dataset.loopLine.remove();
                delete element.dataset.loopLine;
            }
        });
        
        this.timeLoops.delete(loopId);
        this.activeEffects.loops--;
        this.updateTemporalPanel();
    }
    
    reverseTimeFlow() {
        this.timeDirection *= -1;
        
        // Обращаем все анимации
        document.body.classList.toggle('time-reversed');
        
        // Обращаем движение маркера времени
        this.updateTimelineMarker();
        
        // Создаем визуальный эффект обращения
        const reverseEffect = document.createElement('div');
        reverseEffect.style.cssText = `
            position: fixed;
            top: 0;
            left: 0;
            width: 100%;
            height: 100%;
            background: linear-gradient(to bottom, 
                transparent, 
                rgba(255, 0, 255, 0.2), 
                transparent
            );
            pointer-events: none;
            z-index: 9999;
            animation: reverseFlash 1s ease-out forwards;
        `;
        
        const style = document.createElement('style');
        style.textContent = `
            @keyframes reverseFlash {
                0% { opacity: 0; transform: scaleY(0); }
                50% { opacity: 1; transform: scaleY(1); }
                100% { opacity: 0; transform: scaleY(1); }
            }
        `;
        document.head.appendChild(style);
        document.body.appendChild(reverseEffect);
        
        setTimeout(() => {
            reverseEffect.remove();
            style.remove();
        }, 1000);
        
        // Создаем темпоральные эхо для всех элементов
        document.querySelectorAll('.consciousness-fragment').forEach(fragment => {
            this.createTemporalEcho(fragment, 'past');
        });
    }
    
    createParadox() {
        if (this.activeEffects.paradoxes >= this.settings.maxParadoxes) {
            console.log('Реальность не выдержит больше парадоксов');
            return;
        }
        
        // Выбираем элемент для парадокса
        const fragments = document.querySelectorAll('.consciousness-fragment');
        const paradoxElement = fragments[Math.floor(Math.random() * fragments.length)];
        
        if (!paradoxElement || this.paradoxes.has(paradoxElement)) return;
        
        // Создаем парадокс существования/несуществования
        paradoxElement.classList.add('temporal-paradox');
        this.paradoxes.add(paradoxElement);
        
        // Дублируем элемент в разных временных состояниях
        const pastVersion = paradoxElement.cloneNode(true);
        const futureVersion = paradoxElement.cloneNode(true);
        
        pastVersion.classList.add('temporal-echo', 'past');
        futureVersion.classList.add('temporal-echo', 'future');
        
        pastVersion.style.color = 'rgba(255, 100, 100, 0.5)';
        futureVersion.style.color = 'rgba(100, 100, 255, 0.5)';
        
        paradoxElement.parentNode.appendChild(pastVersion);
        paradoxElement.parentNode.appendChild(futureVersion);
        
        // Создаем каузальную петлю
        this.createCausalLoop(paradoxElement, pastVersion, futureVersion);
        
        this.activeEffects.paradoxes++;
        this.updateTemporalPanel();
        
        // Парадокс разрешается через некоторое время
        setTimeout(() => {
            this.resolveParadox(paradoxElement, pastVersion, futureVersion);
        }, 10000);
    }
    
    resolveParadox(element, pastVersion, futureVersion) {
        element.classList.remove('temporal-paradox');
        this.paradoxes.delete(element);
        
        // Коллапс временных версий
        pastVersion.style.opacity = '0';
        futureVersion.style.opacity = '0';
        
        setTimeout(() => {
            pastVersion.remove();
            futureVersion.remove();
        }, 1000);
        
        this.activeEffects.paradoxes--;
        this.updateTemporalPanel();
        
        // Эффект разрешения парадокса
        this.createChronoclasmAt(element);
    }
    
    createTemporalRift() {
        const x = Math.random() * window.innerWidth;
        
        const rift = document.createElement('div');
        rift.className = 'temporal-rift';
        rift.style.left = x + 'px';
        rift.style.top = '0';
        
        document.body.appendChild(rift);
        
        // Эффект искажения пространства вокруг разрыва
        const nearbyElements = document.elementsFromPoint(x, window.innerHeight / 2);
        nearbyElements.forEach(element => {
            if (element.classList.contains('consciousness-fragment')) {
                const distance = Math.abs(element.getBoundingClientRect().left - x);
                if (distance < 100) {
                    // Временная дилатация
                    this.applyTimeDilation(element, 2 + Math.random() * 3);
                }
            }
        });
        
        // Разрыв закрывается
        setTimeout(() => {
            rift.style.opacity = '0';
            setTimeout(() => rift.remove(), 1000);
        }, 5000);
    }
    
    triggerChronoclasm() {
        const x = window.innerWidth / 2 + (Math.random() - 0.5) * 400;
        const y = window.innerHeight / 2 + (Math.random() - 0.5) * 400;
        
        this.createChronoclasmAt({ 
            getBoundingClientRect: () => ({ 
                left: x - 50, 
                top: y - 50, 
                width: 100, 
                height: 100 
            }) 
        });
        
        // Хроноклазм влияет на все элементы в радиусе
        const fragments = document.querySelectorAll('.consciousness-fragment');
        fragments.forEach(fragment => {
            const rect = fragment.getBoundingClientRect();
            const distance = Math.sqrt(
                Math.pow(rect.left + rect.width/2 - x, 2) + 
                Math.pow(rect.top + rect.height/2 - y, 2)
            );
            
            if (distance < 250) {
                // Временной сдвиг
                const shift = (250 - distance) / 250;
                fragment.style.animationDelay = `${shift * 2}s`;
                
                // Создаем темпоральное эхо
                this.createTemporalEcho(fragment, Math.random() > 0.5 ? 'past' : 'future');
            }
        });
    }
    
    createChronoclasmAt(element) {
        const rect = element.getBoundingClientRect();
        const chronoclasm = document.createElement('div');
        chronoclasm.className = 'chronoclasm';
        chronoclasm.style.left = (rect.left + rect.width / 2 - 250) + 'px';
        chronoclasm.style.top = (rect.top + rect.height / 2 - 250) + 'px';
        
        document.body.appendChild(chronoclasm);
        
        setTimeout(() => chronoclasm.remove(), 2000);
    }
    
    createTemporalEcho(element, type = null) {
        const echoType = type || (Math.random() > 0.5 ? 'past' : 'future');
        
        for (let i = 1; i <= this.settings.echoCount; i++) {
            setTimeout(() => {
                const echo = element.cloneNode(true);
                echo.classList.add('temporal-echo', echoType);
                echo.style.opacity = 0.3 / i;
                
                if (echoType === 'past') {
                    echo.style.transform = `translateX(${-20 * i}px)`;
                } else {
                    echo.style.transform = `translateX(${20 * i}px)`;
                }
                
                element.parentNode.appendChild(echo);
                
                // Эхо исчезает
                setTimeout(() => {
                    echo.style.opacity = '0';
                    setTimeout(() => echo.remove(), 1000);
                }, 2000);
            }, i * this.settings.echoDelay);
        }
    }
    
    applyTimeDilation(element, factor) {
        element.classList.add(factor > 1 ? 'time-dilated' : 'time-compressed');
        
        // Сохраняем фактор дилатации
        this.activeEffects.dilations.set(element, factor);
        
        // Визуальный индикатор
        const indicator = document.createElement('div');
        indicator.style.cssText = `
            position: absolute;
            top: -20px;
            left: 50%;
            transform: translateX(-50%);
            font-size: 0.7rem;
            color: rgba(255, 200, 100, 0.8);
            pointer-events: none;
        `;
        indicator.textContent = `${factor.toFixed(1)}x`;
        
        element.style.position = 'relative';
        element.appendChild(indicator);
        
        // Дилатация проходит
        setTimeout(() => {
            element.classList.remove('time-dilated', 'time-compressed');
            this.activeEffects.dilations.delete(element);
            indicator.remove();
        }, 10000);
    }
    
    startTimeFlow() {
        // Движение по временной линии
        setInterval(() => {
            if (this.timeDirection === 1) {
                this.currentTimePosition += 0.01 * this.timeRate;
                if (this.currentTimePosition > 1) this.currentTimePosition = 0;
            } else {
                this.currentTimePosition -= 0.01 * this.timeRate;
                if (this.currentTimePosition < 0) this.currentTimePosition = 1;
            }
            
            this.updateTimelineMarker();
        }, 100);
        
        // Случайные временные аномалии
        setInterval(() => {
            if (Math.random() < 0.1) {
                this.createRandomAnomaly();
            }
        }, 10000);
    }
    
    updateTimelineMarker() {
        const marker = document.getElementById('timeline-marker');
        marker.style.left = (this.currentTimePosition * 100) + '%';
        
        // Цвет маркера зависит от направления времени
        if (this.timeDirection === -1) {
            marker.style.background = 'rgba(255, 100, 100, 0.8)';
        } else {
            marker.style.background = 'rgba(255, 255, 255, 0.8)';
        }
    }
    
    createRandomAnomaly() {
        const anomalies = [
            () => this.createTemporalVortex(),
            () => this.createCausalityGlitch(),
            () => this.createTimeStutter(),
            () => this.createChronoWave()
        ];
        
        const randomAnomaly = anomalies[Math.floor(Math.random() * anomalies.length)];
        randomAnomaly();
    }
    
    createTemporalVortex() {
        const vortex = document.createElement('div');
        vortex.className = 'temporal-vortex';
        
        const x = Math.random() * (window.innerWidth - 200);
        const y = Math.random() * (window.innerHeight - 200);
        
        vortex.style.left = x + 'px';
        vortex.style.top = y + 'px';
        
        document.getElementById('container').appendChild(vortex);
        
        // Вортекс притягивает элементы
        const fragments = document.querySelectorAll('.consciousness-fragment');
        fragments.forEach(fragment => {
            const rect = fragment.getBoundingClientRect();
            const distance = Math.sqrt(
                Math.pow(rect.left - x - 100, 2) + 
                Math.pow(rect.top - y - 100, 2)
            );
            
            if (distance < 200) {
                const angle = Math.atan2(y + 100 - rect.top, x + 100 - rect.left);
                const force = (200 - distance) / 200;
                
                fragment.style.transform = `translate(${Math.cos(angle) * force * 20}px, ${Math.sin(angle) * force * 20}px)`;
                
                setTimeout(() => {
                    fragment.style.transform = '';
                }, 3000);
            }
        });
        
        setTimeout(() => {
            vortex.style.opacity = '0';
            setTimeout(() => vortex.remove(), 1000);
        }, 5000);
    }

        const y1 = rect1.top + rect1.height / 2;

        const y2 = rect2.top + rect2.height / 2;
        
        const distance = Math.sqrt(Math.pow(x2 - x1, 2) + Math.pow(y2 - y1, 2));
        const angle = Math.atan2(y2 - y1, x2 - x1);

        line.style.top = y1 + 'px';

            z-index: 1;

        canvas.width = window.innerWidth;
        canvas.height = window.innerHeight;
        
        // Анимация временного поля
        const animateField = () => {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            const time = Date.now() * 0.001 * this.timeRate * this.timeDirection;
            
            // Рисуем временные линии
            ctx.strokeStyle = 'rgba(255, 255, 255, 0.5)';
            ctx.lineWidth = 1;
            
            for (let i = 0; i < 20; i++) {
                ctx.beginPath();
                const y = (time * 50 + i * 50) % canvas.height;
                ctx.moveTo(0, y);
                
                for (let x = 0; x < canvas.width; x += 10) {
                    const offset = Math.sin(x * 0.01 + time) * 20;
                    ctx.lineTo(x, y + offset);
                }
                
                ctx.stroke();
            }
            
            requestAnimationFrame(animateField);
        };
        
        animateField();
    }
```

## Использование

После добавления кода:

1. Справа внизу появится панель управления временными аномалиями
2. Можно создавать временные петли, парадоксы, разрывы времени
3. Время можно обратить вспять - все анимации пойдут в обратную сторону
4. Элементы оставляют темпоральные эхо из прошлого и будущего
5. Внизу отображается временная линия с текущей позицией

## Эффекты

- **Временные петли**: Элементы застревают в повторяющихся циклах
- **Парадоксы**: Элементы существуют в нескольких временных состояниях одновременно
- **Хроноклазм**: Взрыв времени, создающий волны темпоральных искажений
- **Временная дилатация**: Замедление или ускорение времени для отдельных элементов
- **Каузальные аномалии**: Нарушение причинно-следственных связей
