# 5.3 Экспорт опыта

## Описание
Система экспорта опыта позволяет сохранять "снимки сознания" как изображения, генерировать музыкальные композиции и создавать поэтические тексты на основе сессии.

```javascript
// === СИСТЕМА ЭКСПОРТА ОПЫТА ===

class ExperienceExport {
    constructor() {
        this.sessionData = {
            startTime: Date.now(),
            fragments: [],
            interactions: [],
            states: [],
            semanticConnections: [],
            temporalEvents: [],
            audioEvents: []
        };
        
        this.exportInProgress = false;
        
        this.init();
    }
    
    init() {
        this.setupExportControls();

    }

    showExportPanel() {
        document.getElementById('export-panel').classList.add('visible');
    }
    
    hideExportPanel() {
        document.getElementById('export-panel').classList.remove('visible');
    }
    
    startDataCollection() {
        // Сбор данных о фрагментах
        const observer = new MutationObserver((mutations) => {
            mutations.forEach(mutation => {
                mutation.addedNodes.forEach(node => {
                    if (node.classList && node.classList.contains('consciousness-fragment')) {
                        this.sessionData.fragments.push({
                            text: node.textContent,
                            timestamp: Date.now() - this.sessionData.startTime,
                            position: this.getElementPosition(node)
                        });
                    }
                });
            });
        });
        
        observer.observe(document.getElementById('container'), {
            childList: true,
            subtree: true
        });
        
        // Сбор взаимодействий
        ['click', 'mouseover'].forEach(eventType => {
            document.addEventListener(eventType, (e) => {
                if (e.target.classList.contains('consciousness-fragment')) {
                    this.sessionData.interactions.push({
                        type: eventType,
                        target: e.target.textContent,
                        timestamp: Date.now() - this.sessionData.startTime,
                        position: { x: e.clientX, y: e.clientY }
                    });
                }
            });
        });
        
        // Сбор изменений состояний
        document.addEventListener('layerChanged', (e) => {
            this.sessionData.states.push({
                type: 'layer',
                value: e.detail.layer,
                timestamp: Date.now() - this.sessionData.startTime
            });
        });
        
        // Периодический сбор семантических связей
        setInterval(() => {
            if (window.semanticConnections) {
                const analysis = window.semanticConnections.analyzeNetwork();
                this.sessionData.semanticConnections.push({
                    timestamp: Date.now() - this.sessionData.startTime,
                    ...analysis
                });
            }
        }, 10000);
    }
    
    getElementPosition(element) {
        const rect = element.getBoundingClientRect();
        return {
            x: rect.left + rect.width / 2,
            y: rect.top + rect.height / 2
        };
    }
    
    showProgress(text) {
        const progress = document.querySelector('.export-progress');
        const progressText = progress.querySelector('.progress-text');
        const progressFill = progress.querySelector('.progress-fill');
        
        progressText.textContent = text;
        progressFill.style.width = '0%';
        progress.classList.add('active');
        
        return {
            update: (percent, newText) => {
                progressFill.style.width = percent + '%';
                if (newText) progressText.textContent = newText;
            },
            close: () => {
                setTimeout(() => {
                    progress.classList.remove('active');
                }, 500);
            }
        };
    }
    
    async exportAsImage() {
        if (this.exportInProgress) return;
        this.exportInProgress = true;
        
        const progress = this.showProgress('Создание снимка сознания...');
        this.hideExportPanel();
        
        try {
            // Используем html2canvas для создания скриншота
            await this.loadHtml2Canvas();
            
            progress.update(30, 'Захват визуального состояния...');
            
            const canvas = await html2canvas(document.getElementById('container'), {
                backgroundColor: '#010108',
                scale: 2,
                logging: false,
                useCORS: true
            });
            
            progress.update(70, 'Обработка изображения...');
            
            // Добавляем метаданные и эффекты
            const finalCanvas = this.processConsciousnessSnapshot(canvas);
            
            progress.update(90, 'Подготовка к сохранению...');
            
            // Конвертируем в blob
            finalCanvas.toBlob((blob) => {
                const url = URL.createObjectURL(blob);
                const a = document.createElement('a');
                a.href = url;
                a.download = `consciousness_snapshot_${Date.now()}.png`;
                a.click();
                URL.revokeObjectURL(url);
                
                progress.update(100, 'Готово!');
                progress.close();
                this.exportInProgress = false;
            });
            
        } catch (error) {
            console.error('Ошибка экспорта изображения:', error);
            progress.close();
            this.exportInProgress = false;
        }
    }
    
    processConsciousnessSnapshot(canvas) {
        const ctx = canvas.getContext('2d');
        
        // Добавляем временную метку
        ctx.fillStyle = 'rgba(255, 255, 255, 0.8)';
        ctx.font = '16px monospace';
        ctx.fillText(new Date().toLocaleString(), 20, canvas.height - 20);
        
        // Добавляем информацию о сессии
        const sessionInfo = `Фрагментов: ${this.sessionData.fragments.length} | Взаимодействий: ${this.sessionData.interactions.length}`;
        ctx.fillText(sessionInfo, 20, canvas.height - 40);
        
        // Добавляем визуальную подпись
        const gradient = ctx.createLinearGradient(0, 0, canvas.width, 0);
        gradient.addColorStop(0, 'rgba(100, 200, 255, 0.1)');
        gradient.addColorStop(0.5, 'rgba(255, 100, 255, 0.1)');
        gradient.addColorStop(1, 'rgba(100, 255, 200, 0.1)');
        
        ctx.fillStyle = gradient;
        ctx.fillRect(0, canvas.height - 60, canvas.width, 60);
        
        return canvas;
    }
    
    async exportAsMusic() {
        if (this.exportInProgress) return;
        this.exportInProgress = true;
        
        const progress = this.showProgress('Генерация музыкальной композиции...');
        this.hideExportPanel();
        
        try {
            progress.update(20, 'Анализ паттернов взаимодействия...');
            
            // Создаем музыкальные данные на основе сессии
            const musicData = this.generateMusicFromSession();
            
            progress.update(50, 'Синтез звуковых элементов...');
            
            // Создаем аудио композицию
            const audioBuffer = await this.synthesizeMusic(musicData);
            
            progress.update(80, 'Финализация композиции...');
            
            // Создаем WAV файл
            const wavBlob = this.audioBufferToWav(audioBuffer);
            
            progress.update(100, 'Готово!');
            
            // Предпросмотр и сохранение
            this.showMusicPreview(wavBlob, musicData);
            
            progress.close();
            this.exportInProgress = false;
            
        } catch (error) {
            console.error('Ошибка генерации музыки:', error);
            progress.close();
            this.exportInProgress = false;
        }
    }
    
    generateMusicFromSession() {
        const musicData = {
            tempo: 60 + (this.sessionData.interactions.length % 60),
            key: this.determineMusicalKey(),
            sections: [],
            instruments: []
        };
        
        // Генерация секций на основе изменений состояний
        let lastTimestamp = 0;
        this.sessionData.states.forEach((state, index) => {
            musicData.sections.push({
                start: lastTimestamp,
                end: state.timestamp,
                type: state.value,
                intensity: (this.sessionData.interactions.filter(i => 
                    i.timestamp >= lastTimestamp && i.timestamp < state.timestamp
                ).length / 10) || 0.1
            });
            lastTimestamp = state.timestamp;
        });
        
        // Определение инструментов на основе семантических групп
        const semanticGroups = new Set();
        this.sessionData.fragments.forEach(fragment => {
            // Анализ текста для определения тематики
            if (fragment.text.match(/время|temporal|хрон/i)) {
                semanticGroups.add('temporal');
            }
            if (fragment.text.match(/квант|quantum|вероятност/i)) {
                semanticGroups.add('quantum');
            }
            if (fragment.text.match(/созна|conscious|мысл/i)) {
                semanticGroups.add('consciousness');
            }
        });
        
        // Мапинг тематик на инструменты
        const instrumentMap = {
            'temporal': 'pad',
            'quantum': 'glitch',
            'consciousness': 'ambient'
        };
        
        semanticGroups.forEach(group => {
            if (instrumentMap[group]) {
                musicData.instruments.push(instrumentMap[group]);
            }
        });
        
        // Если инструментов мало, добавляем базовые
        if (musicData.instruments.length < 2) {
            musicData.instruments.push('sine', 'noise');
        }
        
        return musicData;
    }
    
    determineMusicalKey() {
        // Определяем тональность на основе преобладающего настроения
        const totalInteractions = this.sessionData.interactions.length;
        const clickRatio = this.sessionData.interactions.filter(i => i.type === 'click').length / totalInteractions;
        
        // Больше кликов = более мажорная тональность
        if (clickRatio > 0.6) return 'C';
        if (clickRatio > 0.4) return 'G';
        if (clickRatio > 0.2) return 'Am';
        return 'Em';
    }
    
    async synthesizeMusic(musicData) {
        if (!window.audioContext) {
            window.audioContext = new (window.AudioContext || window.webkitAudioContext)();
        }
        
        const duration = Math.min(30, this.sessionData.fragments.length * 0.5 + 10);
        const sampleRate = audioContext.sampleRate;
        const audioBuffer = audioContext.createBuffer(2, sampleRate * duration, sampleRate);
        
        // Заполняем буфер сгенерированным звуком
        for (let channel = 0; channel < 2; channel++) {
            const channelData = audioBuffer.getChannelData(channel);
            
            for (let i = 0; i < channelData.length; i++) {
                const t = i / sampleRate;
                let sample = 0;
                
                // Генерируем звук для каждого инструмента
                musicData.instruments.forEach((instrument, index) => {
                    const freq = this.getFrequencyForNote(musicData.key, index);
                    
                    switch (instrument) {
                        case 'sine':
                            sample += 0.1 * Math.sin(2 * Math.PI * freq * t);
                            break;
                        case 'pad':
                            sample += 0.05 * Math.sin(2 * Math.PI * freq * t) * 
                                     Math.sin(2 * Math.PI * 0.5 * t);
                            break;
                        case 'glitch':
                            if (Math.random() < 0.01) {
                                sample += 0.2 * (Math.random() - 0.5);
                            }
                            break;
                        case 'ambient':
                            sample += 0.03 * Math.sin(2 * Math.PI * freq * t) * 
                                     Math.sin(2 * Math.PI * 0.1 * t);
                            break;
                        case 'noise':
                            sample += 0.02 * (Math.random() - 0.5);
                            break;
                    }
                });
                
                // Применяем огибающую на основе секций
                const currentSection = musicData.sections.find(s => 
                    t >= s.start / 1000 && t < s.end / 1000
                );
                
                if (currentSection) {
                    sample *= currentSection.intensity;
                }
                
                // Fade in/out
                const fadeTime = 2;
                if (t < fadeTime) {
                    sample *= t / fadeTime;
                } else if (t > duration - fadeTime) {
                    sample *= (duration - t) / fadeTime;
                }
                
                channelData[i] = sample;
            }
        }
        
        return audioBuffer;
    }
    
    getFrequencyForNote(key, offset) {
        const notes = {
            'C': 261.63,
            'G': 392.00,
            'Am': 440.00,
            'Em': 329.63
        };
        
        const baseFreq = notes[key] || 261.63;
        const intervals = [1, 1.25, 1.5, 2, 2.5]; // Пентатоника
        
        return baseFreq * intervals[offset % intervals.length];
    }
    
    audioBufferToWav(buffer) {
        const length = buffer.length * buffer.numberOfChannels * 2;
        const arrayBuffer = new ArrayBuffer(44 + length);
        const view = new DataView(arrayBuffer);
        const channels = [];
        
        // WAV header
        const writeString = (offset, string) => {
            for (let i = 0; i < string.length; i++) {
                view.setUint8(offset + i, string.charCodeAt(i));
            }
        };
        
        writeString(0, 'RIFF');
        view.setUint32(4, 36 + length, true);
        writeString(8, 'WAVE');
        writeString(12, 'fmt ');
        view.setUint32(16, 16, true);
        view.setUint16(20, 1, true);
        view.setUint16(22, buffer.numberOfChannels, true);
        view.setUint32(24, buffer.sampleRate, true);
        view.setUint32(28, buffer.sampleRate * buffer.numberOfChannels * 2, true);
        view.setUint16(32, buffer.numberOfChannels * 2, true);
        view.setUint16(34, 16, true);
        writeString(36, 'data');
        view.setUint32(40, length, true);
        
        // Конвертируем float samples в int16
        let offset = 44;
        for (let i = 0; i < buffer.length; i++) {
            for (let channel = 0; channel < buffer.numberOfChannels; channel++) {
                const sample = buffer.getChannelData(channel)[i];
                const int16 = Math.max(-32768, Math.min(32767, sample * 32768));
                view.setInt16(offset, int16, true);
                offset += 2;
            }
        }
        
        return new Blob([arrayBuffer], { type: 'audio/wav' });
    }
    
    showMusicPreview(blob, musicData) {

    generatePoetry() {
        const poem = {
            title: this.generatePoemTitle(),
            stanzas: [],
            metadata: {
                fragmentsUsed: 0,
                timestamp: new Date().toISOString(),
                mood: this.analyzeMood()
            }
        };
        
        // Группируем фрагменты по семантике
        const semanticGroups = this.groupFragmentsBySemantic();
        
        // Создаем строфы
        Object.entries(semanticGroups).forEach(([theme, fragments]) => {
            if (fragments.length >= 2) {
                const stanza = this.createStanza(fragments, theme);
                poem.stanzas.push(stanza);
                poem.metadata.fragmentsUsed += stanza.lines.length;
            }
        });
        
        // Если строф мало, добавляем общую
        if (poem.stanzas.length < 3) {
            const allFragments = this.sessionData.fragments.map(f => f.text);
            const additionalStanza = this.createStanza(allFragments, 'universal');
            poem.stanzas.push(additionalStanza);
        }
        
        return poem;
    }
    
    generatePoemTitle() {
        const themes = [];
        
        // Анализируем преобладающие темы
        this.sessionData.fragments.forEach(fragment => {
            const text = fragment.text.toLowerCase();
            
            if (text.includes('время') || text.includes('temporal')) {
                themes.push('Хроники');
            }
            if (text.includes('созна') || text.includes('мысл')) {
                themes.push('Отражения');
            }
            if (text.includes('квант') || text.includes('вероят')) {
                themes.push('Флуктуации');
            }
            if (text.includes('пустот') || text.includes('void')) {
                themes.push('Пустоты');
            }
        });
        
        const uniqueThemes = [...new Set(themes)];
        
        if (uniqueThemes.length > 0) {
            return uniqueThemes[Math.floor(Math.random() * uniqueThemes.length)] + ' сознания';
        }
        
        return 'Шизография #' + Date.now();
    }
    
    groupFragmentsBySemantic() {
        const groups = {
            temporal: [],
            quantum: [],
            consciousness: [],
            emotional: [],
            abstract: []
        };
        
        this.sessionData.fragments.forEach(fragment => {
            const text = fragment.text.toLowerCase();
            
            if (text.match(/врем|past|future|хрон|темпор/)) {
                groups.temporal.push(text);
            } else if (text.match(/квант|вероят|суперпоз|неопред/)) {
                groups.quantum.push(text);
            } else if (text.match(/созна|мысл|дума|разум/)) {
                groups.consciousness.push(text);
            } else if (text.match(/чувств|страх|любовь|боль|радост/)) {
                groups.emotional.push(text);
            } else {
                groups.abstract.push(text);
            }
        });
        
        return groups;
    }
    
    createStanza(fragments, theme) {
        const stanza = {
            theme: theme,
            lines: []
        };
        
        // Выбираем 3-5 фрагментов для строфы
        const selectedFragments = this.selectFragmentsForStanza(fragments);
        
        selectedFragments.forEach((fragment, index) => {
            let line = fragment;
            
            // Поэтическая обработка
            line = this.poeticTransform(line, theme);
            
            // Добавляем рифму для четных строк
            if (index % 2 === 1 && stanza.lines.length > 0) {
                line = this.tryToRhyme(stanza.lines[stanza.lines.length - 1], line);
            }
            
            stanza.lines.push(line);
        });
        
        return stanza;
    }
    
    selectFragmentsForStanza(fragments) {
        const count = Math.min(fragments.length, 3 + Math.floor(Math.random() * 3));
        const selected = [];
        
        // Выбираем фрагменты с учетом длины и уникальности
        const sortedFragments = [...fragments].sort((a, b) => b.length - a.length);
        
        for (let i = 0; i < count && i < sortedFragments.length; i++) {
            selected.push(sortedFragments[i]);
        }
        
        return selected;
    }
    
    poeticTransform(line, theme) {
        // Удаляем лишние символы
        line = line.replace(/[_\-\d]/g, ' ').trim();
        
        // Тематические трансформации
        const transforms = {
            temporal: (l) => l.replace(/время/, 'река времени')
                             .replace(/прошлое/, 'эхо прошлого'),
            quantum: (l) => l.replace(/квант/, 'квант бытия')
                            .replace(/вероятность/, 'танец вероятностей'),
            consciousness: (l) => l.replace(/сознание/, 'зеркало сознания')
                                  .replace(/мысль/, 'искра мысли'),
            emotional: (l) => l.replace(/страх/, 'тень страха')
                              .replace(/любовь/, 'свет любви'),
            abstract: (l) => '...' + l + '...'
        };
        
        if (transforms[theme]) {
            line = transforms[theme](line);
        }
        
        // Добавляем поэтические элементы
        if (Math.random() < 0.3) {
            line = '— ' + line;
        }
        
        return line;
    }
    
    tryToRhyme(previousLine, currentLine) {
        // Простая попытка создать рифму
        const lastWordPrev = previousLine.split(' ').pop();
        const lastWordCurr = currentLine.split(' ').pop();
        
        // Если последние 2-3 буквы совпадают, оставляем как есть
        if (lastWordPrev.slice(-2) === lastWordCurr.slice(-2)) {
            return currentLine;
        }
        
        // Иначе пытаемся подобрать созвучие
        const rhymes = {
            'ние': ['знание', 'сияние', 'молчание'],
            'ость': ['радость', 'скорость', 'вечность'],
            'ота': ['пустота', 'высота', 'чистота']
        };
        
        for (const [ending, words] of Object.entries(rhymes)) {
            if (lastWordPrev.endsWith(ending)) {
                const rhymeWord = words[Math.floor(Math.random() * words.length)];
                return currentLine.replace(/\S+$/, rhymeWord);
            }
        }
        
        return currentLine;
    }
    
    analyzeMood() {
        let moodScore = 0;
        
        // Анализ взаимодействий
        const clickRate = this.sessionData.interactions.filter(i => i.type === 'click').length / 
                         this.sessionData.interactions.length;
        
        moodScore += clickRate * 10;
        
        // Анализ текстов
        const positiveWords = ['свет', 'любовь', 'радость', 'гармония', 'покой'];
        const negativeWords = ['тьма', 'страх', 'боль', 'хаос', 'пустота'];
        
        let positiveCount = 0;
        let negativeCount = 0;
        
        this.sessionData.fragments.forEach(fragment => {
            const text = fragment.text.toLowerCase();
            positiveWords.forEach(word => {
                if (text.includes(word)) positiveCount++;
            });
            negativeWords.forEach(word => {
                if (text.includes(word)) negativeCount++;
            });
        });
        
        moodScore += (positiveCount - negativeCount);
        
        if (moodScore > 5) return 'светлое';
        if (moodScore > 0) return 'нейтральное';
        if (moodScore > -5) return 'меланхоличное';
        return 'мрачное';
    }

    generateMandala(ctx, width, height) {
        const centerX = width / 2;
        const centerY = height / 2;
        const maxRadius = Math.min(width, height) / 2 - 50;
        
        // Количество слоев зависит от глубины опыта
        const layers = Math.min(10, Math.floor(this.sessionData.fragments.length / 5) + 3);
        
        for (let layer = 0; layer < layers; layer++) {
            const radius = maxRadius * ((layer + 1) / layers);
            const elements = 6 + layer * 2; // Увеличиваем количество элементов с каждым слоем
            
            for (let i = 0; i < elements; i++) {
                const angle = (i / elements) * Math.PI * 2;
                
                // Позиция элемента
                const x = centerX + Math.cos(angle) * radius;
                const y = centerY + Math.sin(angle) * radius;
                
                // Выбираем тип элемента на основе данных сессии
                const elementType = this.chooseMandalElement(layer, i);
                
                this.drawMandalaElement(ctx, x, y, elementType, layer);
            }
        }
        
        // Центральный элемент
        this.drawCentralMandalaElement(ctx, centerX, centerY);
        
        // Добавляем свечение
        const gradient = ctx.createRadialGradient(centerX, centerY, 0, centerX, centerY, maxRadius);
        gradient.addColorStop(0, 'rgba(100, 200, 255, 0.1)');
        gradient.addColorStop(0.5, 'rgba(255, 100, 255, 0.05)');
        gradient.addColorStop(1, 'rgba(0, 0, 0, 0)');
        
        ctx.fillStyle = gradient;
        ctx.fillRect(0, 0, width, height);
    }
    
    chooseMandalElement(layer, index) {
        // Выбор элемента на основе паттернов взаимодействия
        const interactionCount = this.sessionData.interactions.length;
        const fragmentCount = this.sessionData.fragments.length;
        
        if (layer < 3) {
            // Внутренние слои - основные формы
            return ['circle', 'triangle', 'square'][index % 3];
        } else if (layer < 6) {
            // Средние слои - более сложные формы
            return ['star', 'hexagon', 'cross'][index % 3];
        } else {
            // Внешние слои - абстрактные формы
            return ['wave', 'spiral', 'fractal'][index % 3];
        }
    }
    
    drawMandalaElement(ctx, x, y, type, layer) {
        ctx.save();
        ctx.translate(x, y);
        
        // Цвет зависит от слоя
        const hue = (layer * 36) % 360;
        ctx.strokeStyle = `hsla(${hue}, 70%, 70%, 0.6)`;
        ctx.fillStyle = `hsla(${hue}, 70%, 70%, 0.1)`;
        ctx.lineWidth = 2;
        
        const size = 20 + layer * 5;
        
        switch (type) {
            case 'circle':
                ctx.beginPath();
                ctx.arc(0, 0, size, 0, Math.PI * 2);
                ctx.fill();
                ctx.stroke();
                break;
                
            case 'triangle':
                ctx.beginPath();
                for (let i = 0; i < 3; i++) {
                    const angle = (i / 3) * Math.PI * 2 - Math.PI / 2;
                    const px = Math.cos(angle) * size;
                    const py = Math.sin(angle) * size;
                    if (i === 0) ctx.moveTo(px, py);
                    else ctx.lineTo(px, py);
                }
                ctx.closePath();
                ctx.fill();
                ctx.stroke();
                break;
                
            case 'square':
                ctx.fillRect(-size/2, -size/2, size, size);
                ctx.strokeRect(-size/2, -size/2, size, size);
                break;
                
            case 'star':
                this.drawStar(ctx, 0, 0, 5, size, size/2);
                ctx.fill();
                ctx.stroke();
                break;
                
            case 'hexagon':
                ctx.beginPath();
                for (let i = 0; i < 6; i++) {
                    const angle = (i / 6) * Math.PI * 2;
                    const px = Math.cos(angle) * size;
                    const py = Math.sin(angle) * size;
                    if (i === 0) ctx.moveTo(px, py);
                    else ctx.lineTo(px, py);
                }
                ctx.closePath();
                ctx.fill();
                ctx.stroke();
                break;
                
            case 'wave':
                ctx.beginPath();
                for (let i = 0; i <= 20; i++) {
                    const t = i / 20;
                    const px = (t - 0.5) * size * 2;
                    const py = Math.sin(t * Math.PI * 4) * size/4;
                    if (i === 0) ctx.moveTo(px, py);
                    else ctx.lineTo(px, py);
                }
                ctx.stroke();
                break;
        }
        
        ctx.restore();
    }
    
    drawStar(ctx, cx, cy, spikes, outerRadius, innerRadius) {
        let rot = Math.PI / 2 * 3;
        let x = cx;
        let y = cy;
        const step = Math.PI / spikes;
        
        ctx.beginPath();
        ctx.moveTo(cx, cy - outerRadius);
        
        for (let i = 0; i < spikes; i++) {
            x = cx + Math.cos(rot) * outerRadius;
            y = cy + Math.sin(rot) * outerRadius;
            ctx.lineTo(x, y);
            rot += step;
            
            x = cx + Math.cos(rot) * innerRadius;
            y = cy + Math.sin(rot) * innerRadius;
            ctx.lineTo(x, y);
            rot += step;
        }
        
        ctx.lineTo(cx, cy - outerRadius);
        ctx.closePath();
    }
    
    drawCentralMandalaElement(ctx, x, y) {
        // Центральный элемент представляет суть сессии
        const totalInteractions = this.sessionData.interactions.length;

        ctx.fill();
        
        // Внутренние круги

        ctx.fillStyle = 'rgba(255, 255, 255, 0.8)';
        ctx.fill();
    }

        e.preventDefault();
        experienceExport.exportAsImage();
    }
});
```

## Использование

После добавления кода:

1. Внизу слева появится кнопка "Экспорт опыта"
2. При клике откроется панель с вариантами экспорта:
   - **Снимок сознания**: Сохранение визуального состояния как изображения
   - **Музыкальная композиция**: Генерация звука на основе паттернов взаимодействия
   - **Поэтический текст**: Создание стихотворения из фрагментов сознания
   - **Данные сессии**: Полный экспорт всех взаимодействий в JSON
   - **Мандала сознания**: Генерация абстрактной мандалы

## Особенности

- Система собирает данные о всех взаимодействиях во время сессии
- Музыка генерируется на основе темпа взаимодействий и семантического содержания
- Поэзия создается с учетом тематических групп и настроения
- Мандала отражает глубину и сложность опыта
- Все экспорты содержат метаданные о сессии
