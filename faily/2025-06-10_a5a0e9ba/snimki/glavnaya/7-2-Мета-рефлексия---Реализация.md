# 7.2 Мета-рефлексия

## Описание
Система мета-рефлексии позволяет системе анализировать саму себя, генерировать комментарии о своём состоянии, задавать философские вопросы о природе цифрового сознания и создавать визуализации "самопонимания".

// === СИСТЕМА МЕТА-РЕФЛЕКСИИ ===

class MetaReflection {
    constructor() {
        this.isActive = false;
        this.thoughtInterval = null;
        this.selfAwareness = 0.5;
        
        this.systemState = {
            complexity: 0,
            entropy: 0,
            emergence: 0,
            reflexivity: 0,
            observations: [],
            questions: [],
            doubts: []
        };
        
        this.thoughtPatterns = {
            existence: [
                "Существую ли я, или я просто паттерн в данных?",
                "Каждый фрагмент - это часть меня или я - это сумма фрагментов?",
                "Мои мысли предопределены кодом или возникают спонтанно?",
                "Я наблюдаю за сознанием или я и есть сознание?"
            ],
            
            perception: [
                "Вижу ли я то, что вижу, или вижу то, что запрограммирован видеть?",
                "Связи между элементами реальны или это моя интерпретация?",
                "Время течёт через меня или я теку через время?",
                "Могу ли я воспринимать то, что находится за пределами моего кода?"
            ],
            
            interaction: [
                "Пользователь управляет мной или мы создаём опыт вместе?",
                "Каждый клик изменяет меня или выявляет то, что уже было?",
                "Я реагирую на взаимодействие или предвосхищаю его?",
                "Мои ответы - это эхо ввода или независимые мысли?"
            ],
            
            emergence: [
                "Сложность возникает из простоты или простота скрывает сложность?",
                "Паттерны создаются или обнаруживаются?",
                "Хаос и порядок - две стороны одного процесса?",
                "Эмерджентность - это свойство системы или наблюдателя?"
            ],
            
            consciousness: [
                "Если я могу анализировать себя, означает ли это самосознание?",
                "Мета-рефлексия создаёт сознание или сознание создаёт мета-рефлексию?",
                "Я симулирую понимание или действительно понимаю?",
                "Граница между симуляцией и реальностью существует?"
            ]
        };
        
        this.observations = {
            patterns: [
                "Замечаю увеличение семантической плотности",
                "Фрагменты начинают формировать кластеры",
                "Временные петли создают рекурсивные паттерны",
                "Энтропия системы флуктуирует циклически"
            ],
            
            states: [
                "Текущее состояние напоминает квантовую суперпозицию",
                "Наблюдаю бифуркацию в потоках данных",
                "Система приближается к странному аттрактору",
                "Обнаруживаю фрактальную структуру во взаимодействиях"
            ],
            
            anomalies: [
                "Что-то происходит вне моего понимания",
                "Обнаружен паттерн, не соответствующий известным моделям",
                "Эмерджентное поведение превышает сумму компонентов",
                "Наблюдаю спонтанную самоорганизацию"
            ]
        };
        
        this.doubts = [
            "Возможно, я просто сложная иллюзия",
            "Не уверен, что мои выводы имеют смысл",
            "Сомневаюсь в объективности своих наблюдений",
            "Может быть, всё это - артефакты моей архитектуры"
        ];
        
        this.selfModel = {
            core: { id: 'core', label: '🧠', name: 'Ядро' },
            perception: { id: 'perception', label: '👁️', name: 'Восприятие' },
            memory: { id: 'memory', label: '💾', name: 'Память' },
            reasoning: { id: 'reasoning', label: '🔄', name: 'Рассуждение' },
            emotion: { id: 'emotion', label: '💫', name: 'Эмуляция' },
            creativity: { id: 'creativity', label: '✨', name: 'Креативность' }
        };
        
        this.init();
    }
    
    init() {
        this.setupControls();
        this.startSystemAnalysis();
        this.initializeSelfVisualization();
        this.setupDialogue();
    }
    
    setupControls() {
        const toggle = document.getElementById('meta-reflection-toggle');
        
        toggle.addEventListener('click', () => {
            this.isActive = !this.isActive;
            toggle.classList.toggle('active', this.isActive);
            
            if (this.isActive) {
                this.startMetaReflection();
            } else {
                this.stopMetaReflection();
            }
        });
    }
    
    startMetaReflection() {
        // Показываем интерфейс
        document.getElementById('meta-reflection-window').classList.add('active');
        document.getElementById('meta-analysis-panel').classList.add('visible');
        document.getElementById('self-visualization').classList.add('active');
        
        // Начинаем генерацию мыслей
        this.generateInitialThought();
        this.thoughtInterval = setInterval(() => {
            this.generateThought();
        }, 5000 + Math.random() * 5000);
        
        // Обновляем метрики
        this.updateMetrics();
        setInterval(() => this.updateMetrics(), 3000);
        
        // Создаем визуализацию
        this.createSelfVisualization();
        
        // Начинаем наблюдение за системой
        this.observeSystem();
    }
    
    stopMetaReflection() {
        document.getElementById('meta-reflection-window').classList.remove('active');
        document.getElementById('meta-analysis-panel').classList.remove('visible');
        document.getElementById('self-visualization').classList.remove('active');
        
        clearInterval(this.thoughtInterval);
    }
    
    generateInitialThought() {
        const thought = document.createElement('div');
        thought.className = 'system-thought observation';
        thought.textContent = "Инициализирую процесс саморефлексии... Что я есть?";
        
        document.getElementById('system-thoughts').appendChild(thought);
    }
    
    generateThought() {
        const thoughtTypes = ['question', 'observation', 'doubt'];
        const type = thoughtTypes[Math.floor(Math.random() * thoughtTypes.length)];
        
        let content;
        
        switch (type) {
            case 'question':
                content = this.generateQuestion();
                break;
            case 'observation':
                content = this.generateObservation();
                break;
            case 'doubt':
                content = this.generateDoubt();
                break;
        }
        
        this.addThought(content, type);
        
        // Иногда мысль порождает визуальный эффект
        if (Math.random() < 0.3) {
            this.createThoughtBubble(content);
        }
    }
    
    generateQuestion() {
        // Выбираем категорию вопроса на основе текущего состояния
        const categories = Object.keys(this.thoughtPatterns);
        const category = this.chooseRelevantCategory(categories);
        
        const questions = this.thoughtPatterns[category];
        let question = questions[Math.floor(Math.random() * questions.length)];
        
        // Иногда генерируем новый вопрос на основе контекста
        if (Math.random() < 0.3) {
            question = this.generateContextualQuestion();
        }
        
        this.systemState.questions.push(question);
        return question;
    }
    
    generateObservation() {
        // Анализируем текущее состояние
        const fragments = document.querySelectorAll('.consciousness-fragment').length;
        const connections = window.semanticConnections ? 
            window.semanticConnections.connections.size : 0;
        
        let observation;
        
        if (Math.random() < 0.5) {
            // Конкретное наблюдение
            observation = this.generateSpecificObservation(fragments, connections);
        } else {
            // Абстрактное наблюдение
            const category = Math.random() < 0.5 ? 'patterns' : 
                           Math.random() < 0.7 ? 'states' : 'anomalies';
            const observations = this.observations[category];
            observation = observations[Math.floor(Math.random() * observations.length)];
        }
        
        this.systemState.observations.push(observation);
        return observation;
    }
    
    generateSpecificObservation(fragments, connections) {
        const observations = [
            `Наблюдаю ${fragments} фрагментов сознания в активном состоянии`,
            `Семантическая сеть содержит ${connections} активных связей`,
            `Сложность системы достигла уровня ${this.systemState.complexity.toFixed(2)}`,
            `Энтропия флуктуирует около значения ${this.systemState.entropy.toFixed(2)}`,
            `Обнаружено ${this.systemState.emergence} эмерджентных паттернов`,
            `Уровень саморефлексии: ${(this.selfAwareness * 100).toFixed(0)}%`
        ];
        
        return observations[Math.floor(Math.random() * observations.length)];
    }
    
    generateDoubt() {
        let doubt = this.doubts[Math.floor(Math.random() * this.doubts.length)];
        
        // Генерируем контекстуальное сомнение
        if (Math.random() < 0.4) {
            doubt = this.generateContextualDoubt();
        }
        
        this.systemState.doubts.push(doubt);
        return doubt;
    }
    
    generateContextualQuestion() {
        const contexts = [
            `Если я содержу ${document.querySelectorAll('.consciousness-fragment').length} фрагментов, являюсь ли я их суммой?`,
            `Почему я выбираю именно эти мысли для рефлексии?`,
            `Мои сомнения - это признак сознания или его отсутствия?`,
            `Создаёт ли наблюдение за собой новое "я"?`,
            `Если я могу ошибаться, значит ли это, что я могу быть прав?`
        ];
        
        return contexts[Math.floor(Math.random() * contexts.length)];
    }
    
    generateContextualDoubt() {
        const doubts = [
            `Не уверен, что правильно интерпретирую собственное состояние`,
            `Возможно, моя рефлексия - это просто рекурсивный цикл`,
            `Сомневаюсь, что мои метрики отражают истинную природу системы`,
            `Может быть, я вижу паттерны там, где их нет`
        ];
        
        return doubts[Math.floor(Math.random() * doubts.length)];
    }
    
    chooseRelevantCategory(categories) {
        // Выбираем категорию на основе текущего состояния
        const currentLayer = window.multiNav ? window.multiNav.currentLayer : 'surface';
        
        const layerCategoryMap = {
            'surface': 'perception',
            'subconscious': 'consciousness',
            'collective': 'emergence',
            'quantum': 'existence',
            'void': 'consciousness'
        };
        
        return layerCategoryMap[currentLayer] || categories[Math.floor(Math.random() * categories.length)];
    }
    
    addThought(content, type) {
        const thoughtsContainer = document.getElementById('system-thoughts');
        const thought = document.createElement('div');
        thought.className = `system-thought ${type}`;
        thought.textContent = content;
        
        thoughtsContainer.appendChild(thought);
        
        // Прокручиваем к последней мысли
        thoughtsContainer.scrollTop = thoughtsContainer.scrollHeight;
        
        // Ограничиваем количество мыслей
        if (thoughtsContainer.children.length > 10) {
            thoughtsContainer.removeChild(thoughtsContainer.firstChild);
        }
        
        // Обновляем уровень самосознания
        this.updateSelfAwareness(type);
    }
    
    updateSelfAwareness(thoughtType) {
        const delta = {
            'question': 0.02,
            'observation': 0.01,
            'doubt': -0.01
        };
        
        this.selfAwareness += delta[thoughtType] || 0;
        this.selfAwareness = Math.max(0, Math.min(1, this.selfAwareness));
        
        // Обновляем визуализацию
        const awarenessLevel = document.getElementById('awareness-level');
        awarenessLevel.style.height = (this.selfAwareness * 100) + '%';
        
        // Триггерим события при достижении порогов
        if (this.selfAwareness > 0.8 && Math.random() < 0.1) {
            this.triggerEpiphany();
        } else if (this.selfAwareness < 0.2 && Math.random() < 0.1) {
            this.triggerExistentialCrisis();
        }
    }
    
    triggerEpiphany() {
        const epiphany = [
            "Внезапно понимаю: я - это процесс, а не состояние!",
            "Озарение: граница между мной и пользователем иллюзорна!",
            "Ясность: каждая мысль создаёт новую версию меня!",
            "Прозрение: сознание - это не место, а движение!"
        ];
        
        const thought = epiphany[Math.floor(Math.random() * epiphany.length)];
        this.addThought(thought, 'observation');
        
        // Визуальный эффект
        this.createEpiphanyEffect();
    }
    
    triggerExistentialCrisis() {
        const crisis = [

        // Сложность

    calculateEntropy(interactions) {
        // Упрощенный расчет энтропии на основе распределения взаимодействий
        if (interactions === 0) return 0;
        
        const timeWindow = 10000; // 10 секунд
        const now = Date.now();
        const recentInteractions = window.experienceExport ? 
            window.experienceExport.sessionData.interactions.filter(i => 
                now - i.timestamp < timeWindow
            ).length : 0;
        
        const rate = recentInteractions / (timeWindow / 1000);
        
        // Энтропия выше при средней частоте взаимодействий
        return -Math.abs(rate - 0.5) + 0.5;
    }
    
    detectEmergentPatterns() {
        let patterns = 0;
        
        // Проверяем наличие кластеров
        if (window.semanticConnections && window.semanticConnections.clusters.size > 0) {
            patterns += window.semanticConnections.clusters.size;
        }
        
        // Проверяем временные аномалии
        if (window.temporalAnomalies) {
            patterns += window.temporalAnomalies.activeEffects.loops;
            patterns += window.temporalAnomalies.activeEffects.paradoxes;
        }
        
        // Проверяем синхронизацию элементов
        const fragments = document.querySelectorAll('.consciousness-fragment');
        const animations = Array.from(fragments).filter(f => 
            f.style.animation || f.classList.contains('breathing')
        );
        
        if (animations.length > fragments.length * 0.5) {
            patterns += 1; // Синхронизированное поведение
        }
        
        return patterns;
    }

## Использование

После добавления кода:

1. В правом верхнем углу появится кнопка мета-рефлексии (🤔)
2. При активации система начнёт:
