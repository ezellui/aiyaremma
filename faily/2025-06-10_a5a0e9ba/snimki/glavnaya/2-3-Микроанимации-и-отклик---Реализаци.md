# 2.3 Микроанимации и отклик

## Описание
Система микроанимаций добавляет тонкие реакции на действия пользователя, создавая ощущение живого, отзывчивого интерфейса.

### 3. JavaScript для микроанимаций (добавить после системы визуальных эффектов)

```javascript
// === СИСТЕМА МИКРОАНИМАЦИЙ И ОТКЛИКА ===

class MicroAnimations {
    constructor() {
        this.mousePosition = { x: 0, y: 0 };
        this.mouseVelocity = { x: 0, y: 0 };
        this.lastMousePosition = { x: 0, y: 0 };
        this.isMouseMoving = false;
        this.mouseStillTimer = null;
        
        this.touchPoints = new Map();
        this.interactionHistory = [];
        
        this.settings = {
            proximityRadius: 150,
            velocityThreshold: 5,
            stillTimeout: 100,
            hapticFeedback: true
        };
        
        this.init();
    }
    
    init() {
        this.setupMouseTracking();
        this.setupTouchTracking();
        this.setupFragmentAnimations();
        this.setupCursorAnimations();
        this.setupMetaPanelAnimations();
        this.setupThoughtStreamAnimations();
        this.setupIdleAnimations();
        this.addBreathingEffect();
    }
    
    setupMouseTracking() {
        let lastTime = Date.now();
        
        document.addEventListener('mousemove', (e) => {
            const currentTime = Date.now();
            const deltaTime = currentTime - lastTime;
            
            // Рассчитываем скорость движения мыши
            this.mouseVelocity = {
                x: (e.clientX - this.lastMousePosition.x) / deltaTime * 100,
                y: (e.clientY - this.lastMousePosition.y) / deltaTime * 100
            };
            
            this.lastMousePosition = { x: this.mousePosition.x, y: this.mousePosition.y };
            this.mousePosition = { x: e.clientX, y: e.clientY };
            
            // Проверяем, движется ли мышь
            this.isMouseMoving = true;
            clearTimeout(this.mouseStillTimer);
            this.mouseStillTimer = setTimeout(() => {
                this.isMouseMoving = false;
                this.onMouseStop();
            }, this.settings.stillTimeout);
            
            // Обновляем эффекты близости
            this.updateProximityEffects();
            
            // Добавляем микроанимации на основе скорости
            this.applyVelocityEffects();
            
            lastTime = currentTime;
        });
        
        // Эффекты при клике
        document.addEventListener('mousedown', (e) => {
            const cursor = document.getElementById('custom-cursor');
            cursor.classList.add('clicking');
            
            // Тактильный отклик
            this.createTactileFeedback(e.clientX, e.clientY);
        });
        
        document.addEventListener('mouseup', () => {
            const cursor = document.getElementById('custom-cursor');
            cursor.classList.remove('clicking');
        });
    }
    
    setupTouchTracking() {
        // Для мобильных устройств
        document.addEventListener('touchstart', (e) => {
            for (const touch of e.touches) {
                this.touchPoints.set(touch.identifier, {
                    x: touch.clientX,
                    y: touch.clientY,
                    startTime: Date.now()
                });
                
                this.createTactileFeedback(touch.clientX, touch.clientY);
            }
        });
        
        document.addEventListener('touchmove', (e) => {
            for (const touch of e.touches) {
                const point = this.touchPoints.get(touch.identifier);
                if (point) {
                    point.x = touch.clientX;
                    point.y = touch.clientY;
                }
            }
        });
        
        document.addEventListener('touchend', (e) => {
            for (const touch of e.changedTouches) {
                this.touchPoints.delete(touch.identifier);
            }
        });
    }
    
    updateProximityEffects() {
        const fragments = document.querySelectorAll('.consciousness-fragment');
        
        fragments.forEach(fragment => {
            const rect = fragment.getBoundingClientRect();
            const centerX = rect.left + rect.width / 2;
            const centerY = rect.top + rect.height / 2;
            
            const distance = Math.sqrt(
                Math.pow(this.mousePosition.x - centerX, 2) +
                Math.pow(this.mousePosition.y - centerY, 2)
            );
            
            // Удаляем все классы близости
            fragment.classList.remove('proximity-near', 'proximity-medium', 'proximity-far');
            
            // Добавляем соответствующий класс
            if (distance < this.settings.proximityRadius / 3) {
                fragment.classList.add('proximity-near');
                
                // Добавляем микроанимацию при очень близком расстоянии
                if (distance < this.settings.proximityRadius / 6 && !fragment.dataset.animated) {
                    this.applyMicroAnimation(fragment, 'pulse');
                    fragment.dataset.animated = 'true';
                    setTimeout(() => delete fragment.dataset.animated, 1000);
                }
            } else if (distance < this.settings.proximityRadius * 2 / 3) {
                fragment.classList.add('proximity-medium');
            } else if (distance < this.settings.proximityRadius) {
                fragment.classList.add('proximity-far');
            }
        });
    }
    
    applyVelocityEffects() {
        const velocity = Math.sqrt(
            this.mouseVelocity.x ** 2 + 
            this.mouseVelocity.y ** 2
        );
        
        // Эффекты на основе скорости движения
        if (velocity > this.settings.velocityThreshold * 10) {
            // Очень быстрое движение - добавляем волны
            const fragments = document.querySelectorAll('.consciousness-fragment');
            fragments.forEach((fragment, index) => {
                setTimeout(() => {
                    this.applyMicroAnimation(fragment, 'float');
                }, index * 20);
            });
        } else if (velocity > this.settings.velocityThreshold * 5) {
            // Среднее движение - легкая тряска ближайших элементов
            const nearbyFragments = this.getNearbyElements(100);
            nearbyFragments.forEach(fragment => {
                this.applyMicroAnimation(fragment, 'shake');
            });
        }
        
        // Изменяем курсор в зависимости от скорости
        const cursor = document.getElementById('custom-cursor');
        const scale = 1 + Math.min(velocity / 200, 0.5);
        cursor.style.transform = `translate(-50%, -50%) scale(${scale})`;
    }
    
    getNearbyElements(radius) {
        const fragments = document.querySelectorAll('.consciousness-fragment');
        const nearby = [];
        
        fragments.forEach(fragment => {
            const rect = fragment.getBoundingClientRect();
            const centerX = rect.left + rect.width / 2;
            const centerY = rect.top + rect.height / 2;
            
            const distance = Math.sqrt(
                Math.pow(this.mousePosition.x - centerX, 2) +
                Math.pow(this.mousePosition.y - centerY, 2)
            );
            
            if (distance < radius) {
                nearby.push(fragment);
            }
        });
        
        return nearby;
    }
    
    applyMicroAnimation(element, type) {
        // Удаляем предыдущие классы анимации
        element.classList.remove('micro-pulse', 'micro-shake', 'micro-float');
        
        // Форсируем перерисовку
        void element.offsetWidth;
        
        // Добавляем новый класс анимации
        element.classList.add(`micro-${type}`);
        
        // Удаляем класс после завершения анимации
        const duration = type === 'float' ? 2000 : 600;
        setTimeout(() => {
            element.classList.remove(`micro-${type}`);
        }, duration);
    }
    
    onMouseStop() {
        // Эффекты когда мышь остановилась
        const nearbyElements = this.getNearbyElements(150);
        
        nearbyElements.forEach((element, index) => {
            setTimeout(() => {
                element.classList.add('wave-hover');
                setTimeout(() => {
                    element.classList.remove('wave-hover');
                }, 1000);
            }, index * 50);
        });
    }
    
    setupFragmentAnimations() {
        // Переопределяем создание фрагментов для добавления анимации появления
        const originalCreate = window.createConsciousnessFragment;
        window.createConsciousnessFragment = function() {
            const fragment = originalCreate();
            fragment.classList.add('fade-in-up');
            
            // Реакция на наведение
            fragment.addEventListener('mouseenter', () => {
                microAnimations.applyMicroAnimation(fragment, 'pulse');
                
                // Записываем взаимодействие
                microAnimations.recordInteraction('hover', fragment);
            });
            
            // Реакция на клик
            fragment.addEventListener('click', () => {
                microAnimations.applyMicroAnimation(fragment, 'shake');
                microAnimations.showProcessingIndicator(500);
                
                // Записываем взаимодействие
                microAnimations.recordInteraction('click', fragment);
            });
            
            return fragment;
        };
    }
    
    setupCursorAnimations() {
        const cursor = document.getElementById('custom-cursor');
        
        // Пульсация курсора при бездействии
        setInterval(() => {
            if (!this.isMouseMoving) {
                cursor.style.width = '25px';
                cursor.style.height = '25px';
                
                setTimeout(() => {
                    cursor.style.width = '20px';
                    cursor.style.height = '20px';
                }, 200);
            }
        }, 2000);
    }
    
    setupMetaPanelAnimations() {
        // Анимация обновления значений
        const originalUpdateMeta = window.updateMetaPanel;
        window.updateMetaPanel = function() {
            originalUpdateMeta();
            
            // Анимируем изменение значений
            const values = document.querySelectorAll('.meta-value');
            values.forEach(value => {
                value.classList.add('updating');
                setTimeout(() => {
                    value.classList.remove('updating');
                }, 500);
            });
        };
    }
    
    setupThoughtStreamAnimations() {
        const thoughtStream = document.getElementById('thought-stream');
        
        // Добавляем индикатор "мышления"
        const originalUpdate = window.updateThoughtStream;
        window.updateThoughtStream = function() {
            thoughtStream.classList.add('thinking');
            originalUpdate();
            
            setTimeout(() => {
                thoughtStream.classList.remove('thinking');
            }, 3000);
        };
    }
    
    setupIdleAnimations() {
        let idleTimer = null;
        let isIdle = false;
        
        const startIdleMode = () => {
            isIdle = true;
            document.body.classList.add('idle-mode');
            
            // Запускаем медленные анимации для всех элементов
            const fragments = document.querySelectorAll('.consciousness-fragment');
            fragments.forEach((fragment, index) => {
                setTimeout(() => {
                    fragment.classList.add('breathing');
                }, index * 100);
            });
        };
        
        const stopIdleMode = () => {
            isIdle = false;
            document.body.classList.remove('idle-mode');
            
            const fragments = document.querySelectorAll('.consciousness-fragment');
            fragments.forEach(fragment => {
                fragment.classList.remove('breathing');
            });
        };
        
        const resetIdleTimer = () => {
            clearTimeout(idleTimer);
            
            if (isIdle) {
                stopIdleMode();
            }
            
            idleTimer = setTimeout(startIdleMode, 30000); // 30 секунд бездействия
        };
        
        // Отслеживаем активность
        ['mousedown', 'mousemove', 'keypress', 'scroll', 'touchstart'].forEach(event => {
            document.addEventListener(event, resetIdleTimer, true);
        });
        
        resetIdleTimer();
    }
    
    addBreathingEffect() {
        // Добавляем эффект дыхания к основным элементам
        const cartograph = document.getElementById('cartograph');
        cartograph.classList.add('breathing');
        
        // Математические элементы тоже "дышат"
        const observer = new MutationObserver((mutations) => {
            mutations.forEach((mutation) => {
                mutation.addedNodes.forEach((node) => {
                    if (node.classList && node.classList.contains('math-element')) {
                        // Случайная задержка для несинхронного дыхания
                        const delay = Math.random() * 4;
                        node.style.animationDelay = `${delay}s`;
                        node.classList.add('breathing');
                    }
                });
            });
        });
        
        observer.observe(document.getElementById('container'), {
            childList: true,
            subtree: true
        });
    }
    
    createTactileFeedback(x, y) {
        const feedback = document.createElement('div');
        feedback.className = 'tactile-feedback';
        feedback.style.left = (x - 20) + 'px';
        feedback.style.top = (y - 20) + 'px';
        
        document.body.appendChild(feedback);
        
        // Вибрация на мобильных устройствах (если поддерживается)
        if (this.settings.hapticFeedback && 'vibrate' in navigator) {
            navigator.vibrate(10);
        }
        
        setTimeout(() => feedback.remove(), 600);
    }
    
    showProcessingIndicator(duration = 1000) {
        const indicator = document.getElementById('processing-indicator');
        indicator.classList.add('active');
        
        setTimeout(() => {
            indicator.classList.remove('active');
        }, duration);
    }
    
    recordInteraction(type, element) {
        // Записываем историю взаимодействий для адаптивного поведения
        this.interactionHistory.push({
            type: type,
            timestamp: Date.now(),
            element: element.textContent,
            position: { ...this.mousePosition }
        });
        
        // Ограничиваем размер истории
        if (this.interactionHistory.length > 100) {
            this.interactionHistory.shift();
        }
        
        // Анализируем паттерны взаимодействия
        this.analyzeInteractionPatterns();
    }
    
    analyzeInteractionPatterns() {
        // Простой анализ частоты взаимодействий
        const recentInteractions = this.interactionHistory.filter(
            i => Date.now() - i.timestamp < 10000
        );
        
        if (recentInteractions.length > 20) {
            // Пользователь очень активен - ускоряем анимации
            document.body.classList.add('high-activity');
        } else if (recentInteractions.length < 5) {
            // Пользователь малоактивен - замедляем анимации
            document.body.classList.add('low-activity');
        } else {
            // Нормальная активность
            document.body.classList.remove('high-activity', 'low-activity');
        }
    }
}

// Инициализируем систему микроанимаций
const microAnimations = new MicroAnimations();

// Экспортируем для глобального доступа
window.microAnimations = microAnimations;

// CSS для режимов активности (добавить в стили)
const activityStyles = document.createElement('style');
activityStyles.textContent = `

document.head.appendChild(activityStyles);
```

## Использование

После добавления всего кода:

1. Элементы будут реагировать на приближение курсора
2. Быстрое движение мыши вызовет волновые эффекты
3. При остановке курсора элементы начнут "дышать"
4. Клики создадут тактильную обратную связь
5. Система адаптируется к активности пользователя

## Настройка чувствительности

Можно изменить параметры через:
```javascript
microAnimations.settings.proximityRadius = 200; // Радиус реакции
microAnimations.settings.velocityThreshold = 10; // Порог скорости
```
