# 2.2 Продвинутые визуальные эффекты

## Описание
Система продвинутых визуальных эффектов добавляет шейдерные эффекты, частицы, искажения пространства и другие визуальные элементы для создания более погруженного опыта.

### 4. JavaScript для визуальных эффектов (добавить после системы адаптивной типографики)

```javascript
// === СИСТЕМА ПРОДВИНУТЫХ ВИЗУАЛЬНЫХ ЭФФЕКТОВ ===

class AdvancedVisualEffects {
    constructor() {
        this.canvas = document.getElementById('effects-canvas');
        this.ctx = this.canvas.getContext('2d');
        this.particles = [];
        this.connections = [];
        this.mouseTrail = [];
        this.energyNodes = new Map();
        
        this.settings = {
            particleCount: 50,
            particleSpeed: 0.5,
            connectionDistance: 150,
            mouseTrailLength: 20,
            energyLineThreshold: 200,
            distortionStrength: 5
        };
        
        this.init();
    }
    
    init() {
        // Настраиваем canvas
        this.resizeCanvas();
        window.addEventListener('resize', () => this.resizeCanvas());
        
        // Создаем частицы
        this.createParticles();
        
        // Настраиваем эффекты мыши
        this.setupMouseEffects();
        
        // Настраиваем энергетические связи
        this.setupEnergyConnections();
        
        // Добавляем эффекты к элементам
        this.applyElementEffects();
        
        // Запускаем анимационный цикл
        this.animate();
        
        // Добавляем периодические эффекты
        this.startPeriodicEffects();
    }
    
    resizeCanvas() {
        this.canvas.width = window.innerWidth;
        this.canvas.height = window.innerHeight;
    }
    
    createParticles() {
        for (let i = 0; i < this.settings.particleCount; i++) {
            this.particles.push({
                x: Math.random() * this.canvas.width,
                y: Math.random() * this.canvas.height,
                vx: (Math.random() - 0.5) * this.settings.particleSpeed,
                vy: (Math.random() - 0.5) * this.settings.particleSpeed,
                size: Math.random() * 3 + 1,
                opacity: Math.random() * 0.5 + 0.3,
                hue: Math.random() * 60 + 180 // Голубоватые оттенки
            });
        }
    }
    
    setupMouseEffects() {
        let mouseX = 0;
        let mouseY = 0;
        
        document.addEventListener('mousemove', (e) => {
            mouseX = e.clientX;
            mouseY = e.clientY;
            
            // Добавляем точку в след мыши
            this.mouseTrail.push({
                x: mouseX,
                y: mouseY,
                opacity: 1,
                size: 5
            });
            
            // Ограничиваем длину следа
            if (this.mouseTrail.length > this.settings.mouseTrailLength) {
                this.mouseTrail.shift();
            }
            
            // Создаем рябь при быстром движении
            const speed = Math.sqrt(e.movementX ** 2 + e.movementY ** 2);
            if (speed > 20) {
                this.createRipple(mouseX, mouseY);
            }
        });
        
        // Эффект при клике
        document.addEventListener('click', (e) => {
            this.createExplosion(e.clientX, e.clientY);
            this.createRipple(e.clientX, e.clientY, 3);
        });
    }
    
    createRipple(x, y, count = 1) {
        for (let i = 0; i < count; i++) {
            setTimeout(() => {
                const ripple = document.createElement('div');
                ripple.className = 'ripple-effect';
                ripple.style.width = '50px';
                ripple.style.height = '50px';
                ripple.style.left = (x - 25) + 'px';
                ripple.style.top = (y - 25) + 'px';
                
                document.body.appendChild(ripple);
                
                setTimeout(() => ripple.remove(), 2000);
            }, i * 100);
        }
    }
    
    createExplosion(x, y) {
        const particleCount = 20;
        
        for (let i = 0; i < particleCount; i++) {
            const angle = (Math.PI * 2 * i) / particleCount;
            const velocity = 2 + Math.random() * 3;
            
            this.particles.push({
                x: x,
                y: y,
                vx: Math.cos(angle) * velocity,
                vy: Math.sin(angle) * velocity,
                size: Math.random() * 4 + 2,
                opacity: 1,
                hue: Math.random() * 360,
                lifetime: 100,
                decay: 0.02
            });
        }
    }
    
    setupEnergyConnections() {
        // Отслеживаем положения важных элементов
        const fragments = document.querySelectorAll('.consciousness-fragment');
        
        fragments.forEach((fragment, index) => {
            this.energyNodes.set(fragment, {
                index: index,
                connections: new Set()
            });
        });
        
        // Создаем MutationObserver для отслеживания новых элементов
        const observer = new MutationObserver((mutations) => {
            mutations.forEach((mutation) => {
                mutation.addedNodes.forEach((node) => {
                    if (node.classList && node.classList.contains('consciousness-fragment')) {
                        this.energyNodes.set(node, {
                            index: this.energyNodes.size,
                            connections: new Set()
                        });
                    }
                });
            });
        });
        
        observer.observe(document.getElementById('container'), {
            childList: true,
            subtree: true
        });
    }
    
    applyElementEffects() {
        // Добавляем хроматическую аберрацию к заголовку
        const title = document.getElementById('title');
        title.classList.add('chromatic-aberration');
        
        // Добавляем искажение пространства к картографу
        const cartograph = document.getElementById('cartograph');
        cartograph.classList.add('space-distortion');
        
        // Добавляем голографический эффект к некоторым фрагментам
        const fragments = document.querySelectorAll('.consciousness-fragment');
        fragments.forEach((fragment, index) => {
            if (index % 5 === 0) {
                fragment.classList.add('holographic');
            }
            
            // Добавляем эффект глубины
            const z = parseFloat(fragment.style.transform.match(/translateZ\(([^)]+)px\)/)?.[1] || 0);
            if (z > 100) {
                fragment.classList.add('depth-blur-far');
            } else if (z > 0) {
                fragment.classList.add('depth-blur-mid');
            } else {
                fragment.classList.add('depth-blur-near');
            }
        });
    }
    
    drawConnections() {
        this.ctx.strokeStyle = 'rgba(100, 200, 255, 0.1)';
        this.ctx.lineWidth = 1;
        
        // Соединяем близкие частицы
        for (let i = 0; i < this.particles.length; i++) {
            for (let j = i + 1; j < this.particles.length; j++) {
                const dx = this.particles[i].x - this.particles[j].x;
                const dy = this.particles[i].y - this.particles[j].y;
                const distance = Math.sqrt(dx * dx + dy * dy);
                
                if (distance < this.settings.connectionDistance) {
                    const opacity = 1 - (distance / this.settings.connectionDistance);
                    this.ctx.globalAlpha = opacity * 0.3;
                    
                    this.ctx.beginPath();
                    this.ctx.moveTo(this.particles[i].x, this.particles[i].y);
                    this.ctx.lineTo(this.particles[j].x, this.particles[j].y);
                    this.ctx.stroke();
                }
            }
        }
        
        this.ctx.globalAlpha = 1;
    }
    
    drawMouseTrail() {
        this.mouseTrail.forEach((point, index) => {
            const opacity = point.opacity * (index / this.mouseTrail.length);
            const size = point.size * (1 - index / this.mouseTrail.length);
            
            this.ctx.beginPath();
            this.ctx.arc(point.x, point.y, size, 0, Math.PI * 2);
            this.ctx.fillStyle = `rgba(255, 255, 255, ${opacity})`;
            this.ctx.fill();
            
            // Уменьшаем прозрачность
            point.opacity *= 0.95;
        });
        
        // Удаляем слишком прозрачные точки
        this.mouseTrail = this.mouseTrail.filter(point => point.opacity > 0.01);
    }
    
    updateParticles() {
        this.particles.forEach((particle, index) => {
            // Обновляем позицию
            particle.x += particle.vx;
            particle.y += particle.vy;
            
            // Отскок от границ
            if (particle.x < 0 || particle.x > this.canvas.width) {
                particle.vx = -particle.vx;
            }
            if (particle.y < 0 || particle.y > this.canvas.height) {
                particle.vy = -particle.vy;
            }
            
            // Обновляем время жизни для временных частиц
            if (particle.lifetime) {
                particle.lifetime--;
                particle.opacity -= particle.decay;
                
                if (particle.lifetime <= 0 || particle.opacity <= 0) {
                    this.particles.splice(index, 1);
                }
            }
        });
    }
    
    drawParticles() {
        this.particles.forEach(particle => {
            this.ctx.beginPath();
            this.ctx.arc(particle.x, particle.y, particle.size, 0, Math.PI * 2);
            
            const gradient = this.ctx.createRadialGradient(
                particle.x, particle.y, 0,
                particle.x, particle.y, particle.size
            );
            
            gradient.addColorStop(0, `hsla(${particle.hue}, 70%, 70%, ${particle.opacity})`);
            gradient.addColorStop(1, `hsla(${particle.hue}, 70%, 70%, 0)`);
            
            this.ctx.fillStyle = gradient;
            this.ctx.fill();
        });
    }
    
    animate() {
        // Очищаем canvas
        this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);
        
        // Обновляем и рисуем элементы
        this.updateParticles();
        this.drawConnections();
        this.drawParticles();
        this.drawMouseTrail();
        
        // Следующий кадр
        requestAnimationFrame(() => this.animate());
    }
    
    startPeriodicEffects() {
        // Периодическое добавление атмосферных частиц
        setInterval(() => {
            if (this.particles.length < this.settings.particleCount * 1.5) {
                this.createParticles();
            }
        }, 5000);
        
        // Периодические волны искажения
        setInterval(() => {
            const distortionLayer = document.getElementById('distortion-layer');
            distortionLayer.style.opacity = '1';
            
            setTimeout(() => {
                distortionLayer.style.opacity = '0';
            }, 3000);
        }, 15000);
        
        // Энергетические импульсы между элементами
        setInterval(() => {
            this.createEnergyPulse();
        }, 8000);
    }
    
    createEnergyPulse() {
        const fragments = Array.from(document.querySelectorAll('.consciousness-fragment'));
        if (fragments.length < 2) return;
        
        // Выбираем два случайных фрагмента
        const index1 = Math.floor(Math.random() * fragments.length);
        let index2 = Math.floor(Math.random() * fragments.length);
        while (index2 === index1) {
            index2 = Math.floor(Math.random() * fragments.length);
        }
        
        const fragment1 = fragments[index1];
        const fragment2 = fragments[index2];
        
        const rect1 = fragment1.getBoundingClientRect();
        const rect2 = fragment2.getBoundingClientRect();
        
        const x1 = rect1.left + rect1.width / 2;
        const y1 = rect1.top + rect1.height / 2;
        const x2 = rect2.left + rect2.width / 2;
        const y2 = rect2.top + rect2.height / 2;
        
        // Создаем линию энергии
        const line = document.createElement('div');
        line.className = 'energy-line';
        
        const distance = Math.sqrt((x2 - x1) ** 2 + (y2 - y1) ** 2);
        const angle = Math.atan2(y2 - y1, x2 - x1);
        
        line.style.width = distance + 'px';
        line.style.left = x1 + 'px';
        line.style.top = y1 + 'px';
        line.style.transform = `rotate(${angle}rad)`;
        
        document.body.appendChild(line);
        
        // Удаляем линию через некоторое время
        setTimeout(() => line.remove(), 2000);
    }
    
    // Метод для изменения настроек эффектов
    updateSettings(newSettings) {
        this.settings = { ...this.settings, ...newSettings };
        
        // Обновляем фильтр искажения
        const filter = document.querySelector('#distortion-filter feDisplacementMap');
        if (filter && newSettings.distortionStrength !== undefined) {
            filter.setAttribute('scale', newSettings.distortionStrength);
        }
    }
}

// Инициализируем систему визуальных эффектов
const visualEffects = new AdvancedVisualEffects();

// Экспортируем для глобального доступа
window.visualEffects = visualEffects;
```

## Использование

После добавления всего кода:

1. Canvas будет отображать анимированные частицы и связи между ними
2. Элементы получат различные визуальные эффекты (хроматическая аберрация, голографический эффект и т.д.)
3. Мышь будет оставлять светящийся след
4. При клике будут создаваться взрывы частиц и рябь
5. Периодически будут появляться энергетические импульсы между элементами

## Оптимизация производительности

- Используйте `requestAnimationFrame` для анимаций
- Ограничивайте количество частиц
- Применяйте `will-change` к анимированным элементам
- Используйте GPU-ускорение через `transform` и `opacity`
