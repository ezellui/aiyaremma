# 2.1 Адаптивная типографика

## Описание
Система адаптивной типографики создает динамические изменения шрифтов в зависимости от контекста, движения курсора и семантического содержания текста.

## Реализация

### 2. JavaScript для адаптивной типографики (добавить в блок `<script>` после инициализации системы состояний)

```javascript
// === СИСТЕМА АДАПТИВНОЙ ТИПОГРАФИКИ ===

class AdaptiveTypography {
    constructor() {
        this.cursorInfluence = {
            radius: 200,
            strength: 0.5
        };
        
        this.semanticWeights = {
            // Философские термины - тяжелые
            'онтология': 'heavy',
            'эпистемология': 'heavy',
            'феноменология': 'heavy',
            'экзистенция': 'heavy',
            
            // Абстрактные понятия - легкие
            'эфемерный': 'light',
            'невесомый': 'light',
            'призрачный': 'light',
            'иллюзорный': 'light',
            
            // Трансцендентные - эфирные
            'трансцендентный': 'ethereal',
            'запредельный': 'ethereal',
            'невыразимый': 'ethereal',
            'бесконечный': 'ethereal'
        };
        
        this.init();
    }
    
    init() {
        // Применяем базовые классы ко всем фрагментам
        this.updateAllFragments();
        
        // Отслеживаем движение мыши для динамической типографики
        this.setupMouseTracking();
        
        // Настраиваем семантический анализ
        this.setupSemanticAnalysis();
        
        // Запускаем анимации типографики
        this.startTypographyAnimations();
    }
    
    updateAllFragments() {
        const fragments = document.querySelectorAll('.consciousness-fragment');
        fragments.forEach(fragment => {
            fragment.classList.add('typography-morphing', 'typography-responsive');
            this.analyzeAndApplyTypography(fragment);
        });
    }
    
    setupMouseTracking() {
        let mouseX = window.innerWidth / 2;
        let mouseY = window.innerHeight / 2;
        
        document.addEventListener('mousemove', (e) => {
            mouseX = e.clientX;
            mouseY = e.clientY;
            
            // Обновляем типографику фрагментов в зависимости от расстояния до курсора
            const fragments = document.querySelectorAll('.consciousness-fragment');
            fragments.forEach(fragment => {
                const rect = fragment.getBoundingClientRect();
                const fragmentX = rect.left + rect.width / 2;
                const fragmentY = rect.top + rect.height / 2;
                
                const distance = Math.sqrt(
                    Math.pow(mouseX - fragmentX, 2) + 
                    Math.pow(mouseY - fragmentY, 2)
                );
                
                if (distance < this.cursorInfluence.radius) {
                    const influence = 1 - (distance / this.cursorInfluence.radius);
                    this.applyDynamicTypography(fragment, influence);
                } else {
                    this.resetTypography(fragment);
                }
            });
        });
    }
    
    applyDynamicTypography(element, influence) {
        const baseWeight = 400;
        const maxWeight = 800;
        const weight = baseWeight + (maxWeight - baseWeight) * influence * this.cursorInfluence.strength;
        
        const baseWidth = 100;
        const maxWidth = 110;
        const width = baseWidth + (maxWidth - baseWidth) * influence * this.cursorInfluence.strength;
        
        const baseSpacing = 0.05;
        const minSpacing = -0.02;
        const spacing = baseSpacing - (baseSpacing - minSpacing) * influence * this.cursorInfluence.strength;
        
        element.style.setProperty('--font-weight', weight);
        element.style.setProperty('--font-width', width);
        element.style.setProperty('--letter-spacing', `${spacing}em`);
        
        // Добавляем легкое увеличение размера
        const scale = 1 + influence * 0.1;
        element.style.transform = `scale(${scale})`;
    }
    
    resetTypography(element) {
        element.style.setProperty('--font-weight', '400');
        element.style.setProperty('--font-width', '100');
        element.style.setProperty('--letter-spacing', '0.05em');
        element.style.transform = 'scale(1)';
    }
    
    setupSemanticAnalysis() {
        // Анализируем новые фрагменты при их создании
        const originalCreateFragment = window.createConsciousnessFragment;
        window.createConsciousnessFragment = () => {
            const fragment = originalCreateFragment();
            setTimeout(() => {
                this.analyzeAndApplyTypography(fragment);
            }, 100);
            return fragment;
        };
    }
    
    analyzeAndApplyTypography(fragment) {
        const text = fragment.textContent.toLowerCase();
        
        // Проверяем семантический вес
        for (const [keyword, weight] of Object.entries(this.semanticWeights)) {
            if (text.includes(keyword)) {
                fragment.classList.add(`semantic-${weight}`);
                break;
            }
        }
        
        // Определяем тип контента
        if (this.isMathematical(text)) {
            fragment.classList.add('fragment-mathematical');
        } else if (this.isPhilosophical(text)) {
            fragment.classList.add('fragment-philosophical');
        } else if (this.isTechnical(text)) {
            fragment.classList.add('fragment-technical');
        } else if (this.isPoetic(text)) {
            fragment.classList.add('fragment-poetic');
        }
        
        // Случайные анимации для некоторых фрагментов
        if (Math.random() < 0.1) {
            fragment.classList.add('typography-pulse');
        } else if (Math.random() < 0.05) {
            fragment.classList.add('typography-wave');
        }
        
        // Глитч-эффект для особых слов
        if (text.includes('квант') || text.includes('глитч') || text.includes('хаос')) {
            fragment.classList.add('typography-glitch');
            fragment.setAttribute('data-text', fragment.textContent);
        }
    }
    
    isMathematical(text) {
        return /[∑∏∫∬∭∮√∛∜∞]/.test(text) || 
               /\d+/.test(text) ||
               text.includes('функц') ||
               text.includes('интеграл');
    }
    
    isPhilosophical(text) {
        const philosophicalTerms = [
            'бытие', 'сущность', 'феномен', 'ноумен', 
            'трансцендент', 'имманент', 'экзистенц'
        ];
        return philosophicalTerms.some(term => text.includes(term));
    }
    
    isTechnical(text) {
        const technicalTerms = [
            'алгоритм', 'функция', 'система', 'процесс',
            'структура', 'данные', 'параметр'
        ];
        return technicalTerms.some(term => text.includes(term));
    }
    
    isPoetic(text) {
        const poeticTerms = [
            'мечта', 'душа', 'небо', 'звезда',
            'вечность', 'любовь', 'красота'
        ];
        return poeticTerms.some(term => text.includes(term));
    }
    
    startTypographyAnimations() {
        // Анимация потока мыслей
        const thoughtStream = document.getElementById('thought-stream');
        let thinkingSpeed = 'normal';
        
        setInterval(() => {
            if (thinkingSpeed === 'normal') {
                thoughtStream.classList.remove('thinking-slow');
                thoughtStream.classList.add('thinking-fast');
                thinkingSpeed = 'fast';
            } else if (thinkingSpeed === 'fast') {
                thoughtStream.classList.remove('thinking-fast');
                thoughtStream.classList.add('thinking-slow');
                thinkingSpeed = 'slow';
            } else {
                thoughtStream.classList.remove('thinking-slow');
                thinkingSpeed = 'normal';
            }
        }, 10000);
        
        // Пульсация заголовка
        const title = document.getElementById('title');
        title.classList.add('typography-morphing');
        
        let titlePulse = 0;
        setInterval(() => {
            titlePulse += 0.1;
            const weight = 400 + Math.sin(titlePulse) * 300;
            const width = 100 + Math.sin(titlePulse * 0.7) * 10;
            const spacing = 3 + Math.sin(titlePulse * 1.3) * 2;
            
            title.style.setProperty('--font-weight', weight);
            title.style.setProperty('--font-width', width);
            title.style.letterSpacing = `${spacing}px`;
        }, 100);
    }
    
    // Метод для изменения глобальных настроек типографики
    setGlobalTypography(settings) {
        const root = document.documentElement;
        
        if (settings.fontSize) {
            root.style.setProperty('--base-font-size', settings.fontSize);
        }
        
        if (settings.weightRange) {
            root.style.setProperty('--font-weight-min', settings.weightRange[0]);
            root.style.setProperty('--font-weight-max', settings.weightRange[1]);
        }
        
        if (settings.cursorInfluence) {
            this.cursorInfluence = { ...this.cursorInfluence, ...settings.cursorInfluence };
        }
    }
}

// Инициализируем систему адаптивной типографики
const adaptiveTypography = new AdaptiveTypography();

// Экспортируем для глобального доступа
window.adaptiveTypography = adaptiveTypography;
```

```javascript
    if (typographySettings[stateName]) {

}
```

## Использование

После добавления всего кода:

1. Все текстовые элементы получат адаптивную типографику
2. Текст будет реагировать на приближение курсора
3. Семантически важные слова получат соответствующий вес
4. Различные типы контента (математический, философский и т.д.) будут стилизованы по-разному
5. Некоторые элементы получат анимированную типографику
