# 1.1 Система состояний сознания

## Описание
Система позволяет переключаться между различными "режимами сознания", каждый из которых кардинально меняет визуальную эстетику, звуковой ландшафт и поведение элементов.

let currentConsciousnessState = 'lucid';
let stateTransitioning = false;

// Инициализация системы состояний
function initConsciousnessStates() {
    const selector = document.getElementById('consciousness-state-selector');
    const buttons = selector.querySelectorAll('.state-button');
    
    // Устанавливаем начальное состояние
    buttons.forEach(button => {
        if (button.dataset.state === currentConsciousnessState) {
            button.classList.add('active');
        }
        
        button.addEventListener('click', function() {
            if (stateTransitioning) return;
            
            const newState = this.dataset.state;
            if (newState !== currentConsciousnessState) {
                transitionToState(newState);
            }
        });
    });
    
    // Применяем начальное состояние
    applyConsciousnessState(currentConsciousnessState);
}

// Переход между состояниями
async function transitionToState(newState) {
    stateTransitioning = true;
    
    // Визуальный эффект перехода
    const overlay = document.createElement('div');
    overlay.style.cssText = `
        position: fixed;
        top: 0;
        left: 0;
        width: 100%;
        height: 100%;
        background: radial-gradient(circle, transparent 0%, black 100%);
        opacity: 0;
        z-index: 9998;
        pointer-events: none;
        transition: opacity 1.5s;
    `;
    document.body.appendChild(overlay);
    
    // Затемнение
    setTimeout(() => {
        overlay.style.opacity = '1';
    }, 10);
    
    // Смена состояния
    setTimeout(() => {
        // Обновляем активную кнопку
        document.querySelectorAll('.state-button').forEach(btn => {
            btn.classList.toggle('active', btn.dataset.state === newState);
        });
        
        currentConsciousnessState = newState;
        applyConsciousnessState(newState);
        
        // Убираем затемнение
        setTimeout(() => {
            overlay.style.opacity = '0';
            setTimeout(() => {
                overlay.remove();
                stateTransitioning = false;
            }, 1500);
        }, 500);
    }, 1500);
}

// Применение состояния сознания
function applyConsciousnessState(stateName) {
    const state = consciousnessStates[stateName];
    
    // Изменяем класс body
    document.body.className = `consciousness-${stateName}`;
    
    // Обновляем цветовую схему
    updateColorScheme(state.colorScheme);
    
    // Обновляем параметры фрагментов
    updateFragmentBehavior(state);
    
    // Обновляем аудио параметры

    // Обновляем скорость мыслей

    // Обновляем визуальные эффекты
    applyVisualEffects(state.effects);
    
    // Обновляем математическую сложность
    updateMathComplexity(state.mathComplexity);
}

// Обновление поведения фрагментов
function updateFragmentBehavior(state) {
    // Изменяем скорость существующих фрагментов
    fragmentAnimations.forEach((params, fragment) => {
        params.vx *= state.fragmentSpeed;
        params.vy *= state.fragmentSpeed;
        params.vz *= state.fragmentSpeed;
        params.vRotX *= state.fragmentSpeed;
        params.vRotY *= state.fragmentSpeed;
        params.vRotZ *= state.fragmentSpeed;
    });
    
    // Регулируем плотность фрагментов
    const currentFragments = document.querySelectorAll('.consciousness-fragment');
    const targetCount = Math.floor(20 * state.fragmentDensity);
    const currentCount = currentFragments.length;
    
    if (currentCount > targetCount) {
        // Удаляем лишние фрагменты
        for (let i = targetCount; i < currentCount; i++) {
            setTimeout(() => {
                const fragment = currentFragments[i];
                if (fragment) {
                    fragment.style.opacity = '0';
                    setTimeout(() => fragment.remove(), 1000);
                }
            }, i * 100);
        }
    } else if (currentCount < targetCount) {
        // Добавляем недостающие фрагменты
        for (let i = 0; i < targetCount - currentCount; i++) {
            setTimeout(() => createConsciousnessFragment(), i * 200);
        }
    }
}

function updateMathComplexity(complexity) {

// Вспомогательная функция для получения hue из цвета
function getHueFromColor(color) {
    if (color.startsWith('#')) {
        const r = parseInt(color.slice(1, 3), 16) / 255;
        const g = parseInt(color.slice(3, 5), 16) / 255;
        const b = parseInt(color.slice(5, 7), 16) / 255;
        
        const max = Math.max(r, g, b);
        const min = Math.min(r, g, b);
        const delta = max - min;
        
        let hue = 0;
        if (delta !== 0) {
            if (max === r) {
                hue = ((g - b) / delta) % 6;
            } else if (max === g) {
                hue = (b - r) / delta + 2;
            } else {
                hue = (r - g) / delta + 4;
            }
        }
        
        return Math.round(hue * 60);
    }
    return 180; // Default hue
}

## Использование

После добавления всех фрагментов кода:

1. Пользователь увидит панель с кнопками состояний в левом верхнем углу
2. Клик по кнопке запустит плавный переход между состояниями
3. Каждое состояние кардинально изменит:
   - Цветовую схему
   - Плотность и скорость фрагментов
   - Параметры звука
   - Визуальные эффекты
   - Скорость генерации контента
