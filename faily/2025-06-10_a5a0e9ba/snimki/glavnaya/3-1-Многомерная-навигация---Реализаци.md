# 3.1 Многомерная навигация

## Описание
Система многомерной навигации позволяет перемещаться между различными "слоями сознания" - от поверхностных мыслей до квантового уровня и пустоты.

## Реализация

### 3. JavaScript для многомерной навигации (добавить после системы микроанимаций)

```javascript
// === СИСТЕМА МНОГОМЕРНОЙ НАВИГАЦИИ ===

class MultidimensionalNavigation {
    constructor() {
        this.layers = {
            'surface': {
                name: 'Поверхность',
                description: 'Уровень повседневных мыслей',
                depth: 0,
                contentDensity: 1.0,
                fragmentTypes: ['обычные', 'простые'],
                particleCount: 30,
                soundFrequency: 'high',
                concepts: ['работа', 'время', 'планы', 'задачи', 'люди']
            },
            'subconscious': {
                name: 'Подсознание',
                description: 'Скрытые желания и страхи',
                depth: 1,
                contentDensity: 0.7,
                fragmentTypes: ['символические', 'метафорические'],
                particleCount: 40,
                soundFrequency: 'medium',
                concepts: ['мечты', 'страхи', 'желания', 'воспоминания', 'эмоции']
            },
            'collective': {
                name: 'Коллективное бессознательное',
                description: 'Архетипы и общечеловеческие паттерны',
                depth: 2,
                contentDensity: 0.5,
                fragmentTypes: ['архетипические', 'мифологические'],
                particleCount: 50,
                soundFrequency: 'low',
                concepts: ['архетипы', 'мифы', 'символы', 'ритуалы', 'традиции']
            },
            'quantum': {
                name: 'Квантовый уровень',
                description: 'Суперпозиция возможностей',
                depth: 3,
                contentDensity: 0.3,
                fragmentTypes: ['вероятностные', 'неопределенные'],
                particleCount: 100,
                soundFrequency: 'chaotic',
                concepts: ['вероятность', 'неопределенность', 'суперпозиция', 'запутанность', 'коллапс']
            },
            'void': {
                name: 'Пустота',
                description: 'Абсолютное ничто и все одновременно',
                depth: 4,
                contentDensity: 0.1,
                fragmentTypes: ['минимальные', 'эфемерные'],
                particleCount: 10,
                soundFrequency: 'silence',
                concepts: ['ничто', 'все', 'единство', 'покой', 'бесконечность']
            }
        };
        
        this.currentLayer = 'surface';
        this.transitioning = false;
        this.layerHistory = ['surface'];
        this.portals = [];
        
        this.init();
    }
    
    init() {
        this.setupNavigationControls();
        this.setupKeyboardNavigation();
        this.createLayerParticles();
        this.startLayerSpecificBehavior();
        this.setupScrollNavigation();
        this.createRandomPortals();
    }
    
    setupNavigationControls() {
        const indicators = document.querySelectorAll('.layer-indicator');
        
        indicators.forEach(indicator => {
            // Установка начального активного индикатора
            if (indicator.dataset.layer === this.currentLayer) {
                indicator.classList.add('active');
            }
            
            indicator.addEventListener('click', () => {
                const targetLayer = indicator.dataset.layer;
                if (targetLayer !== this.currentLayer && !this.transitioning) {
                    this.navigateToLayer(targetLayer);
                }
            });
            
            // Добавляем подсказки при наведении
            indicator.addEventListener('mouseenter', () => {
                const layer = this.layers[indicator.dataset.layer];
                this.showLayerPreview(layer);
            });
        });
    }
    
    setupKeyboardNavigation() {
        document.addEventListener('keydown', (e) => {
            if (this.transitioning) return;
            
            const layerKeys = Object.keys(this.layers);
            const currentIndex = layerKeys.indexOf(this.currentLayer);
            
            if (e.key === 'ArrowDown' || e.key === 's') {
                // Погружение глубже
                if (currentIndex < layerKeys.length - 1) {
                    this.navigateToLayer(layerKeys[currentIndex + 1]);
                }
            } else if (e.key === 'ArrowUp' || e.key === 'w') {
                // Подъем к поверхности
                if (currentIndex > 0) {
                    this.navigateToLayer(layerKeys[currentIndex - 1]);
                }
            } else if (e.key >= '1' && e.key <= '5') {
                // Прямой переход по номеру
                const index = parseInt(e.key) - 1;
                if (index < layerKeys.length) {
                    this.navigateToLayer(layerKeys[index]);
                }
            }
        });
    }
    
    navigateToLayer(targetLayer) {
        if (this.transitioning || targetLayer === this.currentLayer) return;
        
        this.transitioning = true;
        const sourceLayer = this.layers[this.currentLayer];
        const destinationLayer = this.layers[targetLayer];
        
        // Запускаем эффект перехода
        this.createTransitionEffect(sourceLayer.depth, destinationLayer.depth);
        
        // Обновляем UI
        setTimeout(() => {
            // Обновляем индикаторы
            document.querySelectorAll('.layer-indicator').forEach(ind => {
                ind.classList.toggle('active', ind.dataset.layer === targetLayer);
            });
            
            // Обновляем отображение текущего слоя
            const display = document.getElementById('current-layer-display');
            display.querySelector('.layer-name').textContent = destinationLayer.name;
            display.querySelector('.layer-description').textContent = destinationLayer.description;
            
            // Обновляем индикатор глубины
            const depthMarker = document.getElementById('depth-marker');
            depthMarker.style.bottom = `${(1 - destinationLayer.depth / 4) * 100}%`;
            
            // Обновляем атрибут body
            document.body.setAttribute('data-layer', targetLayer);
            
            // Применяем изменения к контенту
            this.applyLayerChanges(destinationLayer);
            
            // Сохраняем историю навигации
            this.layerHistory.push(targetLayer);
            if (this.layerHistory.length > 10) {
                this.layerHistory.shift();
            }
            
            this.currentLayer = targetLayer;
            this.transitioning = false;
            
            // Уведомляем другие системы о смене слоя
            this.notifyLayerChange(targetLayer);
        }, 1000);
    }
    
    createTransitionEffect(fromDepth, toDepth) {
        const transition = document.getElementById('layer-transition');
        const effect = document.createElement('div');
        effect.className = 'layer-transition-effect';
        
        // Настраиваем эффект в зависимости от направления
        if (toDepth > fromDepth) {
            // Погружение
            effect.style.background = 'radial-gradient(circle at center, black 0%, transparent 100%)';
        } else {
            // Подъем
            effect.style.background = 'radial-gradient(circle at center, white 0%, transparent 100%)';
        }
        
        transition.appendChild(effect);
        
        // Звуковой эффект перехода
        if (window.audioContext && window.audioEnabled) {
            this.playLayerTransitionSound(fromDepth, toDepth);
        }
        
        setTimeout(() => effect.remove(), 2000);
    }
    
    applyLayerChanges(layer) {
        // Изменяем плотность фрагментов
        const currentFragments = document.querySelectorAll('.consciousness-fragment').length;
        const targetFragments = Math.floor(20 * layer.contentDensity);
        
        if (currentFragments > targetFragments) {
            // Удаляем лишние
            const fragments = Array.from(document.querySelectorAll('.consciousness-fragment'));
            for (let i = targetFragments; i < currentFragments; i++) {
                if (fragments[i]) {
                    fragments[i].style.opacity = '0';
                    setTimeout(() => fragments[i].remove(), 1000);
                }
            }
        } else {
            // Добавляем новые
            for (let i = 0; i < targetFragments - currentFragments; i++) {
                setTimeout(() => {
                    const fragment = createConsciousnessFragment();
                    this.adaptFragmentToLayer(fragment, layer);
                }, i * 100);
            }
        }
        
        // Обновляем существующие фрагменты
        const remainingFragments = document.querySelectorAll('.consciousness-fragment');
        remainingFragments.forEach(fragment => {
            this.adaptFragmentToLayer(fragment, layer);
        });
        
        // Обновляем частицы слоя
        this.updateLayerParticles(layer);
        
        // Изменяем математическую сложность
        this.updateMathComplexity(layer);
    }
    
    adaptFragmentToLayer(fragment, layer) {
        // Генерируем контент в соответствии со слоем
        const concept = layer.concepts[Math.floor(Math.random() * layer.concepts.length)];
        
        // Модифицируем текст фрагмента
        if (layer.name === 'Пустота') {
            fragment.textContent = '...';
        } else if (layer.name === 'Квантовый уровень') {
            // Добавляем неопределенность
            if (Math.random() < 0.5) {
                fragment.style.opacity = '0.5';
                fragment.textContent = fragment.textContent + '?';
            }
        } else if (layer.name === 'Коллективное бессознательное') {
            // Добавляем архетипические символы
            const archetypes = ['☉', '☽', '♾', '◯', '△', '□'];
            fragment.textContent += ' ' + archetypes[Math.floor(Math.random() * archetypes.length)];
        }
        
        // Применяем визуальные изменения
        const hue = parseInt(getComputedStyle(document.body).getPropertyValue('--layer-hue'));
        fragment.style.color = `hsl(${hue + Math.random() * 60 - 30}, 70%, 70%)`;
    }
    
    createLayerParticles() {
        setInterval(() => {
            const layer = this.layers[this.currentLayer];
            
            if (Math.random() < 0.3) {
                const particle = document.createElement('div');
                particle.className = `layer-particle ${this.currentLayer}`;
                
                const size = 5 + Math.random() * 15;
                particle.style.width = size + 'px';
                particle.style.height = size + 'px';
                particle.style.left = Math.random() * window.innerWidth + 'px';
                particle.style.top = Math.random() * window.innerHeight + 'px';
                
                document.getElementById('container').appendChild(particle);
                
                // Удаляем через некоторое время
                const lifetime = 5000 + Math.random() * 10000;
                setTimeout(() => particle.remove(), lifetime);
            }
        }, 1000);
    }
    
    updateLayerParticles(layer) {
        // Удаляем старые частицы
        const oldParticles = document.querySelectorAll('.layer-particle');
        oldParticles.forEach(particle => {
            particle.style.opacity = '0';
            setTimeout(() => particle.remove(), 1000);
        });
        
        // Создаем новые частицы для текущего слоя
        for (let i = 0; i < layer.particleCount; i++) {
            setTimeout(() => {
                const particle = document.createElement('div');
                particle.className = `layer-particle ${this.currentLayer}`;
                
                const size = 5 + Math.random() * 20;
                particle.style.width = size + 'px';
                particle.style.height = size + 'px';
                particle.style.left = Math.random() * window.innerWidth + 'px';
                particle.style.top = Math.random() * window.innerHeight + 'px';
                
                document.getElementById('container').appendChild(particle);
            }, i * 50);
        }
    }
    
    startLayerSpecificBehavior() {
        setInterval(() => {
            const layer = this.layers[this.currentLayer];
            
            switch (this.currentLayer) {
                case 'surface':
                    // Быстрые, хаотичные движения
                    this.createSurfaceRipple();
                    break;
                    
                case 'subconscious':
                    // Волнообразные движения
                    this.createSubconsciousWave();
                    break;
                    
                case 'collective':
                    // Синхронизированные паттерны
                    this.createCollectivePattern();
                    break;
                    
                case 'quantum':
                    // Квантовые флуктуации
                    this.createQuantumFluctuation();
                    break;
                    
                case 'void':
                    // Минимальная активность
                    this.createVoidEcho();
                    break;
            }
        }, 5000);
    }
    
    createSurfaceRipple() {
        const x = Math.random() * window.innerWidth;
        const y = Math.random() * window.innerHeight;
        
        if (window.visualEffects) {
            window.visualEffects.createRipple(x, y, 2);
        }
    }
    
    createSubconsciousWave() {
        const fragments = document.querySelectorAll('.consciousness-fragment');
        const waveOrigin = Math.random() * fragments.length;
        
        fragments.forEach((fragment, index) => {
            const delay = Math.abs(index - waveOrigin) * 50;
            setTimeout(() => {
                fragment.style.transform = 'translateY(-10px)';
                setTimeout(() => {
                    fragment.style.transform = 'translateY(0)';
                }, 500);
            }, delay);
        });
    }
    
    createCollectivePattern() {
        // Все фрагменты синхронно пульсируют
        const fragments = document.querySelectorAll('.consciousness-fragment');
        fragments.forEach(fragment => {
            fragment.style.transform = 'scale(1.1)';
            setTimeout(() => {
                fragment.style.transform = 'scale(1)';
            }, 1000);
        });
    }
    
    createQuantumFluctuation() {
        // Случайные элементы появляются и исчезают
        const fragments = document.querySelectorAll('.consciousness-fragment');
        const count = Math.floor(Math.random() * 5) + 1;
        
        for (let i = 0; i < count; i++) {
            const fragment = fragments[Math.floor(Math.random() * fragments.length)];
            if (fragment) {
                const originalOpacity = fragment.style.opacity;
                fragment.style.opacity = '0';
                
                setTimeout(() => {
                    fragment.style.opacity = originalOpacity || '1';
                }, Math.random() * 1000);
            }
        }
    }
    
    createVoidEcho() {
        // Очень медленное, едва заметное движение
        const element = document.querySelector('.consciousness-fragment');
        if (element) {
            element.style.transition = 'all 10s ease-in-out';
            element.style.transform = 'translateX(5px)';
            
            setTimeout(() => {
                element.style.transform = 'translateX(-5px)';
            }, 10000);
        }
    }
    
    setupScrollNavigation() {
        // Навигация колесом мыши
        let scrollAccumulator = 0;
        
        document.addEventListener('wheel', (e) => {
            if (this.transitioning) return;
            
            scrollAccumulator += e.deltaY;
            
            if (Math.abs(scrollAccumulator) > 100) {
                const layerKeys = Object.keys(this.layers);
                const currentIndex = layerKeys.indexOf(this.currentLayer);
                
                if (scrollAccumulator > 0 && currentIndex < layerKeys.length - 1) {
                    // Вниз - погружение
                    this.navigateToLayer(layerKeys[currentIndex + 1]);
                } else if (scrollAccumulator < 0 && currentIndex > 0) {
                    // Вверх - подъем
                    this.navigateToLayer(layerKeys[currentIndex - 1]);
                }
                
                scrollAccumulator = 0;
            }
            
            // Сброс аккумулятора через время
            clearTimeout(this.scrollResetTimer);
            this.scrollResetTimer = setTimeout(() => {
                scrollAccumulator = 0;
            }, 200);
        });
    }
    
    createRandomPortals() {
        // Создаем порталы между слоями
        setInterval(() => {
            if (this.portals.length < 3 && Math.random() < 0.1) {
                const portal = document.createElement('div');
                portal.className = 'layer-portal';
                
                const x = Math.random() * (window.innerWidth - 100);
                const y = Math.random() * (window.innerHeight - 100);
                
                portal.style.left = x + 'px';
                portal.style.top = y + 'px';
                
                // Определяем целевой слой
                const layerKeys = Object.keys(this.layers);
                const currentIndex = layerKeys.indexOf(this.currentLayer);
```
