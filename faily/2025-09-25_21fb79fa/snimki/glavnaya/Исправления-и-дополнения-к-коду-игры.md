// Функция для проверки конвергенции судеб
function checkConvergenceEvent() {
    // Условия: множество материалов, высокие характеристики, особые события
    if (gameState.materials.length >= 4 && 
        gameState.silmarils.light >= 50 && 
        gameState.silmarils.purity >= 50 && 
        gameState.silmarils.power >= 50 &&
        gameState.events.length >= 10) {
        
        return {
            text: "Все твои выборы привели к этому моменту! Материалы, благословения, проклятия - всё сходится в одной точке. Реальность трещит по швам! Сильмариллы резонируют с самой тканью бытия!",
            choices: [
                {
                    text: "Позволить судьбе идти своим чередом",
                    reaction: "Ты отступаешь и наблюдаешь. Хаос энергий сам находит баланс. Результат непредсказуем, но естественен.",
                    effect: () => {
                        // Случайное значительное изменение всех параметров
                        const modifier = Math.random() > 0.5 ? 1 : -1;
                        gameState.silmarils.light += modifier * Math.floor(Math.random() * 40);
                        gameState.silmarils.purity += modifier * Math.floor(Math.random() * 40);
                        gameState.silmarils.power += modifier * Math.floor(Math.random() * 40);
                        gameState.events.push('fate_convergence');
                    }
                },
                {
                    text: "Попытаться контролировать конвергенцию",
                    reaction: () => {
                        if (gameState.feanor.skill >= 80) {
                            return "Твоё мастерство позволяет направить хаотические энергии! Ты создаёшь идеальную гармонию!";
                        } else {
                            return "Энергии слишком сильны! Взрыв отбрасывает тебя, но Сильмариллы... изменились.";
                        }
                    },
                    effect: () => {
                        if (gameState.feanor.skill >= 80) {
                            gameState.silmarils.light += 50;
                            gameState.silmarils.purity += 50;
                            gameState.silmarils.power += 50;
                            gameState.events.push('perfect_convergence');
                        } else {
                            gameState.feanor.wisdom -= 20;
                            gameState.silmarils.power += 80;
                            gameState.silmarils.purity -= 30;
                            gameState.events.push('convergence_explosion');
                        }
                    }
                },
                {
                    text: "Разделить судьбу на три пути - по одному для каждого Сильмарилла",
                    reaction: "Гениальное решение! Каждый Сильмарилл получает уникальную судьбу: один - чистый свет, другой - абсолютная сила, третий - совершенная гармония.",

                        gameState.silmarils.light += 60;

                        gameState.silmarils.power += 60;
                        // Третий Сильмарилл
                        gameState.silmarils.purity += 60;
                    }
                }
            ]
        };
    }
    return null;
}

function checkEventConditions(stageId) {
    const triggers = [];

    if (stageId === 6) {
        const natureMaterials = ['telperion_dew', 'star_dust', 'eldest_sap'];
        const hasNature = natureMaterials.some(m => gameState.materials.includes(m));
        if (hasNature && Math.random() > 0.4) {
            triggers.push('yavanna_intervention');
        }
    }
    
    // Последнее искушение при высокой гордости
    if (stageId === 8 && gameState.feanor.pride >= 70) {
        if (Math.random() > 0.3) {
            triggers.push('final_temptation');
        }
    }
    
    // Конвергенция при множестве факторов
    if (stageId >= 6 && stageId <= 8) {
        const complexity = gameState.materials.length + 
                          gameState.events.length / 3 +
                          gameState.blessings.length * 2 +
                          gameState.curses.length;
        
        if (complexity >= 15 && Math.random() > 0.6) {
            triggers.push('fate_convergence');
        }
    }
    
    return triggers;
