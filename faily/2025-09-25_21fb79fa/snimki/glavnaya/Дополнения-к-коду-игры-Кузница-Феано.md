## 2. Новые функции для механик

```javascript
// Система репутации
function modifyReputation(faction, amount) {
    if (gameState.reputation[faction] !== undefined) {
        gameState.reputation[faction] = Math.max(0, Math.min(100, 
            gameState.reputation[faction] + amount));
        
        // Триггер событий на основе репутации
        checkReputationEvents(faction);
    }
}

// Система временных баффов
function addTemporaryBuff(buff) {
    gameState.temporaryBuffs.push({
        name: buff.name,
        effect: buff.effect,
        duration: buff.duration || 1,
        description: buff.description
    });
}

function processBuffs() {
    gameState.temporaryBuffs = gameState.temporaryBuffs.filter(buff => {
        buff.duration--;
        if (buff.duration > 0) {
            buff.effect(); // Применяем эффект
            return true;
        }
        return false;
    });
}

// Проверка условий для особых событий
function checkSpecialEventConditions(stageId) {
    // Проверка цепочек событий
    if (gameState.eventChains['twins_prophecy'] && stageId === 7) {
        return getSpecialTwinsEvent();
    }
    
    // Проверка комбинаций материалов
    if (gameState.materials.length >= 3 && Math.random() > 0.7) {
        return getConvergenceEvent();
    }
    
    // Мета-событие (очень редкое)
    if (Math.random() > 0.98 && gameState.feanor.wisdom >= 70) {
        return getMetaEvent();
    }
    
    return null;
}

// Проверка репутационных событий
function checkReputationEvents(faction) {
    if (faction === 'valar' && gameState.reputation.valar >= 90) {
        gameState.events.push('valar_champion');
        gameState.silmarils.purity += 10;
    } else if (faction === 'noldor' && gameState.reputation.noldor <= 20) {
        gameState.events.push('noldor_outcast');
        gameState.feanor.pride += 20;
    }
}
```

function renderSpecialEvent(event) {
    const textDiv = document.getElementById('gameText');
    const choicesDiv = document.getElementById('choices');
    
    textDiv.innerHTML = `<div class="special-event">${event.text}</div>`;
    
    choicesDiv.innerHTML = '';
    event.choices.forEach(choice => {
        const btn = document.createElement('button');
        btn.className = 'choice-btn';
        btn.textContent = choice.text;
        
        // Применяем усиление выборов если есть благословение
        btn.onclick = () => {
            if (gameState.blessings.includes('enhanced_choices')) {
                // Усиливаем положительные эффекты на 10%
                const originalEffect = choice.effect;
                choice.effect = () => {
                    originalEffect();
                    gameState.feanor.skill = Math.floor(gameState.feanor.skill * 1.1);
                    gameState.silmarils.light = Math.floor(gameState.silmarils.light * 1.1);
                };
            }
            
            choice.effect();
            updateStats();
            
            if (choice.reaction) {
                showReactionAndProceed(choice.reaction, choice.next || 9);
            } else {
                renderStage(choice.next || 9);
            }
        };
        choicesDiv.appendChild(btn);
    });
}
