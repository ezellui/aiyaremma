## 7. Улучшенный парсер блазона

### Недостающие элементы:
- Поддержка всех конструкций языка
- Обработка сложных описаний
- Правильный порядок элементов

### Фрагмент кода:
**Заменить функцию `parseBlazon()` на расширенную версию:**

```javascript
function parseBlazon() {
    const blazonText = document.getElementById('blazonInput').value.trim();
    
    if (!blazonText) {
        showParseError('Пожалуйста, введите блазон');
        return;
    }
    
    try {
        // Создаём парсер с токенизацией
        const parser = new BlazonParser(blazonText);
        const parsed = parser.parse();
        
        // Применяем распарсенные данные
        applyParsedBlazon(parsed);
        
        // Переключение на режим блазона и отрисовка
        currentMode = 'blazon';
        drawShield();
        
        document.getElementById('parseError').style.display = 'none';
        
    } catch (error) {
        showParseError('Ошибка парсинга: ' + error.message);
    }
}

class BlazonParser {
    constructor(blazon) {
        this.blazon = blazon;
        this.tokens = this.tokenize(blazon);
        this.position = 0;
    }
    
    tokenize(blazon) {
        // Разбиваем на токены с учётом специальных конструкций
        const patterns = [
            /^(Per\s+(?:pale|fess|bend|chevron|saltire))/i,
            /^(Quarterly|Gyronny|Barry|Paly|Bendy|Chequy|Lozengy)/i,
            /^(Or|Argent|Azure|Gules|Sable|Vert|Purpure|Ermine|Vair)/i,
            /^(a|an|two|three|four|five|six|seven|eight|nine|ten)/i,
            /^(between|on|within|overall|charged with)/i,
            /^(chief|pale|fess|bend|chevron|cross|saltire|bordure)/i,
            /^(lion|eagle|rose|fleur-de-lis|mullet|crescent)/i,
            /^(rampant|passant|displayed|volant)/i,
            /^(guardant|regardant)/i,
            /^(armed|langued|crowned|gorged)/i,
            /^(and)/i,
            /^(,)/,
            /^(\w+)/
        ];
        
        const tokens = [];
        let remaining = blazon;
        
        while (remaining.length > 0) {
            remaining = remaining.trimStart();
            let matched = false;
            
            for (const pattern of patterns) {
                const match = remaining.match(pattern);
                if (match) {
                    tokens.push({
                        type: this.getTokenType(match[0]),
                        value: match[0]
                    });
                    remaining = remaining.substring(match[0].length);
                    matched = true;
                    break;
                }
            }
            
            if (!matched) {
                remaining = remaining.substring(1); // Skip unknown character
            }
        }
        
        return tokens;
    }
    
    getTokenType(value) {
        const lower = value.toLowerCase();
        
        if (/^per\s+/.test(lower)) return 'DIVISION';
        if (/^(quarterly|gyronny|barry|paly|bendy|chequy|lozengy)$/i.test(lower)) return 'PARTITION';
        if (/^(or|argent|azure|gules|sable|vert|purpure|ermine|vair)$/i.test(lower)) return 'TINCTURE';
        if (/^(a|an|two|three|four|five|six|seven|eight|nine|ten)$/i.test(lower)) return 'NUMBER';
        if (/^(between|on|within|overall|charged with)$/i.test(lower)) return 'CONSTRUCTION';
        if (/^(chief|pale|fess|bend|chevron|cross|saltire|bordure)$/i.test(lower)) return 'ORDINARY';
        if (/^(lion|eagle|rose|fleur-de-lis|mullet|crescent)$/i.test(lower)) return 'CHARGE';
        if (/^(rampant|passant|displayed|volant)$/i.test(lower)) return 'ATTITUDE';
        if (/^(guardant|regardant)$/i.test(lower)) return 'FACING';
        if (/^(armed|langued|crowned|gorged)$/i.test(lower)) return 'ATTRIBUTE';
        if (lower === 'and') return 'AND';
        if (lower === ',') return 'COMMA';
        
        return 'WORD';
    }
    
    parse() {
        const result = {
            field: this.parseField(),
            ordinaries: [],
            charges: []
        };
        
        // Парсим остальные элементы
        while (this.position < this.tokens.length) {
            if (this.peek()?.type === 'COMMA') {
                this.consume('COMMA');
            }
            
            if (this.peek()?.type === 'ORDINARY') {
                result.ordinaries.push(this.parseOrdinary());
            } else if (this.peek()?.type === 'NUMBER' || 
                      (this.peek()?.type === 'WORD' && /^(a|an)$/i.test(this.peek().value))) {
                result.charges.push(this.parseCharge());
            } else {
                this.position++;
            }
        }
        
        return result;
    }
    
    parseField() {
        if (this.peek()?.type === 'DIVISION') {
            return this.parseDividedField();
        } else if (this.peek()?.type === 'PARTITION') {
            return this.parsePartitionedField();
        } else if (this.peek()?.type === 'TINCTURE') {
            return {
                type: 'plain',
                tincture: this.consume('TINCTURE').value
            };
        }
        
        throw new Error('Ожидалось описание поля');
    }
    
    // ... остальные методы парсера
}
```

function drawWreath(svg) {
    const wreathG = document.createElementNS('http://www.w3.org/2000/svg', 'g');
    wreathG.setAttribute('transform', 'translate(200, 20)');
    
    // Бурелет - скрученная ткань из 6 сегментов
    const segments = 6;
    const width = 80;
    const height = 15;
    
    for (let i = 0; i < segments; i++) {
        const segment = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
        segment.setAttribute('x', -width/2 + i * (width/segments));
        segment.setAttribute('y', -height/2);
        segment.setAttribute('width', width/segments);
        segment.setAttribute('height', height);
        
        // Чередуем цвета
        const color = i % 2 === 0 ? 
            getTinctureFill(shieldData.field.tincture1, 'colored') : 
            getTinctureFill(shieldData.field.tincture2 || 'argent', 'colored');
        segment.setAttribute('fill', color);
        
        wreathG.appendChild(segment);
    }
    
    // Добавляем скручивание

    svg.appendChild(wreathG);
}

function checkHeraldricRules() {
    const violations = [];
    
    // Правило тинктуры
    violations.push(...checkTinctureRule());
    
    // Правило каденции
    if (shieldData.cadency !== 'none') {
        violations.push(...checkCadencyRules());
    }
    
    // Правило маршаллинга
    if (shieldData.marshalling !== 'none') {
        violations.push(...checkMarshallingRules());
    }
    
    // Правило размещения зарядов
    violations.push(...checkChargePositioning());
    
    // Правило уникальности
    violations.push(...checkUniqueness());

}

function checkCadencyRules() {
    const violations = [];
    
    // Label должен быть в верхней части
    if (shieldData.cadency === 'label' && shieldData.ordinary.type === 'chief') {
        violations.push('Label не может использоваться с chief');
    }
    
    // Размер знаков каденции
    if (shieldData.charges.length > 0) {
        const totalElements = shieldData.charges.reduce((sum, charge) => sum + charge.count, 0);
        if (totalElements > 3) {
            violations.push('Знак каденции может быть неразличим при большом количестве зарядов');
        }
    }
    
    return violations;
}

function checkMarshallingRules() {
    const violations = [];
    
    // Проверка правильности четвертования
    if (shieldData.marshalling === 'quartering') {
        if (!shieldData.marshallingData || shieldData.marshallingData.quarters.length < 4) {
            violations.push('Четвертование требует минимум 4 герба');
        }
    }
    
    // Escutcheon of pretence только для наследниц
    if (shieldData.marshalling === 'escutcheonPretence' && !shieldData.marshallingData?.heiress) {
        violations.push('Escutcheon of pretence используется только для геральдических наследниц');
    }
    
    return violations;
}

function createRandomValidBlazon() {
    // Генератор случайных корректных блазонов
    const fields = [
        'Argent', 'Or', 'Azure', 'Gules', 'Sable', 'Vert',
        'Per pale argent and azure', 'Per fess Or and gules',
        'Quarterly azure and argent', 'Gyronny Or and sable'
    ];
    
    const ordinaries = [
        '', // может не быть
        ', a chief {}',
        ', a fess {}',
        ', a pale {}',
        ', a bend {}',
        ', a chevron {}',
        ', a cross {}',
        ', a saltire {}',
        ', a bordure {}'
    ];
    
    const charges = [
        '', // может не быть
        ', {} lion{} rampant {}',
        ', {} rose{} {}',
        ', {} mullet{} {}',
        ', {} fleur{}-de-lis {}',
        ', {} eagle{} displayed {}',
        ', {} crescent{} {}'
    ];
    
    const numbers = ['a', 'two', 'three'];
    const plurals = ['', 's', 's'];
    const tinctures = ['Or', 'argent', 'azure', 'gules', 'sable', 'vert'];
    
    // Выбираем случайные элементы
    let blazon = fields[Math.floor(Math.random() * fields.length)];
    
    // Добавляем ординарий с вероятностью 50%
    if (Math.random() > 0.5) {
        const ordinary = ordinaries[Math.floor(Math.random() * ordinaries.length)];
        if (ordinary) {
            // Выбираем тинктуру, соблюдая правило
            const fieldTincture = blazon.includes('argent') || blazon.includes('Or') ? 
                ['azure', 'gules', 'sable', 'vert'] : ['Or', 'argent'];
            const ordinaryTincture = fieldTincture[Math.floor(Math.random() * fieldTincture.length)];
            blazon += ordinary.replace('{}', ordinaryTincture);
        }
    }
    
    // Добавляем заряды с вероятностью 70%
    if (Math.random() > 0.3) {
        const chargeTemplate = charges[Math.floor(Math.random() * charges.length)];
        if (chargeTemplate) {
            const numIndex = Math.floor(Math.random() * numbers.length);
            const number = numbers[numIndex];
            const plural = plurals[numIndex];
            
            // Выбираем тинктуру для заряда
            const fieldMainTincture = blazon.split(/[\s,]/)[0].toLowerCase();
            const chargeTincture = ['or', 'argent'].includes(fieldMainTincture) ?
                tinctures.filter(t => !['Or', 'argent'].includes(t)) :
                ['Or', 'argent'];
            
            const charge = chargeTemplate
                .replace(/{}/g, (match, offset) => {
                    const matches = chargeTemplate.match(/{}/g);
                    const index = matches.slice(0, chargeTemplate.indexOf(match, offset)).length;
                    if (index === 0) return number;
                    if (index === 1) return plural;
                    if (index === 2) return plural; // для fleurs-de-lis
                    return chargeTincture[Math.floor(Math.random() * chargeTincture.length)];
                });
            
            blazon += charge;
        }
    }
    
    return blazon;
}
