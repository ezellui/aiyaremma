## Ручной ввод параметров визуализации с поддержкой математических выражений

### 1. Добавление полей ввода к существующим ползункам

Добавьте следующую функцию в основной JavaScript-код:

```javascript
// Функция для добавления полей ввода к существующим ползункам
function addInputFieldsToRangeInputs() {
    // Получаем все элементы ползунков
    const rangeInputs = document.querySelectorAll('.range-input');
    
    rangeInputs.forEach(rangeInput => {
        const container = rangeInput.parentElement;
        
        // Создаем поле ввода
        const inputField = document.createElement('input');
        inputField.type = 'text';
        inputField.className = 'range-value-input';
        inputField.value = rangeInput.value;
        inputField.placeholder = 'Значение или формула';
        
        // Вставляем поле ввода после ползунка
        if (rangeInput.nextElementSibling && rangeInput.nextElementSibling.className === 'range-values') {
            container.insertBefore(inputField, rangeInput.nextElementSibling);
        } else {
            container.insertBefore(inputField, rangeInput.nextSibling);
        }
        
        // Связываем поле ввода с ползунком
        rangeInput.addEventListener('input', function() {
            inputField.value = this.value;
        });
        
        // Связываем ползунок с полем ввода
        inputField.addEventListener('input', function() {
            try {
                const value = parseExpressionValue(this.value);
                if (value >= parseFloat(rangeInput.min) && value <= parseFloat(rangeInput.max)) {
                    rangeInput.value = value;
                    
                    // Вызываем событие input у ползунка чтобы обновить связанные значения
                    const event = new Event('input', { bubbles: true });
                    rangeInput.dispatchEvent(event);
                }
            } catch (error) {
                // Игнорируем ошибку, позволяя пользователю продолжить ввод
            }
        });
        
        // Обработка потери фокуса, чтобы исправить неверные значения
        inputField.addEventListener('blur', function() {
            try {
                const value = parseExpressionValue(this.value);
                const min = parseFloat(rangeInput.min);
                const max = parseFloat(rangeInput.max);
                
                if (value < min) this.value = min;
                else if (value > max) this.value = max;
                else this.value = value;
                
                rangeInput.value = this.value;
                
                // Вызываем событие input у ползунка чтобы обновить связанные значения
                const event = new Event('input', { bubbles: true });
                rangeInput.dispatchEvent(event);
            } catch (error) {
                // Для неверных выражений сбрасываем на текущее значение ползунка
                this.value = rangeInput.value;
            }
        });
    });
}
```

### 3. Обновление полей ввода для пределов комплексной плоскости

Модифицируйте блок с вводом границ домена, чтобы он поддерживал математические выражения:

```javascript
// Обработчики для полей ввода пределов комплексной плоскости
function setupDomainBoundsInputs() {
    const xMinInput = document.getElementById('x-min');
    const xMaxInput = document.getElementById('x-max');
    const yMinInput = document.getElementById('y-min');
    const yMaxInput = document.getElementById('y-max');
    
    // Функция для добавления поддержки математических выражений для поля ввода
    function enhanceInput(input, stateProperty) {
        input.addEventListener('input', function() {
            try {
                // Пытаемся распарсить выражение, но не обновляем состояние сразу
                parseExpressionValue(this.value);
            } catch (error) {
                // Игнорируем ошибку, позволяя продолжить ввод
            }
        });
        
        input.addEventListener('blur', function() {
            try {
                const value = parseExpressionValue(this.value);
                state[stateProperty] = value;
                this.value = value;
                updateDomainDisplay();
            } catch (error) {
                // Для неверных выражений сбрасываем на предыдущее значение
                this.value = state[stateProperty];
            }
        });
    }
    
    enhanceInput(xMinInput, 'xMin');
    enhanceInput(xMaxInput, 'xMax');
    enhanceInput(yMinInput, 'yMin');
    enhanceInput(yMaxInput, 'yMax');
}
```

### 5. Функция для парсинга математических выражений

Если вы ещё не добавили функцию `parseExpressionValue` из предыдущего артефакта, добавьте её:

```javascript
// Функция для парсинга математических выражений
function parseExpressionValue(expression) {
    try {
        // Заменяем математические константы
        const processedExpression = expression
            .replace(/\bpi\b/gi, 'Math.PI')
            .replace(/\be\b/gi, 'Math.E');
        
        // Используем Function вместо eval для безопасности
        // eslint-disable-next-line no-new-func
        const result = new Function('Math', 'return ' + processedExpression)(Math);
        
        if (typeof result !== 'number' || isNaN(result)) {
            throw new Error('Результат не является числом');
        }
        
        return result;
    } catch (error) {
        console.error('Ошибка при парсинге выражения:', error);
        throw error;
    }
}
```
