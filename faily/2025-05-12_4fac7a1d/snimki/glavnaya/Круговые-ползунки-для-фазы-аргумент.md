### 3. Добавление JavaScript-кода для управления

Добавьте функцию для обработки математических выражений (добавьте перед функцией инициализации):

```javascript
// Функция для парсинга математических выражений
function parseExpressionValue(expression) {
    try {
        // Заменяем pi и e на их значения
        const processedExpression = expression
            .replace(/pi/gi, 'Math.PI')
            .replace(/e/gi, 'Math.E');
        
        // Используем Function вместо eval для большей безопасности
        // eslint-disable-next-line no-new-func
        const result = new Function('return ' + processedExpression)();
        
        if (typeof result !== 'number' || isNaN(result)) {
            throw new Error('Результат не является числом');
        }
        
        return result;
    } catch (error) {
        console.error('Ошибка при парсинге выражения:', error);
        throw error;
    }
}
```

Добавьте следующую функцию для инициализации круговых ползунков (добавьте к инициализации в основном скрипте):

```javascript
// Функция для работы с круговыми ползунками фазы
function setupCircularSliders() {
    // Получаем элементы DOM
    const argPhaseSlider = document.getElementById('argument-phase-slider');
    const argPhaseHandle = document.getElementById('argument-phase-handle');
    const argPhaseInput = document.getElementById('argument-phase-input');
    
    const funcPhaseSlider = document.getElementById('function-phase-slider');
    const funcPhaseHandle = document.getElementById('function-phase-handle');
    const funcPhaseInput = document.getElementById('function-phase-input');
    
    // Функция для обработки изменения положения ручки
    function handleCircularSliderChange(slider, handle, input, stateProperty) {
        let isDragging = false;
        
        // Функция для обновления положения ручки по углу
        function updateHandlePosition(angle) {
            const radians = (angle * Math.PI) / 180;
            state[stateProperty] = radians;
            handle.style.transform = `rotate(${angle}deg)`;
            input.value = radians.toFixed(2);
        }
        
        // Функция для вычисления угла по координатам
        function calculateAngle(event) {
            const rect = slider.getBoundingClientRect();
            const centerX = rect.left + rect.width / 2;
            const centerY = rect.top + rect.height / 2;
            const angle = Math.atan2(event.clientY - centerY, event.clientX - centerX) * (180 / Math.PI);
            return (angle + 90) % 360; // Корректируем угол, чтобы 0 был сверху
        }
        
        // Обработка начала перетаскивания
        handle.addEventListener('mousedown', function(event) {
            event.preventDefault();
            isDragging = true;
        });
        
        // Обработка перемещения мыши
        document.addEventListener('mousemove', function(event) {
            if (isDragging) {
                const angle = calculateAngle(event);
                updateHandlePosition(angle);
            }
        });
        
        // Обработка окончания перетаскивания
        document.addEventListener('mouseup', function() {
            isDragging = false;
        });
        
        // Обработка клика по слайдеру
        slider.addEventListener('click', function(event) {
            const angle = calculateAngle(event);
            updateHandlePosition(angle);
        });
        
        // Обработка ввода значения
        input.addEventListener('input', function() {
            try {
                const value = parseExpressionValue(this.value);
                // Нормализуем значение в диапазон [0, 2π]
                const normalizedValue = ((value % (2 * Math.PI)) + 2 * Math.PI) % (2 * Math.PI);
                const angle = (normalizedValue * 180) / Math.PI;
                handle.style.transform = `rotate(${angle}deg)`;
                state[stateProperty] = normalizedValue;
            } catch (error) {
                console.error('Ошибка при обработке ввода:', error);
            }
        });
        
        // Инициализация начальной позиции
        updateHandlePosition(0);
    }
    
    // Инициализация слайдеров
    handleCircularSliderChange(argPhaseSlider, argPhaseHandle, argPhaseInput, 'argumentPhase');
    handleCircularSliderChange(funcPhaseSlider, funcPhaseHandle, funcPhaseInput, 'functionPhase');
}
```

### 5. Модификация worker'а для учёта фаз

В функции worker'а, которая вычисляет значения функции (`domainColoring`), добавьте обработку фаз аргумента и результата.

В блок параметров в начале функции добавьте:

```javascript
       colorScheme, showGrid, gridDensity, showAxes,

```

И внутри цикла, перед вычислением функции, добавьте:
```javascript
// Применяем фазу аргумента, если она задана
let zWithPhase;
if (argumentPhase) {
    const r = math.abs(z);
    let theta = math.arg(z);
    theta += argumentPhase;
    zWithPhase = math.complex({
        r: r,
        phi: theta
    });
} else {
    zWithPhase = z;
}

// Вычисляем значение функции
let w;
try {
    w = complexFunction(zWithPhase); // Используем zWithPhase вместо z
    
    // Применяем фазу функции, если она задана
    if (functionPhase) {
        const r = math.abs(w);
        let theta = math.arg(w);
        theta += functionPhase;
        w = math.complex({
            r: r,
            phi: theta
        });
    }
} catch (error) {
    // В случае ошибки используем NaN
    w = math.complex(NaN, NaN);
}
```
