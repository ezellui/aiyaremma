```javascript
// JavaScript для переключения темы
function setupThemeSwitch() {
    const themeSwitch = document.getElementById('theme-switch-checkbox');
    
    // Проверяем сохранённую тему или системные настройки
    if (localStorage.getItem('theme') === 'dark' || 
        (window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches && 
         !localStorage.getItem('theme'))) {
        document.documentElement.setAttribute('data-theme', 'dark');
        themeSwitch.checked = true;
    }
    
    themeSwitch.addEventListener('change', function() {
        if (this.checked) {
            document.documentElement.setAttribute('data-theme', 'dark');
            localStorage.setItem('theme', 'dark');
        } else {
            document.documentElement.setAttribute('data-theme', 'light');
            localStorage.setItem('theme', 'light');
        }
    });
}

// Вызов в инициализации
```

```javascript
// Функция для настройки миникарты комплексной плоскости
function setupMinimap() {
    const minimapCanvas = document.getElementById('minimap-canvas');
    const ctx = minimapCanvas.getContext('2d');
    const zoomInButton = document.getElementById('minimap-zoom-in');
    const zoomOutButton = document.getElementById('minimap-zoom-out');
    const resetButton = document.getElementById('minimap-reset');
    
    // Глобальные границы для миникарты
    const globalBounds = {
        xMin: -10,
        xMax: 10,
        yMin: -10,
        yMax: 10
    };
    
    // Функция для отрисовки миникарты
    function drawMinimap() {
        const width = minimapCanvas.width;
        const height = minimapCanvas.height;
        
        // Очищаем холст
        ctx.fillStyle = '#f8f9fa';
        ctx.fillRect(0, 0, width, height);
        
        // Рисуем сетку
        ctx.strokeStyle = '#e0e0e0';
        ctx.lineWidth = 0.5;
        
        // Вертикальные линии
        for (let x = Math.ceil(globalBounds.xMin); x <= Math.floor(globalBounds.xMax); x++) {
            if (x === 0) continue; // Пропускаем оси
            const canvasX = (x - globalBounds.xMin) / (globalBounds.xMax - globalBounds.xMin) * width;
            ctx.beginPath();
            ctx.moveTo(canvasX, 0);
            ctx.lineTo(canvasX, height);
            ctx.stroke();
        }
        
        // Горизонтальные линии
        for (let y = Math.ceil(globalBounds.yMin); y <= Math.floor(globalBounds.yMax); y++) {
            if (y === 0) continue; // Пропускаем оси
            const canvasY = height - (y - globalBounds.yMin) / (globalBounds.yMax - globalBounds.yMin) * height;
            ctx.beginPath();
            ctx.moveTo(0, canvasY);
            ctx.lineTo(width, canvasY);
            ctx.stroke();
        }
        
        // Рисуем оси
        ctx.strokeStyle = '#666';
        ctx.lineWidth = 1;
        
        // Ось X
        const yAxisPos = height * (globalBounds.yMax / (globalBounds.yMax - globalBounds.yMin));
        ctx.beginPath();
        ctx.moveTo(0, yAxisPos);
        ctx.lineTo(width, yAxisPos);
        ctx.stroke();
        
        // Ось Y
        const xAxisPos = width * (-globalBounds.xMin / (globalBounds.xMax - globalBounds.xMin));
        ctx.beginPath();
        ctx.moveTo(xAxisPos, 0);
        ctx.lineTo(xAxisPos, height);
        ctx.stroke();
        
        // Рисуем прямоугольник текущей области просмотра
        const viewportX1 = (state.xMin - globalBounds.xMin) / (globalBounds.xMax - globalBounds.xMin) * width;
        const viewportY1 = height - (state.yMax - globalBounds.yMin) / (globalBounds.yMax - globalBounds.yMin) * height;
        const viewportWidth = (state.xMax - state.xMin) / (globalBounds.xMax - globalBounds.xMin) * width;
        const viewportHeight = (state.yMax - state.yMin) / (globalBounds.yMax - globalBounds.yMin) * height;
        
        ctx.strokeStyle = '#3498db';
        ctx.lineWidth = 2;
        ctx.strokeRect(viewportX1, viewportY1, viewportWidth, viewportHeight);
        
        // Заливка области просмотра полупрозрачным цветом
        ctx.fillStyle = 'rgba(52, 152, 219, 0.2)';
        ctx.fillRect(viewportX1, viewportY1, viewportWidth, viewportHeight);
    }
    
    // Обработчики кнопок
    zoomInButton.addEventListener('click', function() {
        // Увеличиваем масштаб (уменьшаем размер области)
        const centerX = (state.xMin + state.xMax) / 2;
        const centerY = (state.yMin + state.yMax) / 2;
        const newWidth = (state.xMax - state.xMin) * 0.7;
        const newHeight = (state.yMax - state.yMin) * 0.7;
        
        state.xMin = centerX - newWidth / 2;
        state.xMax = centerX + newWidth / 2;
        state.yMin = centerY - newHeight / 2;
        state.yMax = centerY + newHeight / 2;
        
        // Обновляем поля ввода
        document.getElementById('x-min').value = state.xMin.toFixed(3);
        document.getElementById('x-max').value = state.xMax.toFixed(3);
        document.getElementById('y-min').value = state.yMin.toFixed(3);
        document.getElementById('y-max').value = state.yMax.toFixed(3);
        
        // Обновляем отображение домена
        updateDomainDisplay();
        
        // Перерисовываем визуализацию и миникарту
        drawMinimap();
        renderVisualization();
    });
    
    zoomOutButton.addEventListener('click', function() {
        // Уменьшаем масштаб (увеличиваем размер области)
        const centerX = (state.xMin + state.xMax) / 2;
        const centerY = (state.yMin + state.yMax) / 2;
        const newWidth = (state.xMax - state.xMin) / 0.7;
        const newHeight = (state.yMax - state.yMin) / 0.7;
        
        state.xMin = centerX - newWidth / 2;
        state.xMax = centerX + newWidth / 2;
        state.yMin = centerY - newHeight / 2;
        state.yMax = centerY + newHeight / 2;
        
        // Обновляем поля ввода
        document.getElementById('x-min').value = state.xMin.toFixed(3);
        document.getElementById('x-max').value = state.xMax.toFixed(3);
        document.getElementById('y-min').value = state.yMin.toFixed(3);
        document.getElementById('y-max').value = state.yMax.toFixed(3);
        
        // Обновляем отображение домена
        updateDomainDisplay();
        
        // Перерисовываем визуализацию и миникарту
        drawMinimap();
        renderVisualization();
    });
    
    resetButton.addEventListener('click', function() {
        // Сбрасываем область к начальным значениям
        state.xMin = -5;
        state.xMax = 5;
        state.yMin = -5;
        state.yMax = 5;
        
        // Обновляем поля ввода
        document.getElementById('x-min').value = state.xMin;
        document.getElementById('x-max').value = state.xMax;
        document.getElementById('y-min').value = state.yMin;
        document.getElementById('y-max').value = state.yMax;
        
        // Обновляем отображение домена
        updateDomainDisplay();
        
        // Перерисовываем визуализацию и миникарту
        drawMinimap();
        renderVisualization();
    });
    
    // Перетаскивание области на миникарте
    let isDragging = false;
    let dragStartX, dragStartY;
    
    minimapCanvas.addEventListener('mousedown', function(event) {
        isDragging = true;
        dragStartX = event.offsetX;
        dragStartY = event.offsetY;
    });
    
    document.addEventListener('mousemove', function(event) {
        if (!isDragging) return;
        
        const rect = minimapCanvas.getBoundingClientRect();
        const offsetX = event.clientX - rect.left;
        const offsetY = event.clientY - rect.top;
        
        const deltaX = offsetX - dragStartX;
        const deltaY = offsetY - dragStartY;
        
        // Преобразуем смещение на холсте в смещение в комплексной плоскости
        const complexDeltaX = deltaX / minimapCanvas.width * (globalBounds.xMax - globalBounds.xMin);
        const complexDeltaY = -deltaY / minimapCanvas.height * (globalBounds.yMax - globalBounds.yMin);
        
        // Обновляем границы области просмотра
        state.xMin -= complexDeltaX;
        state.xMax -= complexDeltaX;
        state.yMin -= complexDeltaY;
        state.yMax -= complexDeltaY;
        
        // Обновляем поля ввода
        document.getElementById('x-min').value = state.xMin.toFixed(3);
        document.getElementById('x-max').value = state.xMax.toFixed(3);
        document.getElementById('y-min').value = state.yMin.toFixed(3);
        document.getElementById('y-max').value = state.yMax.toFixed(3);
        
        // Обновляем начальную позицию для следующего смещения
        dragStartX = offsetX;
        dragStartY = offsetY;
        
        // Обновляем отображение домена
        updateDomainDisplay();
        
        // Перерисовываем миникарту
        drawMinimap();
    });
    
    document.addEventListener('mouseup', function() {
        if (isDragging) {
            isDragging = false;
            renderVisualization(); // Обновляем визуализацию только при окончании перетаскивания
        }
    });
    
    // Инициализация
    drawMinimap();
    
    // Обновление миникарты при изменении состояния
    function setupMinimapUpdating() {
        // Находим функцию обновления домена
        const originalUpdateDomainDisplay = updateDomainDisplay;
        
        // Переопределяем функцию, чтобы также обновлять миникарту
        updateDomainDisplay = function() {
            originalUpdateDomainDisplay();
            drawMinimap();
        };
    }
    
    setupMinimapUpdating();
}
```

```javascript
// Функция для управления пользовательскими предустановками
function setupUserPresets() {
    const userPresetsList = document.getElementById('user-presets-list');
    const savePresetButton = document.getElementById('save-preset-button');
    const importPresetsButton = document.getElementById('import-presets-button');
    const exportPresetsButton = document.getElementById('export-presets-button');
    
    // Загрузка сохранённых предустановок из localStorage
    function loadUserPresets() {
        const presets = JSON.parse(localStorage.getItem('userPresets') || '[]');
        return presets;
    }
    
    // Сохранение предустановок в localStorage
    function saveUserPresets(presets) {
        localStorage.setItem('userPresets', JSON.stringify(presets));
    }
    
    // Отображение предустановок в списке
    function displayUserPresets() {
        userPresetsList.innerHTML = '';
        const presets = loadUserPresets();
        
        if (presets.length === 0) {
            userPresetsList.innerHTML = '<div class="user-preset-item">Нет сохранённых предустановок</div>';
            return;
        }
        
        presets.forEach((preset, index) => {
            const presetItem = document.createElement('div');
            presetItem.className = 'user-preset-item';
            presetItem.textContent = preset.name;
            presetItem.setAttribute('data-index', index);
            
            // Кнопка удаления
            const deleteButton = document.createElement('button');
            deleteButton.className = 'delete-preset';
            deleteButton.innerHTML = '×';
            deleteButton.title = 'Удалить предустановку';
            
            deleteButton.addEventListener('click', function(event) {
                event.stopPropagation(); // Предотвращаем всплытие события
                
                if (confirm(`Удалить предустановку "${preset.name}"?`)) {
                    const presets = loadUserPresets();
                    presets.splice(index, 1);
                    saveUserPresets(presets);
                    displayUserPresets();
                }
            });
            
            presetItem.appendChild(deleteButton);
            
            // Обработчик клика для загрузки предустановки
            presetItem.addEventListener('click', function() {
                // Обновляем состояние приложения
                state.functionExpression = preset.function;
                state.xMin = preset.domain.xMin;
                state.xMax = preset.domain.xMax;
                state.yMin = preset.domain.yMin;
                state.yMax = preset.domain.yMax;
                state.colorScheme = preset.colorScheme || 'standard';
                
                // Обновляем элементы интерфейса
                document.getElementById('function-input').value = preset.function;
                document.getElementById('x-min').value = preset.domain.xMin;
                document.getElementById('x-max').value = preset.domain.xMax;
                document.getElementById('y-min').value = preset.domain.yMin;
                document.getElementById('y-max').value = preset.domain.yMax;
                
                // Обновляем отображение текущей функции
                document.getElementById('current-function').textContent = preset.function;
                
                // Обновляем отображение домена
                updateDomainDisplay();
                
                // Обновляем цветовую схему
                const colorSchemeOptions = document.querySelectorAll('.color-scheme-option');
                colorSchemeOptions.forEach(opt => {
                    if (opt.getAttribute('data-scheme') === preset.colorScheme) {
                        opt.click(); // Имитируем клик по нужной опции
                    }
                });
                
                // Перерисовываем визуализацию
                renderVisualization();
            });
            
            userPresetsList.appendChild(presetItem);
        });
    }
    
    // Обработчик кнопки сохранения предустановки
    savePresetButton.addEventListener('click', function() {
        // Создаём модальное окно для ввода имени предустановки
        const modal = document.createElement('div');
        modal.className = 'modal';
        modal.innerHTML = `
            <div class="modal-content">
                <span class="close-modal">&times;</span>
                <h3 class="modal-title">Сохранить предустановку</h3>
                <input type="text" id="preset-name-input" class="modal-input" placeholder="Введите имя предустановки">
                <div class="modal-buttons">
                    <button class="modal-button modal-cancel">Отмена</button>
                    <button class="modal-button modal-confirm">Сохранить</button>
                </div>
            </div>
        `;
        
        document.body.appendChild(modal);
        modal.style.display = 'block';
        
        const closeModal = () => {
            modal.style.display = 'none';
            modal.remove();
        };
        
        // Обработчики для модального окна
        modal.querySelector('.close-modal').addEventListener('click', closeModal);
        modal.querySelector('.modal-cancel').addEventListener('click', closeModal);
        
        modal.querySelector('.modal-confirm').addEventListener('click', function() {
            const presetName = document.getElementById('preset-name-input').value.trim();
            
            if (!presetName) {
                alert('Введите имя предустановки');
                return;
            }
            
            // Создаём объект предустановки
            const preset = {
                name: presetName,
                function: state.functionExpression,
                domain: {
                    xMin: state.xMin,
                    xMax: state.xMax,
                    yMin: state.yMin,
                    yMax: state.yMax
                },
                colorScheme: state.colorScheme
            };
            
            // Загружаем существующие предустановки
            const presets = loadUserPresets();
            
            // Добавляем новую предустановку
            presets.push(preset);
            
            // Сохраняем обновлённый список
            saveUserPresets(presets);
            
            // Обновляем отображение
            displayUserPresets();
            
            // Закрываем модальное окно
            closeModal();
        });
    });
    
    // Обработчик кнопки импорта предустановок
    importPresetsButton.addEventListener('click', function() {
        // Создаём модальное окно для импорта
        const modal = document.createElement('div');
        modal.className = 'modal';
        modal.innerHTML = `
            <div class="modal-content">
                <span class="close-modal">&times;</span>
                <h3 class="modal-title">Импорт предустановок</h3>
                <p>Вставьте JSON с предустановками:</p>
                <textarea id="import-json" class="modal-textarea"></textarea>
                <div class="modal-buttons">
                    <button class="modal-button modal-cancel">Отмена</button>
                    <button class="modal-button modal-confirm">Импортировать</button>
                </div>
            </div>
        `;
        
        document.body.appendChild(modal);
        modal.style.display = 'block';
        
        const closeModal = () => {
            modal.style.display = 'none';
            modal.remove();
        };
        
        // Обработчики для модального окна
        modal.querySelector('.close-modal').addEventListener('click', closeModal);
        modal.querySelector('.modal-cancel').addEventListener('click', closeModal);
        
        modal.querySelector('.modal-confirm').addEventListener('click', function() {
            const importJson = document.getElementById('import-json').value.trim();
            
            if (!importJson) {
                alert('Вставьте JSON с предустановками');
                return;
            }
            
            try {
                // Парсим JSON
                const importedPresets = JSON.parse(importJson);
                
                // Проверяем, что это массив
                if (!Array.isArray(importedPresets)) {
                    throw new Error('Неверный формат данных. Ожидается массив объектов.');
                }
                
                // Проверяем валидность каждой предустановки
                importedPresets.forEach(preset => {
                    if (!preset.name || !preset.function || !preset.domain) {
                        throw new Error('Неверный формат предустановки. Отсутствуют обязательные поля.');
                    }
                });
                
                // Загружаем существующие предустановки
                const existingPresets = loadUserPresets();
                
                // Добавляем новые предустановки
                const mergedPresets = [...existingPresets, ...importedPresets];
                
                // Сохраняем обновлённый список
                saveUserPresets(mergedPresets);
                
                // Обновляем отображение
                displayUserPresets();
                
                // Закрываем модальное окно
                closeModal();
                
                alert(`Успешно импортировано ${importedPresets.length} предустановок`);
            } catch (error) {
                alert(`Ошибка импорта: ${error.message}`);
            }
        });
    });
    
    // Обработчик кнопки экспорта предустановок
    exportPresetsButton.addEventListener('click', function() {
        const presets = loadUserPresets();
        
        if (presets.length === 0) {
            alert('Нет сохранённых предустановок для экспорта');
            return;
        }
        
        // Создаём модальное окно для экспорта
        const modal = document.createElement('div');
        modal.className = 'modal';
        modal.innerHTML = `
            <div class="modal-content">
                <span class="close-modal">&times;</span>
                <h3 class="modal-title">Экспорт предустановок</h3>
                <p>Скопируйте JSON с предустановками:</p>
                <textarea id="export-json" class="modal-textarea" readonly></textarea>
                <div class="modal-buttons">
                    <button class="modal-button modal-cancel">Закрыть</button>
                    <button class="modal-button modal-confirm">Скопировать</button>
                </div>
            </div>
        `;
        
        document.body.appendChild(modal);
        modal.style.display = 'block';
        
        // Заполняем текстовое поле JSON-данными
        document.getElementById('export-json').value = JSON.stringify(presets, null, 2);
        
        const closeModal = () => {
            modal.style.display = 'none';
            modal.remove();
        };
        
        // Обработчики для модального окна
        modal.querySelector('.close-modal').addEventListener('click', closeModal);
        modal.querySelector('.modal-cancel').addEventListener('click', closeModal);
        
        modal.querySelector('.modal-confirm').addEventListener('click', function() {
            const exportJson = document.getElementById('export-json');
            exportJson.select();
            document.execCommand('copy');
            alert('Предустановки скопированы в буфер обмена');
        });
    });
    
    // Инициализация отображения предустановок
    displayUserPresets();
}
```
