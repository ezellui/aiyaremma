## Расширение высших трансцендентных функций

### Куда добавлять:
Эти функции нужно добавить в объект `ComplexMath` в JavaScript-код workers'а. 
Найдите фрагмент кода, начинающийся с `const ComplexMath = {` внутри `workerCode` в функции `initWorker()`.

```javascript
// W-функция Ламберта (главная ветвь)
lambertW: function(z) {
    // Начальное приближение
    let w;
    if (math.abs(z) < 1) {
        w = z;
    } else {
        const lnz = math.log(z);
        w = math.subtract(lnz, math.log(lnz));
    }
    
    // Итерационный метод Ньютона
    const maxIter = 20;
    for (let i = 0; i < maxIter; i++) {
        const ew = math.exp(w);
        const wew = math.multiply(w, ew);
        const delta = math.divide(
            math.subtract(wew, z),
            math.add(ew, math.multiply(w, ew))
        );
        w = math.subtract(w, delta);
        
        // Проверка сходимости
        if (math.abs(delta) < 1e-10) break;
    }
    
    return w;
},

// Функции Бесселя
besselJ: function(nu, z) {
    // Приближение для целых nu >= 0
    if (Math.abs(Math.round(math.re(nu)) - math.re(nu)) < 1e-10 && math.re(nu) >= 0 && math.im(nu) === 0) {
        const n = Math.round(math.re(nu));
        let sum = math.complex(0);
        const maxIter = 50;
        
        for (let m = 0; m < maxIter; m++) {
            const term = math.multiply(
                math.pow(math.complex(-1), m),
                math.divide(
                    math.pow(math.divide(z, 2), math.add(n, math.multiply(2, m))),
                    math.multiply(
                        ComplexMath.gamma(math.add(m, 1)),
                        ComplexMath.gamma(math.add(n, m, 1))
                    )
                )
            );
            sum = math.add(sum, term);
            
            // Проверка сходимости
            if (math.abs(term) < 1e-10) break;
        }
        
        return sum;
    }
    
    // Общий случай для произвольных nu и z
    let sum = math.complex(0);
    const maxIter = 50;
    
    for (let m = 0; m < maxIter; m++) {
        const term = math.multiply(
            math.pow(math.complex(-1), m),
            math.divide(
                math.pow(math.divide(z, 2), math.add(nu, math.multiply(2, m))),
                math.multiply(
                    ComplexMath.gamma(math.add(m, 1)),
                    ComplexMath.gamma(math.add(nu, m, 1))
                )
            )
        );
        sum = math.add(sum, term);
        
        // Проверка сходимости
        if (math.abs(term) < 1e-10) break;
    }
    
    return sum;
},

besselY: function(nu, z) {
    // Y_nu(z) = (J_nu(z)cos(nu*pi) - J_{-nu}(z))/sin(nu*pi)
    const nuPi = math.multiply(nu, math.pi);
    return math.divide(
        math.subtract(
            math.multiply(ComplexMath.besselJ(nu, z), math.cos(nuPi)),
            ComplexMath.besselJ(math.multiply(nu, -1), z)
        ),
        math.sin(nuPi)
    );
},

// Эллиптические функции Якоби
ellipticSn: function(z, k) {
    // Аппроксимация на основе разложения в ряд
    let sum = z;
    const k2 = math.multiply(k, k);
    const z2 = math.multiply(z, z);
    let term = math.multiply(
        math.divide(math.multiply(k2, z), 6),
        math.multiply(z2, math.subtract(1, z2))
    );
    sum = math.add(sum, term);
    
    // Дополнительные члены для более точной аппроксимации
    term = math.multiply(
        math.divide(math.multiply(k2, math.multiply(k2, z)), 120),
        math.multiply(
            z2,
            math.add(
                math.subtract(
                    math.multiply(3, math.pow(z2, 2)),
                    math.multiply(10, z2)
                ),
                7
            )
        )
    );
    sum = math.add(sum, term);
    
    return sum;
},

ellipticCn: function(z, k) {
    // Связь с sn: cn²(z, k) + sn²(z, k) = 1
    const sn = ComplexMath.ellipticSn(z, k);
    return math.sqrt(math.subtract(1, math.multiply(sn, sn)));
},

ellipticDn: function(z, k) {
    // Связь с sn: dn²(z, k) + k² sn²(z, k) = 1
    const sn = ComplexMath.ellipticSn(z, k);
    const k2 = math.multiply(k, k);
    return math.sqrt(math.subtract(1, math.multiply(k2, math.multiply(sn, sn))));
},

// Тета-функции
theta: function(z, tau, type) {
    const q = math.exp(math.multiply(math.pi, math.multiply(math.i, tau)));
    let sum = math.complex(0);
    const maxIter = 20;
    
    if (type === 1) {
        // θ₁(z, τ)
        for (let n = 0; n < maxIter; n++) {
            const qn2 = math.pow(q, math.pow(n + 0.5, 2));
            const term = math.multiply(
                math.pow(math.complex(-1), n),
                math.multiply(
                    qn2,
                    math.sin(math.multiply(math.complex(2*n + 1), z))
                )
            );
            sum = math.add(sum, term);
        }
        return math.multiply(2, sum);
    } else if (type === 2) {
        // θ₂(z, τ)
        for (let n = 0; n < maxIter; n++) {
            const qn2 = math.pow(q, math.pow(n + 0.5, 2));
            const term = math.multiply(
                qn2,
                math.cos(math.multiply(math.complex(2*n + 1), z))
            );
            sum = math.add(sum, term);
        }
        return math.multiply(2, sum);
    } else if (type === 3) {
        // θ₃(z, τ)
        sum = math.complex(1);
        for (let n = 1; n < maxIter; n++) {
            const qn2 = math.pow(q, math.pow(n, 2));
            const term = math.multiply(
                math.multiply(2, qn2),
                math.cos(math.multiply(math.complex(2*n), z))
            );
            sum = math.add(sum, term);
        }
        return sum;
    } else if (type === 4) {
        // θ₄(z, τ)
        sum = math.complex(1);
        for (let n = 1; n < maxIter; n++) {
            const qn2 = math.pow(q, math.pow(n, 2));
            const term = math.multiply(
                math.multiply(2, math.pow(math.complex(-1), n), qn2),
                math.cos(math.multiply(math.complex(2*n), z))
            );
            sum = math.add(sum, term);
        }
        return sum;
    }
    
    return math.complex(NaN, NaN);
},

// Полином Лежандра
legendreP: function(n, z) {
    if (n === 0) return math.complex(1);
    if (n === 1) return z;
    
    // Рекуррентная формула
    let p_prev = math.complex(1);  // P_0(z)
    let p_curr = z;                // P_1(z)
    
    for (let k = 1; k < n; k++) {
        const p_next = math.divide(
            math.subtract(
                math.multiply(math.multiply(2*k + 1, z), p_curr),
                math.multiply(k, p_prev)
            ),
            k + 1
        );
        p_prev = p_curr;
        p_curr = p_next;
    }
    
    return p_curr;
},

// Комплексное сопряжение
conj: function(z) {
    return math.complex(math.re(z), -math.im(z));
},
```
