### 3. Добавление JavaScript-кода для выделения области

```javascript
// Функция для добавления возможности выделения области на холсте
function setupZoomSelection() {
    const canvas = document.getElementById('visualization-canvas');
    const zoomButton = document.getElementById('zoom-selection-button');
    
    // Добавляем блок с инструкцией
    const instructionElement = document.createElement('div');
    instructionElement.id = 'zoom-instruction';
    instructionElement.className = 'success-message';
    instructionElement.textContent = 'Выделите область на холсте для увеличения';
    
    const canvasContainer = canvas.parentElement;
    canvasContainer.appendChild(instructionElement);
    
    let isSelecting = false;
    let selectionStart = { x: 0, y: 0 };
    let selectionCurrent = { x: 0, y: 0 };
    let selectionRect = null;
    
    // Функция для создания прямоугольника выделения
    function createSelectionRect() {
        if (selectionRect) return;
        
        selectionRect = document.createElement('div');
        selectionRect.className = 'selection-rect';
        canvasContainer.appendChild(selectionRect);
    }
    
    // Функция для обновления прямоугольника выделения
    function updateSelectionRect() {
        if (!selectionRect) return;
        
        const canvasRect = canvas.getBoundingClientRect();
        const containerRect = canvasContainer.getBoundingClientRect();
        
        // Вычисляем координаты относительно контейнера
        const left = Math.min(selectionStart.x, selectionCurrent.x);
        const top = Math.min(selectionStart.y, selectionCurrent.y);
        const width = Math.abs(selectionCurrent.x - selectionStart.x);
        const height = Math.abs(selectionCurrent.y - selectionStart.y);
        
        // Устанавливаем размеры и позицию прямоугольника
        selectionRect.style.left = left + 'px';
        selectionRect.style.top = top + 'px';
        selectionRect.style.width = width + 'px';
        selectionRect.style.height = height + 'px';
    }
    
    // Функция для удаления прямоугольника выделения
    function removeSelectionRect() {
        if (selectionRect) {
            selectionRect.remove();
            selectionRect = null;
        }
    }
    
    // Функция для преобразования координат холста в координаты комплексной плоскости
    function canvasToComplex(x, y) {
        const canvasRect = canvas.getBoundingClientRect();
        const canvasX = (x - canvasRect.left) / canvasRect.width * canvas.width;
        const canvasY = (y - canvasRect.top) / canvasRect.height * canvas.height;
        
        const re = state.xMin + (canvasX / canvas.width) * (state.xMax - state.xMin);
        const im = state.yMax - (canvasY / canvas.height) * (state.yMax - state.yMin);
        
        return { re, im };
    }
    
    // Обработчик нажатия на кнопку выделения
    zoomButton.addEventListener('click', function() {
        isSelecting = !isSelecting;
        
        if (isSelecting) {
            // Активируем режим выделения
            zoomButton.textContent = 'Отменить выделение';
            zoomButton.classList.add('active');
            instructionElement.style.display = 'block';
            
            // Изменяем стиль курсора
            canvas.style.cursor = 'crosshair';
        } else {
            // Деактивируем режим выделения
            zoomButton.textContent = 'Выделить область для увеличения';
            zoomButton.classList.remove('active');
            instructionElement.style.display = 'none';
            
            // Сбрасываем курсор
            canvas.style.cursor = '';
            
            // Удаляем прямоугольник выделения, если есть
            removeSelectionRect();
        }
    });
    
    // Обработчик начала выделения
    canvas.addEventListener('mousedown', function(event) {
        if (!isSelecting) return;
        
        // Запоминаем начальную точку
        selectionStart.x = event.clientX;
        selectionStart.y = event.clientY;
        selectionCurrent.x = event.clientX;
        selectionCurrent.y = event.clientY;
        
        // Создаем прямоугольник выделения
        createSelectionRect();
        updateSelectionRect();
    });
    
    // Обработчик перемещения мыши
    document.addEventListener('mousemove', function(event) {
        if (!isSelecting || !selectionRect) return;
        
        // Обновляем текущую точку
        selectionCurrent.x = event.clientX;
        selectionCurrent.y = event.clientY;
        
        // Обновляем прямоугольник выделения
        updateSelectionRect();
    });
    
    // Обработчик окончания выделения
    document.addEventListener('mouseup', function(event) {
        if (!isSelecting || !selectionRect) return;
        
        // Получаем координаты выделенной области в комплексной плоскости
        const startComplex = canvasToComplex(selectionStart.x, selectionStart.y);
        const endComplex = canvasToComplex(selectionCurrent.x, selectionCurrent.y);
        
        // Проверяем, что выделена достаточно большая область
        const minSizeX = (state.xMax - state.xMin) * 0.01;
        const minSizeY = (state.yMax - state.yMin) * 0.01;
        
        if (Math.abs(endComplex.re - startComplex.re) < minSizeX || 
            Math.abs(endComplex.im - startComplex.im) < minSizeY) {
            // Слишком маленькая область, отменяем выделение
            removeSelectionRect();
            return;
        }
        
        // Определяем новые границы области
        const newXmin = Math.min(startComplex.re, endComplex.re);
        const newXmax = Math.max(startComplex.re, endComplex.re);
        const newYmin = Math.min(startComplex.im, endComplex.im);
        const newYmax = Math.max(startComplex.im, endComplex.im);
        
        // Обновляем состояние и элементы интерфейса
        state.xMin = newXmin;
        state.xMax = newXmax;
        state.yMin = newYmin;
        state.yMax = newYmax;
        
        // Обновляем поля ввода границ домена
        document.getElementById('x-min').value = newXmin.toFixed(3);
        document.getElementById('x-max').value = newXmax.toFixed(3);
        document.getElementById('y-min').value = newYmin.toFixed(3);
        document.getElementById('y-max').value = newYmax.toFixed(3);
        
        // Обновляем отображение домена
        updateDomainDisplay();
        
        // Удаляем прямоугольник выделения
        removeSelectionRect();
        
        // Выключаем режим выделения
        isSelecting = false;
        zoomButton.textContent = 'Выделить область для увеличения';
        zoomButton.classList.remove('active');
        instructionElement.style.display = 'none';
        canvas.style.cursor = '';
        
        // Перерисовываем визуализацию
        renderVisualization();
    });
    
    // Обработчик отмены выделения при клике правой кнопкой мыши
    canvas.addEventListener('contextmenu', function(event) {
        if (!isSelecting) return;
        
        event.preventDefault();
        
        // Отменяем выделение
        removeSelectionRect();
    });
    
    // Обработчик нажатия Escape для отмены выделения
    document.addEventListener('keydown', function(event) {
        if (event.key === 'Escape' && isSelecting) {
            // Отменяем выделение
            isSelecting = false;
            zoomButton.textContent = 'Выделить область для увеличения';
            zoomButton.classList.remove('active');
            instructionElement.style.display = 'none';
            canvas.style.cursor = '';
            
            // Удаляем прямоугольник выделения
            removeSelectionRect();
        }
    });
}
```

### 5. Обработчик сброса масштаба

Добавьте следующий код к функции `setupZoomSelection()`:

```javascript
// Обработчик сброса масштаба
const resetZoomButton = document.getElementById('reset-zoom-button');
resetZoomButton.addEventListener('click', function() {
    // Сбрасываем домен к значениям по умолчанию
    state.xMin = -5;
    state.xMax = 5;
    state.yMin = -5;
    state.yMax = 5;
    
    // Обновляем поля ввода
    document.getElementById('x-min').value = state.xMin;
    document.getElementById('x-max').value = state.xMax;
    document.getElementById('y-min').value = state.yMin;
    document.getElementById('y-max').value = state.yMax;
    
    // Обновляем ползунок масштаба
    document.getElementById('scale-range').value = 5;
    
    // Обновляем отображение домена
    updateDomainDisplay();
    
    // Перерисовываем визуализацию
    renderVisualization();
});
```

### 7. Добавление двойного клика для быстрого увеличения

Добавьте следующий обработчик к функции `setupZoomSelection()`:

```javascript
// Обработчик двойного клика для быстрого увеличения
canvas.addEventListener('dblclick', function(event) {
    // Получаем координаты точки в комплексной плоскости
    const point = canvasToComplex(event.clientX, event.clientY);
    
    // Рассчитываем новые границы (увеличение в 2 раза)
    const zoomFactor = 0.5; // Уменьшаем область в 2 раза
    const currentWidth = state.xMax - state.xMin;
    const currentHeight = state.yMax - state.yMin;
    const newWidth = currentWidth * zoomFactor;
    const newHeight = currentHeight * zoomFactor;
    
    // Устанавливаем новые границы вокруг точки
    state.xMin = point.re - newWidth / 2;
    state.xMax = point.re + newWidth / 2;
    state.yMin = point.im - newHeight / 2;
    state.yMax = point.im + newHeight / 2;
    
    // Обновляем поля ввода
    document.getElementById('x-min').value = state.xMin.toFixed(3);
    document.getElementById('x-max').value = state.xMax.toFixed(3);
    document.getElementById('y-min').value = state.yMin.toFixed(3);
    document.getElementById('y-max').value = state.yMax.toFixed(3);
    
    // Обновляем отображение домена
    updateDomainDisplay();
    
    // Перерисовываем визуализацию
    renderVisualization();
});
```
