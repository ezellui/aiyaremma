FROM node:18-alpine

WORKDIR /app

# Устанавливаем nodemon глобально
RUN npm install -g nodemon

# Копируем shared модуль
COPY shared /app/shared
WORKDIR /app/shared
RUN npm install

# Копируем API Gateway
WORKDIR /app/api-gateway
COPY api-gateway/package*.json ./
RUN npm install

COPY api-gateway .

EXPOSE 4000

CMD ["npm", "run", "dev"]
