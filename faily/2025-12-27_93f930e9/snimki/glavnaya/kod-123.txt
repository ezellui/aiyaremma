FROM node:18-alpine

WORKDIR /app

# Устанавливаем nodemon глобально
RUN npm install -g nodemon

# Копируем shared модуль
COPY shared /app/shared
WORKDIR /app/shared
RUN npm install

# Копируем User Service
WORKDIR /app/user-service
COPY user-service/package*.json ./
RUN npm install

COPY user-service .

EXPOSE 4001

CMD ["npm", "run", "dev"]
