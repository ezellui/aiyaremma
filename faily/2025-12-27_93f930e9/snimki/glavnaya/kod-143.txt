FROM node:18-alpine

WORKDIR /app

# Устанавливаем nodemon глобально
RUN npm install -g nodemon

# Копируем shared модуль
COPY shared /app/shared
WORKDIR /app/shared
RUN npm install

# Копируем Claude Service
WORKDIR /app/claude-service
COPY claude-service/package*.json ./
RUN npm install

COPY claude-service .

EXPOSE 4006

CMD ["npm", "run", "dev"]
