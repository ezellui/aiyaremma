## 3. Добавление альтернативного текстового представления сети

Добавьте следующий JavaScript-код в конец файла `<script>`:

```javascript
/**
 * Генерирует текстовое представление сети для скринридеров
 * @param {Object} data - Данные сети
 * @returns {string} - HTML для текстового представления
 */
function generateAccessibleNetworkText(data) {
  const nodes = data.nodes;
  const links = data.links;
  
  // Группируем узлы по группам (школам/периодам)
  const groupedNodes = {};
  
  nodes.forEach(node => {
    if (!groupedNodes[node.group]) {
      groupedNodes[node.group] = [];
    }
    groupedNodes[node.group].push(node);
  });
  
  // Начинаем создавать HTML-разметку
  let html = '<div id="accessible-network" class="accessible-network" aria-hidden="true" tabindex="-1">';
  html += '<h2>Текстовое представление сети российской философии</h2>';
  
  // Добавляем информацию о количестве узлов и связей
  html += `<p>Сеть содержит ${nodes.length} философов и ${links.length} связей между ними.</p>`;
  
  // Добавляем список философов по группам
  html += '<div class="group-list">';
  
  // Для каждой группы создаем раздел
  Object.keys(groupedNodes).forEach(group => {
    const groupNodes = groupedNodes[group];
    const groupName = getGroupDisplayName(group);
    
    html += `<div class="group-section">`;
    html += `<h3>${groupName} (${groupNodes.length} философов)</h3>`;
    
    // Создаем список философов в группе
    html += `<ul>`;
    groupNodes.forEach(node => {
      html += `<li>`;
      html += `<button class="accessible-node-button" data-node-id="${node.id}" tabindex="-1">${node.name} (${node.years})</button>`;
      
      // Добавляем связи для каждого философа
      const nodeLinks = links.filter(link => 
        link.source.id === node.id || link.target.id === node.id
      );
      
      if (nodeLinks.length > 0) {
        html += `<ul class="relations-list">`;
        
        // Группируем связи по типам
        const relationsByType = {};
        
        nodeLinks.forEach(link => {
          if (!relationsByType[link.type]) {
            relationsByType[link.type] = [];
          }
          
          // Определяем, является ли текущий философ источником или целью связи
          const isSource = link.source.id === node.id;
          const relatedNode = isSource ? link.target : link.source;
          
          relationsByType[link.type].push({
            relatedNode,
            isSource
          });
        });
        
        // Для каждого типа связи создаем подраздел
        Object.keys(relationsByType).forEach(type => {
          const relationsOfType = relationsByType[type];
          const relationTypeName = getRelationshipTypeName(type);
          
          html += `<li>${relationTypeName}: `;
          
          // Список связанных философов
          const relatedNodeNames = relationsOfType.map(relation => {
            const directionText = relation.isSource ? 'на' : 'от';
            
            return `${relation.relatedNode.name} (${directionText})`;
          }).join(', ');
          
          html += relatedNodeNames;
          html += `</li>`;
        });
        
        html += `</ul>`;
      } else {
        html += ` <em>(нет связей)</em>`;
      }
      
      html += `</li>`;
    });
    html += `</ul>`;
    html += `</div>`;
  });
  
  html += '</div>'; // Закрываем .group-list
  
  // Добавляем кнопку для переключения между текстовым и визуальным представлением
  html += `
    <div class="accessibility-controls">
      <button id="toggle-visualization" tabindex="-1" aria-hidden="true">Переключить на визуальное представление</button>
    </div>
  `;
  
  html += '</div>'; // Закрываем #accessible-network
  
  return html;
}

/**
 * Инициализирует альтернативное текстовое представление
 * @param {Object} data - Данные сети
 */
function initAccessibleTextRepresentation(data) {
  // Генерируем текстовое представление
  const accessibleHTML = generateAccessibleNetworkText(data);
  
  // Добавляем кнопку для переключения на текстовое представление
  const networkContainer = document.querySelector('.network-container');
  
  const textViewButton = document.createElement('button');
  textViewButton.id = 'show-text-view';
  textViewButton.className = 'accessibility-button';
  textViewButton.setAttribute('aria-label', 'Переключить на текстовое представление');
  textViewButton.innerHTML = '<span aria-hidden="true">Aa</span>';
  textViewButton.style.position = 'absolute';
  textViewButton.style.top = '10px';
  textViewButton.style.right = '10px';
  textViewButton.style.zIndex = '100';
  textViewButton.style.backgroundColor = 'white';
  textViewButton.style.border = '1px solid #ccc';
  textViewButton.style.borderRadius = '4px';
  textViewButton.style.padding = '5px 10px';
  textViewButton.style.fontWeight = 'bold';
  
  networkContainer.appendChild(textViewButton);
  
  // Создаем контейнер для текстового представления
  const accessibleContainer = document.createElement('div');
  accessibleContainer.id = 'accessible-view-container';
  accessibleContainer.className = 'accessible-view-container';
  accessibleContainer.style.position = 'absolute';
  accessibleContainer.style.top = '0';
  accessibleContainer.style.left = '0';
  accessibleContainer.style.width = '100%';
  accessibleContainer.style.height = '100%';
  accessibleContainer.style.backgroundColor = 'white';
  accessibleContainer.style.padding = '20px';
  accessibleContainer.style.overflowY = 'auto';
  accessibleContainer.style.display = 'none';
  accessibleContainer.style.zIndex = '200';
  accessibleContainer.innerHTML = accessibleHTML;
  
  networkContainer.appendChild(accessibleContainer);
  
  // Добавляем обработчики событий для переключения представлений
  textViewButton.addEventListener('click', function() {
    const isTextViewVisible = accessibleContainer.style.display !== 'none';
    
    if (isTextViewVisible) {
      accessibleContainer.style.display = 'none';
      textViewButton.setAttribute('aria-label', 'Переключить на текстовое представление');
      textViewButton.setAttribute('aria-pressed', 'false');
    } else {
      accessibleContainer.style.display = 'block';
      textViewButton.setAttribute('aria-label', 'Переключить на визуальное представление');
      textViewButton.setAttribute('aria-pressed', 'true');
      
      // Устанавливаем фокус на текстовое представление
      accessibleContainer.setAttribute('tabindex', '0');
      accessibleContainer.focus();
    }
  });
  
  // Добавляем обработчик для кнопки внутри текстового представления
  const toggleButton = accessibleContainer.querySelector('#toggle-visualization');
  if (toggleButton) {
    toggleButton.addEventListener('click', function() {
      accessibleContainer.style.display = 'none';
      textViewButton.setAttribute('aria-label', 'Переключить на текстовое представление');
      textViewButton.setAttribute('aria-pressed', 'false');
    });
  }
  
  // Добавляем обработчики для кнопок философов
  const nodeButtons = accessibleContainer.querySelectorAll('.accessible-node-button');
  nodeButtons.forEach(button => {
    button.addEventListener('click', function() {
      const nodeId = this.dataset.nodeId;
      const node = data.nodes.find(n => n.id === nodeId);
      
      if (node) {
        // Отображаем информацию о философе
        showNodeInfo(node);
        
        // Опционально открываем детальную информацию
        const detailPopup = document.querySelector('.detail-popup');
        showDetailedInfo(node, detailPopup, {links: data.links});
      }
    });
  });
}

/**
 * Создает озвучиваемый ARIA live region для объявлений
 */
function createAriaLiveRegion() {
  const liveRegion = document.createElement('div');
  liveRegion.id = 'aria-live-announcer';
  liveRegion.className = 'sr-only';
  liveRegion.setAttribute('aria-live', 'polite');
  liveRegion.setAttribute('aria-atomic', 'true');
  document.body.appendChild(liveRegion);
  
  // Функция для объявления сообщений
  window.announce = function(message, priority = 'polite') {
    liveRegion.setAttribute('aria-live', priority);
    liveRegion.textContent = message;
    
    // Очищаем сообщение через определенное время
    setTimeout(() => {
      liveRegion.textContent = '';
    }, 3000);
  };
}

/**
 * Добавляет атрибуты доступности к существующим элементам SVG
 * @param {Object} network - Объект сети
 */
function enhanceSVGAccessibility(network) {
  // Добавляем атрибуты к SVG-элементу
  const svg = d3.select('#network svg');
  
  svg.attr('role', 'img')
    .attr('aria-labelledby', 'network-description');
  
  // Устанавливаем атрибуты для групп и слоев
  svg.select('g').attr('aria-label', 'Группа элементов сети');
  
  // Добавляем атрибуты к узлам
  d3.selectAll('.node')
    .attr('role', 'button')
    .attr('aria-label', d => `Философ ${d.name}, ${d.years}, группа ${d.groupName}`)
    .attr('tabindex', '0');
  
  // Добавляем метки узлов для улучшения доступности
  d3.selectAll('.node').each(function(d) {
    // Создаем title-элемент для SVG-элементов (всплывающая подсказка и доступные описания)
    const nodeGroup = d3.select(this);
    
    // Проверяем, есть ли уже title
    if (nodeGroup.select('title').empty()) {
      nodeGroup.append('title')
        .text(d => `${d.name}, ${d.years}, ${d.groupName}`);
    }
  });
  
  // Добавляем описания для связей
  d3.selectAll('.links line').each(function(d) {
    const link = d3.select(this);
    
    // Добавляем title для связи, если его еще нет
    if (link.select('title').empty()) {
      const sourceNode = d.source.name || d.source;
      const targetNode = d.target.name || d.target;
      const relationshipType = getRelationshipTypeName(d.type);
      
      link.append('title')
        .text(`${relationshipType} между ${sourceNode} и ${targetNode}`);
    }
  });
}

/**
 * Улучшает доступность тултипов и всплывающих подсказок
 */
function enhanceTooltipAccessibility() {
  // Улучшаем доступность тултипа
  const tooltip = document.getElementById('tooltip');
  tooltip.setAttribute('role', 'status');
  tooltip.setAttribute('aria-live', 'polite');
  
  // Заменяем обработчики событий для узлов, чтобы объявлять информацию
  const originalNodeMouseover = d3.selectAll('.node').on('mouseover');
  
  d3.selectAll('.node').on('mouseover', function(event, d) {
    // Вызываем оригинальный обработчик
    if (originalNodeMouseover) {
      originalNodeMouseover.call(this, event, d);
    }
    
    // Объявляем для скринридеров
    if (window.announce) {
      window.announce(`Философ ${d.name}, ${d.years}, ${d.groupName}`);
    }
  });
  
  // Аналогично для кластеров
  d3.selectAll('.cluster').on('mouseover', function(event, d) {
    // Объявляем для скринридеров
    if (window.announce) {
      window.announce(`Кластер ${d.name} содержит ${d.nodes.length} философов`);
    }
  });
}

/**
```
```javascript
 * Обновляет семантическую структуру страницы для скринридеров
 */
function enhancePageSemantics() {
  // Добавляем главный элемент навигации
  const controls = document.querySelector('.controls');
  controls.setAttribute('role', 'navigation');
  controls.setAttribute('aria-label', 'Фильтры и поиск');

  const sections = [
    { selector: '.timeline-container', label: 'Временная шкала' },
    { selector: '.network-container', label: 'Интерактивная сеть философов' },
    { selector: '.sidebar', label: 'Легенда и информация' }
  ];
  
  sections.forEach(section => {
    const element = document.querySelector(section.selector);
    if (element) {
      element.setAttribute('role', 'region');
      element.setAttribute('aria-label', section.label);
    }
  });
  
  // Добавляем метки для элементов управления
  const inputs = document.querySelectorAll('input, select, button');
  inputs.forEach(input => {
    if (!input.hasAttribute('aria-label') && !input.hasAttribute('aria-labelledby')) {
      // Если у элемента нет метки, добавляем на основе текста или placeholder
      if (input.placeholder) {
        input.setAttribute('aria-label', input.placeholder);
      } else if (input.textContent) {
        input.setAttribute('aria-label', input.textContent.trim());
      }
    }
  });
  
  // Добавляем подсказки для модальных окон
  const modal = document.querySelector('.detail-popup');
  if (modal) {
    modal.setAttribute('role', 'dialog');
    modal.setAttribute('aria-modal', 'true');
    modal.setAttribute('aria-labelledby', 'detail-popup-title');
  }
  
  // Устанавливаем правильные атрибуты для элементов легенды
  const legendItems = document.querySelectorAll('.legend-item');
  legendItems.forEach(item => {
    const colorBox = item.querySelector('.legend-color');
    const text = item.querySelector('span').textContent;
    
    if (colorBox) {
      colorBox.setAttribute('aria-hidden', 'true');
      item.setAttribute('role', 'listitem');
      item.setAttribute('aria-label', text);
    }
  });
}

// Вызываем функцию после загрузки DOM
document.addEventListener('DOMContentLoaded', enhancePageSemantics);
```

## 5. Генерация текстовых описаний для графических элементов

Добавьте следующий код для создания текстовых описаний на основе визуализации:

```javascript
/**
 * Генерирует текстовое описание сети на основе текущего состояния
 * @param {Object} data - Данные сети
 * @returns {string} - Текстовое описание
 */
function generateNetworkDescription(data) {
  const nodes = data.nodes;
  const links = data.links;
  
  // Базовое описание
  let description = `Сеть состоит из ${nodes.length} философов и ${links.length} связей между ними. `;
  
  // Группировка философов по школам
  const schoolCounts = {};
  nodes.forEach(node => {
    if (!schoolCounts[node.group]) {
      schoolCounts[node.group] = 0;
    }
    schoolCounts[node.group]++;
  });
  
  // Добавляем описание распределения по школам
  description += 'Философы распределены по следующим школам и периодам: ';
  
  const schoolDescriptions = Object.keys(schoolCounts).map(group => {
    const groupName = getGroupDisplayName(group);
    const count = schoolCounts[group];
    return `${groupName} (${count} философов)`;
  });
  
  description += schoolDescriptions.join(', ') + '. ';
  
  // Добавляем описание типов связей
  const connectionTypes = {};
  links.forEach(link => {
    if (!connectionTypes[link.type]) {
      connectionTypes[link.type] = 0;
    }
    connectionTypes[link.type]++;
  });
  
  description += 'Типы связей между философами: ';
  
  const connectionDescriptions = Object.keys(connectionTypes).map(type => {
    const typeName = getRelationshipTypeName(type);
    const count = connectionTypes[type];
    return `${typeName} (${count} связей)`;
  });
  
  description += connectionDescriptions.join(', ') + '. ';
  
  // Добавляем информацию о самых влиятельных философах
  const influenceMap = {};
  
  links.forEach(link => {
    if (link.type === 'influence' || link.type === 'mutual') {
      const sourceId = link.source.id || link.source;
      
      if (!influenceMap[sourceId]) {
        influenceMap[sourceId] = 0;
      }
      
      influenceMap[sourceId]++;
    }
  });
  
  // Сортируем философов по степени влияния
  const influentialPhilosophers = Object.keys(influenceMap)
    .sort((a, b) => influenceMap[b] - influenceMap[a])
    .slice(0, 5)
    .map(id => {
      const node = nodes.find(n => n.id === id);
      return node ? node.name : id;
    });
  
  if (influentialPhilosophers.length > 0) {
    description += 'Наиболее влиятельные философы: ' + influentialPhilosophers.join(', ') + '. ';
  }
  
  // Добавляем описание текущего фильтра, если он применен
  const periodFilter = document.getElementById('filter-period');
  const relationFilter = document.getElementById('filter-relation');
  
  if (periodFilter && periodFilter.value !== 'all') {
    const periodLabel = periodFilter.options[periodFilter.selectedIndex].text;
    description += `Применен фильтр по периоду: ${periodLabel}. `;
  }
  
  if (relationFilter && relationFilter.value !== 'all') {
    const relationLabel = relationFilter.options[relationFilter.selectedIndex].text;
    description += `Применен фильтр по типу связи: ${relationLabel}. `;
  }
  
  // Добавляем информацию о временной шкале
  description += 'Сеть охватывает период с XI века до наших дней. ';
  
  return description;
}

/**
 * Обновляет текстовое описание для скринридеров
 * @param {Object} data - Данные сети
 */
function updateAccessibleDescription(data) {
  const descriptionElement = document.getElementById('network-description');
  if (!descriptionElement) return;
  
  // Генерируем описание
  const description = generateNetworkDescription(data);
  
  // Обновляем элемент с описанием
  descriptionElement.textContent = description;
  
  // Объявляем о значительных изменениях
  if (window.announce) {
    window.announce('Визуализация сети обновлена');
  }
}

document.addEventListener('DOMContentLoaded', function() {
```

```javascript
  `;

  } else {
    contrastButton.setAttribute('aria-pressed', 'false');

    document.body.classList.add('high-contrast-mode');
    contrastButton.setAttribute('aria-pressed', 'true');
  }
}
```
