## 5. Реализация функций управления доступностью цветов

Добавьте следующий JavaScript-код в конец файла `<script>`:

```javascript
/**
 * Класс для управления цветовой доступностью
 */
class ColorAccessibilityManager {
  /**
   * Инициализирует менеджер доступности цветов
   */
  constructor() {
    this.colorSchemes = [
      { id: 'standard', name: 'Стандартная цветовая схема', class: '' },
      { id: 'protanopia', name: 'Протанопия (красный дальтонизм)', class: 'color-protanopia' },
      { id: 'deuteranopia', name: 'Дейтеранопия (зеленый дальтонизм)', class: 'color-deuteranopia' },
      { id: 'tritanopia', name: 'Тританопия (синий дальтонизм)', class: 'color-tritanopia' },
      { id: 'monochromacy', name: 'Монохромное зрение', class: 'color-monochromacy' }
    ];
    
    this.currentScheme = 'standard';
    this.usePatterns = false;
    this.largeNodes = false;
    this.thickLines = false;
    this.enhancedLabels = false;
    
    // Инициализация элементов управления
    this.setupControls();
    
    // Определение начальных настроек доступности
    this.detectAccessibilityNeeds();
  }
  
  /**
   * Настраивает элементы управления доступностью
   */
  setupControls() {
    // Кнопка переключения панели
    const toggleButton = document.getElementById('toggle-accessibility-toolbar');
    const controlsPanel = document.querySelector('.accessibility-controls');
    
    if (toggleButton && controlsPanel) {
      toggleButton.addEventListener('click', () => {
        const isVisible = controlsPanel.style.display !== 'none';
        controlsPanel.style.display = isVisible ? 'none' : 'flex';
        toggleButton.setAttribute('aria-expanded', !isVisible);
      });
    }
    
    // Кнопки цветовых схем
    const colorButtons = document.querySelectorAll('.color-scheme-btn');
    colorButtons.forEach(button => {
      button.addEventListener('click', () => {
        const schemeId = button.id.replace('color-', '');
        this.setColorScheme(schemeId);
        
        // Обновляем активную кнопку
        colorButtons.forEach(btn => btn.classList.remove('active'));
        button.classList.add('active');
      });
    });
    
    // Кнопка переключения паттернов
    const patternButton = document.getElementById('toggle-patterns');
    if (patternButton) {
      patternButton.addEventListener('click', () => {
        this.usePatterns = !this.usePatterns;
        this.updateAccessibility();
        patternButton.classList.toggle('active', this.usePatterns);
        
        // Объявляем для скринридеров
        if (window.announce) {
          window.announce(this.usePatterns ? 'Текстурные паттерны включены' : 'Текстурные паттерны отключены');
        }
      });
    }
    
    // Кнопка переключения размера узлов
    const nodeButton = document.getElementById('toggle-large-nodes');
    if (nodeButton) {
      nodeButton.addEventListener('click', () => {
        // Переключение между тремя состояниями: обычное -> большое -> очень большое -> обычное
        if (!this.largeNodes) {
          this.largeNodes = 'large';
        } else if (this.largeNodes === 'large') {
          this.largeNodes = 'larger';
        } else {
          this.largeNodes = false;
        }
        
        this.updateAccessibility();
        nodeButton.classList.toggle('active', this.largeNodes);
        
        // Объявляем для скринридеров
        if (window.announce) {
          const sizeText = !this.largeNodes ? 'Стандартный размер узлов' : 
            (this.largeNodes === 'large' ? 'Увеличенный размер узлов' : 'Максимальный размер узлов');
          window.announce(sizeText);
        }
      });
    }
    
    // Кнопка переключения толщины линий
    const lineButton = document.getElementById('toggle-thick-lines');
    if (lineButton) {
      lineButton.addEventListener('click', () => {

    } else if (this.thickLines === 'thick') {
      document.body.classList.add('line-width-thick');

    document.body.classList.toggle('node-label-enhanced', this.enhancedLabels);

  /**
   * Определяет настройки доступности на основе системных предпочтений

  detectAccessibilityNeeds() {

    const savedScheme = localStorage.getItem('color-scheme');
    const savedPatterns = localStorage.getItem('use-patterns') === 'true';
    const savedNodes = localStorage.getItem('large-nodes');

    // Проверяем системные настройки высокой контрастности
    if (window.matchMedia && window.matchMedia('(prefers-contrast: high)').matches) {
      this.usePatterns = true;
      this.thickLines = 'thick';
      this.enhancedLabels = true;

      const patternButton = document.getElementById('toggle-patterns');

      this.usePatterns = savedPatterns;
      this.thickLines = savedLines;
      this.largeNodes = savedNodes;
      this.enhancedLabels = savedLabels;
      
      // Активируем соответствующие кнопки на основе сохраненных настроек
      const patternButton = document.getElementById('toggle-patterns');
      const lineButton = document.getElementById('toggle-thick-lines');
      const nodeButton = document.getElementById('toggle-large-nodes');

      if (patternButton) patternButton.classList.toggle('active', savedPatterns);
      if (lineButton) lineButton.classList.toggle('active', savedLines);
      if (nodeButton) nodeButton.classList.toggle('active', savedNodes);
      if (labelButton) labelButton.classList.toggle('active', savedLabels);
    }
    
    // Проверяем системные настройки цветовой схемы (если пользователь не сохранил свои)
    if (!savedScheme && window.matchMedia) {
      if (window.matchMedia('(prefers-color-scheme: dark)').matches) {
        // Если пользователь предпочитает темную тему, усиливаем контраст
        this.thickLines = 'medium';
        this.enhancedLabels = true;
      }
      
      if (window.matchMedia('(prefers-contrast: more)').matches) {
        // Если пользователь предпочитает повышенный контраст, усиливаем еще больше
        this.thickLines = 'thick';
        this.enhancedLabels = true;
      }
      
      if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
        // Отключаем анимации при предпочтении уменьшенного движения
        document.body.classList.add('reduced-motion');
      }
    }
    
    // Применяем настройки
    this.updateAccessibility();
  }
  
  /**
   * Обновляет цвета узлов на основе выбранной схемы
   * @param {Object} network - Объект сети

  updateNodeColors(network) {
    if (!network) return;
    
    // Обновляем цвета узлов
    d3.selectAll('.node circle').each(function(d) {
      const node = d3.select(this);
      const group = d.group;
      
      // Получаем цвет узла на основе текущей цветовой схемы
      const fillVarName = `--${group}-fill`;
      const strokeVarName = `--${group}-stroke`;
      
      const fillColor = getComputedStyle(document.documentElement).getPropertyValue(fillVarName).trim();
      const strokeColor = getComputedStyle(document.documentElement).getPropertyValue(strokeVarName).trim();
      
      // Применяем цвета
      if (!document.body.classList.contains('use-patterns')) {
        node.style('fill', fillColor || null);
      }
      node.style('stroke', strokeColor || null);
    });
    
    // Обновляем цвета связей
    d3.selectAll('.links line').each(function(d) {
      const link = d3.select(this);
      const type = d.type;
      
      // Для разных типов связей разные цвета
      let color;
      switch (type) {
        case 'acquaintance':
          color = '#4CAF50';
          break;
        case 'false':
          color = '#F44336';
          break;
        case 'indirect':
          color = '#9E9E9E';
          break;
        default:
          color = '#333333';
      }
      
      // Для монохромного режима используем оттенки серого
      if (document.body.classList.contains('color-monochromacy')) {
        if (type === 'mutual' || type === 'polemic') {
          color = '#000000';
        } else {
          color = '#555555';
        }
      }
      
      // Применяем цвета
      link.style('stroke', color);
    });
  }
}

function updateLegendForAccessibility() {

    }
    
    if (groupClass) {

      const fillVarName = `--${groupClass}-fill`;
      const strokeVarName = `--${groupClass}-stroke`;
      
      const fillColor = getComputedStyle(document.documentElement).getPropertyValue(fillVarName).trim();
      const strokeColor = getComputedStyle(document.documentElement).getPropertyValue(strokeVarName).trim();

      el.style.backgroundColor = fillColor || null;
      el.style.borderColor = strokeColor || null;

  });

  document.querySelectorAll('.legend-item svg line').forEach(el => {
    const classes = el.classList;

    } else {
      el.style.stroke = document.body.classList.contains('color-monochromacy') ? '#000000' : '#333333';

      el.style.strokeWidth = parseInt(el.getAttribute('stroke-width')) + 1 + 'px';

    }
  });

  const colorManager = new ColorAccessibilityManager();
  
  // Сохраняем ссылку на менеджер в объекте сети
  network.colorManager = colorManager;
  
  // Обновляем цвета узлов при смене схемы
  const observer = new MutationObserver(mutations => {
    // Проверяем изменения классов тела документа
    const classChanges = mutations.filter(m => 
      m.type === 'attributes' && 
      m.attributeName === 'class' && 
      m.target === document.body
    );
    
    if (classChanges.length > 0) {
      // Обновляем цвета узлов и легенду
      colorManager.updateNodeColors(network);
      updateLegendForAccessibility();
    }
  });
  
  // Начинаем наблюдение за изменениями классов тела документа
  observer.observe(document.body, { attributes: true });
  
  // Выполняем начальное обновление цветов
  colorManager.updateNodeColors(network);
  updateLegendForAccessibility();
  
  // Обновляем цвета при изменении данных
  network.simulation.on('end', () => {
    colorManager.updateNodeColors(network);
    updateLegendForAccessibility();
  });
  
  return colorManager;
}
```

## 6. Добавление метрики контрастности и анализатора доступности цветов

Добавьте следующий код в конец файла `<script>`:

```javascript
/**
 * Класс для анализа доступности цветов и контрастности
 */
class ColorAccessibilityAnalyzer {
  /**
   * Инициализирует анализатор доступности цветов
   */
  constructor() {
    this.contrastThreshold = 4.5; // Минимальный контраст для текстов по WCAG
    this.smallContrastThreshold = 3; // Минимальный контраст для больших текстов
    this.distinguishableThreshold = 0.27; // Порог различимости цветов
  }
  
  /**
   * Анализирует контраст между двумя цветами
   * @param {string} color1 - Первый цвет в формате HEX или RGB
   * @param {string} color2 - Второй цвет в формате HEX или RGB
   * @returns {Object} - Результат анализа контрастности
   */
  analyzeContrast(color1, color2) {
    // Преобразуем цвета в RGB
    const rgb1 = this.parseColor(color1);
    const rgb2 = this.parseColor(color2);
    
    if (!rgb1 || !rgb2) {
      return { contrast: 0, passes: false, message: 'Недопустимый формат цвета' };
    }
    
    // Вычисляем относительную яркость
    const luminance1 = this.calculateLuminance(rgb1);
    const luminance2 = this.calculateLuminance(rgb2);
    
    // Вычисляем контраст
    const contrast = this.calculateContrast(luminance1, luminance2);
    
    // Определяем, соответствует ли контраст требованиям WCAG
    const passesAAText = contrast >= this.contrastThreshold;
    const passesAALargeText = contrast >= this.smallContrastThreshold;
    const passesAAAText = contrast >= 7;
    const passesAAALargeText = contrast >= 4.5;
    
    return {
      contrast: contrast.toFixed(2),
      passesAAText,
      passesAALargeText,
      passesAAAText,
      passesAAALargeText,
      message: this.getContrastMessage(contrast)
    };
  }
  
  /**
   * Анализирует различимость цветов для разных типов дальтонизма
   * @param {string} color1 - Первый цвет в формате HEX или RGB
   * @param {string} color2 - Второй цвет в формате HEX или RGB
   * @returns {Object} - Результат анализа различимости
   */
  analyzeColorDistinguishability(color1, color2) {
    // Преобразуем цвета в RGB
    const rgb1 = this.parseColor(color1);
    const rgb2 = this.parseColor(color2);
    
    if (!rgb1 || !rgb2) {
      return { distinguishable: false, message: 'Недопустимый формат цвета' };
    }
    
    // Анализируем различимость для разных типов дальтонизма
    const normalDiff = this.calculateColorDifference(rgb1, rgb2);
    const protanDiff = this.calculateColorDifferenceProtanopia(rgb1, rgb2);
    const deuteranDiff = this.calculateColorDifferenceDeuteranopia(rgb1, rgb2);
    const tritanDiff = this.calculateColorDifferenceTritanopia(rgb1, rgb2);
    const monochromeDiff = this.calculateColorDifferenceMonochromacy(rgb1, rgb2);
    
    // Определяем проблемы с различимостью
    const issues = [];
    
    if (normalDiff < this.distinguishableThreshold) {
      issues.push('Недостаточная различимость цветов для обычного зрения');
    }
    
    if (protanDiff < this.distinguishableThreshold) {
      issues.push('Недостаточная различимость для протанопии (красный дальтонизм)');
    }
    
    if (deuteranDiff < this.distinguishableThreshold) {
      issues.push('Недостаточная различимость для дейтеранопии (зеленый дальтонизм)');
    }
    
    if (tritanDiff < this.distinguishableThreshold) {
      issues.push('Недостаточная различимость для тританопии (синий дальтонизм)');
    }
    
    if (monochromeDiff < this.distinguishableThreshold) {
      issues.push('Недостаточная различимость для монохромного зрения');
    }
    
    return {
      distinguishable: issues.length === 0,
      normalDiff: normalDiff.toFixed(2),
      protanDiff: protanDiff.toFixed(2),
      deuteranDiff: deuteranDiff.toFixed(2),
      tritanDiff: tritanDiff.toFixed(2),
      monochromeDiff: monochromeDiff.toFixed(2),
      issues
    };
  }
  
  /**
   * Анализирует доступность цветовой схемы
   * @param {Object} data - Данные сети
   * @returns {Object} - Результаты анализа
   */
  analyzeColorScheme(data) {
    const issues = [];
    const nodeGroups = new Set();
    
    // Собираем все группы узлов
    data.nodes.forEach(node => {
      nodeGroups.add(node.group);
    });
    
    // Анализируем контраст для каждой группы
    nodeGroups.forEach(group => {
      const fillVarName = `--${group}-fill`;
      const strokeVarName = `--${group}-stroke`;
      
      const fillColor = getComputedStyle(document.documentElement).getPropertyValue(fillVarName).trim();
      const strokeColor = getComputedStyle(document.documentElement).getPropertyValue(strokeVarName).trim();
      
      // Проверяем контраст между заливкой и обводкой
      const contrast = this.analyzeContrast(fillColor, strokeColor);
      
      if (!contrast.passesAALargeText) {
        issues.push(`Недостаточный контраст для группы ${group}: ${contrast.contrast}`);
      }
    });
    
    // Анализируем различимость между группами
    const groups = Array.from(nodeGroups);
    for (let i = 0; i < groups.length; i++) {
      for (let j = i + 1; j < groups.length; j++) {
        const group1 = groups[i];
        const group2 = groups[j];
        
        const fill1 = getComputedStyle(document.documentElement).getPropertyValue(`--${group1}-fill`).trim();
        const fill2 = getComputedStyle(document.documentElement).getPropertyValue(`--${group2}-fill`).trim();
        
        const distinguishability = this.analyzeColorDistinguishability(fill1, fill2);
        
        if (!distinguishability.distinguishable) {
          issues.push(`Недостаточная различимость между группами ${group1} и ${group2}`);
        }
      }
    }
    
    return {
      issues,
      hasIssues: issues.length > 0,
      recommendations: this.getRecommendations(issues)
    };
  }
  
  /**
   * Возвращает рекомендации по улучшению доступности на основе проблем
   * @param {Array} issues - Список проблем
   * @returns {Array} - Список рекомендаций
   */
  getRecommendations(issues) {
    const recommendations = [];
    
    if (issues.some(issue => issue.includes('контраст'))) {
      recommendations.push('Рекомендуется использовать режим высокой контрастности');
    }
    
    if (issues.some(issue => issue.includes('различимость'))) {
      recommendations.push('Рекомендуется использовать текстурные паттерны для лучшей различимости групп');
    }
    
    if (issues.length > 0) {
      recommendations.push('Рекомендуется увеличить размер узлов и толщину линий для улучшения видимости');
    }
    
    return recommendations;
  }
  
  /**
   * Возвращает сообщение о контрастности
   * @param {number} contrast - Уровень контраста
   * @returns {string} - Текстовое сообщение
   */
  getContrastMessage(contrast) {
    if (contrast < this.smallContrastThreshold) {
      return 'Недостаточный контраст';
    } else if (contrast < this.contrastThreshold) {
      return 'Минимальный контраст для крупного текста';
    } else if (contrast < 7) {
      return 'Хороший контраст (AA)';
    } else {
      return 'Отличный контраст (AAA)';
    }
  }
  
  /**
   * Преобразует цвет в RGB
   * @param {string} color - Цвет в формате HEX или RGB
   * @returns {Object|null} - RGB-компоненты или null
   */
  parseColor(color) {
    // Для HEX цветов (#RRGGBB или #RGB)
    if (color.startsWith('#')) {
      if (color.length === 4) {
        // #RGB формат
        const r = parseInt(color[1] + color[1], 16);
        const g = parseInt(color[2] + color[2], 16);
        const b = parseInt(color[3] + color[3], 16);
        return { r, g, b };
      } else if (color.length === 7) {
        // #RRGGBB формат
        const r = parseInt(color.substring(1, 3), 16);
        const g = parseInt(color.substring(3, 5), 16);
        const b = parseInt(color.substring(5, 7), 16);
        return { r, g, b };
      }
    }
    
    // Для RGB цветов (rgb(r, g, b))
    const rgbMatch = color.match(/rgb\(\s*(\d+)\s*,\s*(\d+)\s*,\s*(\d+)\s*\)/i);
    if (rgbMatch) {
      const r = parseInt(rgbMatch[1], 10);
      const g = parseInt(rgbMatch[2], 10);
      const b = parseInt(rgbMatch[3], 10);
      return { r, g, b };
    }
    
    return null;
  }
  
  /**
   * Вычисляет относительную яркость цвета (по формуле WCAG)
   * @param {Object} rgb - RGB-компоненты цвета
   * @returns {number} - Относительная яркость
   */
  calculateLuminance(rgb) {
    const { r, g, b } = rgb;
    
    // Нормализуем RGB-компоненты
    const rNorm = r / 255;
    const gNorm = g / 255;
    const bNorm = b / 255;
    
    // Преобразуем в линейные значения
    const rLinear = rNorm <= 0.03928 ? rNorm / 12.92 : Math.pow((rNorm + 0.055) / 1.055, 2.4);
    const gLinear = gNorm <= 0.03928 ? gNorm / 12.92 : Math.pow((gNorm + 0.055) / 1.055, 2.4);
    const bLinear = bNorm <= 0.03928 ? bNorm / 12.92 : Math.pow((bNorm + 0.055) / 1.055, 2.4);
    
    // Вычисляем яркость
    return 0.2126 * rLinear + 0.7152 * gLinear + 0.0722 * bLinear;
  }
  
  /**
   * Вычисляет контраст между двумя значениями яркости
   * @param {number} l1 - Первая яркость
   * @param {number} l2 - Вторая яркость
   * @returns {number} - Уровень контраста
   */
  calculateContrast(l1, l2) {
    // Определяем светлый и темный цвета
    const lighter = Math.max(l1, l2);
    const darker = Math.min(l1, l2);
    
    // Вычисляем контраст
    return (lighter + 0.05) / (darker + 0.05);
  }
  
  /**
   * Вычисляет разницу между цветами для обычного зрения
   * @param {Object} rgb1 - Первый цвет
   * @param {Object} rgb2 - Второй цвет
   * @returns {number} - Нормализованная разница (0-1)
   */
  calculateColorDifference(rgb1, rgb2) {
    const { r: r1, g: g1, b: b1 } = rgb1;
    const { r: r2, g: g2, b: b2 } = rgb2;
    
    // Вычисляем евклидово расстояние в RGB-пространстве
    const rDiff = Math.abs(r1 - r2) / 255;
    const gDiff = Math.abs(g1 - g2) / 255;
    const bDiff = Math.abs(b1 - b2) / 255;
    
    // Нормализуем на [0, 1]
    return Math.sqrt(rDiff * rDiff + gDiff * gDiff + bDiff * bDiff) / Math.sqrt(3);
  }
  
  /**
   * Симуляция протанопии (красный дальтонизм)
   * @param {Object} rgb - RGB-компоненты цвета
   * @returns {Object} - Симулированные RGB-компоненты
   */
  simulateProtanopia(rgb) {
    const { r, g, b } = rgb;
    
    // Симуляция протанопии (упрощенная модель)
    const rp = 0.0 * r + 0.9 * g + 0.1 * b;
    const gp = 0.0 * r + 1.0 * g + 0.0 * b;
    const bp = 0.0 * r + 0.0 * g + 1.0 * b;
    
    return { r: rp, g: gp, b: bp };
  }
  
  /**
   * Симуляция дейтеранопии (зеленый дальтонизм)
   * @param {Object} rgb - RGB-компоненты цвета
   * @returns {Object} - Симулированные RGB-компоненты
   */
  simulateDeuteranopia(rgb) {
    const { r, g, b } = rgb;
    
    // Симуляция дейтеранопии (упрощенная модель)
    const rd = 1.0 * r + 0.0 * g + 0.0 * b;
    const gd = 0.7 * r + 0.3 * g + 0.0 * b;
    const bd = 0.0 * r + 0.0 * g + 1.0 * b;
    
    return { r: rd, g: gd, b: bd };
  }
  
  /**
   * Симуляция тританопии (синий дальтонизм)
   * @param {Object} rgb - RGB-компоненты цвета
   * @returns {Object} - Симулированные RGB-компоненты
   */
  simulateTritanopia(rgb) {
    const { r, g, b } = rgb;
    
    // Симуляция тританопии (упрощенная модель)
    const rt = 1.0 * r + 0.0 * g + 0.0 * b;
    const gt = 0.0 * r + 1.0 * g + 0.0 * b;
    const bt = 0.1 * r + 0.9 * g + 0.0 * b;
    
    return { r: rt, g: gt, b: bt };
  }
  
  /**
   * Преобразование в оттенки серого
   * @param {Object} rgb - RGB-компоненты цвета
   * @returns {Object} - Симулированные RGB-компоненты
   */
  simulateMonochromacy(rgb) {
    const { r, g, b } = rgb;
    
    // Преобразование в оттенки серого
    const gray = 0.299 * r + 0.587 * g + 0.114 * b;
    
    return { r: gray, g: gray, b: gray };
  }
  
  /**
   * Вычисляет разницу между цветами для протанопии
   */
  calculateColorDifferenceProtanopia(rgb1, rgb2) {
    const p1 = this.simulateProtanopia(rgb1);
    const p2 = this.simulateProtanopia(rgb2);
    
    return this.calculateColorDifference(p1, p2);
  }
  
  /**
   * Вычисляет разницу между цветами для дейтеранопии
   */
  calculateColorDifferenceDeuteranopia(rgb1, rgb2) {
    const d1 = this.simulateDeuteranopia(rgb1);
    const d2 = this.simulateDeuteranopia(rgb2);
    
    return this.calculateColorDifference(d1, d2);
  }
  
  /**
   * Вычисляет разницу между цветами для тританопии
   */
  calculateColorDifferenceTritanopia(rgb1, rgb2) {
    const t1 = this.simulateTritanopia(rgb1);
    const t2 = this.simulateTritanopia(rgb2);
    
    return this.calculateColorDifference(t1, t2);
  }
  
  /**
   * Вычисляет разницу между цветами для монохромного зрения
   */
  calculateColorDifferenceMonochromacy(rgb1, rgb2) {
    const m1 = this.simulateMonochromacy(rgb1);
    const m2 = this.simulateMonochromacy(rgb2);
    
    return this.calculateColorDifference(m1, m2);
  }
}

/**
 * Инициализирует анализатор доступности цветов
 * @param {Object} network - Объект сети
 * @param {Object} data - Данные сети
 */
function initColorAccessibilityAnalyzer(network, data) {
  const analyzer = new ColorAccessibilityAnalyzer();
  
  // Анализируем текущую цветовую схему
  const analysis = analyzer.analyzeColorScheme(data);
  
  // Если есть проблемы с доступностью, показываем рекомендации
  if (analysis.hasIssues && analysis.recommendations.length > 0) {
    // Показываем уведомление с рекомендациями
    setTimeout(() => {
      // Проверяем, не включена ли уже одна из рекомендованных опций
      const currentAccessibility = {
        patterns: document.body.classList.contains('use-patterns'),
        highContrast: document.body.classList.contains('color-protanopia') || 
                      document.body.classList.contains('color-deuteranopia') || 
                      document.body.classList.contains('color-tritanopia') || 
                      document.body.classList.contains('color-monochromacy'),
        largeNodes: document.body.classList.contains('node-size-large') || 
                   document.body.classList.contains('node-size-larger'),
        thickLines: document.body.classList.contains('line-width-medium') || 
                   document.body.classList.contains('line-width-thick')
      };
      
      if (!currentAccessibility.patterns && !currentAccessibility.highContrast &&
          !currentAccessibility.largeNodes && !currentAccessibility.thickLines) {
        if (window.announce) {
          window.announce('Обнаружены возможные проблемы с доступностью цветов. Рекомендуется использовать настройки доступности.');
        }
        
        // Автоматически открываем панель доступности
        const toggleButton = document.getElementById('toggle-accessibility-toolbar');
        const controlsPanel = document.querySelector('.accessibility-controls');
        
        if (toggleButton && controlsPanel) {
          controlsPanel.style.display = 'flex';
          toggleButton.setAttribute('aria-expanded', 'true');
        }
      }
    }, 2000);
  }
  
  // Сохраняем анализатор в объекте сети
  network.colorAnalyzer = analyzer;
  
  return analyzer;
}
```
