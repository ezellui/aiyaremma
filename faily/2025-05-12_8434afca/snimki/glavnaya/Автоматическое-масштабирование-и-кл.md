# Автоматическое масштабирование и кластеризация

Реализация автоматического масштабирования и кластеризации позволит эффективно работать с очень большими сетями, объединяя узлы в метаузлы при масштабировании.

## 4. Реализация автоматического масштабирования и кластеризации

Добавьте следующий код в конец файла `<script>`:

```javascript
/**
 * Класс для управления автоматической кластеризацией и масштабированием
 */
class AutoClusteringManager {
  /**
   * Создает менеджер кластеризации
   * @param {Object} network - Объект сети с узлами и связями
   * @param {Object} data - Исходные данные сети
   */
  constructor(network, data) {
    this.network = network;
    this.svg = network.svg;
    this.g = network.g;
    this.data = data;
    this.nodes = data.nodes;
    this.links = data.links;
    
    // Параметры кластеризации
    this.enabled = true;
    this.mode = 'smart'; // smart, aggressive, minimal, off
    
    // Подготовка переменных
    this.clusterGroups = new Map(); // group -> nodes
    this.clusters = [];  // Список активных кластеров
    this.originalNodeVisibility = new Map(); // node -> visibility
    this.clusterLayer = null;
    this.currentZoomLevel = 1;
    
    // Инициализация
    this.prepareClusterGroups();
    this.createClusterLayer();
    this.setupEventListeners();
    this.setupZoomIndicator();
    
    // Флаг первоначальной инициализации
    this.initialized = false;
    
    // Инициализация советника по кластеризации
    this.clusteringAdvisor = new ClusteringAdvisor(this.nodes, this.links);
    
    // Классы для хранения состояния кластеризации
    this.viewportNodes = new Set(this.nodes.map(n => n.id));
    this.activeClusterIds = new Set();
  }
  
  /**
   * Подготавливает группы кластеров на основе групп философов
   */
  prepareClusterGroups() {
    // Группируем узлы по их типам/группам
    this.nodes.forEach(node => {
      if (!this.clusterGroups.has(node.group)) {
        this.clusterGroups.set(node.group, []);
      }
      this.clusterGroups.get(node.group).push(node);
    });
    
    // Создаем начальные данные кластеров
    this.clusterGroups.forEach((nodes, group) => {
      // Вычисляем центр кластера как среднее положение узлов
      const center = this.calculateClusterCenter(nodes);
      
      // Создаем кластер
      const cluster = {
        id: `cluster-${group}`,
        group: group,
        name: getGroupDisplayName(group),
        x: center.x,
        y: center.y,
        nodes: nodes,
        radius: Math.sqrt(nodes.length) * 15, // Радиус зависит от количества узлов
        nodeIds: new Set(nodes.map(n => n.id))
      };
      
      this.clusters.push(cluster);
    });
  }
  
  /**
   * Вычисляет центр кластера по его узлам
   * @param {Array} nodes - Массив узлов кластера
   * @returns {Object} - Координаты центра {x, y}
   */
  calculateClusterCenter(nodes) {
    if (nodes.length === 0) return { x: 0, y: 0 };
    
    let sumX = 0, sumY = 0;
    let validNodes = 0;
    
    nodes.forEach(node => {
      if (node.x !== undefined && node.y !== undefined) {
        sumX += node.x;
        sumY += node.y;
        validNodes++;
      }
    });
    
    if (validNodes === 0) return { x: 0, y: 0 };
    
    return {
      x: sumX / validNodes,
      y: sumY / validNodes
    };
  }
  
  /**
   * Создает слой для отображения кластеров
   */
  createClusterLayer() {
    // Создаем группу для кластеров
    this.clusterLayer = this.g.append("g")
      .attr("class", "cluster-layer")
      .style("opacity", 0); // Изначально скрыта
  }
  
  /**
   * Настраивает слушатели событий
   */
  setupEventListeners() {
    // Обработчик событий масштабирования
    const manager = this;
    
    // Переопределяем обработчик события масштабирования
    this.svg.on("zoom", null); // Удаляем старый обработчик
    this.svg.call(d3.zoom()
      .scaleExtent([0.1, 5])
      .on("zoom", function(event) {
        // Обновляем масштабирование
        manager.g.attr("transform", event.transform);
        
        // Обновляем текущий уровень масштабирования
        manager.currentZoomLevel = event.transform.k;
        
        // Обновляем кластеризацию на основе уровня масштабирования
        manager.updateClustering();
        
        // Обновляем индикатор масштабирования
        manager.updateZoomIndicator();
      }));
    
    // Инициализируем переключатели и селекторы
    const autoClusteringCheckbox = document.getElementById("auto-clustering");
    const clusteringLevelSelect = document.getElementById("clustering-level");
    
    // Устанавливаем начальные значения
    if (autoClusteringCheckbox) {
      autoClusteringCheckbox.checked = this.enabled;
      autoClusteringCheckbox.addEventListener("change", function() {
        manager.enabled = this.checked;
        manager.updateClustering();
      });
    }
    
    if (clusteringLevelSelect) {
      clusteringLevelSelect.value = this.mode;
      clusteringLevelSelect.addEventListener("change", function() {
        manager.mode = this.value;
        manager.updateClustering();
      });
    }
  }
  
  /**
   * Настраивает индикатор уровня масштабирования
   */
  setupZoomIndicator() {
    this.zoomIndicator = document.querySelector(".zoom-level-indicator");
    this.zoomLevelText = document.getElementById("zoom-level");
    this.clusteringStatus = document.getElementById("clustering-status");
    
    // Автоматически скрываем индикатор через несколько секунд
    let hideTimeout;
    
    const showIndicator = () => {
      clearTimeout(hideTimeout);
      this.zoomIndicator.classList.remove("auto-hidden");
      hideTimeout = setTimeout(() => {
        this.zoomIndicator.classList.add("auto-hidden");
      }, 3000);
    };
    
    // Показываем индикатор при наведении мыши на сетку
    this.svg.node().addEventListener("mousemove", showIndicator);
    
    // Показываем индикатор при масштабировании
    this.svg.on("zoom.indicator", showIndicator);
    
    // Начальное обновление индикатора
    this.updateZoomIndicator();
    showIndicator();
  }
  
  /**
   * Обновляет индикатор уровня масштабирования
   */
  updateZoomIndicator() {
    if (!this.zoomLevelText) return;
    
    // Обновляем текст масштаба
    const zoomPercent = Math.round(this.currentZoomLevel * 100);
    this.zoomLevelText.textContent = `${zoomPercent}%`;
    
    // Обновляем статус кластеризации
    if (this.clusteringStatus) {
      const activeCount = this.activeClusterIds.size;
      
      if (activeCount > 0 && this.enabled) {
        this.clusteringStatus.innerHTML = `<span class="clustering-badge">Кластеров: ${activeCount}</span>`;
      } else {
        this.clusteringStatus.innerHTML = '';
      }
    }
  }
  
  /**
   * Обновляет кластеризацию на основе текущего масштаба
   */
  updateClustering() {
    if (!this.enabled || this.mode === 'off') {
      // Если кластеризация отключена, показываем все узлы
      this.resetClusterView();
      return;
    }
    
    // Получаем пороги кластеризации в зависимости от режима
    const thresholds = this.getClusteringThresholds();
    
    // Проверяем, нужна ли кластеризация в зависимости от масштаба
    if (this.currentZoomLevel < thresholds.maxDetailedZoom) {
      // При малом масштабе активируем кластеризацию
      this.activateClustering();
    } else {
      // При большом масштабе показываем отдельные узлы
      this.resetClusterView();
    }
  }
  
  /**
   * Получает пороги кластеризации в зависимости от режима
   * @returns {Object} - Объект с порогами
   */
  getClusteringThresholds() {
    // Размер сети влияет на пороги
    const nodeCount = this.nodes.length;
    const baseThreshold = Math.min(1, Math.max(0.4, 1 - nodeCount / 500));
    
    // Разные режимы имеют разные пороги
    switch (this.mode) {
      case 'aggressive':
        return {
          maxDetailedZoom: baseThreshold + 0.5,
          minClusterNodeCount: 3
        };
      case 'minimal':
        return {
          maxDetailedZoom: baseThreshold - 0.2,
          minClusterNodeCount: 8
        };
      case 'smart':
      default:
        return {
          maxDetailedZoom: baseThreshold,
          minClusterNodeCount: 5
        };
    }
  }
  
  /**
   * Активирует кластеризацию
   */
  activateClustering() {
    // Определяем, какие узлы видны в текущем видимом окне
    const visibleNodes = this.getVisibleNodes();
    
    // Получаем рекомендации по кластеризации
    const clusteringRecommendation = this.clusteringAdvisor.getRecommendation(
      this.currentZoomLevel,
      visibleNodes.size,
      this.mode
    );
    
    // Если рекомендаций нет, сбрасываем вид
    if (!clusteringRecommendation.shouldCluster) {
      this.resetClusterView();
      return;
    }
    
    // Очищаем текущие кластеры
    this.clusterLayer.selectAll("*").remove();
    this.activeClusterIds.clear();
    
    // Показываем слой кластеров
    this.clusterLayer.style("opacity", 1);
    
    // Пороговое значение для кластеризации (минимальное количество узлов в кластере)
    const minClusterNodeCount = clusteringRecommendation.minClusterSize;
    
    // Обновляем кластеры
    this.clusters.forEach(cluster => {
      // Определяем, какие узлы кластера видны
      const visibleClusterNodes = [...cluster.nodeIds].filter(id => visibleNodes.has(id));
      
      // Применяем кластеризацию только если количество видимых узлов превышает порог
      if (visibleClusterNodes.length >= minClusterNodeCount) {
        this.renderCluster(cluster, visibleClusterNodes);
        
        // Скрываем узлы, входящие в кластер
        visibleClusterNodes.forEach(nodeId => {
          // Сохраняем оригинальную видимость, если еще не сохранена
          if (!this.originalNodeVisibility.has(nodeId)) {
            const nodeElement = d3.select(`.node[data-id="${nodeId}"]`);
            this.originalNodeVisibility.set(nodeId, nodeElement.style("opacity"));
          }
          
          // Скрываем узел
          const nodeElement = d3.select(`.node[data-id="${nodeId}"]`);
          nodeElement.classed("fading-out", true);
          setTimeout(() => {
            nodeElement.style("display", "none");
          }, 300);
        });
        
        this.activeClusterIds.add(cluster.id);
      } else {
        // Для маленьких кластеров показываем отдельные узлы
        visibleClusterNodes.forEach(nodeId => {
          const nodeElement = d3.select(`.node[data-id="${nodeId}"]`);
          nodeElement.style("display", "block");
          nodeElement.classed("fading-out", false);
          nodeElement.classed("fading-in", true);
          
          setTimeout(() => {
            nodeElement.classed("fading-in", false);
          }, 300);
        });
      }
    });
    
    // Если нет активных кластеров, сбрасываем вид
    if (this.activeClusterIds.size === 0) {
      this.resetClusterView();
    }
  }
  
  /**
   * Отрисовывает кластер на основе данных
   * @param {Object} cluster - Данные кластера
   * @param {Array} visibleNodeIds - Массив ID видимых узлов кластера
   */
  renderCluster(cluster, visibleNodeIds) {
    // Обновляем положение кластера на основе видимых узлов
    const visibleNodes = visibleNodeIds.map(id => this.nodes.find(n => n.id === id));
    const center = this.calculateClusterCenter(visibleNodes);
    
    // Обновляем координаты кластера
    cluster.x = center.x;
    cluster.y = center.y;
    
    // Создаем группу для кластера
    const clusterGroup = this.clusterLayer.append("g")
      .attr("class", "cluster")
      .attr("transform", `translate(${cluster.x},${cluster.y})`)
      .attr("data-cluster-id", cluster.id)
      .attr("data-group", cluster.group);
    
    // Добавляем обработчики событий для кластера
    clusterGroup.on("click", (event) => this.handleClusterClick(event, cluster))
      .on("mouseover", (event) => this.showClusterTooltip(event, cluster, visibleNodeIds))
      .on("mouseout", () => this.hideClusterTooltip());
    
    // Отрисовываем кружок кластера
    clusterGroup.append("circle")
      .attr("class", `cluster-node ${cluster.group}`)
      .attr("r", Math.sqrt(visibleNodeIds.length) * 6 + 10)
      .attr("fill", getNodeColor(cluster.group));
    
    // Добавляем метку с названием группы
    clusterGroup.append("text")
      .attr("class", "cluster-label")
      .attr("dy", -5)
      .text(cluster.name);
    
    // Добавляем метку с количеством узлов
    clusterGroup.append("text")
      .attr("class", "cluster-count")
      .attr("dy", 10)
      .text(visibleNodeIds.length);
  }
  
  /**
   * Возвращает множество ID узлов, видимых в текущем видовом окне
   * @returns {Set} - Множество ID видимых узлов
   */
  getVisibleNodes() {
    // Если нет инициализации, возвращаем все узлы
    if (!this.initialized) {
      this.initialized = true;
      return new Set(this.nodes.map(n => n.id));
    }
    
    const visibleNodes = new Set();
    
    // Получаем текущие размеры и преобразование видового окна
    const transform = d3.zoomTransform(this.svg.node());
    const width = this.svg.node().clientWidth;
    const height = this.svg.node().clientHeight;
    
    // Вычисляем границы видимой области
    const visibleX0 = -transform.x / transform.k;
    const visibleY0 = -transform.y / transform.k;
    const visibleX1 = visibleX0 + width / transform.k;
    const visibleY1 = visibleY0 + height / transform.k;
    
    // Получаем узлы в видимой области
    this.nodes.forEach(node => {
      if (node.x !== undefined && node.y !== undefined) {
        if (node.x >= visibleX0 && node.x <= visibleX1 &&
            node.y >= visibleY0 && node.y <= visibleY1) {
          visibleNodes.add(node.id);
        }
      }
    });
    
    return visibleNodes;
  }
  
  /**
   * Обработчик клика по кластеру
   * @param {Event} event - Событие клика
   * @param {Object} cluster - Данные кластера
   */
  handleClusterClick(event, cluster) {
    // Получаем координаты кластера
    const x = cluster.x;
    const y = cluster.y;
    
    // Увеличиваем масштаб до кластера
    const currentTransform = d3.zoomTransform(this.svg.node());
    
    // Определяем новый масштаб (увеличиваем достаточно, чтобы видеть отдельные узлы)
    const newScale = Math.max(currentTransform.k * 1.5, this.getClusteringThresholds().maxDetailedZoom + 0.1);
    
    // Создаем новое преобразование
    const width = this.svg.node().clientWidth;
    const height = this.svg.node().clientHeight;
    
    const newX = -x * newScale + width / 2;
    const newY = -y * newScale + height / 2;
    
    // Применяем новое преобразование с анимацией
    this.svg.transition()
      .duration(750)
      .call(this.network.zoom.transform, d3.zoomIdentity.translate(newX, newY).scale(newScale));
    
    // Предотвращаем всплытие события
    event.stopPropagation();
  }
  
  /**
   * Показывает всплывающую подсказку для кластера
   * @param {Event} event - Событие наведения
   * @param {Object} cluster - Данные кластера
   * @param {Array} visibleNodeIds - Массив ID видимых узлов кластера
   */
  showClusterTooltip(event, cluster, visibleNodeIds) {
    // Проверяем существование тултипа
    let tooltip = document.querySelector(".auto-cluster-tooltip");
    
    if (!tooltip) {
      tooltip = document.createElement("div");
      tooltip.className = "auto-cluster-tooltip";
      document.body.appendChild(tooltip);
    }
    
    // Получаем имена узлов в кластере
    const nodeNames = visibleNodeIds
      .map(id => this.nodes.find(n => n.id === id))
      .filter(node => node)
      .map(node => node.name);
    
    // Максимальное количество имен для отображения
    const maxNamesToShow = 7;
    let tooltipContent = `<strong>${cluster.name}</strong> (${visibleNodeIds.length} философов)`;
    
    // Если узлов немного, показываем их имена
    if (nodeNames.length <= maxNamesToShow) {
      tooltipContent += "<ul>";
      nodeNames.forEach(name => {
        tooltipContent += `<li>${name}</li>`;
      });
      tooltipContent += "</ul>";
    } else {
      // Иначе показываем только первые несколько
      tooltipContent += "<ul>";
      for (let i = 0; i < maxNamesToShow - 1; i++) {
        tooltipContent += `<li>${nodeNames[i]}</li>`;
      }
      tooltipContent += `<li>и ещё ${nodeNames.length - maxNamesToShow + 1} философов...</li>`;
      tooltipContent += "</ul>";
    }
    
    // Добавляем подсказку о действии
    tooltipContent += "<small><em>Нажмите для увеличения</em></small>";
    
    // Устанавливаем контент и позицию тултипа
    tooltip.innerHTML = tooltipContent;
    tooltip.style.display = "block";
    tooltip.style.left = (event.pageX + 10) + "px";
    tooltip.style.top = (event.pageY - 10) + "px";
  }
  
  /**
   * Скрывает всплывающую подсказку для кластера
   */
  hideClusterTooltip() {
    const tooltip = document.querySelector(".auto-cluster-tooltip");
    if (tooltip) {
      tooltip.style.display = "none";
    }
  }
  
  /**
   * Сбрасывает вид к отображению отдельных узлов (без кластеризации)
   */
  resetClusterView() {
    // Скрываем слой кластеров
    this.clusterLayer.style("opacity", 0);
    this.clusterLayer.selectAll("*").remove();
    this.activeClusterIds.clear();
    
    // Восстанавливаем видимость всех узлов
    this.originalNodeVisibility.forEach((originalOpacity, nodeId) => {
      const nodeElement = d3.select(`.node[data-id="${nodeId}"]`);
      nodeElement.style("display", "block");
      nodeElement.classed("fading-out", false);
      nodeElement.classed("fading-in", true);
      
      // Восстанавливаем оригинальную непрозрачность
      nodeElement.style("opacity", originalOpacity);
      
      setTimeout(() => {
        nodeElement.classed("fading-in", false);
      }, 300);
    });
    
    // Очищаем сохраненные состояния видимости
    this.originalNodeVisibility.clear();
  }
}

/**
 * Класс для определения оптимальных параметров кластеризации
 */
class ClusteringAdvisor {
  /**
   * Создает советника по кластеризации
   * @param {Array} nodes - Массив узлов сети
   * @param {Array} links - Массив связей сети
   */
  constructor(nodes, links) {
    this.nodes = nodes;
    this.links = links;
    
    // Пороговые значения для разных режимов
    this.thresholds = {
      // Режим, масштаб, видимые узлы, минимальный размер кластера
      aggressive: [
        { zoomLevel: 0.3, visibleNodes: 30, minClusterSize: 3 },
        { zoomLevel: 0.5, visibleNodes: 50, minClusterSize: 4 },
        { zoomLevel: 0.8, visibleNodes: 80, minClusterSize: 5 }
      ],
      smart: [
        { zoomLevel: 0.3, visibleNodes: 40, minClusterSize: 5 },
        { zoomLevel: 0.5, visibleNodes: 70, minClusterSize: 6 },
        { zoomLevel: 0.8, visibleNodes: 100, minClusterSize: 7 }
      ],
      minimal: [
        { zoomLevel: 0.2, visibleNodes: 50, minClusterSize: 8 },
        { zoomLevel: 0.4, visibleNodes: 80, minClusterSize: 10 },
        { zoomLevel: 0.6, visibleNodes: 120, minClusterSize: 12 }
      ]
    };
  }
  
  /**
   * Получает рекомендации по кластеризации
   * @param {number} zoomLevel - Текущий уровень масштабирования
   * @param {number} visibleNodeCount - Количество видимых узлов
   * @param {string} mode - Режим кластеризации
   * @returns {Object} - Рекомендации по кластеризации
   */
  getRecommendation(zoomLevel, visibleNodeCount, mode) {
    // Выбираем пороговые значения в зависимости от режима
    const thresholds = this.thresholds[mode] || this.thresholds.smart;
    
    // Ищем подходящие пороговые значения
    let recommendation = { shouldCluster: false, minClusterSize: 5 };
    
    for (const threshold of thresholds) {
      if (zoomLevel <= threshold.zoomLevel && visibleNodeCount >= threshold.visibleNodes) {
        recommendation = {
          shouldCluster: true,
          minClusterSize: threshold.minClusterSize
        };
        break;
      }
    }
    
    // Если мы в режиме "off", отключаем кластеризацию
    if (mode === "off") {
      recommendation.shouldCluster = false;
    }
    
    return recommendation;
  }
}

/**
 * Инициализирует систему автоматического масштабирования и кластеризации
 * @param {Object} network - Объект сети
 * @param {Object} data - Данные сети
 */
function initAutoClusteringSystem(network, data) {
  // Добавляем атрибуты data-id к узлам для быстрого поиска
  d3.selectAll(".node").attr("data-id", d => d.id);
  
  // Создаем менеджер кластеризации
  const clusteringManager = new AutoClusteringManager(network, data);
  
  // Сохраняем ссылку на менеджер кластеризации для доступа из других функций
  network.clusteringManager = clusteringManager;
  
  // Добавляем обработчик события изменения размера окна
  window.addEventListener("resize", () => {
    clusteringManager.updateClustering();
  });
  
  return clusteringManager;
}

// Добавляем вызов функции в конец функции integrateComponents
// Найдите функцию integrateComponents и добавьте в её конец:
function integrateComponents(graphData) {
  // Существующий код...
  
  // Добавьте в конец функции:
  initAutoClusteringSystem(network, graphData);
}
```

## 6. Динамическое распределение нагрузки при большом количестве данных

Добавьте следующую функцию в конец файла `<script>`:

```javascript
/**
 * Динамически оптимизирует визуализацию в зависимости от размера данных
 * @param {Object} network - Объект сети
 * @param {Object} data - Данные сети
 */
function optimizeForLargeDatasets(network, data) {
  const nodeCount = data.nodes.length;
  const linkCount = data.links.length;
  
  // Определяем размер датасета
  let datasetSize;
  if (nodeCount > 500 || linkCount > 1000) {
    datasetSize = 'very-large';
  } else if (nodeCount > 200 || linkCount > 500) {
    datasetSize = 'large';
  } else if (nodeCount > 100 || linkCount > 200) {
    datasetSize = 'medium';
  } else {
    datasetSize = 'small';
  }
  
  // Устанавливаем различные оптимизации в зависимости от размера
  switch (datasetSize) {
    case 'very-large':
      // Для очень больших сетей применяем максимальные оптимизации
      
      // 1. Уменьшаем детализацию отрисовки
      d3.selectAll(".links line").style("stroke-opacity", 0.5);
      
      // 2. Применяем WebGL принудительно, если доступен
      const useWebGLCheckbox = document.getElementById('use-webgl');
      if (useWebGLCheckbox && !useWebGLCheckbox.checked) {
        useWebGLCheckbox.checked = true;
        useWebGLCheckbox.dispatchEvent(new Event('change'));
      }
      
      // 3. Включаем агрессивную кластеризацию
      if (network.clusteringManager) {
        const clusteringSelect = document.getElementById('clustering-level');
        if (clusteringSelect && clusteringSelect.value !== 'aggressive') {
          clusteringSelect.value = 'aggressive';
          clusteringSelect.dispatchEvent(new Event('change'));
        }
      }
      
      // 4. Прячем метки для повышения производительности
      d3.selectAll(".node text").style("display", "none");
      
      // 5. Отображаем системное сообщение для пользователя
      showPerformanceWarning("Обнаружен очень большой набор данных. Включены автоматические оптимизации производительности.");
      break;
      
    case 'large':
      // Для больших сетей применяем умеренные оптимизации
      
      // 1. Умеренно уменьшаем детализацию
      d3.selectAll(".links line").style("stroke-opacity", 0.7);
      
      // 2. Рекомендуем WebGL, но не принуждаем
      const webGLCheckbox = document.getElementById('use-webgl');
      if (webGLCheckbox && !webGLCheckbox.checked) {
        showPerformanceWarning("Обнаружен большой набор данных. Рекомендуется включить режим WebGL для лучшей производительности.");
      }
      
      // 3. Включаем умную кластеризацию
      if (network.clusteringManager) {
        const clusteringModeSelect = document.getElementById('clustering-level');
        if (clusteringModeSelect && clusteringModeSelect.value !== 'smart') {
          clusteringModeSelect.value = 'smart';
          clusteringModeSelect.dispatchEvent(new Event('change'));
        }
      }
      
      // 4. Показываем метки только для важных узлов
      d3.selectAll(".node text").style("display", function(d) {
        // Показываем метки только для узлов с большим количеством связей
        const nodeLinks = network.link.filter(l => l.source.id === d.id || l.target.id === d.id);
        return nodeLinks.size() > 3 ? "block" : "none";
      });
      break;
      
    case 'medium':
      // Для средних сетей легкие оптимизации
      
      // 1. Включаем обычную кластеризацию
      if (network.clusteringManager) {
        const clusteringSelect = document.getElementById('clustering-level');
        if (clusteringSelect && clusteringSelect.value === 'off') {
          clusteringSelect.value = 'smart';
          clusteringSelect.dispatchEvent(new Event('change'));
        }
      }
      break;
      
    case 'small':
    default:
      // Для малых сетей оптимизации не требуются
      break;
  }
}

// Добавляем вызов функции в конец функции integrateComponents
// Найдите функцию integrateComponents и добавьте в её конец:
function integrateComponents(graphData) {
  // Существующий код...
  
  // Добавьте в конец функции:
  optimizeForLargeDatasets(network, graphData);
}
```

Эти изменения добавят продвинутую автоматическую кластеризацию, которая будет динамически адаптироваться к размеру сети и уровню масштабирования. При малом масштабе узлы будут объединяться в кластеры по группам, которые можно развернуть кликом. При увеличении масштаба будут показываться отдельные узлы. Также добавлена оптимизация для работы с большими сетями, включая динамическое распределение нагрузки.
