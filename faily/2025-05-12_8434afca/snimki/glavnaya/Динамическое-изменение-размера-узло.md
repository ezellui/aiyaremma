# Динамическое изменение размера узлов в зависимости от значимости

## 4. Реализация класса управления размерами узлов

Добавьте следующий JavaScript-код в конец файла `<script>`:

```javascript
/**
 * Класс для управления динамическим изменением размера узлов
 */
class NodeSizeManager {
  /**
   * Создает менеджер размеров узлов
   * @param {Object} network - Объект сети
   * @param {Object} data - Данные сети
   */
  constructor(network, data) {
    this.network = network;
    this.data = data;
    this.nodes = data.nodes;
    this.links = data.links;
    this.storage = window.localStorage;
    
    // Настройки размеров узлов
    this.sizeMode = this.storage.getItem('rpw_node_size_mode') || 'uniform';
    this.sizeScale = this.storage.getItem('rpw_node_size_scale') || 'linear';
    
    // Метрики важности узлов
    this.metrics = {};
    
    // Инициализация
    this.init();
  }
  
  /**
   * Инициализирует менеджер размеров узлов
   */
  init() {
    // Инициализируем элементы управления
    this.initControls();
    
    // Вычисляем метрики важности
    this.calculateImportanceMetrics();
    
    // Применяем начальные размеры
    this.applyNodeSizes();
    
    // Добавляем всплывающие подсказки с информацией о важности
    this.addImportanceTooltips();
  }
  
  /**
   * Инициализирует элементы управления размерами узлов
   */
  initControls() {
    // Находим элементы управления
    this.toggleButton = document.getElementById('node-size-toggle');
    this.controlsPanel = document.getElementById('node-size-controls');
    this.sizeOptions = document.querySelectorAll('input[name="node-size-mode"]');
    this.scaleSelect = document.getElementById('size-scale');
    this.legendElement = document.getElementById('node-importance-legend');
    this.legendDesc = document.getElementById('node-importance-legend-desc');
    
    // Устанавливаем начальные значения
    for (const option of this.sizeOptions) {
      if (option.value === this.sizeMode) {
        option.checked = true;
      }
    }
    
    if (this.scaleSelect) {
      this.scaleSelect.value = this.sizeScale;
    }
    
    // Показываем легенду, если используется не uniform режим
    if (this.sizeMode !== 'uniform' && this.legendElement) {
      this.legendElement.style.display = 'block';
      this.updateLegendDescription();
    }
    
    // Добавляем обработчики событий
    
    // Переключение панели настроек
    if (this.toggleButton) {
      this.toggleButton.addEventListener('click', () => {
        this.toggleControlsPanel();
      });
    }
    
    // Изменение режима размеров
    for (const option of this.sizeOptions) {
      option.addEventListener('change', () => {
        this.sizeMode = option.value;
        this.storage.setItem('rpw_node_size_mode', this.sizeMode);
        this.applyNodeSizes();
        this.updateLegendVisibility();
      });
    }
    
    // Изменение масштаба размеров
    if (this.scaleSelect) {
      this.scaleSelect.addEventListener('change', () => {
        this.sizeScale = this.scaleSelect.value;
        this.storage.setItem('rpw_node_size_scale', this.sizeScale);
        this.applyNodeSizes();
      });
    }
  }
  
  /**
   * Переключает отображение панели настроек
   */
  toggleControlsPanel() {
    if (!this.controlsPanel) return;
    
    const isVisible = this.controlsPanel.classList.contains('visible');
    
    if (isVisible) {
      this.controlsPanel.classList.remove('visible');
    } else {
      this.controlsPanel.classList.add('visible');
    }
  }
  
  /**
   * Обновляет видимость легенды размеров узлов
   */
  updateLegendVisibility() {
    if (!this.legendElement) return;
    
    if (this.sizeMode === 'uniform') {
      this.legendElement.style.display = 'none';
    } else {
      this.legendElement.style.display = 'block';
      this.updateLegendDescription();
    }
  }
  
  /**
   * Обновляет описание в легенде
   */
  updateLegendDescription() {
    if (!this.legendDesc) return;
    
    switch (this.sizeMode) {
      case 'connections':
        this.legendDesc.textContent = 'Количество связей';
        break;
      case 'influence':
        this.legendDesc.textContent = 'Влиятельность философа';
        break;
      case 'centrality':
        this.legendDesc.textContent = 'Центральность в сети';
        break;
      default:
        this.legendDesc.textContent = '';
    }
  }
  
  /**
   * Вычисляет метрики важности узлов
   */
  calculateImportanceMetrics() {
    // Метрики по количеству связей
    this.calculateConnectionsMetric();
    
    // Метрика влиятельности
    this.calculateInfluenceMetric();
    
    // Метрика центральности
    this.calculateCentralityMetric();
  }
  
  /**
   * Вычисляет метрику по количеству связей
   */
  calculateConnectionsMetric() {
    const connectionsCount = {};
    
    // Подсчитываем количество связей для каждого узла
    this.links.forEach(link => {
      const sourceId = typeof link.source === 'object' ? link.source.id : link.source;
      const targetId = typeof link.target === 'object' ? link.target.id : link.target;
      
      if (!connectionsCount[sourceId]) connectionsCount[sourceId] = 0;
      if (!connectionsCount[targetId]) connectionsCount[targetId] = 0;
      
      connectionsCount[sourceId]++;
      connectionsCount[targetId]++;
    });
    
    // Нормализуем значения от 1 до 5
    const counts = Object.values(connectionsCount);
    const maxCount = Math.max(...counts);
    const minCount = Math.min(...counts);
    
    const connections = {};
    for (const [nodeId, count] of Object.entries(connectionsCount)) {
      // Нормализация от 1 до 5
      const normalized = this.normalizeValue(count, minCount, maxCount, 1, 5);
      connections[nodeId] = {
        raw: count,
        normalized: Math.round(normalized),
        max: maxCount
      };
    }
    
    this.metrics.connections = connections;
  }
  
  /**
   * Вычисляет метрику влиятельности
   */
  calculateInfluenceMetric() {
    const influenceScores = {};
    
    // Инициализируем начальные значения
    this.nodes.forEach(node => {
      influenceScores[node.id] = 0;
    });
    
    // Подсчитываем количество исходящих влияний
    this.links.forEach(link => {
      const sourceId = typeof link.source === 'object' ? link.source.id : link.source;
      const targetId = typeof link.target === 'object' ? link.target.id : link.target;
      
      // Увеличиваем счет влиятельности для источника связи
      if (link.type === 'influence' || link.type === 'mutual') {
        if (!influenceScores[sourceId]) influenceScores[sourceId] = 0;
        influenceScores[sourceId] += 1;
      }
      
      // Для взаимного влияния увеличиваем счет для цели
      if (link.type === 'mutual') {
        if (!influenceScores[targetId]) influenceScores[targetId] = 0;
        influenceScores[targetId] += 0.5; // Меньший вес для обратного влияния
      }
    });
    
    // Нормализуем значения от 1 до 5
    const scores = Object.values(influenceScores);
    const maxScore = Math.max(...scores);
    const minScore = Math.min(...scores);
    
    const influence = {};
    for (const [nodeId, score] of Object.entries(influenceScores)) {
      // Нормализация от 1 до 5
      const normalized = this.normalizeValue(score, minScore, maxScore, 1, 5);
      influence[nodeId] = {
        raw: score,
        normalized: Math.round(normalized),
        max: maxScore
      };
    }
    
    this.metrics.influence = influence;
  }
  
  /**
   * Вычисляет метрику центральности
   */
  calculateCentralityMetric() {
    // Создаем граф для расчета центральности
    const graph = {};
    
    // Инициализируем граф
    this.nodes.forEach(node => {
      graph[node.id] = [];
    });
    
    // Добавляем связи
    this.links.forEach(link => {
      const sourceId = typeof link.source === 'object' ? link.source.id : link.source;
      const targetId = typeof link.target === 'object' ? link.target.id : link.target;
      
      if (!graph[sourceId]) graph[sourceId] = [];
      if (!graph[targetId]) graph[targetId] = [];
      
      graph[sourceId].push(targetId);
      graph[targetId].push(sourceId);
    });
    
    // Вычисляем центральность по близости
    const centralityScores = this.calculateClosenessCentrality(graph);
    
    // Нормализуем значения от 1 до 5
    const scores = Object.values(centralityScores);
    const maxScore = Math.max(...scores);
    const minScore = Math.min(...scores);
    
    const centrality = {};
    for (const [nodeId, score] of Object.entries(centralityScores)) {
      // Нормализация от 1 до 5
      const normalized = this.normalizeValue(score, minScore, maxScore, 1, 5);
      centrality[nodeId] = {
        raw: score,
        normalized: Math.round(normalized),
        max: maxScore
      };
    }
    
    this.metrics.centrality = centrality;
  }
  
  /**
   * Вычисляет центральность по близости для всех узлов
   * @param {Object} graph - Граф в виде списка смежности
   * @returns {Object} - Значения центральности для каждого узла
   */
  calculateClosenessCentrality(graph) {
    const nodes = Object.keys(graph);
    const centrality = {};
    
    // Для каждого узла вычисляем сумму кратчайших путей до всех остальных узлов
    nodes.forEach(node => {
      const distances = this.breadthFirstSearch(graph, node);
      
      // Сумма всех расстояний
      let sum = 0;
      let reachableNodes = 0;
      
      for (const [other, distance] of Object.entries(distances)) {
        if (other !== node && distance !== Infinity) {
          sum += distance;
          reachableNodes++;
        }
      }
      
      // Центральность определяется как обратное значение средней длины пути
      centrality[node] = reachableNodes > 0 ? reachableNodes / sum : 0;
    });
    
    return centrality;
  }
  
  /**
   * Выполняет поиск в ширину для нахождения кратчайших путей
   * @param {Object} graph - Граф в виде списка смежности
   * @param {string} start - Начальный узел
   * @returns {Object} - Расстояния до всех узлов
   */
  breadthFirstSearch(graph, start) {
    const distances = {};
    const queue = [start];
    const visited = new Set([start]);
    
    // Инициализируем расстояния
    Object.keys(graph).forEach(node => {
      distances[node] = node === start ? 0 : Infinity;
    });
    
    // Поиск в ширину
    while (queue.length > 0) {
      const current = queue.shift();
      
      // Обрабатываем соседей
      graph[current].forEach(neighbor => {
        if (!visited.has(neighbor)) {
          visited.add(neighbor);
          queue.push(neighbor);
          distances[neighbor] = distances[current] + 1;
        }
      });
    }
    
    return distances;
  }
  
  /**
   * Нормализует значение в заданном диапазоне
   * @param {number} value - Исходное значение
   * @param {number} min - Минимальное значение исходного диапазона
   * @param {number} max - Максимальное значение исходного диапазона
   * @param {number} newMin - Минимальное значение нового диапазона
   * @param {number} newMax - Максимальное значение нового диапазона
   * @returns {number} - Нормализованное значение
   */
  normalizeValue(value, min, max, newMin, newMax) {
    // Проверка на деление на ноль
    if (max === min) return newMin;
    
    return ((value - min) / (max - min)) * (newMax - newMin) + newMin;
  }
  
  /**
   * Применяет размеры узлов в зависимости от выбранного режима
   */
  applyNodeSizes() {
    // Удаляем предыдущие классы масштабирования
    document.body.classList.remove('importance-scale-linear', 'importance-scale-logarithmic');
    
    // Добавляем новый класс масштабирования
    document.body.classList.add(`importance-scale-${this.sizeScale}`);
    
    // Применяем размеры в зависимости от режима
    if (this.sizeMode === 'uniform') {
      // Сброс к одинаковому размеру
      d3.selectAll('.node')
        .attr('data-importance', null);
    } else {
      // Получаем метрики в зависимости от режима
      const metricType = this.metrics[this.sizeMode];
      
      if (!metricType) return;
      
      // Применяем размеры на основе метрик
      d3.selectAll('.node').each(function(d) {
        const nodeId = d.id;
        if (metricType[nodeId]) {
          const importance = metricType[nodeId].normalized;
          d3.select(this).attr('data-importance', importance);
        } else {
          d3.select(this).attr('data-importance', '1');
        }
      });
    }
  }
  
  /**
   * Добавляет всплывающие подсказки с информацией о важности
   */
  addImportanceTooltips() {
    // Находим элемент для всплывающей подсказки
    const tooltip = document.getElementById('importance-tooltip');
    
    if (!tooltip) return;
    
    // Добавляем обработчики событий для узлов
    d3.selectAll('.node').on('mouseover.importance', (event, d) => {
      // Пропускаем, если выбран режим одинаковых размеров
      if (this.sizeMode === 'uniform') return;
      
      // Получаем метрики для узла
      const metricType = this.metrics[this.sizeMode];
      
      if (!metricType || !metricType[d.id]) return;
      
      const metric = metricType[d.id];
      
      // Формируем содержимое подсказки
      let content = `
        <div class="importance-tooltip-title">${d.name}</div>
      `;
      
      // Добавляем разные типы метрик
      if (this.sizeMode === 'connections') {
        content += `
          <div class="importance-tooltip-metric">
            <span>Всего связей:</span>
            <span class="importance-tooltip-metric-value">${metric.raw}</span>

          <div class="importance-tooltip-bar">
            <div class="importance-tooltip-bar-fill" style="width: ${(metric.raw / metric.max) * 100}%"></div>
          </div>
        `;
      } else if (this.sizeMode === 'influence') {
        content += `
          <div class="importance-tooltip-metric">
            <span>Влиятельность:</span>
            <span class="importance-tooltip-metric-value">${metric.raw.toFixed(1)}</span>
          </div>
          <div class="importance-tooltip-bar">
            <div class="importance-tooltip-bar-fill" style="width: ${(metric.raw / metric.max) * 100}%"></div>
          </div>
        `;
      } else if (this.sizeMode === 'centrality') {
        content += `
          <div class="importance-tooltip-metric">
            <span>Центральность:</span>
            <span class="importance-tooltip-metric-value">${metric.raw.toFixed(3)}</span>
          </div>
          <div class="importance-tooltip-bar">
            <div class="importance-tooltip-bar-fill" style="width: ${(metric.raw / metric.max) * 100}%"></div>
          </div>
        `;
      }
      
      // Устанавливаем содержимое подсказки
      tooltip.innerHTML = content;
      
      // Позиционируем подсказку около курсора
      tooltip.style.left = `${event.pageX + 10}px`;
      tooltip.style.top = `${event.pageY + 10}px`;
      
      // Показываем подсказку
      tooltip.classList.add('show');
    });
    
    // Скрываем подсказку при уходе курсора
    d3.selectAll('.node').on('mouseout.importance', () => {
      tooltip.classList.remove('show');
    });
    
    // Обновляем позицию подсказки при движении курсора
    d3.selectAll('.node').on('mousemove.importance', (event) => {
      if (!tooltip.classList.contains('show')) return;
      
      tooltip.style.left = `${event.pageX + 10}px`;
      tooltip.style.top = `${event.pageY + 10}px`;
    });
  }
  
  /**
   * Обновляет метрики и размеры узлов
   */
  updateMetricsAndSizes() {
    // Пересчитываем метрики
    this.calculateImportanceMetrics();
    
    // Применяем размеры

  }
```

## 6. Интеграция с системой кластеризации

Добавьте следующий код в конец функции `initNodeSizeManager`:

```javascript
// Интегрируем с системой кластеризации
if (network.clusteringManager) {
  // Добавляем обработчик события изменения кластеризации
  const originalUpdateClustering = network.clusteringManager.updateClustering;
  
  network.clusteringManager.updateClustering = function() {
    // Вызываем оригинальный метод
    originalUpdateClustering.call(this);
    
    // Обновляем размеры кластеров на основе важности узлов
    if (network.nodeSizeManager && this.activeClusterIds.size > 0) {
      updateClusterSizes(network.nodeSizeManager);
    }
  };
}

/**
 * Обновляет размеры кластеров на основе важности узлов
 * @param {Object} nodeSizeManager - Менеджер размеров узлов
 */
function updateClusterSizes(nodeSizeManager) {
  // Пропускаем, если выбран режим одинаковых размеров
  if (nodeSizeManager.sizeMode === 'uniform') return;
  
  // Получаем метрики в зависимости от режима
  const metricType = nodeSizeManager.metrics[nodeSizeManager.sizeMode];
  
  if (!metricType) return;
  
  // Обновляем размеры кластеров
  d3.selectAll('.cluster').each(function(d) {
    if (!d || !d.nodes) return;
    
    // Вычисляем среднюю важность узлов в кластере
    let totalImportance = 0;
    let count = 0;
    
    d.nodes.forEach(node => {
      if (metricType[node.id]) {
        totalImportance += metricType[node.id].normalized;
        count++;
      }
    });
    
    const avgImportance = count > 0 ? Math.round(totalImportance / count) : 3;
    
    // Устанавливаем размер кластера на основе средней важности
    d3.select(this).attr('data-importance', avgImportance);
    
    // Обновляем радиус круга кластера
    const circle = d3.select(this).select('circle');
    
    if (circle.empty()) return;
    
    const baseRadius = Math.sqrt(d.nodes.length) * 6 + 10;
    const importanceFactor = avgImportance / 3; // 3 - средняя важность
    
    circle.attr('r', baseRadius * importanceFactor);
  });
}
```
