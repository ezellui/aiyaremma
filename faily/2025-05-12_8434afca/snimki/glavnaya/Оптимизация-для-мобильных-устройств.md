## Улучшение поддержки сенсорных жестов

Добавьте следующий JavaScript код в конец файла `<script>` перед закрывающим тегом `</script>`:

```javascript
/**
 * Улучшает взаимодействие с сенсорными устройствами
 * @param {Object} network - Объект с компонентами сети
 */
function enhanceTouchInteractions(network) {
  // Получаем элементы управления для сенсорных устройств
  const zoomInBtn = document.getElementById("zoom-in");
  const zoomOutBtn = document.getElementById("zoom-out");
  const resetPanBtn = document.getElementById("reset-pan");
  
  // Обработчики нажатий на кнопки
  zoomInBtn.addEventListener("click", () => {
    const currentTransform = d3.zoomTransform(network.svg.node());
    const newScale = currentTransform.k * 1.3;
    
    network.svg.transition().duration(300).call(
      network.zoom.transform,
      d3.zoomIdentity
        .translate(currentTransform.x, currentTransform.y)
        .scale(newScale)
    );
  });
  
  zoomOutBtn.addEventListener("click", () => {
    const currentTransform = d3.zoomTransform(network.svg.node());
    const newScale = currentTransform.k / 1.3;
    
    network.svg.transition().duration(300).call(
      network.zoom.transform,
      d3.zoomIdentity
        .translate(currentTransform.x, currentTransform.y)
        .scale(newScale)
    );
  });
  
  resetPanBtn.addEventListener("click", () => {
    network.svg.transition().duration(750).call(
      network.zoom.transform,
      d3.zoomIdentity
    );
  });
  
  // Определяем и настраиваем обработчики касания
  const svg = network.svg.node();
  let touchStartX, touchStartY;
  let touchMoved = false;
  
  svg.addEventListener("touchstart", (event) => {
    // Предотвращаем прокрутку страницы при взаимодействии с графом
    if (event.target.tagName === "svg" || 
        event.target.parentNode.classList.contains("node") ||
        event.target.parentNode.parentNode.classList.contains("node")) {
      event.preventDefault();
    }
    
    const touch = event.touches[0];
    touchStartX = touch.clientX;
    touchStartY = touch.clientY;
    touchMoved = false;
  }, { passive: false });
  
  svg.addEventListener("touchmove", (event) => {
    if (event.touches.length !== 1) return;
    
    const touch = event.touches[0];
    const deltaX = touch.clientX - touchStartX;
    const deltaY = touch.clientY - touchStartY;
    
    // Если достаточное смещение, считаем это движением (не кликом)
    if (Math.abs(deltaX) > 5 || Math.abs(deltaY) > 5) {
      touchMoved = true;
    }
  }, { passive: true });
  
  svg.addEventListener("touchend", (event) => {
    // Обрабатываем клик по узлу при касании
    if (!touchMoved && event.target.parentNode.classList.contains("node")) {
      const nodeElement = event.target.parentNode;
      const nodeData = d3.select(nodeElement).datum();
      
      // Показываем информацию о выбранном узле
      showNodeInfo(nodeData);
      
      // Выделяем выбранный узел
      d3.selectAll(".node circle").classed("selected", false);
      d3.select(nodeElement).select("circle").classed("selected", true);
    }
  }, { passive: true });
  
  // Оптимизация двойного тапа для открытия детальной информации
  let lastTapTime = 0;
  svg.addEventListener("touchend", (event) => {
    const currentTime = new Date().getTime();
    const tapLength = currentTime - lastTapTime;
    
    if (tapLength < 500 && tapLength > 0 && !touchMoved) {
      // Двойной тап обнаружен
      if (event.target.parentNode.classList.contains("node")) {
        const nodeElement = event.target.parentNode;
        const nodeData = d3.select(nodeElement).datum();
        
        // Показываем детальную информацию
        showDetailedInfo(nodeData, document.querySelector(".detail-popup"), {links: data.links});
        
        event.preventDefault();
      }
    }
    
    lastTapTime = currentTime;
  });
}

// Добавляем вызов функции в конец функции integrateComponents
// Найдите функцию integrateComponents и в конце добавьте:
function integrateComponents(graphData) {
  // Существующий код...
  
  // Добавьте в конец функции:
  enhanceTouchInteractions(network);
}
```

## Улучшение отзывчивости интерфейса для мобильных устройств

Измените существующую функцию `initMobileFiltersToggle()` на следующий улучшенный вариант:

```javascript
/**
 * Инициализирует улучшенный мобильный интерфейс
 */
function initMobileFiltersToggle() {
  const toggleFiltersBtn = document.getElementById("toggle-filters");
  if (toggleFiltersBtn) {
    const mobileFilters = document.querySelector(".mobile-filters");
    
    toggleFiltersBtn.addEventListener("click", function() {
      mobileFilters.style.display = mobileFilters.style.display === "flex" ? "none" : "flex";
      this.textContent = mobileFilters.style.display === "flex" ? "Скрыть фильтры" : "Показать фильтры";
    });
  }
  
  // Добавляем оптимизацию для мобильного просмотра
  if (window.innerWidth <= 768) {
    // Автоматически сворачиваем боковую панель при загрузке на мобильных устройствах
    const sidebar = document.querySelector(".sidebar");
    const mainContent = document.querySelector(".main-content");
    
    if (sidebar && mainContent) {
      const toggleSidebarBtn = document.createElement("button");
      toggleSidebarBtn.id = "toggle-sidebar";
      toggleSidebarBtn.textContent = "Показать легенду";
      toggleSidebarBtn.classList.add("mobile-toggle-btn");
      toggleSidebarBtn.style.position = "fixed";
      toggleSidebarBtn.style.bottom = "10px";
      toggleSidebarBtn.style.left = "10px";
      toggleSidebarBtn.style.zIndex = "100";
      toggleSidebarBtn.style.padding = "8px 12px";
      toggleSidebarBtn.style.borderRadius = "4px";
      toggleSidebarBtn.style.backgroundColor = "rgba(255, 255, 255, 0.8)";
      toggleSidebarBtn.style.border = "1px solid #ccc";
      toggleSidebarBtn.style.boxShadow = "0 2px 5px rgba(0, 0, 0, 0.2)";
      
      sidebar.style.display = "none";
      document.body.appendChild(toggleSidebarBtn);
      
      toggleSidebarBtn.addEventListener("click", function() {
        const isVisible = sidebar.style.display !== "none";
        sidebar.style.display = isVisible ? "none" : "block";
        this.textContent = isVisible ? "Показать легенду" : "Скрыть легенду";
      });
      
      // Автоматическое скрытие элементов управления при прокрутке
      let lastScrollPosition = 0;
      let scrollTimeout;
      
      window.addEventListener("scroll", function() {
        clearTimeout(scrollTimeout);
        
        const controlsElement = document.querySelector(".controls");
        const currentScroll = window.pageYOffset || document.documentElement.scrollTop;
        
        if (currentScroll > lastScrollPosition) {
          // Прокрутка вниз - скрываем элементы управления
          controlsElement.style.transform = "translateY(-100%)";
          controlsElement.style.transition = "transform 0.3s ease";
        } else {
          // Прокрутка вверх - показываем элементы управления
          controlsElement.style.transform = "translateY(0)";
        }
        
        lastScrollPosition = currentScroll;
        
        // Показать элементы управления при остановке прокрутки
        scrollTimeout = setTimeout(() => {
          controlsElement.style.transform = "translateY(0)";
        }, 1000);
      });
    }
  }
}
```

## Оптимизация производительности для мобильных устройств

Добавьте следующую функцию в конец файла `<script>`:

```javascript
/**
 * Оптимизирует производительность для мобильных устройств
 * @param {Object} network - Объект с компонентами сети
 */
function optimizeForMobilePerformance(network) {
  // Определяем, является ли устройство мобильным
  const isMobile = window.innerWidth <= 768;
  
  if (isMobile) {
    // Уменьшаем сложность отрисовки
    network.simulation.alphaDecay(0.03); // Быстрее завершать симуляцию
    
    // Уменьшаем количество показываемых меток на мобильных устройствах
    const nodeLabels = d3.selectAll(".node text");
    
    // Показываем метки только для важных узлов или когда число узлов небольшое
    if (network.node.size() > 20) {
      nodeLabels.style("display", function(d) {
        // Показываем метки только для узлов с большим количеством связей
        const nodeLinks = network.link.filter(l => l.source.id === d.id || l.target.id === d.id);
        return nodeLinks.size() > 3 ? "block" : "none";
      });
    }
    
    // Добавляем показ метки при нажатии на узел
    network.node.on("click.showLabel", function(event, d) {
      const textElement = d3.select(this).select("text");
      const isVisible = textElement.style("display") !== "none";
      
      // Скрываем все метки
      d3.selectAll(".node text").style("display", "none");
      
      // Показываем метку выбранного узла
      if (!isVisible) {
        textElement.style("display", "block");
      }
    });
    
    // Оптимизируем память, удаляя ненужные элементы 
    // для очень больших графов на мобильных устройствах
    if (network.node.size() > 50) {
      // Уменьшаем количество отрисовываемых элементов
      d3.selectAll(".group-labels circle").remove(); // Удаляем фоновые круги групп
    }
  }
}

// Добавляем вызов функции в конец функции integrateComponents
// Найдите функцию integrateComponents и в конце добавьте:
function integrateComponents(graphData) {
  // Существующий код...
  
  // Добавьте после вызова enhanceTouchInteractions(network):
  optimizeForMobilePerformance(network);
}
```

Эти изменения значительно улучшат работу приложения на мобильных устройствах, добавив адаптивные стили, специальные элементы управления для сенсорных экранов и оптимизацию производительности.
