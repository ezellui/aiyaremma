Реализация полной поддержки клавиатурной навигации обеспечит доступность приложения для пользователей, которые не могут или предпочитают не использовать мышь.

## 4. Реализация клавиатурной навигации

Добавьте следующий код в конец файла `<script>`:

```javascript
/**
 * Класс для реализации клавиатурной навигации по сети
 */
class KeyboardNavigationManager {
  /**
   * Создает менеджера клавиатурной навигации
   * @param {Object} network - Объект сети
   * @param {Object} data - Данные сети
   */
  constructor(network, data) {
    this.network = network;
    this.data = data;
    this.nodes = data.nodes;
    this.links = data.links;
    
    // Элемент для отображения индикатора фокуса
    this.focusIndicator = document.getElementById('focus-indicator');
    
    // Текущий индекс в списке узлов
    this.currentNodeIndex = -1;
    
    // Режим навигации (узлы, кластеры, элементы управления)
    this.navigationMode = 'nodes';
    
    // Добавление обработчиков событий
    this.setupEventListeners();
    
    // Инициализация подсказки о навигации
    this.setupNavigationHelp();
    
    // Флаг режима клавиатурной навигации
    this.keyboardNavigationActive = false;
  }
  
  /**
   * Настраивает обработчики событий для клавиатурной навигации
   */
  setupEventListeners() {
    // Обработчик событий клавиатуры
    document.addEventListener('keydown', this.handleKeyDown.bind(this));
    
    // Обработчик для переключения между мышью и клавиатурой
    document.addEventListener('mousedown', () => {
      this.keyboardNavigationActive = false;
      this.hideFocusIndicator();
    });
    
    document.addEventListener('keydown', (event) => {
      // При любом нажатии клавиши активируем режим клавиатурной навигации
      if (!this.keyboardNavigationActive && event.key !== 'Tab') {
        this.keyboardNavigationActive = true;
      }
    });
    
    // Делаем узлы фокусируемыми
    this.makeNodesFocusable();
    
    // Добавляем обработчики событий для элементов управления
    this.setupControlsNavigation();
  }
  
  /**
   * Делает узлы доступными для фокуса с клавиатуры
   */
  makeNodesFocusable() {
    // Добавляем атрибут tabindex к узлам
    d3.selectAll('.node')
      .attr('tabindex', '0')
      .attr('role', 'button')
      .attr('aria-label', d => `Философ ${d.name}, ${d.years}, группа ${d.groupName}`);
    
    // Добавляем обработчики событий фокуса
    d3.selectAll('.node')
      .on('focus', (event, d) => {
        if (this.keyboardNavigationActive) {
          this.highlightNode(d);
          this.showNodeInfo(d);
          this.updateFocusIndicator(event.currentTarget);
        }
      })
      .on('blur', () => {
        if (this.keyboardNavigationActive) {
          this.hideFocusIndicator();
        }
      })
      .on('keydown', (event, d) => {
        if (event.key === 'Enter' || event.key === ' ') {
          // Показываем детальную информацию при нажатии Enter или пробела
          event.preventDefault();
          showDetailedInfo(d, document.querySelector('.detail-popup'), {links: this.links});
        }
      });
    
    // Делаем кластеры также доступными для фокуса
    this.network.svg.on('clustersCreated', () => {
      d3.selectAll('.cluster')
        .attr('tabindex', '0')
        .attr('role', 'button')
        .attr('aria-label', d => `Кластер ${d.name}, содержит ${d.nodes.length} философов`);
      
      d3.selectAll('.cluster')
        .on('focus', (event, d) => {
          if (this.keyboardNavigationActive) {
            this.updateFocusIndicator(event.currentTarget);
          }
        })
        .on('keydown', (event, d) => {
          if (event.key === 'Enter' || event.key === ' ') {
            event.preventDefault();
            // Обработка клика по кластеру при нажатии Enter или пробела
            if (this.network.clusteringManager) {
              this.network.clusteringManager.handleClusterClick(event, d);
            }
          }
        });
    });
  }
  
  /**
   * Настраивает обработчики событий для элементов управления
   */
  setupControlsNavigation() {
    // Добавляем aria-метки к элементам управления
    document.getElementById('filter-period').setAttribute('aria-label', 'Фильтр по периоду');
    document.getElementById('filter-relation').setAttribute('aria-label', 'Фильтр по типу связи');
    document.getElementById('search').setAttribute('aria-label', 'Поиск философа');
    document.getElementById('reset-zoom').setAttribute('aria-label', 'Сбросить масштаб');
    document.getElementById('toggle-clusters').setAttribute('aria-label', 'Группировать по школам');
    document.getElementById('toggle-labels').setAttribute('aria-label', 'Показать названия школ');
    
    // Добавляем метку группы для элементов управления
    const controlsLabel = document.createElement('span');
    controlsLabel.id = 'controls-label';
    controlsLabel.className = 'sr-only';
    controlsLabel.textContent = 'Элементы управления визуализацией';
    document.querySelector('.controls').prepend(controlsLabel);
    document.querySelector('.controls').setAttribute('aria-labelledby', 'controls-label');
    
    // Для визуализации фокуса на элементах управления
    const controlElements = document.querySelectorAll('.controls button, .controls select, .controls input, .controls-extra input, .controls-extra select');
    controlElements.forEach(element => {
      element.addEventListener('focus', () => {
        this.navigationMode = 'controls';
      });
    });
  }
  
  /**
   * Настраивает подсказку о навигации с клавиатуры
   */
  setupNavigationHelp() {
    const helpDialog = document.getElementById('keyboard-help');
    
    // Добавляем обработчик для показа/скрытия подсказки
    document.addEventListener('keydown', (event) => {
      if (event.key === '?' || event.key === '/') {
        event.preventDefault();
        helpDialog.classList.toggle('visible');
      } else if (event.key === 'Escape' && helpDialog.classList.contains('visible')) {
        helpDialog.classList.remove('visible');
      }
    });
    
    // Добавляем кнопку показа подсказки в интерфейс
    const helpButton = document.createElement('button');
    helpButton.textContent = '?';
    helpButton.id = 'show-keyboard-help';
    helpButton.setAttribute('aria-label', 'Показать справку по клавиатурной навигации');
    helpButton.style.position = 'fixed';
    helpButton.style.bottom = '10px';
    helpButton.style.right = '10px';
    helpButton.style.width = '30px';
    helpButton.style.height = '30px';
    helpButton.style.borderRadius = '50%';
    helpButton.style.backgroundColor = '#f0f0f0';
    helpButton.style.border = '1px solid #ccc';
    helpButton.style.cursor = 'pointer';
    helpButton.style.zIndex = '1000';
    
    helpButton.addEventListener('click', () => {
      helpDialog.classList.toggle('visible');
    });
    
    document.body.appendChild(helpButton);
  }
  
  /**
   * Обработчик нажатий клавиш
   * @param {KeyboardEvent} event - Событие нажатия клавиши
   */
  handleKeyDown(event) {
    // Не перехватываем нажатия в поле ввода поиска
    if (event.target.tagName === 'INPUT' && event.target.type === 'text') {
      return;
    }
    
    // Обработка основных навигационных клавиш
    switch (event.key) {
      case 'ArrowUp':
      case 'ArrowDown':
      case 'ArrowLeft':
      case 'ArrowRight':
        event.preventDefault();
        this.navigateGraph(event.key);
        break;
      
      case 'Home':
        // Сброс масштаба и позиции
        event.preventDefault();
        this.resetView();
        break;
      
      case '+':
      case '=':
        // Увеличение масштаба
        event.preventDefault();
        this.zoom(1.2);
        break;
      
      case '-':
      case '_':
        // Уменьшение масштаба
        event.preventDefault();
        this.zoom(0.8);
        break;
      
      case 'Tab':
        // При нажатии Tab переключаемся в режим навигации по элементам интерфейса
        this.keyboardNavigationActive = true;
        break;
      
      case 'Escape':
        // Закрытие всех модальных окон
        this.closeAllModals();
        break;
    }
  }
  
  /**
   * Навигация по графу с помощью клавиш со стрелками
   * @param {string} direction - Направление навигации (ArrowUp, ArrowDown, ArrowLeft, ArrowRight)
   */
  navigateGraph(direction) {
    this.keyboardNavigationActive = true;
    
    if (this.navigationMode !== 'nodes') {
      // Переключаемся в режим навигации по узлам
      this.navigationMode = 'nodes';
      this.currentNodeIndex = 0;
      this.focusNodeByIndex(this.currentNodeIndex);
      return;
    }
    
    // Если нет выбранного узла, выбираем первый
    if (this.currentNodeIndex === -1) {
      this.currentNodeIndex = 0;
      this.focusNodeByIndex(this.currentNodeIndex);
      return;
    }
    
    // Получаем текущий узел
    const currentNode = this.nodes[this.currentNodeIndex];
    
    if (!currentNode) return;
    
    // Находим соседние узлы
    const connectedNodes = this.links
      .filter(link => link.source.id === currentNode.id || link.target.id === currentNode.id)
      .map(link => link.source.id === currentNode.id ? link.target : link.source);
    
    // Если нет соседних узлов, переходим к следующему/предыдущему
    if (connectedNodes.length === 0) {
      switch (direction) {
        case 'ArrowRight':
        case 'ArrowDown':
          this.currentNodeIndex = (this.currentNodeIndex + 1) % this.nodes.length;
          break;
        case 'ArrowLeft':
        case 'ArrowUp':
          this.currentNodeIndex = (this.currentNodeIndex - 1 + this.nodes.length) % this.nodes.length;
          break;
      }
      
      this.focusNodeByIndex(this.currentNodeIndex);
      return;
    }
    
    // Находим ближайшие узлы в указанном направлении
    let nearestNode = null;
    let minDistance = Infinity;
    
    connectedNodes.forEach(node => {
      // Вычисляем угол между текущим узлом и соседним
      const dx = node.x - currentNode.x;
      const dy = node.y - currentNode.y;
      const angle = Math.atan2(dy, dx) * 180 / Math.PI;
      
      // Определяем, подходит ли этот узел для выбранного направления
      let isInDirection = false;
      
      switch (direction) {
        case 'ArrowRight':
          isInDirection = (angle > -45 && angle < 45);
          break;
        case 'ArrowDown':
          isInDirection = (angle > 45 && angle < 135);
          break;
        case 'ArrowLeft':
          isInDirection = (angle > 135 || angle < -135);
          break;
        case 'ArrowUp':
          isInDirection = (angle > -135 && angle < -45);
          break;
      }
      
      if (isInDirection) {
        // Вычисляем расстояние до узла
        const distance = Math.sqrt(dx * dx + dy * dy);
        
        if (distance < minDistance) {
          minDistance = distance;
          nearestNode = node;
        }
      }
    });
    
    // Если найден ближайший узел в указанном направлении, переходим к нему
    if (nearestNode) {
      this.currentNodeIndex = this.nodes.findIndex(n => n.id === nearestNode.id);
      this.focusNodeByIndex(this.currentNodeIndex);
    } else {
      // Если нет узла в указанном направлении, переходим к следующему/предыдущему
      switch (direction) {
        case 'ArrowRight':
        case 'ArrowDown':
          this.currentNodeIndex = (this.currentNodeIndex + 1) % this.nodes.length;
          break;
        case 'ArrowLeft':
        case 'ArrowUp':
          this.currentNodeIndex = (this.currentNodeIndex - 1 + this.nodes.length) % this.nodes.length;
          break;
      }
      
      this.focusNodeByIndex(this.currentNodeIndex);
    }
  }
  
  /**
   * Фокусируется на узле по его индексу
   * @param {number} index - Индекс узла
   */
  focusNodeByIndex(index) {
    if (index < 0 || index >= this.nodes.length) return;
    
    // Получаем узел
    const node = this.nodes[index];
    
    // Находим соответствующий DOM-элемент
    const nodeElement = d3.select(`.node[data-id="${node.id}"]`).node();
    
    if (!nodeElement) return;
    
    // Уделяем фокус элементу
    nodeElement.focus();
    
    // Прокручиваем вид, чтобы узел был видимым
    this.panToNode(node);
  }
  
  /**
   * Прокручивает вид так, чтобы узел был видимым
   * @param {Object} node - Узел для отображения
   */
  panToNode(node) {
    if (!node || node.x === undefined || node.y === undefined) return;
    
    // Получаем текущее преобразование
    const currentTransform = d3.zoomTransform(this.network.svg.node());
    
    // Вычисляем новые координаты центра просмотра
    const width = this.network.svg.node().clientWidth;
    const height = this.network.svg.node().clientHeight;
    
    const newX = -node.x * currentTransform.k + width / 2;
    const newY = -node.y * currentTransform.k + height / 2;
    
    // Плавно переходим к новому центру
    this.network.svg.transition()
      .duration(500)
      .call(
        this.network.zoom.transform,
        d3.zoomIdentity
          .translate(newX, newY)
          .scale(currentTransform.k)
      );
  }
  
  /**
   * Сбрасывает вид (масштаб и позицию)
   */
  resetView() {
    this.network.svg.transition()
      .duration(750)
      .call(
        this.network.zoom.transform,
        d3.zoomIdentity
      );
    
    // Сброс текущего индекса узла
    this.currentNodeIndex = -1;
  }
  
  /**
   * Изменяет масштаб на указанный коэффициент
   * @param {number} factor - Коэффициент масштабирования
   */
  zoom(factor) {
    // Получаем текущее преобразование
    const currentTransform = d3.zoomTransform(this.network.svg.node());
    
    // Вычисляем новый масштаб
    const newScale = currentTransform.k * factor;
    
    // Проверяем, что новый масштаб находится в допустимых пределах
    const minScale = 0.1;
    const maxScale = 5;
    
    if (newScale < minScale || newScale > maxScale) return;
    
    // Определяем центр масштабирования
    const width = this.network.svg.node().clientWidth;
    const height = this.network.svg.node().clientHeight;
    
    // Плавно изменяем масштаб с сохранением центра
    this.network.svg.transition()
      .duration(300)
      .call(
        this.network.zoom.transform,
        d3.zoomIdentity
          .translate(currentTransform.x, currentTransform.y)
          .scale(newScale)
      );
  }
  
  /**
   * Закрывает все модальные окна
   */
  closeAllModals() {
    // Скрываем всплывающую подсказку с клавиатурной навигацией
    document.getElementById('keyboard-help').classList.remove('visible');
    
    // Скрываем всплывающее окно с детальной информацией
    const detailPopup = document.querySelector('.detail-popup');
    const detailPopupBackdrop = document.querySelector('.detail-popup-backdrop');
    
    if (detailPopup) detailPopup.style.display = 'none';
    if (detailPopupBackdrop) detailPopupBackdrop.style.display = 'none';
    
    // Скрываем другие возможные модальные окна
    const quotePopup = document.querySelector('.quote-popup');
    const quotePopupBackdrop = document.querySelector('.quote-popup-backdrop');
    
    if (quotePopup) quotePopup.style.display = 'none';
    if (quotePopupBackdrop) quotePopupBackdrop.style.display = 'none';
  }
  
  /**
   * Выделяет узел визуально
   * @param {Object} node - Узел для выделения
   */
  highlightNode(node) {
    // Снимаем предыдущее выделение
    d3.selectAll('.node circle').classed('selected', false);
    
    // Выделяем выбранный узел
    d3.select(`.node[data-id="${node.id}"] circle`).classed('selected', true);
  }
  
  /**
   * Показывает информацию об узле
   * @param {Object} node - Узел для отображения информации
   */
  showNodeInfo(node) {
    // Вызываем существующую функцию отображения информации
    window.showNodeInfo(node);
  }
  
  /**
   * Обновляет индикатор фокуса
   * @param {HTMLElement} element - Элемент, на котором установлен фокус
   */
  updateFocusIndicator(element) {
    if (!this.focusIndicator || !element) return;
    
    // Получаем координаты элемента
    const rect = element.getBoundingClientRect();
    
    // Устанавливаем размеры и положение индикатора
    this.focusIndicator.style.display = 'block';
    this.focusIndicator.style.width = `${rect.width + 10}px`;
    this.focusIndicator.style.height = `${rect.height + 10}px`;
    this.focusIndicator.style.left = `${rect.left + rect.width / 2}px`;
    this.focusIndicator.style.top = `${rect.top + rect.height / 2}px`;
  }
  
  /**
   * Скрывает индикатор фокуса
   */
  hideFocusIndicator() {
    if (!this.focusIndicator) return;
    
    this.focusIndicator.style.display = 'none';
  }
}

/**
 * Добавляет визуально скрытые элементы для скринридеров
 */
function addScreenReaderOnlyElements() {
  // Создаем стили для визуально скрытых элементов
  const style = document.createElement('style');
  style.textContent = `
    .sr-only {
      position: absolute;
      width: 1px;
      height: 1px;
      margin: -1px;
      padding: 0;
      overflow: hidden;
      clip: rect(0, 0, 0, 0);
      border: 0;
    }
  `;
  document.head.appendChild(style);
  
  // Добавляем заголовок для области сети
  const networkHeading = document.createElement('h2');
  networkHeading.textContent = 'Сеть российской философии (интерактивная визуализация)';
  networkHeading.className = 'sr-only';
  
  const networkContainer = document.querySelector('.network-container');
  networkContainer.setAttribute('role', 'application');
  networkContainer.setAttribute('aria-label', 'Визуализация связей между российскими философами');
  networkContainer.prepend(networkHeading);
  
  // Добавляем описание для временной шкалы
  const timelineContainer = document.querySelector('.timeline-container');

## 5. Реализация доступных модальных окон

Обновите функцию `initDetailPopup` для обеспечения правильной работы с клавиатурой:

```javascript
/**
 * Инициализирует доступное всплывающее окно с детальной информацией о философах
 * @param {Object} graphData - Объект с узлами и связями
 */
function initDetailPopup(graphData) {
  // Создаем элементы попапа, если они еще не существуют
  let popupBackdrop = document.querySelector(".detail-popup-backdrop");
  let popup = document.querySelector(".detail-popup");
  
  if (!popupBackdrop) {
    popupBackdrop = document.createElement("div");
    popupBackdrop.className = "detail-popup-backdrop";
    popupBackdrop.setAttribute("role", "presentation");
    document.body.appendChild(popupBackdrop);

  if (!popup) {
    popup = document.createElement("div");
    popup.className = "detail-popup";
    popup.setAttribute("role", "dialog");
    popup.setAttribute("aria-modal", "true");
    popup.setAttribute("aria-labelledby", "detail-popup-title");
    popup.innerHTML = `
      <button class="detail-popup-close" aria-label="Закрыть">&times;</button>
      <div class="detail-popup-content">
        <h2 id="detail-popup-title">Информация о философе</h2>
        <div id="detail-popup-body"></div>
      </div>
    `;
    document.body.appendChild(popup);
  }
  
  // Обработчик закрытия
  const closeButton = popup.querySelector(".detail-popup-close");
  closeButton.addEventListener("click", function() {
    popup.style.display = "none";
    popupBackdrop.style.display = "none";
    
    // Возвращаем фокус на элемент, который открыл попап
    if (popup.lastFocusedElement) {
      popup.lastFocusedElement.focus();
      popup.lastFocusedElement = null;
    }
  });
  
  popupBackdrop.addEventListener("click", function() {
    popup.style.display = "none";
    popupBackdrop.style.display = "none";
    
    // Возвращаем фокус на элемент, который открыл попап
    if (popup.lastFocusedElement) {
      popup.lastFocusedElement.focus();
      popup.lastFocusedElement = null;
    }
  });
  
  // Добавляем обработчик клавиши Escape
  popup.addEventListener("keydown", function(event) {
    if (event.key === "Escape") {
      popup.style.display = "none";
      popupBackdrop.style.display = "none";
      
      // Возвращаем фокус на элемент, который открыл попап
      if (popup.lastFocusedElement) {
        popup.lastFocusedElement.focus();
        popup.lastFocusedElement = null;
      }
    }
  });
  
  // Двойной клик по узлу открывает детальную информацию
  d3.selectAll(".node").on("dblclick", function(event, d) {
    // Сохраняем элемент, который открыл попап
    popup.lastFocusedElement = document.activeElement;
    
    showDetailedInfo(d, popup, graphData);
    
    // Устанавливаем фокус на кнопку закрытия
    closeButton.focus();
  });
  
  // Обновляем функцию отображения детальной информации
  window.showDetailedInfo = function(philosopher, popup, graphData) {
    const content = popup.querySelector(".detail-popup-content");
    const title = popup.querySelector("#detail-popup-title");
    const body = popup.querySelector("#detail-popup-body");

    title.textContent = philosopher.name;
    
    // Используем расширенную базу данных, если доступна информация о философе
    if (philosophersDetails[philosopher.id]) {
      showPhilosopherDetails(philosopher.id, body);
    } else {
      // Иначе используем существующий код для отображения информации
      // Создаем HTML-контент с информацией о философе
      let html = `<p><strong>Годы жизни:</strong> ${philosopher.years}</p>`;
      html += `<p><strong>Группа:</strong> ${philosopher.groupName}</p>`;
      
      // Добавляем информацию о связях
      // ... (остальной код без изменений)
      
      // Устанавливаем контент
      body.innerHTML = html;
    }
    
    // Отображаем попап
    popup.style.display = "block";
    popupBackdrop.style.display = "block";
    
    // Устанавливаем ловушку фокуса
    trapFocus(popup);
  };
  
  /**
   * Создает ловушку фокуса внутри элемента
   * @param {HTMLElement} element - Элемент для ловушки фокуса
   */
  function trapFocus(element) {
    // Находим все фокусируемые элементы внутри
    const focusableElements = element.querySelectorAll(
      'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])'
    );
    
    if (focusableElements.length === 0) return;
    
    const firstElement = focusableElements[0];
    const lastElement = focusableElements[focusableElements.length - 1];
    
    // Добавляем обработчик Tab для зацикливания фокуса
    element.addEventListener('keydown', function(event) {
      if (event.key === 'Tab') {
        if (event.shiftKey && document.activeElement === firstElement) {
          // Shift+Tab на первом элементе -> переход к последнему
          event.preventDefault();
          lastElement.focus();
        } else if (!event.shiftKey && document.activeElement === lastElement) {
          // Tab на последнем элементе -> переход к первому
          event.preventDefault();
          firstElement.focus();
        }
      }
    });
  }
}

Также добавлены доступные модальные окна с корректной работой фокуса и ловушкой фокуса для модальных диалогов. Пользователи с клавиатурой смогут видеть индикатор фокуса и получать информацию о доступных комбинациях клавиш.
