```javascript
  /**
   * Инициализирует менеджер подсказок
   */
  init() {
    // Инициализируем элементы управления
    this.initControls();
    
    // Проверяем, завершено ли обучение
    const tutorialCompleted = this.storage.getItem(this.tutorialKey) === 'true';
    const tipsDismissed = this.storage.getItem(this.tipsKey) === 'true';
    
    // Если обучение не завершено и советы не отклонены, показываем панель советов
    if (!tutorialCompleted && !tipsDismissed) {
      setTimeout(() => {
        this.showTipsPanel();
      }, 2000);
    }
  }
  
  /**
   * Инициализирует элементы управления подсказками
   */
  initControls() {
    // Кнопка запуска обучения
    if (this.startBtn) {
      this.startBtn.addEventListener('click', () => {
        this.startTutorial();
      });
    }
    
    // Кнопка закрытия панели советов
    const closePanel = this.panel ? this.panel.querySelector('.close-panel') : null;
    if (closePanel) {
      closePanel.addEventListener('click', () => {
        this.hideTipsPanel();
      });
    }
    
    // Кнопка запуска обучения из панели советов
    const startTutorialBtn = document.getElementById('start-tutorial-btn');
    if (startTutorialBtn) {
      startTutorialBtn.addEventListener('click', () => {
        this.hideTipsPanel();
        this.startTutorial();
      });
    }
    
    // Кнопка отклонения советов
    const dismissTipsBtn = document.getElementById('dismiss-tips-btn');
    if (dismissTipsBtn) {
      dismissTipsBtn.addEventListener('click', () => {
        this.hideTipsPanel();
        this.storage.setItem(this.tipsKey, 'true');
      });
    }
    
    // Добавляем всплывающие подсказки к основным элементам управления
    this.addFeatureTooltips();
  }
  
  /**
   * Показывает панель советов
   */
  showTipsPanel() {
    if (this.panel) {
      this.panel.classList.add('show');
    }
  }
  
  /**
   * Скрывает панель советов
   */
  hideTipsPanel() {
    if (this.panel) {
      this.panel.classList.remove('show');
    }
  }
  
  /**
   * Запускает обучение
   */
  startTutorial() {
    // Отображаем оверлей
    if (this.overlay) {
      this.overlay.style.display = 'block';
    }
    
    // Скрываем кнопку запуска
    if (this.startBtn) {
      this.startBtn.style.display = 'none';
    }
    
    // Сбрасываем индекс на первый шаг
    this.currentStepIndex = 0;
    
    // Показываем первый шаг
    this.showStep(this.currentStepIndex);
    
    // Объявляем для скринридеров
    if (window.announce) {
      window.announce('Началось интерактивное обучение. Нажимайте "Далее" для перехода к следующему шагу.');
    }
  }
  
  /**
   * Завершает обучение
   */
  finishTutorial() {
    // Скрываем оверлей
    if (this.overlay) {
      this.overlay.style.display = 'none';
    }
    
    // Удаляем подсветку и подсказку
    this.removeHighlight();
    this.removeTooltip();
    
    // Отображаем кнопку запуска
    if (this.startBtn) {
      this.startBtn.style.display = 'flex';
      this.startBtn.classList.remove('pulse');
    }
    
    // Отмечаем обучение как завершенное
    this.storage.setItem(this.tutorialKey, 'true');
    
    // Объявляем для скринридеров
    if (window.announce) {
      window.announce('Обучение завершено. Теперь вы можете использовать все возможности приложения.');
    }
  }
  
  /**
   * Показывает определенный шаг обучения
   * @param {number} index - Индекс шага
   */
  showStep(index) {
    if (index < 0 || index >= this.steps.length) return;
    
    const step = this.steps[index];
    
    // Удаляем предыдущую подсветку и подсказку
    this.removeHighlight();
    this.removeTooltip();
    
    // Создаем подсветку для текущего элемента
    if (step.highlight) {
      this.createHighlight(step);
    }
    
    // Создаем подсказку для текущего шага
    this.createTooltip(step, index);
    
    // Выполняем действие для шага, если оно определено
    if (step.action && typeof step.action === 'function') {
      step.action.call(this);
    }
  }
  
  /**
   * Создает подсветку для элемента
   * @param {Object} step - Информация о шаге
   */
  createHighlight(step) {
    // Находим целевой элемент
    const targetEl = document.querySelector(step.target);
    if (!targetEl) return;
    
    // Создаем элемент подсветки
    const highlight = document.createElement('div');
    highlight.className = 'tutorial-highlight';
    
    // Получаем размеры и позицию целевого элемента
    const rect = targetEl.getBoundingClientRect();
    
    // Устанавливаем размеры и позицию подсветки
    if (step.highlightRadius) {
      // Круглая подсветка с заданным радиусом
      const radius = step.highlightRadius;
      highlight.style.width = `${radius * 2}px`;
      highlight.style.height = `${radius * 2}px`;
      highlight.style.left = `${rect.left + rect.width / 2 - radius}px`;
      highlight.style.top = `${rect.top + rect.height / 2 - radius}px`;
      highlight.style.borderRadius = '50%';
    } else {
      // Подсветка по размеру элемента
      highlight.style.width = `${rect.width + 10}px`;
      highlight.style.height = `${rect.height + 10}px`;
      highlight.style.left = `${rect.left - 5}px`;
      highlight.style.top = `${rect.top - 5}px`;
    }
    
    // Добавляем подсветку на страницу
    document.body.appendChild(highlight);
    this.currentHighlight = highlight;
    
    // Если элемент кликабельный, добавляем указатель
    if (step.clickable) {
      this.createPointer(rect);
    }
  }
  
  /**
   * Создает анимированный указатель для элемента
   * @param {DOMRect} rect - Размеры и позиция элемента
   */
  createPointer(rect) {
    const pointer = document.createElement('div');
    pointer.className = 'tutorial-pointer';
    pointer.textContent = '👆';
    
    // Устанавливаем позицию указателя
    pointer.style.left = `${rect.left + rect.width / 2}px`;
    pointer.style.top = `${rect.top + rect.height / 2}px`;
    
    // Добавляем указатель на страницу
    document.body.appendChild(pointer);
    this.currentPointer = pointer;
  }
  
  /**
   * Удаляет подсветку
   */
  removeHighlight() {
    if (this.currentHighlight) {
      document.body.removeChild(this.currentHighlight);
      this.currentHighlight = null;
    }
    
    if (this.currentPointer) {
      document.body.removeChild(this.currentPointer);
      this.currentPointer = null;
    }
  }
  
  /**
   * Создает подсказку для шага
   * @param {Object} step - Информация о шаге
   * @param {number} index - Индекс текущего шага
   */
  createTooltip(step, index) {
    // Создаем элемент подсказки
    const tooltip = document.createElement('div');
    tooltip.className = 'tutorial-tooltip';
    
    // Создаем заголовок
    const title = document.createElement('h3');
    title.textContent = step.title;
    tooltip.appendChild(title);
    
    // Создаем содержимое
    const content = document.createElement('p');
    content.textContent = step.content;
    tooltip.appendChild(content);
    
    // Добавляем подсказку, если она есть
    if (step.hint) {
      const hint = document.createElement('div');
      hint.className = 'hint';
      hint.innerHTML = `<span class="hint-icon">💡</span> ${step.hint}`;
      tooltip.appendChild(hint);
    }
    
    // Создаем блок с кнопками
    const actions = document.createElement('div');
    actions.className = 'actions';
    
    // Кнопка "Назад" (кроме первого шага)
    if (index > 0) {
      const prevBtn = document.createElement('button');
      prevBtn.className = 'prev-btn';
      prevBtn.textContent = 'Назад';
      prevBtn.addEventListener('click', () => {
        this.showStep(index - 1);
      });
      actions.appendChild(prevBtn);
    } else {
      // Кнопка "Пропустить" для первого шага
      const skipBtn = document.createElement('button');
      skipBtn.className = 'skip-btn';
      skipBtn.textContent = 'Пропустить';
      skipBtn.addEventListener('click', () => {
        this.finishTutorial();
      });
      actions.appendChild(skipBtn);
    }
    
    // Кнопка "Далее" или "Завершить"
    const nextBtn = document.createElement('button');
    
    if (step.isFinish) {
      nextBtn.className = 'done-btn';
      nextBtn.textContent = 'Завершить';
      nextBtn.addEventListener('click', () => {
        this.finishTutorial();
      });
    } else {
      nextBtn.className = 'next-btn';
      nextBtn.textContent = 'Далее';
      nextBtn.addEventListener('click', () => {
        this.showStep(index + 1);
      });
    }
    
    actions.appendChild(nextBtn);
    tooltip.appendChild(actions);
    
    // Добавляем индикатор прогресса
    const progress = document.createElement('div');
    progress.className = 'progress';
    
    for (let i = 0; i < this.steps.length; i++) {
      const dot = document.createElement('div');
      dot.className = `progress-dot ${i === index ? 'active' : ''}`;
      progress.appendChild(dot);
    }
    
    tooltip.appendChild(progress);
    
    // Добавляем подсказку на страницу
    document.body.appendChild(tooltip);
    this.currentTooltip = tooltip;
    
    // Позиционируем подсказку относительно целевого элемента
    this.positionTooltip(tooltip, step);
    
    // Устанавливаем обработчик клавиши Escape
    document.addEventListener('keydown', this.handleEscapeKey);
    
    // Устанавливаем обработчики клавиш для навигации
    document.addEventListener('keydown', this.handleArrowKeys);
  }
  
  /**
   * Позиционирует подсказку относительно целевого элемента
   * @param {HTMLElement} tooltip - Элемент подсказки
   * @param {Object} step - Информация о шаге
   */
  positionTooltip(tooltip, step) {
    // Находим целевой элемент
    const targetEl = document.querySelector(step.target);
    if (!targetEl) {
      // Центрируем подсказку на экране, если элемент не найден
      const windowWidth = window.innerWidth;
      const windowHeight = window.innerHeight;
      const tooltipRect = tooltip.getBoundingClientRect();
      
      tooltip.style.left = `${(windowWidth - tooltipRect.width) / 2}px`;
      tooltip.style.top = `${(windowHeight - tooltipRect.height) / 2}px`;
      return;
    }
    
    const targetRect = targetEl.getBoundingClientRect();
    const tooltipRect = tooltip.getBoundingClientRect();
    
    // Позиционирование в зависимости от указанной позиции
    let left, top;
    
    if (step.position === 'center') {
      // Центрирование по экрану
      left = (window.innerWidth - tooltipRect.width) / 2;
      top = (window.innerHeight - tooltipRect.height) / 2;
    } else {
      // Позиционирование относительно целевого элемента
      switch (step.position) {
        case 'top':
          left = targetRect.left + (targetRect.width - tooltipRect.width) / 2;
          top = targetRect.top - tooltipRect.height - 15;
          break;
        case 'bottom':
          left = targetRect.left + (targetRect.width - tooltipRect.width) / 2;
          top = targetRect.bottom + 15;
          break;
        case 'left':
          left = targetRect.left - tooltipRect.width - 15;
          top = targetRect.top + (targetRect.height - tooltipRect.height) / 2;
          break;
        case 'right':
          left = targetRect.right + 15;
          top = targetRect.top + (targetRect.height - tooltipRect.height) / 2;
          break;
        default:
          left = targetRect.right + 15;
          top = targetRect.top + (targetRect.height - tooltipRect.height) / 2;
      }
    }
    
    // Проверка и корректировка, чтобы подсказка не выходила за пределы экрана
    if (left < 10) left = 10;
    if (left + tooltipRect.width > window.innerWidth - 10) {
      left = window.innerWidth - tooltipRect.width - 10;
    }
    
    if (top < 10) top = 10;
    if (top + tooltipRect.height > window.innerHeight - 10) {
      top = window.innerHeight - tooltipRect.height - 10;
    }
    
    // Устанавливаем позицию
    tooltip.style.left = `${left}px`;
    tooltip.style.top = `${top}px`;
  }
  
  /**
   * Удаляет подсказку
   */
  removeTooltip() {
    if (this.currentTooltip) {
      document.body.removeChild(this.currentTooltip);
      this.currentTooltip = null;
    }
    
    // Удаляем обработчики клавиш
    document.removeEventListener('keydown', this.handleEscapeKey);
    document.removeEventListener('keydown', this.handleArrowKeys);
  }
  
  /**
   * Обработчик нажатия клавиши Escape
   * @param {KeyboardEvent} event - Событие нажатия клавиши
   */
  handleEscapeKey = (event) => {
    if (event.key === 'Escape') {
      this.finishTutorial();
    }
  }
  
  /**
   * Обработчик нажатия клавиш со стрелками
   * @param {KeyboardEvent} event - Событие нажатия клавиши
   */
  handleArrowKeys = (event) => {
    if (event.key === 'ArrowRight' || event.key === 'ArrowDown') {
      event.preventDefault();
      
      if (this.currentStepIndex < this.steps.length - 1) {
        this.showStep(this.currentStepIndex + 1);
      } else {
        this.finishTutorial();
      }
    } else if (event.key === 'ArrowLeft' || event.key === 'ArrowUp') {
      event.preventDefault();
      
      if (this.currentStepIndex > 0) {
        this.showStep(this.currentStepIndex - 1);
      }
    }
  }
  
  /**
   * Имитирует нажатие на элемент
   * @param {string} selector - CSS-селектор элемента
   */
  simulateClick(selector) {
    const element = document.querySelector(selector);
    if (!element || !this.currentPointer) return;
    
    // Анимируем указатель
    this.currentPointer.classList.add('click');
    
    // Удаляем класс через 300мс
    setTimeout(() => {
      if (this.currentPointer) {
        this.currentPointer.classList.remove('click');
      }
    }, 300);
  }
  
  /**
   * Добавляет всплывающие подсказки к основным элементам управления
   */
  addFeatureTooltips() {
    // Создаем элемент для всплывающей подсказки
    const tooltip = document.createElement('div');
```

## 4. Добавление динамических подсказок для контекстной помощи

Добавьте следующий JavaScript-код в конец файла `<script>`:

```javascript
/**
 * Класс для управления контекстными подсказками
 */
class ContextualHelpManager {
  /**
   * Создает менеджер контекстных подсказок
   * @param {Object} network - Объект сети
   */
  constructor(network) {
    this.network = network;
    this.storage = window.localStorage;
    this.helpKey = 'rpw_contextual_help_disabled';
    
    // Определение контекстных ситуаций и подсказок
    this.helpContexts = this.defineHelpContexts();

  /**
   * Определяет контексты помощи
   * @returns {Array} - Массив контекстов помощи

  defineHelpContexts() {
    return [
      {
        id: 'first-node-selection',
        condition: () => {
          // Условие: пользователь выбрал узел впервые

                 document.querySelector('.node circle.selected');
        },
        tip: 'Вы выбрали философа! Информация о нем отображается в боковой панели. Дважды щелкните, чтобы увидеть подробную информацию.',

          const infoPanel = document.querySelector('.info-panel');
          if (infoPanel) {

        }
      },

        id: 'first-zoom',
        condition: () => {
          // Условие: пользователь увеличил масштаб впервые

                 this.network.currentZoomLevel > 1.5;
        },
        tip: 'Вы увеличили масштаб! Используйте колесико мыши для масштабирования или кнопку "Сбросить масштаб", чтобы вернуться к общему виду.',

          this.storage.setItem('rpw_zoom_used', 'true');

          const resetBtn = document.getElementById('reset-zoom');
          if (resetBtn) {
            this.highlightElement(resetBtn, 'Нажмите для сброса масштаба');
          }
        }
      },
      {
        id: 'first-search',
        condition: () => {
          // Условие: пользователь начал поиск
          const searchField = document.getElementById('search');
          return !this.storage.getItem('rpw_search_used') && 
                 searchField && searchField.value.length > 0;
        },
        tip: 'Вы используете поиск! Введите имя философа, чтобы найти его в сети. Подсветятся все совпадения.',
        action: () => {
          // Отмечаем, что пользователь использовал поиск
          this.storage.setItem('rpw_search_used', 'true');
        }
      },
      {
        id: 'first-filter',
        condition: () => {
          // Условие: пользователь применил фильтр

        tip: 'Вы применили фильтр! Теперь отображаются только философы и связи, соответствующие выбранным критериям.',

          if (filterPeriod) {

          }
        }
      }
    ];
  }
  
  /**
   * Инициализирует менеджер контекстных подсказок
   */
  init() {
    // Проверяем, отключены ли контекстные подсказки
    if (this.storage.getItem(this.helpKey) === 'true') return;
    
    // Создаем элемент для отображения контекстных подсказок
    this.createTooltipElement();
    
    // Запускаем проверку контекстов каждую секунду
    this.checkInterval = setInterval(() => {
      this.checkContexts();
    }, 1000);
    
    // Добавляем обработчик события зума для обновления уровня масштабирования
    if (this.network.svg) {
      this.network.svg.on('zoom.contextHelp', (event) => {
        this.network.currentZoomLevel = event.transform.k;
      });
    }
  }
  
  /**
   * Создает элемент для отображения контекстных подсказок
   */
  createTooltipElement() {
    // Проверяем, существует ли уже элемент
    if (document.getElementById('contextual-tooltip')) return;
    
    // Создаем элемент подсказки
    const tooltip = document.createElement('div');
    tooltip.id = 'contextual-tooltip';
    tooltip.className = 'tutorial-tooltip';
    tooltip.style.display = 'none';
    tooltip.style.zIndex = '8000';
    
    // Создаем заголовок
    const title = document.createElement('h3');
    title.textContent = 'Подсказка';
    tooltip.appendChild(title);
    
    // Создаем содержимое
    const content = document.createElement('p');
    tooltip.appendChild(content);
    
    // Создаем блок с кнопками
    const actions = document.createElement('div');
    actions.className = 'actions';
    
    // Кнопка "Понятно"
    const okBtn = document.createElement('button');
    okBtn.className = 'next-btn';
    okBtn.textContent = 'Понятно';
    okBtn.addEventListener('click', () => {
      this.hideTooltip();
    });
    actions.appendChild(okBtn);
    
    // Кнопка "Отключить подсказки"
    const disableBtn = document.createElement('button');
    disableBtn.className = 'skip-btn';
    disableBtn.textContent = 'Отключить подсказки';
    disableBtn.addEventListener('click', () => {
      this.disableContextualHelp();
    });
    actions.appendChild(disableBtn);
    
    tooltip.appendChild(actions);
    
    // Добавляем подсказку на страницу
    document.body.appendChild(tooltip);
    this.tooltip = tooltip;
  }
  
  /**
   * Проверяет все контексты помощи
   */
  checkContexts() {
    // Проверяем, не идет ли сейчас обучение
    const tutorialOverlay = document.getElementById('tutorial-overlay');
    if (tutorialOverlay && tutorialOverlay.style.display === 'block') return;
    
    // Не проверяем, если уже отображается подсказка
    if (this.tooltip && this.tooltip.style.display === 'block') return;
    
    // Проверяем каждый контекст
    for (const context of this.helpContexts) {
      if (context.condition()) {
        this.showContextualTip(context);
        break;
      }
    }
  }
  
  /**
   * Показывает контекстную подсказку
   * @param {Object} context - Контекст подсказки
   */
  showContextualTip(context) {
    if (!this.tooltip) return;
    
    // Устанавливаем текст подсказки
    const title = this.tooltip.querySelector('h3');
    const content = this.tooltip.querySelector('p');
    
    title.textContent = 'Подсказка';
    content.textContent = context.tip;
    
    // Позиционируем подсказку в центре экрана
    const windowWidth = window.innerWidth;
    const windowHeight = window.innerHeight;
    
    this.tooltip.style.display = 'block';
    const tooltipRect = this.tooltip.getBoundingClientRect();
    
    this.tooltip.style.left = `${(windowWidth - tooltipRect.width) / 2}px`;
    this.tooltip.style.top = `${(windowHeight - tooltipRect.height) / 2}px`;
    
    // Выполняем действие контекста
    if (context.action) {
      context.action();
    }
    
    // Объявляем для скринридеров
    if (window.announce) {
      window.announce(context.tip);
    }
  }
  
  /**
   * Скрывает контекстную подсказку
   */
  hideTooltip() {
    if (this.tooltip) {
      this.tooltip.style.display = 'none';
    }
    
    // Удаляем подсветку, если она есть
    this.removeHighlight();
  }
  
  /**
   * Отключает контекстные подсказки
   */
  disableContextualHelp() {
    this.hideTooltip();
    
    // Останавливаем интервал проверки
    clearInterval(this.checkInterval);
    
    // Сохраняем настройку
    this.storage.setItem(this.helpKey, 'true');
    
    // Объявляем для скринридеров
    if (window.announce) {
      window.announce('Контекстные подсказки отключены');
    }
  }
  
  /**
   * Подсвечивает элемент и показывает текст
   * @param {HTMLElement} element - Элемент для подсветки
   * @param {string} text - Текст подсказки
   */
  highlightElement(element, text) {
    // Удаляем предыдущую подсветку, если она есть
    this.removeHighlight();
    
    // Получаем размеры и позицию элемента
    const rect = element.getBoundingClientRect();
    
    // Создаем подсветку
    const highlight = document.createElement('div');
    highlight.className = 'tutorial-highlight';
    highlight.style.width = `${rect.width + 10}px`;
    highlight.style.height = `${rect.height + 10}px`;
    highlight.style.left = `${rect.left - 5}px`;
    highlight.style.top = `${rect.top - 5}px`;
    highlight.style.position = 'absolute';
    highlight.style.zIndex = '8010';
    highlight.style.pointerEvents = 'none';
    highlight.style.boxShadow = '0 0 0 2000px rgba(0, 0, 0, 0.3)';
    
    // Создаем указатель
    const pointer = document.createElement('div');
    pointer.className = 'tutorial-pointer';
    pointer.textContent = '👆';
    pointer.style.left = `${rect.left + rect.width / 2}px`;
    pointer.style.top = `${rect.top + rect.height / 2}px`;
    pointer.style.position = 'absolute';
    pointer.style.zIndex = '8030';
    
    // Создаем текстовую подсказку
    const tooltip = document.createElement('div');
    tooltip.className = 'feature-tooltip';
    tooltip.textContent = text;
    tooltip.style.left = `${rect.left + rect.width / 2}px`;
    tooltip.style.top = `${rect.bottom + 10}px`;
    tooltip.style.position = 'absolute';
    tooltip.style.zIndex = '8020';
    tooltip.style.opacity = '1';
    tooltip.style.backgroundColor = 'rgba(0, 0, 0, 0.8)';
    tooltip.style.color = 'white';
    tooltip.style.padding = '8px 12px';
    tooltip.style.borderRadius = '4px';
    tooltip.style.maxWidth = '200px';
    tooltip.style.textAlign = 'center';
    
    // Добавляем элементы на страницу
    document.body.appendChild(highlight);
    document.body.appendChild(pointer);
    document.body.appendChild(tooltip);
    
    // Сохраняем ссылки на элементы
    this.currentHighlight = highlight;
    this.currentPointer = pointer;
    this.currentFeatureTooltip = tooltip;
  }
  
  /**
   * Удаляет подсветку и указатель
   */
  removeHighlight() {
    // Удаляем подсветку
    if (this.currentHighlight) {
      document.body.removeChild(this.currentHighlight);
      this.currentHighlight = null;
    }
    
    // Удаляем указатель
    if (this.currentPointer) {
      document.body.removeChild(this.currentPointer);
      this.currentPointer = null;
    }
    
    // Удаляем подсказку
    if (this.currentFeatureTooltip) {
      document.body.removeChild(this.currentFeatureTooltip);
      this.currentFeatureTooltip = null;
    }
  }
}

/**
```

## 5. Реализация функциональности динамических советов и подсказок

Добавьте следующий JavaScript-код в конец файла `<script>`:

```javascript
/**
 * Класс для управления динамическими советами
 */
class DynamicTipsManager {
  /**
   * Создает менеджер динамических советов
   * @param {Object} network - Объект сети

  constructor(network) {

  /**

      },
      {
        id: 'save-state',
        title: 'Сохраняйте состояние',
        content: 'Вы можете сохранить текущее состояние визуализации, чтобы вернуться к нему позже.',
        condition: () => !this.isShown('save-state') && this.userSpentEnoughTime(),
        priority: 'low'
      },
      {
        id: 'accessibility-options',
        title: 'Настройки доступности',
        content: 'Если у вас есть нарушения цветовосприятия, используйте кнопку настроек доступности.',
        condition: () => !this.isShown('accessibility-options'),
        priority: 'medium'
      },
      {
        id: 'export-visualization',
        title: 'Экспорт визуализации',
        content: 'Вы можете экспортировать текущую визуализацию в формате PNG или SVG для использования в презентациях или публикациях.',

        priority: 'medium'

  /**
   * Инициализирует менеджер динамических советов

    // Создаем панель советов
    this.initTipsPanel();
    
    // Запускаем таймер проверки советов
    setTimeout(() => {
      this.checkForTips();

    // Запускаем периодическую проверку советов
    setInterval(() => {
      this.checkForTips();

  /**
   * Инициализирует панель советов

    if (existingPanel) return;

    const panel = document.createElement('div');

    closeBtn.addEventListener('click', () => {
      this.hideTipPanel();
    });
    
    dismissBtn.addEventListener('click', () => {
      this.hideTipPanel();
      
      // Помечаем текущий совет как показанный
      if (this.currentTip) {
        this.markTipAsShown(this.currentTip.id);
      }
    });
    
    disableBtn.addEventListener('click', () => {
      this.hideTipPanel();
      
      // Отключаем динамические советы
      this.storage.setItem('rpw_dynamic_tips_disabled', 'true');
      
      // Объявляем для скринридеров
      if (window.announce) {
        window.announce('Динамические советы отключены');
      }
    });

  /**
   * Проверяет наличие подходящих советов для отображения
   */
  checkForTips() {
    // Проверяем, отключены ли советы
    if (this.storage.getItem('rpw_dynamic_tips_disabled') === 'true') return;
    
    // Проверяем, не идет ли сейчас обучение
    const tutorialOverlay = document.getElementById('tutorial-overlay');
    if (tutorialOverlay && tutorialOverlay.style.display === 'block') return;
    
    // Проверяем, не отображается ли уже какой-то совет
    if (this.tipsPanel && this.tipsPanel.style.opacity === '1') return;
    
    // Получаем доступные советы, удовлетворяющие условиям
    const availableTips = this.tips.filter(tip => tip.condition());
    
    if (availableTips.length === 0) return;
    
    // Сортируем советы по приоритету
    availableTips.sort((a, b) => {
      const priorityValues = { high: 3, medium: 2, low: 1 };
      return priorityValues[b.priority] - priorityValues[a.priority];
    });
    
    // Показываем совет с наивысшим приоритетом
    this.showTip(availableTips[0]);
  }
  
  /**
   * Показывает совет
   * @param {Object} tip - Объект совета
   */
  showTip(tip) {
    if (!this.tipsPanel) return;
    
    // Сохраняем текущий совет
    this.currentTip = tip;
    
    // Устанавливаем содержимое совета
    const title = this.tipsPanel.querySelector('h3');
    const content = this.tipsPanel.querySelector('.tip-content');
    
    title.textContent = tip.title;
    content.textContent = tip.content;
    
    // Отображаем панель
    this.tipsPanel.style.display = 'block';
    
    // Задержка для анимации
    setTimeout(() => {
      this.tipsPanel.style.transform = 'translateY(0)';
      this.tipsPanel.style.opacity = '1';
    }, 10);
    
    // Объявляем для скринридеров
    if (window.announce) {
      window.announce(`Совет: ${tip.title}. ${tip.content}`);
    }
    
    // Автоматически скрываем совет через 15 секунд
    this.autoHideTimeout = setTimeout(() => {
      this.hideTipPanel();
    }, 15000);
  }
  
  /**
   * Скрывает панель советов
   */
  hideTipPanel() {
    if (!this.tipsPanel) return;
    
    // Анимация скрытия
    this.tipsPanel.style.transform = 'translateY(20px)';
    this.tipsPanel.style.opacity = '0';
    
    // Очищаем таймер автоматического скрытия
    clearTimeout(this.autoHideTimeout);
    
    // Полностью скрываем панель после анимации
    setTimeout(() => {
      this.tipsPanel.style.display = 'none';
    }, 300);
  }
  
  /**
   * Проверяет, был ли показан совет
   * @param {string} tipId - Идентификатор совета
   * @returns {boolean} - true, если совет был показан
   */
  isShown(tipId) {
    return this.shownTips.includes(tipId);
  }
  
  /**
   * Получает список показанных советов
   * @returns {Array} - Массив идентификаторов показанных советов
   */
  getShownTips() {
    const tipsJSON = this.storage.getItem(this.tipsKey);
    return tipsJSON ? JSON.parse(tipsJSON) : [];
  }
  
  /**
   * Помечает совет как показанный
   * @param {string} tipId - Идентификатор совета
   */
  markTipAsShown(tipId) {
    // Добавляем совет в список показанных
    this.shownTips.push(tipId);
    
    // Сохраняем обновленный список
    this.storage.setItem(this.tipsKey, JSON.stringify(this.shownTips));
  }
  
  /**
   * Проверяет, провел ли пользователь достаточно времени на странице
   * @returns {boolean} - true, если пользователь провел достаточно времени
   */
  userSpentEnoughTime() {
    // Получаем время начала сессии
    const sessionStart = this.storage.getItem('rpw_session_start');
    
    if (!sessionStart) {
      // Если сессия только началась, сохраняем текущее время
      this.storage.setItem('rpw_session_start', Date.now().toString());
      return false;
    }
    
    // Проверяем, прошло ли достаточно времени (3 минуты)
    const timeSpent = Date.now() - parseInt(sessionStart);
    return timeSpent >= 3 * 60 * 1000;
  }
  
  /**
   * Получает количество взаимодействий пользователя
   * @returns {number} - Количество взаимодействий
   */
  interactionsCount() {
    // Получаем счетчик взаимодействий
    const count = parseInt(this.storage.getItem('rpw_interactions_count') || '0');
    
    // Увеличиваем счетчик при каждой проверке
    this.storage.setItem('rpw_interactions_count', (count + 1).toString());
    
    return count;
  }
}

/**
 * Инициализирует сессию пользователя
 */
function initSession() {
  // Проверяем, инициализирована ли уже сессия
  if (!localStorage.getItem('rpw_session_start')) {
    localStorage.setItem('rpw_session_start', Date.now().toString());
    localStorage.setItem('rpw_interactions_count', '0');
  }
  
  // Увеличиваем счетчик взаимодействий при различных действиях пользователя
  const incrementInteraction = () => {
    const count = parseInt(localStorage.getItem('rpw_interactions_count') || '0');
    localStorage.setItem('rpw_interactions_count', (count + 1).toString());
  };
  
  // Регистрируем различные события пользователя
  document.addEventListener('click', incrementInteraction);
  document.addEventListener('keydown', incrementInteraction);
  document.addEventListener('wheel', incrementInteraction);
  
  // Регистрируем события на основных элементах управления
  const controls = [
    'filter-period', 'filter-relation', 'search', 'reset-zoom',
    'toggle-clusters', 'toggle-labels', 'auto-clustering', 'clustering-level',
    'save-state-btn', 'load-state-btn'
  ];
  
  controls.forEach(id => {
    const element = document.getElementById(id);
    if (element) {
      element.addEventListener('click', incrementInteraction);
      element.addEventListener('change', incrementInteraction);
    }
  });
}
```
